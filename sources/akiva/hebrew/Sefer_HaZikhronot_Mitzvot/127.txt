~ שֶׁהִיא מֵעֵבֶר לְכַדּוּר הָאָרֶץ מִזֶּה, וְהַשֶּׁמֶשׁ נִכְחָהּ מֵעֵבֶר הַשֵּׁנִי, וְכַדּוּר הָאָרֶץ בֵּינְתַיִם מוֹנֵעַ אוֹר הַשֶּׁמֶשׁ מֵהַשְׁפִּיעַ עַל הַיָּרֵחַ — אָז, אִם יֶאֱרַע זֶה בַּלַּיְלָה, יִהְיֶה לִקּוּי הַלְּבָנָה. וְזֶה אִי אֶפְשָׁר אֶלָּא בְּאֶמְצַע הַחֹדֶשׁ: כִּי בִּתְחִלַּת הַחֹדֶשׁ הַלְּבָנָה מִתְדַּבֶּקֶת עִם הַשֶּׁמֶשׁ, אֲשֶׁר זֶהוּ סִבַּת מְנִיעַת אוֹרָהּ; וְעַד חֲצִי הַחֹדֶשׁ — הוֹלֶכֶת וּמִתְרַחֶקֶת מִמֶּנָּה, עַד שֶׁבְּרֶגַע הַחֲצִי חֹדֶשׁ — אָז הִיא בְּתַכְלִית הָרִחוּק מֵהַשֶּׁמֶשׁ, עַד שֶׁהִיא נֶגְדִּית לָהּ; וּמִשָּׁם וְאֵילָךְ — חוֹזֶרֶת לְהִתְקָרֵב בַּחֲצִי חֹדֶשׁ אַחֲרוֹן, עַד סָמוּךְ לְרֹאשׁ חֹדֶשׁ, שֶׁאָז מִתְדַּבֶּקֶת וְנִמְנָע אוֹרָהּ. וְנִמְצָא, רֶגַע נִגּוּדָהּ לַשֶּׁמֶשׁ — הוּא רֶגַע חֲצִי חֹדֶשׁ הָאֲמִתִּי:

**וּבְסֵפֶר** פ״מ כָּתַב, וְזֶה לְשׁוֹנוֹ: "וְקָשֶׁה לִי: אִם יִהְיֶה הַלִּקּוּי — שֶׁהוּא וְכוּלֵי — אֶמְצַע כ״ט י״ב תשצ״ג, אִם כֵּן אֵיךְ נַעֲשֶׂה בַּחֶשְׁבּוֹן" וְכוּלֵי, עַד כָּאן. וְלָא קַשְׁיָא מִידֵּי; וּכְבָר נִתְבָּאֵר לְעֵיל (פֶּרֶק ב׳ וּפֶרֶק ג׳), כִּי חֶשְׁבּוֹן כ״ט י״ב תשצ״ג אֵינוֹ אֶלָּא חֶשְׁבּוֹן הָאֶמְצָעִי; אֲבָל חֶשְׁבּוֹן אֲמִתִּי, שֶׁהָיוּ חוֹשְׁבִין בּוֹ כְּשֶׁהָיוּ מְקַדְּשִׁים עַל פִּי הָרְאִיָּה — הוּא אַחֵר, וְאֵינוֹ מְכֻוָּן לְחֶשְׁבּוֹן זֶה. אֲבָל מִכָּל מָקוֹם, אָנוּ עוֹשִׂים הַכֹּל עַל פִּי חֶשְׁבּוֹן כ״ט י״ב תשצ״ג, כִּי אִי אֶפְשָׁר לִמְסוֹר לַכֹּל חֶשְׁבּוֹן הָאֲמִתִּי הֶעָמֹק; וְסָמְכוּ עַל מַה שֶּׁאָמְרוּ: "אַתֶּם — אֲפִלּוּ מְזִידִים" וְכוּלֵי. אֲבָל לְעִנְיַן קִדּוּשׁ הַלְּבָנָה, שֶׁצָּרִיךְ לָדַעַת חִדּוּשָׁהּ הָאֲמִתִּי — וַדַּאי, כֵּיוָן דְּרָאִינוּ לִקּוּי, וַדַּאי אָז הוּא חֲצִי חֹדֶשׁ אֲמִתִּי, וְלֹא יְבָרֵךְ מִשָּׁם וָהָלְאָה:

**וְנִרְאֶה** לְפִי עֲנִיּוּת דַּעְתִּי: הַבָּקִי בַּחֶשְׁבּוֹן, וְיוֹדֵעַ תָּמִיד אֶת חֹדֶשׁ הָאֲמִתִּי — הַתְחָלָתוֹ וְסוֹפוֹ — לֹא יְבָרֵךְ אֶלָּא עַד חֲצִי חֹדֶשׁ אֲמִתִּי. מִיהוּ, צָרִיךְ עִיּוּן בָּזֶה, דְּיֵשׁ לוֹמַר סְבָרָא לְבָרֵךְ — דְּכָךְ תִּקְּנוּ חֲכָמִים לִהְיוֹתֵנוּ עוֹשִׂים עַל פִּי חֶשְׁבּוֹן אֶמְצָעִי כָּל מִילֵּי; וְרַק בְּלִקּוּי, דְּנִרְאֶה לְעֵין כֹּל — שַׁפִּיר דָּמֵי לִסְמוֹךְ עַל זֶה; לֹא עַל חֶשְׁבּוֹן הַיָּדוּעַ רַק לַיּוֹדְעִים. וְנִרְאֶה לְבָרֵךְ בְּלִי שֵׁם וּמַלְכוּת. וְכֵן, אִם חֲצִי חֹדֶשׁ אֲמִתִּי נִמְשָׁךְ יוֹתֵר מֵחֲצִי כ״ט י״ב תשצ״ג — נִרְאֶה דִּיבָרֵךְ בְּלִי שֵׁם וּמַלְכוּת:

**וְאֵין** מְבָרְכִין עַל הַיָּרֵחַ אֶלָּא בְּמוֹצָאֵי שַׁבָּת אוֹ יוֹם טוֹב, כְּשֶׁהוּא מְבֻשָּׂם וּבְגָדָיו נָאִים. אֲבָל אִם אֵין עוֹד זְמַן ה׳ לֵילוֹת עִם לֵיל מוֹצָאֵי שַׁבָּת לְקַדֵּשׁ — לֹא יַמְתִּינוּ עַד מוֹצָאֵי שַׁבָּת, אֶלָּא יְקַדְּשׁוּ בְּחֹל. וְר״י עֶמְדִּין כָּתַב, דְּכָל שֶׁמּוֹצָאֵי שַׁבָּת יִהְיֶה יוֹתֵר מִז׳ — אֵין צָרִיךְ לְהַמְתִּין עַד מוֹצָאֵי שַׁבָּת; אֶלָּא אִם כֵּן הוּא תּוֹךְ ז׳ — דְּאָז יַמְתִּינוּ עַד מוֹצָאֵי שַׁבָּת. וְיֵשׁ לִנְהוֹג כֵּן בִּימוֹת הַגְּשָׁמִים, דִּשְׁכִיחֵי עֲנָנִים, וְיֵשׁ לָחוּשׁ שֶׁמָּא יְכֻסֶּה בָּעֲנָנִים. אֲבָל בְּלֵיל שַׁבָּת וְיוֹם טוֹב — אֵין לְקַדֵּשׁ, אֲפִלּוּ חָל יוֹם טוֹב בְּמוֹצָאֵי שַׁבָּת; אִם לֹא כְּשֶׁאִי אֶפְשָׁר לְקַדֵּשׁ אַחַר כָּךְ. וְכֵן אֵין
