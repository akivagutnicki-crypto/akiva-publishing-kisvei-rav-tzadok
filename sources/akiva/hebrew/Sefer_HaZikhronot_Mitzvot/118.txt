~ לְאָמְרָהּ, לְפִי שֶׁעֲתִידִין לְקָרוֹתָהּ בְּסֵפֶר תּוֹרָה; וְלָכֵן מִי שֶׁנִּמְנַע לְאָמְרָהּ — אֵין מַזְנִיחִים אוֹתוֹ. וְאַחַר הַתְּפִלָּה — מִנְהַג קַדְמוֹנִים לוֹמַר הַלֵּל (וְאֵין הַנָּשִׁים חַיָּבוֹת; וְכָתַב ר״י עֶמְדִּין, דְּאַף עַל פִּי שֶׁמַּחְמִירוֹת עַל עַצְמָן — רָאוּי שֶׁלֹּא יְבָרְכוּ, כֵּיוָן דִּבְלָאו הָכֵי מַחֲלֹקֶת אִם לְבָרֵךְ. וּלְפִיכָךְ אֵין יְכוֹלִים לְהוֹצִיא אֲחֵרִים, אִם לֹא שֶׁעוֹנֶה אַחֲרֵיהֶם; "וְתָבוֹא לוֹ מְאֵרָה" — שֶׁלֹּא לָמַד, אוֹ שֶׁעוֹשֶׂה שְׁלוּחִים כָּאֵלֶּה. וְדִין הַלֵּל בְּבֵית אָבֵל — עַיֵּן בִּמְקוֹמוֹ), בְּדִלּוּג פָּרָשַׁת "לֹא לָנוּ" חֶצְיוֹ, וּפָרָשַׁת "אָהַבְתִּי" חֶצְיוֹ. וְכוֹפְלִין פְּסוּקֵי "אוֹדְךָ" וְכוּלֵי, וּפְסוּקֵי "הוֹדוּ" וְכוּלֵי "כִּי לְעוֹלָם חַסְדּוֹ", וְ"אָנָּא" וְכוּלֵי — הַשְּׁלִיחַ צִבּוּר אוֹמֵר, וְהַקָּהָל עוֹנִים אַחֲרָיו. וְצָרִיךְ לִשְׁמוֹעַ מִפִּי הַשְּׁלִיחַ צִבּוּר, וְאִם לֹא שָׁמַע — לֹא יָצָא; לָכֵן יֹאמַר גַּם כֵּן פְּסוּקִים שֶׁאוֹמֵר הַשְּׁלִיחַ צִבּוּר, דְּשֶׁמָּא לֹא יִשְׁמַע יָפֶה. וּמִי שֶׁאֵינוֹ בָּקִי — אַחֵר יָכוֹל לְהוֹצִיאוֹ, אֲפִלּוּ בְּכָל הַהַלֵּל; אֲבָל הַבָּקִי — אֵין מוֹצִיאוֹ, אֶלָּא אִם כֵּן עוֹנֶה אַחֲרָיו "הַלְלוּיָהּ":

**וּלְבָרֵךְ** עָלָיו מִקֹּדֶם — יֵשׁ אוֹמְרִים דְּאֵין מְבָרֵךְ; וְיֵשׁ אוֹמְרִים דְּצִבּוּר מְבָרְכִים וְלֹא יָחִיד; וְיֵשׁ אוֹמְרִים דְּאַף יָחִיד מְבָרֵךְ בַּתְּחִלָּה: "אֲשֶׁר קִדְּשָׁנוּ בְּמִצְוֹתָיו וְצִוָּנוּ לִקְרוֹת אֶת הַהַלֵּל"; וְאִם בֵּרַךְ "לִגְמוֹר" — יָצָא; וּלְבַסּוֹף "יְהַלְלוּךָ"; וְכֵן נוֹהֲגִים. וּמִכָּל מָקוֹם, יִזָּהֵר לִקְרוֹת בְּצִבּוּר, כְּדֵי לְבָרֵךְ עָלָיו; וּבְלָאו הָכֵי רָאוּי לִקְרוֹת פְּסוּקֵי "הוֹדוּ" וְכוּלֵי בְּצִבּוּר. וְלָכֵן, אֲפִלּוּ לְהַפְסִיק בְּאֶמְצַע פְּסוּקֵי דְּזִמְרָה — רַשַּׁאי, כְּדֵי לִקְרוֹת עִם הַצִּבּוּר; אֶלָּא שֶׁאָז לֹא יְבָרֵךְ, דְּיָצָא בְּבִרְכַּת "בָּרוּךְ שֶׁאָמַר" וְ"יִשְׁתַּבַּח". וְאִם הוּא קֹדֶם הַתְּפִלָּה — יֹאמַר עִמָּהֶם וִיבָרֵךְ, וְאַחַר כָּךְ יִתְפַּלֵּל. וּבְיָחִיד — יֵשׁ אוֹמְרִים דְּיִקְרָא לְב׳ שֶׁיֹּאמְרוּ עִמּוֹ כָּל פָּסוּק, דְּאָז הָוֵי כְּרַבִּים; וְנָהֲגוּ כֵן בְּ"הוֹדוּ", דִּבְלָאו הָכֵי רָאוּי לִהְיוֹת עוֹד ב׳ — דְּאִי לָאו הָכֵי, לְמִי יֹאמַר "הוֹדוּ"? וְלֹא בְּ"אָנָּא". וְאִם אֵין מְצוּיִים עוֹד ב׳ — אֵין צָרִיךְ לְהַטְרִיחַ אַחֲרֵיהֶם:

**וְלֹא** יַפְסִיק, אֶלָּא אִם כֵּן לְהָשִׁיב שָׁלוֹם לְכָל אָדָם (נִרְאֶה — אֲפִלּוּ לְנָכְרִי), אוֹ לִשְׁאוֹל שָׁלוֹם לְאָדָם שֶׁצָּרִיךְ לִנְהוֹג בּוֹ כָּבוֹד, דְּמֻתָּר אֲפִלּוּ בִּלְשׁוֹן לַעַ״ז; וּבִלְבַד כְּשֶׁדֶּרֶךְ לִתֵּן לוֹ שָׁלוֹם וּלְהָשִׁיב. אֲבָל אִם הוּא בְּבֵית הַכְּנֶסֶת, שֶׁאֵין דֶּרֶךְ לִתֵּן שָׁלוֹם וּלְהַחֲזִיר שָׁלוֹם כְּלָל — אָסוּר לְהַפְסִיק. וְאִם הִפְסִיק — בֵּין בְּדִבּוּר בֵּין בִּשְׁתִיקָה — וְשָׁהָה, אִם כְּדֵי לִגְמוֹר אֶת כֻּלָּהּ — חוֹזֵר לָרֹאשׁ; וְיֵשׁ אוֹמְרִים דְּאֵין צָרִיךְ לַחְזוֹר, וְכֵן עִקָּר; וְהַמַּחְמִיר — יִקְרָא בְּלֹא בְּרָכָה. וְלֹא יִקְרָא לְמַפְרֵעַ, וְאִם קָרָא — לֹא יָצָא; לְפִיכָךְ, אִם טָעָה בְּפָסוּק אֶחָד — אֵינוֹ אוֹמְרוֹ בְּמָקוֹם שֶׁעוֹמֵד, דַּהֲוֵי לֵיהּ לְמַפְרֵעַ, אֶלָּא יַחְזוֹר לְמָקוֹם שֶׁטָּעָה. וְסָפֵק אִם קָרָא הַלֵּל — נִרְאֶה לִי דְּיַחְזוֹר בְּלֹא בְּרָכָה; אִם לֹא שֶׁהָיָה זָכוּר, בֵּין בְּשַׁחֲרִית בֵּין בְּמוּסָף, שֶׁהוּא רֹאשׁ חֹדֶשׁ, וְלֹא שָׁהָה בֵּינְתַיִם, בְּעִנְיָן שֶׁבְּוַדַּאי הָיָה זָכוּר כָּל אוֹתוֹ הָעֵת שֶׁהוּא רֹאשׁ חֹדֶשׁ — אָז אָמְרִינַן דְּוַדַּאי אֲמָרוֹ;
