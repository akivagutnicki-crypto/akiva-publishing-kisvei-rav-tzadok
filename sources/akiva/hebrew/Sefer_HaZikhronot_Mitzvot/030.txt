~ בְּבֵיתוֹ, וְהוּא גַּם כֵּן סְעוּדַת מִצְוָה. וְנִרְאֶה לִי: הוֹאִיל וְלֹא נָהֲגוּ בִּסְעוּדָה גְּמוּרָה — שֶׁאֵין הַסְּעוּדָה נִקְרֵאת אֶלָּא עַל שֵׁם הַלֶּחֶם, וְעַל כָּל פָּנִים כְּהוֹרָאַת הַגְּאוֹנִים לְעִנְיַן קִדּוּשׁ בִּמְקוֹם סְעוּדָה, בִּכְזַיִת מֵחֲמֵשֶׁת הַמִּינִין שֶׁמְּבָרְכִים עָלָיו "בּוֹרֵא מִינֵי מְזוֹנוֹת", אוֹ רְבִיעִית יַיִן שֶׁמְּבָרְכִין עָלָיו "בּוֹרֵא פְּרִי הַגָּפֶן" — וְהוֹאִיל וְלֹא נָהֲגוּ אֶלָּא בְּמִינֵי פֵּרוֹת, וְכַהַאי גַּוְנָא דְּלָאו מֵעִקְּרֵי סְעוּדָה — אֵין לִמְחוֹת בְּמִי שֶׁאֵינוֹ הוֹלֵךְ לִסְעוּדָה זוֹ; כֵּן נִרְאֶה. וְעוֹד נִרְאֶה: דְּאִם תָּאֵב לִישׁוֹן לְעֹנֶג שַׁבָּת, וְהַהֲלִיכָה קָשָׁה לוֹ — אֶפְשָׁר לְהָקֵל נַמִּי מִשּׁוּם עֹנֶג שַׁבָּת, שֶׁלֹּא יֵלֵךְ כְּשֶׁזִּמְּנוּהוּ, אֲפִלּוּ יִהְיֶה סְעוּדָה גְּמוּרָה. וּמְבַטְּלִין סְעוּדַת מִצְוָה בִּשְׁבִיל מִצְוֹת אֲחֵרוֹת — תְּרוּמַת הַדֶּשֶׁן סִימָן ק״י, מִגְּמָרָא דְּפֶרֶק "אוֹתוֹ וְאֶת בְּנוֹ":

# מִן פָּרָשַׁת וַיֵּרָא עַד הֵנָּה אֵין בּוֹ מִצְוָה

# מִצְוַת לֹא תַעֲשֶׂה א׳ — פֶּרֶק א׳
# פָּרָשַׁת וַיִּשְׁלַח

(ג) **מִצְוַת לֹא תַעֲשֶׂה** שֶׁלֹּא לֶאֱכוֹל גִּיד הַנָּשֶׁה, שֶׁנֶּאֱמַר: "עַל כֵּן לֹא יֹאכְלוּ בְנֵי יִשְׂרָאֵל אֶת גִּיד הַנָּשֶׁה"; וְאֵינוֹ דֶּרֶךְ סִפּוּר, אֶלָּא אַזְהָרַת הַתּוֹרָה שֶׁלֹּא יֹאכְלוּהוּ. וְנֶאֶמְרָה הָאַזְהָרָה בְּסִינַי, אֶלָּא שֶׁנִּכְתְּבָה בִּמְקוֹמָהּ (עַיֵּן בְּסֵפֶר חֲסִידִים סִימָן רל״א, שֶׁכָּתַב: כָּךְ אֵירַע לוֹ בִּשְׁבִילֵנוּ, לְכָךְ נָדְרוּ שֶׁלֹּא לֶאֱכוֹל גִּיד הַנָּשֶׁה וְכוּלֵי, עַיֵּן שָׁם, וּדְבָרָיו צָרִיךְ עִיּוּן). וְכָתַב הָרַ״ם הַבַּבְלִי: כַּאֲשֶׁר נֶאֱכַל הַגִּיד מִתְחַזֵּק כֹּחַ שָׂרוֹ שֶׁל עֵשָׂו, כִּי שָׁם רְבִיצַת סַמָּאֵ״ל וּמַחֲנוֹתָיו; וּשְׁסַ״ה לָאוִין הֵם נֶגֶד שְׁסַ״ה יְמוֹת הַשָּׁנָה, וְיַגִּיעַ לָאו דְּלֹא יֹאכַל גִּיד הַנָּשֶׁה נֶגֶד ט׳ בְּאָב (לְדִבְרָיו יוֹתֵר נָכוֹן: שְׁסַ״ה גִּידִים נֶגֶד שְׁסַ״ה יָמִים, וּמוּבָן). וְנוֹהֶגֶת מִצְוָה זוֹ בְּכָל מָקוֹם וּבְכָל זְמַן, בִּזְכָרִים וּבִנְקֵבוֹת. וְעוֹבֵר עַל זֶה וְאָכַל גִּיד שָׁלֵם — אֲפִלּוּ הוּא פָּחוֹת מִכְּזַיִת — אוֹ שֶׁאָכַל כְּזַיִת מִגִּיד הַנָּשֶׁה גָּדוֹל — הֲרֵי זֶה לוֹקֶה. וְעַיֵּן בַּזֹּהַר וַיִּשְׁלַח (ק״ע עַמּוּד ב׳): "וְאַמַּאי אִקְרֵי גִּיד הַנָּשֶׁה? כְּלוֹמַר: גִּיד דְּאִיהוּ מְנַשֶּׁה (מַשְׁכֵּחַ) לִבְנֵי נָשָׁא לְפוּלְחָנָא דְּמָארֵיהוֹן, וְתַמָּן הוּא יֵצֶר הָרָע רָבִיץ" וְכוּלֵי; "וְאִית בְּבַר נָשׁ שְׁסַ״ה גִּידִין, וּלְקָבְלֵיהוֹן שְׁסַ״ה פִּקּוּדִין" וְכוּלֵי, "וּלְקָבֵיל שְׁסַ״ה יוֹמֵי שַׁתָּא, וְהָא ט׳ בְּאָב חַד מִנְּהוֹן" וְכוּלֵי, עַד כָּאן. מַשְׁמָע כְּמוֹ שֶׁכָּתַבְתִּי. וְעַיֵּן עוֹד שָׁם (קע״א עַמּוּד א׳): "תָּא חֲזֵי, מֵהַאי נָחָשׁ כַּמָּה חֵילִין מִתְפָּרְשָׁן לְכָל סְטַר, וְאִשְׁתַּכַּח בְּעָלְמָא גַּבֵּי בַּר נָשׁ, וּבָעָא לְקִיּוּמָא לְהַהוּא גִּיד הַנָּשֶׁה" וְכוּלֵי, עַיֵּן שָׁם. וְכָתַב הַפְּרִי מְגָדִים: דְּאִם אָכַל גִּיד הַנָּשֶׁה כְּזַיִת בִּכְדֵי אֲכִילַת פְּרָס — אֵין לוֹקֶה אֶלָּא כְּזַיִת כְּאַחַת; וּמִזֶּה יָרַד לְסַפֵּק אִם חֲצִי שִׁעוּר אָסוּר בְּגִיד הַנָּשֶׁה, עַיֵּן שָׁם:
