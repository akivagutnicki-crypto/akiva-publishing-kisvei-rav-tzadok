~ כְּבַאֲכִילָה. וְאָסוּר לִתְּנוֹ לֶחָתוּל וּלְכֶלֶב שֶׁמְּזוֹנָתָן עָלָיו; וְכֵן אָסוּר לִשְׁלוֹחַ מַתָּנָה לְעוֹבֵד כּוֹכָבִים בִּשְׁלֵמָה, בְּעִנְיָן שֶׁהַגּוֹי מַחֲזִיק לוֹ טוֹבָה יוֹתֵר בִּשְׁבִיל הַגִּיד שֶׁאֵין בּוֹ טַעַם (אִם לֹא שֶׁיֵּדַע שֶׁנּוֹתֵן מָעוֹת יוֹתֵר, כְּגוֹן שֶׁצָּרִיךְ הַגִּיד לְדָבָר מָה, אוֹ לִתְּנוֹ לְכַלְבּוֹ וְחָתוּלוֹ, וְכַהַאי גַּוְנָא — דְּאָז וַדַּאי דְּאָסוּר), וְגַם אֵין מַחֲזִיק לוֹ טוֹבָה כֵּיוָן דִּמְזַכְּרוֹ. וְיֵשׁ לְהִסְתַּפֵּק אִם גַּם בְּגִיד הַחִיצוֹן, וְהַפְּנִימִי לְמַעְלָה מֵהַכַּף, וּקְנוּקְנוֹת, דְּמִדְּרַבָּנָן — אִם אָסוּר בַּהֲנָאָה (וְעַיֵּן רֵישׁ סִימָן ס״ז בַּבַּיִת חָדָשׁ). אֲבָל שׁוּמָנוֹ — פָּשׁוּט דְּמֻתָּר, דְּיִשְׂרָאֵל קְדוֹשִׁים נָהֲגוּ אִסּוּר אֲכִילָה, לֹא הֲנָאָה (כַּנִּזְכָּר לְעֵיל בְּסַכִּין כַּהַאי גַּוְנָא; וְכֵן כָּתַב הַפְּרִי חָדָשׁ). וּלְעִנְיַן תַּעֲרוֹבְתּוֹ כְּשֶׁאֵין מַכִּירוֹ, דַּהֲוֵי בְּרִיָּה — יֵשׁ לְצַדֵּד לְהָרַמְבַּ״ם דְּאֵין לוֹקִין עַל הֲנָאָה, וְהֵיכִי דְּאֵין לוֹקִין לָא שַׁיָּךְ בְּרִיָּה (עַיֵּן פְּרִי מְגָדִים שַׁעַר הַתַּעֲרוֹבוֹת רפ״ב), וְאִם כֵּן לְעִנְיַן אִסּוּר הֲנָאָה לָא שַׁיָּךְ בְּרִיָּה. אֲבָל כְּבָר נִתְבָּאֵר דִּלְכוּלֵּי עָלְמָא לוֹקִין כָּאן. וְנָכוֹן לְהַחְמִיר כְּדֵעָה זוֹ בְּאִסּוּר דְּאוֹרַיְתָא, בִּפְרָט דְּבַזֹּהַר הַקָּדוֹשׁ אִיתָא כֵּן, וַאֲפִלּוּ בְּגִיד הַנָּשֶׁה דִּטְרֵפָה — שֶׁלֹּא לֵהָנוֹת. וְיֵשׁ לְסַפֵּק בִּטְרֵפָה מִן הַבֶּטֶן, לְמַאן דְּאָמַר דְּאֵין נוֹהֵג בִּטְמֵאִים; וְלִכְאוֹרָה תַּלְיָא בְּמַחְלֹקֶת דְּשָׁלִיל, אִי מִשְּׁעַת יְצִירָה אִסּוּר גִּיד. מִיהוּ לְהַמְעַיֵּן בְּרֵישׁ פֶּרֶק גִּיד הַנָּשֶׁה (דַּף צ׳ עַמּוּד א׳), עַיֵּן שָׁם בַּתּוֹסָפוֹת דִּבּוּר הַמַּתְחִיל "אַלְמָא" — מַשְׁמָע דְּאֵין נוֹהֵג, דִּטְרֵפָה קָדִים; וְכֵן כָּתַב הַפְּרִי מְגָדִים:

# מִן וַיֵּשֶׁב עַד הֵנָּה אֵין שׁוּם מִצְוָה

# מִצְוַת עֲשֵׂה ג׳ — פֶּרֶק א׳
# פָּרָשַׁת בֹּא

(ד) **מִצְוַת עֲשֵׂה** לְקַדֵּשׁ חֳדָשִׁים, שֶׁנֶּאֱמַר: "הַחֹדֶשׁ הַזֶּה לָכֶם רֹאשׁ חֳדָשִׁים" — כְּלוֹמַר: כְּשֶׁתִּרְאוּ חִדּוּשָׁהּ שֶׁל לְבָנָה, תִּקְבְּעוּ לָכֶם רֹאשׁ חֹדֶשׁ. וְכֵן תַּכְלוּל מִצְוָה זוֹ מִצְוַת עִבּוּר שָׁנָה, לְפִי שֶׁיְּסוֹד מִצְוַת קִדּוּשׁ הַחֹדֶשׁ כְּדֵי שֶׁיַּעֲשׂוּ הַמּוֹעֲדִים בְּמוֹעֲדָם, וְכֵן מִצְוַת עִבּוּר שָׁנִים מִזֶּה הַיְסוֹד. וּבָאוּ הֶעָרוֹת בַּכְּתוּבִים יוֹרוּ עַל עִבּוּר שָׁנָה, כְּמוֹ "וְשָׁמַרְתָּ אֶת הַחֻקָּה הַזֹּאת לְמוֹעֲדָהּ", וְכֵן "שָׁמוֹר אֶת חֹדֶשׁ הָאָבִיב". זֶהוּ דַּעַת הָרַמְבַּ״ם; אֲבָל בַּעַל הֲלָכוֹת גְּדוֹלוֹת וְהָרַמְבַּ״ן מוֹנִים קִדּוּשׁ הַחֹדֶשׁ מִצְוָה אַחַת, וְעִבּוּר הַשָּׁנָה מִצְוָה אַחַת, וְהַפָּסוּק הַמּוֹרֶה עַל מִצְוַת הָעִבּוּר — שֶׁהוּא חִשּׁוּב הַתְּקוּפוֹת — הוּא מַה שֶּׁנֶּאֱמַר "וְשָׁמַרְתָּ אֶת הַחֻקָּה" וְ"שָׁמוֹר אֶת חֹדֶשׁ הָאָבִיב", כַּנִּזְכָּר. וְעוֹבֵר עָלֶיהָ וְלֹא עֲשָׂאָהּ, אִם הוּא חָכָם הָרָאוּי לָהּ, כְּמוֹ שֶׁיִּתְבָּאֵר — בִּטֵּל עֲשֵׂה, וְעָנְשׁוֹ גָּדוֹל מְאֹד, שֶׁגּוֹרֵם קִלְקוּל הַמּוֹעֲדוֹת. וְעַיֵּן
