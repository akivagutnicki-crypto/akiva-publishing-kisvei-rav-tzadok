~ תָּמוּהַּ, וְלֹא דַּק יָפֶה. דְּשָׁם הֵתִיב ריב״ס: וְהָתְנִינַן, שֶׁעַל מַהֲלַךְ לַיְלָה מְחַלְּלִין אֶת הַשַּׁבָּת וְכוּלֵי — ור״ל: וַהֲרֵי אֵין מְקַדְּשִׁין בַּלַּיְלָה, וּמַדּוּעַ מְחַלְּלִין בַּלַּיְלָה? וּמְשַׁנֵּי הָתָם: וְאַחַר כָּךְ אָמַר ר״י בר״כ: וְלֹא לְמַפְרֵעַ הוּא קָדוֹשׁ? וְכֵיוָן שֶׁקָּדוֹשׁ לְמַפְרֵעַ — הוּא יוֹם, הוּא לַיְלָה, עַד כָּאן. וְעַיֵּן שָׁם פֵּרוּשׁוֹ, דְּהוּא קֻשְׁיָא וְתֵרוּץ — וְאֵין לוֹ שַׁחַר; דְּאֵיךְ סָלְקָא דַּעְתָּךְ דְּרִיב״ס לְהַקְשׁוֹת טְפֵי אַלַּיְלָה מֵעַל עִקַּר חִלּוּל שַׁבָּת, אַף דִּימָמָא? אֲבָל וַדַּאי דְּמִצְוַת קִדּוּשׁ בִּזְמַנּוֹ, וְאַחַר כָּךְ מְאַיְּמִין. וְתוּ קָשֶׁה: מַאי שְׁנָא דְּר״י בר״כ הָכָא? אֲבָל פָּשׁוּט, דְּכַוָּנַת ר״י בר״כ לְיַשֵּׁב גַּם כֵּן קֻשְׁיַת ר״י בר״ס, וְהַכֹּל מִדִּבְרֵי ר״י בר״כ, דְּקָאָמַר: וַהֲלֹא לְמַפְרֵעוֹ הוּא קָדוֹשׁ — ר״ל: אַחַר כָּךְ, כְּשֶׁמְּקַדְּשִׁים — הֲרֵי לְמַפְרֵעַ הוּא קָדוֹשׁ, וְגַם הַלַּיְלָה קְדוֹשָׁה לְמַפְרֵעַ, לִהְיוֹת מֵרֹאשׁ חֹדֶשׁ; אִם כֵּן — הוּא יוֹם, הוּא לַיְלָה. מַה שֶּׁאֵין כֵּן שָׁם, לְעִנְיַן עֹמֶר, עַיֵּן שָׁם — תִּרְאֶה שֶׁכֵּן הוּא הָאֱמֶת, וְאֵין רְאָיָה כְּלָל:

**וְדַע**, דְּרָאִיתִי בְּסֵפֶר טוּרֵי אֶבֶן (כ׳ עַמּוּד א׳), דִּבּוּר הַמַּתְחִיל "רָבָא אָמַר", כָּתַב: דְּרָבָא רַק לְיַשֵּׁב "תָּנֵי רַבָּה בַּר שְׁמוּאֵל" מְהַדַּר (עַיֵּן שָׁם, לְפִי גִּירְסָתוֹ: מִקֹּדֶם "רַבָּה"; וּבְלֶחֶם מִשְׁנֶה אִיתָא "רַב". מִיהוּ לְפָנֵינוּ אִיתָא מִקֹּדֶם גַּם כֵּן "רָבָא", וְאִם כֵּן לֵיתָא לָזֶה). אֲבָל לְדִידָן, דְּקַיְימָא לַן דְּלָא כַּאֲחֵרִים — סָבִירָא לֵיהּ נַמִּי דְּאַבָּיֵי לְיַשֵּׁב דְּרִיב״ל אַדְּרַבִּי יְהוּדָה נְשִׂיאָה, אִי נַמִּי דְּרַב דִּימִי, עַד כָּאן. וּלְפִי זֶה יָצָא לָנוּ דִּינִים מְחֻדָּשִׁים — אַךְ הוֹאִיל וְהָרַמְבַּ״ם לֹא מְפָרֵשׁ כֵּן, כַּנִּזְכָּר לְעֵיל, וּמֵהָרִאשׁוֹנִים לֹא נִמְצָא הֶפְכּוֹ — לֹא אַאֲרִיךְ בָּזֶה. דְּרַק בְּקִדּוּשׁ לְמַפְרֵעַ פְּלִיגֵי עָלָיו חִדּוּשֵׁי הָרַשְׁבָּ״א, וְכֵן רַשִׁ״י בְּפֵרוּשׁ "מְאַיְּמִין" — לֹא בְּזוּלַת:

**שָׁנָה** מְעֻבֶּרֶת — הִיא שָׁנָה שֶׁמּוֹסִיפִין בָּהּ חֹדֶשׁ, כְּמוֹ שֶׁנִּתְבָּאֵר לְעֵיל. וְאֵין מוֹסִיפִין לְעוֹלָם אֶלָּא אֲדָר, וְעוֹשִׂין אוֹתָהּ שָׁנָה ב׳ אֲדָרִין: אֲדָר רִאשׁוֹן וַאֲדָר שֵׁנִי. וּמִפְּנֵי מָה מוֹסִיפִין חֹדֶשׁ זֶה? מִפְּנֵי זְמַן הָאָבִיב, כְּדֵי שֶׁיִּהְיֶה הַפֶּסַח בְּאוֹתוֹ זְמַן, שֶׁנֶּאֱמַר: "שָׁמוֹר אֶת חֹדֶשׁ הָאָבִיב" — שֶׁיִּהְיֶה חֹדֶשׁ זֶה בִּזְמַן אָבִיב. וְלוּלֵי הוֹסָפַת הַחֹדֶשׁ הַזֶּה, הַפֶּסַח בָּא פְּעָמִים בִּימוֹת הַחַמָּה וּפְעָמִים בִּימוֹת הַגְּשָׁמִים. וְעַל ג׳ סִימָנִים מְעַבְּרִין, וּשְׁלָשְׁתָּן מִדְּאוֹרַיְתָא (תּוֹסָפוֹת סַנְהֶדְרִין י״א עַמּוּד ב׳): עַל הָאָבִיב — שֶׁרָאוּ בֵּית דִּין שֶׁעֲדַיִן לֹא בִּשְּׁלָה הַתְּבוּאָה, שֶׁנֶּאֱמַר "חֹדֶשׁ הָאָבִיב"; וְעַל פֵּרוֹת הָאִילָן — שֶׁלֹּא צָמְחוּ, וְיִתְאַחֵר בִּשּׁוּלָם יוֹתֵר מִזְּמַן הָעֲצֶרֶת, וַעֲצֶרֶת זְמַן הֲבָאַת בִּכּוּרִים, שֶׁנֶּאֱמַר: "חַג הַקָּצִיר בִּכּוּרֵי מַעֲשֶׂיךָ"; וְעַל הַתְּקוּפָה — שֶׁחִשְּׁבוּ בֵּית דִּין וְיָדְעוּ שֶׁתִּהְיֶה תְּקוּפַת נִיסָן בְּט״ז בְּנִיסָן אוֹ אַחַר זְמַן זֶה, שֶׁנֶּאֱמַר: "שָׁמוֹר אֶת חֹדֶשׁ הָאָבִיב" — שֶׁיִּהְיֶה הַפֶּסַח בְּחֹדֶשׁ הָאָבִיב; וְאִם תִּמָּשֵׁךְ תְּקוּפַת טֵבֵת עַד ט״ז בְּנִיסָן — לֹא יִהְיֶה יוֹם א׳ שֶׁל פֶּסַח בְּחֹדֶשׁ הָאָבִיב שֶׁל תְּקוּפָה (לְבוּשׁ מֵהָרַמְבַּ״ם, וְעַיֵּן הַגָּהוֹת):
