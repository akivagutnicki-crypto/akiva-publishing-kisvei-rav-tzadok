~ רֵישׁ פֶּרֶק ד׳ (וְעַיֵּן בְּמַשְׂאַת בִּנְיָמִין (סִימָן ק״א) הֵבִיא גַּם כֵּן רְאָיָה זוֹ דְּהַמְּפָרֵשׁ, וְהִיא תְּמוּהָה כַּנִּזְכָּר לְעֵיל, וּמִשָּׁם הֵם נוֹבְעִים. וְעַיֵּן מַה שֶּׁכָּתַבְתִּי לְמַעְלָה בְּפֶרֶק א׳ מִזֶּה, בְּסִיַּעְתָּא דִּשְׁמַיָּא). וְעוֹד, כַּד דַּיְקַתְּ מַשְׁכַּחַתְּ — לְפִי סֵדֶר גו״ח אדז״ט — גַּם לְדַעַת רַב אַדָּא שָׁנָה תִּפּוֹל בָּהּ הַתְּקוּפָה בְּט״ז לַחֹדֶשׁ (כְּשֶׁלֹּא יִדָּחֶה הַחֹדֶשׁ מִקְּבִיעוּתוֹ), וְהִיא שְׁנַת הַט״ז לְמַחֲזוֹר קָטָן: שֶׁמִּתְּקוּפַת נִיסָן שֶׁל שָׁנָה א׳ עַד אוֹתָהּ — יֵשׁ ט״ו שָׁנִים שְׁלֵמִים; וְנַחְשׁוֹב ט״ו פְּעָמִים י׳ כ״א קכ״א מ״ח, וְקִבּוּצָם עוֹלֶה קס״ג יָמִים, ד׳ שָׁעוֹת, תשמ״ד חֲלָקִים, ל״ו רְגָעִים; וְנִגְרַע מִזֶּה ה׳ עִבּוּרִים שֶׁל גו״ח א״ד — ה׳ פְּעָמִים כ״ט י״ב תשצ״ג — שֶׁקִּבּוּצָם קמ״ז ט״ו תשכ״ה, וְגַם ט׳ תרמ״ב; וְיִשָּׁאֵר ט״ו יָמִים, ג׳ שָׁעוֹת, תנ״ז חֲלָקִים, ל״ו רְגָעִים. וְאִם כֵּן, הֲרֵי שֶׁהָיָה הַמּוֹלָד אֲפִלּוּ בִּתְחִלַּת הַלַּיְלָה — נִמְנֶה ט״ו יָמִים שְׁלֵמִים, וְעוֹד ג׳ שָׁעוֹת עַל יוֹם אַחֵר — עַל יוֹם ט״ז:

**וְעַל** חֶשְׁבּוֹן תְּקוּפָה זוֹ דְּרַב אַדָּא הָיוּ סוֹמְכִין לְעִנְיַן עִבּוּר שָׁנִים בְּבֵית דִּין הַגָּדוֹל, שֶׁהָיוּ מְעַבְּרִין מִפְּנֵי הַזְּמַן אוֹ מִפְּנֵי הַצֹּרֶךְ; לְפִי שֶׁחֶשְׁבּוֹן זֶה הוּא הָאֱמֶת יוֹתֵר מֵחֶשְׁבּוֹן דִּשְׁמוּאֵל הַנִּזְכָּר לְעֵיל (זֶה דַּעַת הָרַמְבַּ״ם, וְכֵן כָּתַב הָרַאֲבַ״ע, וְכֵן ברד״א הֱבִיאוֹ. אֲבָל כָּל חַכְמֵי הָעִבּוּר נִמְשָׁכִים אַחַר תְּקוּפָה רִאשׁוֹנָה דִּשְׁמוּאֵל; וְכֵן מַשְׁמַע מִמַּה שֶּׁכָּתְבוּ הַפּוֹסְקִים לְבָרֵךְ עַל הַחַמָּה בְּכָל מַחֲזוֹר גָּדוֹל, וְיִתְבָּאֵר בִּמְקוֹמוֹ):

**וְחֶשְׁבּוֹן** ב׳ תְּקוּפוֹת הָאֵלּוּ שֶׁבֵּאַרְנוּ דַּרְכָּם — הַכֹּל בְּקֵרוּב הוּא; כְּמוֹ שֶׁכֵּן הוּא כָּל הַחֶשְׁבּוֹן הַזֶּה — לֹא בַּחֶשְׁבּוֹן הָאֲמִתִּי, בִּמְקוֹם הַשֶּׁמֶשׁ הָאֲמִתִּי שֶׁבְּכָל תְּקוּפָה; כִּי בְּחֶשְׁבּוֹן זֶה נִרְצֶה לְהַשְׁווֹת תָּמִיד, כְּדֵי שֶׁלֹּא נִתְקַלְקֵל בַּחֶשְׁבּוֹן, וּבֶאֱמֶת אֵינוֹ מִשְׁתַּוֶּה תָּמִיד. (וּבְסֵפֶר יְסוֹד עוֹלָם (הוּבָא בְּסֵפֶר יב״י, פֶּרֶק א׳, אוֹת ג׳) כָּתַב בְּשֵׁם רַבֵּינוּ סַעֲדְיָה גָּאוֹן וַאֲחֵרִים שֶׁהוֹדוּ לוֹ, כִּי מֵעוֹלָם — אֲפִלּוּ בִּזְמַן שֶׁבֵּית הַמִּקְדָּשׁ קַיָּם — לֹא קִדְּשׁוּ בֵּית דִּין עַל פִּי רְאִיָּה, רַק עַל פִּי חֶשְׁבּוֹן הַמָּסוּר עַתָּה בְּיָדֵינוּ; וְהוּא הִשִּׂיג עָלָיו, עַיֵּן שָׁם בַּאֹרֶךְ. וְעַיֵּן בְּפֵירוּשׁ הַמִּשְׁנָיוֹת לְהָרַמְבַּ״ם (פֶּרֶק ב׳ דְּרֹאשׁ הַשָּׁנָה), מִשְׁנַת "כֵּיצַד בּוֹדְקִין" וְכוּלֵי, כָּתַב: "וַאֲנִי תָּמֵהַּ מֵאָדָם יַכְחִישׁ הַרְאוּת, וְיֹאמַר כִּי דַּת הַיְּהוּדִים אֵינָהּ בְּנוּיָה עַל רְאִיַּת הַלְּבָנָה אֶלָּא עַל הַחֶשְׁבּוֹן בִּלְבַד" וְכוּלֵי, עַיֵּן שָׁם בַּאֹרֶךְ. נִרְאֶה כַּוָּנָתוֹ לְרַס״ג הַנִּזְכָּר לְעֵיל. וּבְפֶרֶק י״ח, מֵהֲלָכָה ה׳ וּלְהַלָּן, כָּתַב הָרַמְבַּ״ם: "שֶׁאִם" וְכוּלֵי, עַיֵּן שָׁם; נִרְאֶה כַּוָּנָתוֹ לִדְחוֹת דִּבְרֵי רַס״ג הַנִּזְכָּר לְעֵיל, עַיֵּן שָׁם בַּהֲלָכָה י׳):
