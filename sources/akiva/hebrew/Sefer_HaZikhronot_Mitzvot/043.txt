~ בְּגֹבַהּ קוֹמָה אַחַת, שֶׁדֶּרֶךְ בְּנֵי אָדָם לִטְעוֹת, וּמְצָרְפִים עֵדוּתָם. וְכָל זֶה כְּשֶׁהָיוּ עוֹמְדִים שְׁנֵיהֶם בְּשָׁוֶה בְּמָקוֹם אֶחָד; אֲבָל הָיָה אֶחָד עוֹמֵד מִלְמַעְלָה וְאֶחָד מִלְמַטָּה — אַף בְּזֶה אוֹמֵר ג׳ וְזֶה אוֹמֵר ה׳, מְשַׁעֲרִים לְפִי גֹּבַהּ מְקוֹם מַעֲמָדוֹ שֶׁל הָאוֹמֵר ג׳ נֶגֶד הָאוֹמֵר ה׳: אֶפְשָׁר שֶׁלְּפִי מַה שֶּׁהוּא עוֹמֵד בְּגֹבַהּ — יִהְיֶה נִרְאֶה נָמוּךְ קוֹמָה אַחַת, וּבְקוֹמָה אַחַת טָעוּ אִינָשֵׁי, וּמִצְטָרְפִים (יְרוּשַׁלְמִי פֶּרֶק ב׳, סוֹף הֲלָכָה ד׳, עַיֵּן שָׁם). וְכֵן נִרְאֶה לְהֵפֶךְ: אִם אֶחָד אוֹמֵר ד׳ וְאֶחָד אוֹמֵר ג׳, וְהָיָה זֶה הָאוֹמֵר ד׳ עוֹמֵד בְּגֹבַהּ מֵהָאוֹמֵר ג׳ — אֵין מִצְטָרְפִים, אַף עַל פִּי שֶׁאֵין בֵּינֵיהֶם אֶלָּא קוֹמָה אַחַת:

(**וְעַיֵּן** שָׁם בְּסֵפֶר טוּרֵי אֶבֶן, כָּתַב דְּיֵשׁ לוֹמַר דִּלְפִי גֹּבַהּ הַקּוֹמוֹת שֶׁאָמְרוּ אָדָם טוֹעֶה שְׁלִישׁ מִלְּבַר; וּלְכָךְ בְּב׳ וְג׳ מִצְטָרְפִים, דְּאִי שֶׁל ב׳ טָעָה — הֲוֵי שְׁלִישׁ מִלְּבַר לִפְחוֹת, וְאִי שֶׁל ג׳ טָעָה — הֲוֵי שְׁלִישׁ מִלְּבַר לְהוֹסִיף. (אֵין דְּבָרָיו מְכֻוָּנִים: וּמֵאַיִן הוּא אוֹמֵר "הַשְּׁלִישׁ מִלְּבַר"? אִם מֵהָאֻמְדָּנָא — נִיחָא אִי בַּעַל ג׳ טָעָה; אֲבָל אִי שֶׁל ב׳ טָעָה — הֲוֵי שְׁלִישׁ מִלְּגָאו. וְאִי מֵהַטָּעוּת שְׁלִישׁ — אִם כֵּן נִיחָא אִי שֶׁל ב׳ טָעָה, אִי שֶׁל ג׳ טָעָה — מַאי אִיכָּא לְמֵימַר.) וּלְפִי זֶה בְּאֶחָד אוֹמֵר ג׳ וְאֶחָד אוֹמֵר ה׳ — דַּוְקָא ה׳, הָא ד׳ וָחֵצִי טָעֵי אִינָשֵׁי בִּשְׁלִישׁ. וְהַדַּעַת נוֹטָה כֵּן, דְּוַדַּאי נִתְרַבָּה הַטָּעוּת לְפִי רִבּוּי הַגֹּבַהּ, עַד כָּאן. וּקְצָת רְאָיָה לִדְבָרָיו מֵהַיְרוּשַׁלְמִי (פֶּרֶק ב׳ סוֹף הֲלָכָה ד׳): בְּזֶה אוֹמֵר מְלֹא מַרְדֵּעַ אַחַת וְזֶה אוֹמֵר מְלֹא ב׳ מַרְדְּעִים — אֵין דִּבְרֵיהֶם קַיָּמִים בְּעוֹמְדִים בְּשָׁוֶה. וְקָשֶׁה, הָא אָדָם טוֹעֶה מְלֹא מַרְדֵּעַ אַחַת! וְעַל כָּרְחֲךָ כַּנִּזְכָּר לְעֵיל, דְּדַוְקָא בִּשְׁלִישׁ — לֹא בְּזֶה אוֹמֵר א׳ וְזֶה אוֹמֵר ב׳. מִיהוּ יִתָּכֵן דְּהַיְרוּשַׁלְמִי פָּלִיג אַתַּלְמוּדִין):

**וְעֵדִים** שֶׁאָמְרוּ "רְאִינוּהוּ בְּלֹא כַּוָּנָה, וְכֵיוָן שֶׁהִתְבּוֹנַנּוּ בּוֹ וְנִתְכַּוַּנּוּ לִרְאוֹתוֹ לְהָעִיד — שׁוּב לֹא רְאִינוּהוּ" — אֵין זוֹ עֵדוּת; שֶׁמָּא עָבִים נִתְקַשְּׁרוּ, וְנִרְאוּ כִּלְבָנָה, וְהָלְכוּ לָהֶם:

**וְעֵדִים** שֶׁאָמְרוּ שֶׁרָאוּ יָרֵחַ בְּכ״ט — אֵין שׁוֹמְעִים לָהֶם, שֶׁאֵין חֲדָשָׁה שֶׁל לְבָנָה פָּחוֹת מִכ״ט יוֹם, כְּמוֹ שֶׁנִּתְבָּאֵר לְעֵיל; אֶלָּא הַיְשָׁנָה רָאוּ, אוֹ דְּמוּת לְבָנָה בֶּעָבִים. וּמִכָּל מָקוֹם אִם אָמְרוּ אוֹתָם עֵדִים שֶׁרָאוּהָ בְּשַׁחֲרִית, קֹדֶם עֲלוֹת הַשֶּׁמֶשׁ, בְּכ״ט בַּמִּזְרָח, וּרְאוּהוּ עַרְבִית אַחַר שְׁקִיעַת הַחַמָּה בַּמַּעֲרָב — וְאַף עַל פִּי שֶׁאִי אֶפְשָׁר שֶׁתֵּרָאֶה הַיְשָׁנָה שַׁחֲרִית וְהַחֲדָשָׁה עַרְבִית, וּכְמוֹ שֶׁכָּתַבְתִּי לְמַעְלָה שֶׁעַל כָּל פָּנִים צָרִיךְ שֶׁיִּהְיֶה בֵּינֵיהֶם כ״ד שָׁעָה — אוֹמְרִים שֶׁדְּמוּת לְבָנָה רָאוּ בְּשַׁחֲרִית מֵעָבִים, וּמְקַדְּשִׁין עַל רְאִיָּה זוֹ שֶׁרָאוּ בְּל׳; וְאֵין אוֹמְרִים שֶׁמָּא גַּם זוֹ
