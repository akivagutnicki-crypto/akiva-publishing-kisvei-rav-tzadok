~ מִן ג׳ — כִּי הַד׳ שָׁעוֹת כְּבָר נַעֲשׂוּ ג׳ בְּהִלָּקַח מֵהֶם א׳ לַחֲלָקִים — וְגַם כֵּן לֹא נוּכַל, וְנָרִים יוֹם אֶחָד, וְנַעֲשֵׂהוּ כ״ד שָׁעוֹת, וּבְחִבּוּר הַג׳ יִהְיוּ כ״ז; וְנִגְרַע י״ב, וְיִשָּׁאֵר ט״ו. וְנָבוֹא לִגְרוֹעַ הַיָּמִים: א׳ מִן ב׳, וְיִשָּׁאֵר א׳. וְנֵדַע מוֹלַד תִּשְׁרֵי הַקּוֹדֵם: א׳ ט״ו תתכ״ז. וְכֵן אַתָּה דָּן לְשָׁנִים פְּשׁוּטוֹת וּמְעֻבָּרוֹת, וְכֵן לְמַחֲזוֹרִים. וְאִם לֹא נוּכַל לִגְרוֹעַ יָמִים מִיָּמִים, כִּי יִהְיֶה הַמְּגוֹרָע יוֹתֵר מִן הַמְּגוֹרָע — נוֹסִיף עַל יְמֵי הַמְּגוֹרָע שָׁבוּעַ אַחַת, וְנַעֲשֵׂנָה יָמִים, וְאָז נִגְרַע יָמִים מִיָּמִים, וּכְמִשְׁפָּט הַנִּזְכָּר לְעֵיל; וְהַמּוֹתָר הוּא מוֹלַד הֶעָבָר:

**וְהִנֵּה** הַמִּנְיָן שֶׁכָּל יִשְׂרָאֵל מוֹנִין עַכְשָׁיו הוּא לִבְרִיאַת הָעוֹלָם. וִיצִירַת אָדָם הָרִאשׁוֹן הָיְתָה בְּיוֹם ו׳, בִּתְחִלַּת שָׁעָה ג׳, כְּפִי קַבָּלַת חֲזַ״ל; וְקִבְּלוּ שֶׁאוֹתוֹ יוֹם הָיָה רֹאשׁ הַשָּׁנָה, וְאָז בִּיצִירָתוֹ הָיָה מוֹלַד הַלְּבָנָה שֶׁל חֹדֶשׁ תִּשְׁרֵי הַהוּא. וּלְפִי שֶׁקָּדְמָה בְּרִיאַת הָעוֹלָם לִבְרִיאָתוֹ ה׳ יָמִים, וְי״ב שָׁעוֹת לֵיל ו׳, וּב׳ שָׁעוֹת שֶׁל הַיּוֹם, עַד תְּחִלַּת שָׁעָה ג׳ שֶׁנּוֹצַר הָאָדָם — הֲרֵי ה׳ יָמִים וְי״ד שָׁעוֹת, לֹא יִתָּכֵן לְבִלְתִּי הַשְׁגִּיחַ עֲלֵיהֶם. וְלָכֵן מוֹנִין אוֹתָן שָׁנָה שְׁלֵמָה, וּמוֹנִין יְצִירַת אָדָם הָרִאשׁוֹן לְשָׁנָה ב׳ לִבְרִיאַת הָעוֹלָם, לְפִי שֶׁהַתְחָלַת הַשָּׁנִים מִתִּשְׁרֵי:

**וּמִכֵּיוָן** שֶׁיָּדַעְנוּ מוֹלַד תִּשְׁרֵי שֶׁל יְצִירַת אָדָם הָרִאשׁוֹן — שֶׁהוּא שְׁנַת ב׳ לִבְרִיאָה — שֶׁהָיָה ה׳ י״ד, כַּנִּזְכָּר לְעֵיל; אִם כֵּן, יָדַעְנוּ מוֹלַד תִּשְׁרֵי שֶׁל שְׁנַת תֹּהוּ — שֶׁאָנוּ מוֹנִין אוֹתוֹ רִאשׁוֹנָה לְמִנְיָנֵנוּ — שֶׁאִלּוּ שִׁמְּשָׁה אָז הַלְּבָנָה, הָיָה מוֹלַדְתָּהּ בְּתִשְׁרֵי: א׳ ה׳ ר״ד; שֶׁזֶּהוּ הַנִּשְׁאָר אַחֲרֵי גִּירְעוֹן ד׳ ח׳ תתע״ו — הַיִּתְרוֹן שֶׁל שָׁנָה פְּשׁוּטָה — מִן ה׳ י״ד, כְּמִשְׁפָּט הָאָמוּר. וְסִימָנוֹ: בהר״ד — רוֹצֶה לוֹמַר, שֶׁהָיָה הַמּוֹלָד בְּלֵיל ב׳, ה׳ שָׁעוֹת בַּלַּיְלָה, ר״ד חֲלָקִים, עַל שָׁעָה ו׳. וְהוּא מוֹלַד תִּשְׁרֵי שֶׁל שָׁנָה רִאשׁוֹנָה שֶׁל מַחֲזוֹר הָרִאשׁוֹן. וּכְפִי מַה שֶּׁבֵּאַרְנוּ — נָקֵל לָדַעַת מוֹלַד כָּל שָׁנָה שֶׁנִּרְצֶה, וְכָל חֹדֶשׁ שֶׁנִּרְצֶה:

**כֵּיצַד?** הֲרֵי שֶׁרָצִיתָ לֵידַע מוֹלַד חֶשְׁוָן שֶׁל שְׁנַת כ״א לִבְרִיאַת עוֹלָם — הִנֵּה תַּחְשׁוֹב הַיִּתְרוֹן מִן מַחֲזוֹר א׳, שֶׁהוּא י״ט שָׁנָה — מוֹתָרוֹ ב׳ י״ו תקצ״ה; וְהַיִּתְרוֹן שֶׁל שָׁנָה א׳ פְּשׁוּטָה, שָׁנָה רִאשׁוֹנָה שֶׁל מַחֲזוֹר קָטָן ב׳ — וְיִתְרוֹנָהּ ד׳ ח׳ תתע״ו; וְהַיִּתְרוֹן שֶׁל חֹדֶשׁ א׳ — וְהוּא א׳ י״ב תשצ״ג. וַחֲבָרֵם יַחַד עַל בהר״ד, וּקְבוּצָתָם יַחַד, עֲשׂוֹתְךָ כָּאָמוּר לְמַעְלָה — יִהְיֶה: ב׳ י״ט ש״ח; וְהוּא הַמּוֹלָד — חֶשְׁוָן כ״א — י״ט שָׁעוֹת, ש״ח חֲלָקִים, עַל יוֹם ג׳. וְכֵן אִם נֹאבֶה לָדַעַת מוֹלַד תִּשְׁרֵי שֶׁל שָׁנָה הַבָּאָה עָלֵינוּ לְטוֹבָה, שֶׁנִּזְכֶּה לְקַדְּשָׁהּ עַל פִּי הָרְאִיָּה בִּירוּשָׁלַיִם
