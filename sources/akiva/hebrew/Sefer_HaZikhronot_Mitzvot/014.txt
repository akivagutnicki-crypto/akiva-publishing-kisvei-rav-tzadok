# מִצְוַת עֲשֵׂה ב׳ — פֶּרֶק א׳
# פָּרָשַׁת לֶךְ לְךָ

(ב) **מִצְוַת עֲשֵׂה** לָאָב לָמוּל אֶת בְּנוֹ, וְכֵן לַבֵּן כְּשֶׁיִּגְדַּל לָמוּל אֶת עַצְמוֹ, לִשְׁמוֹנַת יָמִים, שֶׁנֶּאֱמַר: "הִמּוֹל לָכֶם כָּל זָכָר" וְגוֹמֵר. וְנִכְפְּלָה מִצְוָה זוֹ בְּסֵדֶר תַּזְרִיעַ: "וּבַיּוֹם הַשְּׁמִינִי יִמּוֹל בְּשַׂר עָרְלָתוֹ". וּמְאוּסָה הִיא הָעָרְלָה, שֶׁנִּתְגַּנּוּ בָהּ רְשָׁעִים, שֶׁנֶּאֱמַר: "כָּל הַגּוֹיִם עֲרֵלִים". וּגְדוֹלָה הַמִּילָה, שֶׁנִּכְרְתוּ עָלֶיהָ י״ג בְּרִיתוֹת, וְשֶׁדּוֹחָה הַשַּׁבָּת וְהַנְּגָעִים, וְשֶׁלֹּא נִתְּלָה לוֹ לְמֹשֶׁה רַבֵּינוּ עָלָיו הַשָּׁלוֹם עָלֶיהָ מְלֹא שָׁעָה, וְכָל זְכֻיּוֹתָיו לֹא הוֹעִילוּ לוֹ כְּשֶׁנִּתְרַשֵּׁל מִמֶּנָּה; וְכָל מִצְוֹת שֶׁעָשָׂה אַבְרָהָם אָבִינוּ עָלָיו הַשָּׁלוֹם לֹא נִקְרָא שָׁלֵם עַד שֶׁמָּל. וּשְׁקוּלָה כְּנֶגֶד כָּל הַמִּצְוֹת שֶׁבַּתּוֹרָה (עַיֵּן בְּבֵית הִלֵּל, דְּהַטּוּר לָא סְבִירָא לֵיהּ כֵּן; רַק כָּתַב מִשְּׁאָר מִצְוֹת, וְרַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה — חוּץ מִצִּיצִית, עַיֵּן שָׁם. וְלֵיתָא, דְּכֵן נִמְצָא כַּמָּה פְּעָמִים בְּדִבְרֵי רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה עַל כַּמָּה מִצְוֹת. וְעַיֵּן בְּחִדּוּשֵׁי הָרַמְבַּ״ן פֶּרֶק קַמָּא דִּיבָמוֹת, כָּתַב אַצִּיצִית גּוּפָא, דְּאֵין לְהַחְשִׁיב זֶה לְחֹמֶר שֶׁבָּהּ, כִּי "אֵין אַתָּה יוֹדֵעַ מַתַּן שְׂכָרָן שֶׁל מִצְוֹת", עַיֵּן שָׁם. וְעַיֵּן מַה שֶּׁכָּתוּב בְּשׁוּד״י (סִימָן כ״א) אַדִּבְרֵי נִמּוּקֵי יוֹסֵף בְּהִלְכוֹת צִיצִית). וְאִלְמָלֵא הִיא לֹא בָּרָא הַקָּדוֹשׁ בָּרוּךְ הוּא עוֹלָמוֹ, שֶׁנֶּאֱמַר: "אִם לֹא בְרִיתִי" וְגוֹמֵר (הַכֹּל סוֹף פֶּרֶק ג׳ דִּנְדָרִים). וּלְמִי שֶׁנִּמּוֹל — אַבְרָהָם מַסִּיק לֵיהּ מִגֵּיהִנֹּם, כִּדְאִיתָא פֶּרֶק ב׳ דְּעֵירוּבִין. וּמִי שֶׁעָבַר וּבִטֵּל מִצְוָה זוֹ, וְלֹא מָל עַצְמוֹ מִי״ג שָׁנָה וּלְמַעְלָה — הוּא חַיָּב כָּרֵת, וְכָל יוֹם שֶׁעוֹבֵר — עוֹמֵד בְּאִסּוּר כָּרֵת. וְעַיֵּן אַגָּדַת בְּרֵאשִׁית (פֶּרֶק ט״ז, י״ז, י״ט) הַרְבֵּה מִשְׁבְּחֵי הַמִּילָה; וְעַיֵּן עוֹד שָׁם (פֶּרֶק ל״ד) שֶׁבִּזְכוּת מִילָה וְכוּלֵי, עַיֵּן שָׁם. וְדַם בְּרִית חָשׁוּב כְּדַם זֶבַח — פִּרְקֵי דְרַבִּי אֱלִיעֶזֶר (פֶּרֶק י׳), עַיֵּן שָׁם:

**בְּנֵי נֹחַ** אֵין מְצֻוִּים עָלֶיהָ, אַף עַל פִּי שֶׁנֶּאֶמְרָה הַמִּצְוָה לִבְנֵי נֹחַ, וְנִשְׁנֵית בְּסִינַי; לְפִי שֶׁנֶּאֱמַר: "אַתָּה וְזַרְעֲךָ אַחֲרֶיךָ". וַאֲפִלּוּ זַרְעוֹ שֶׁל יִשְׁמָעֵאל, שֶׁבָּא מֵאַבְרָהָם — אֵין נִמּוֹלִים, שֶׁנֶּאֱמַר: "כִּי בְיִצְחָק יִקָּרֵא לְךָ זָרַע". וְכֵן זַרְעוֹ שֶׁל עֵשָׂו, שֶׁנֶּאֱמַר: "וְיִתֶּן לְךָ אֶת בִּרְכַּת אַבְרָהָם לְךָ וּלְזַרְעֲךָ" — מִכְּלָל שֶׁהוּא לְבַדּוֹ זֶרַע אַבְרָהָם, וְהֵן הַמְחֻיָּבִים בְּמִילָה. וּבְנֵי קְטוּרָה אָמְרוּ חֲכָמִים שֶׁחַיָּבִים בְּמִילָה; וְהוֹאִיל וְנִתְעָרְבוּ עַכְשָׁיו בְּנֵי יִשְׁמָעֵאל בִּבְנֵי קְטוּרָה, נִתְחַיְּבוּ הַכֹּל בְּמִילָה בַּח׳, וְאֵין נֶהֱרָגִים עָלֶיהָ (רַמְבַּ״ם פֶּרֶק י׳ מִמְּלָכִים). וְלָכֵן אִם מְבַקְשִׁים הַיִּשְׁמְעֵאלִים לְאֶחָד מִיִּשְׂרָאֵל לָמוּל בְּנֵיהֶם — מִצְוָה לְמוּלוֹ, כֵּיוָן דִּמְצֻוִּים; וְכָל שֶׁכֵּן שֶׁאָסוּר לְמָנְעָם מִלָּמוּל, דְּעוֹבְרִים עַל "לִפְנֵי עִוֵּר". וְכֵן הַיִּשְׁמְעֵאלִים עַכְשָׁיו מַחְזִיקִים בְּמִילָה (וְלֹא בִּפְרִיעָה). אֲבָל רַשִׁ״י סַנְהֶדְרִין (נ״ט עַמּוּד ב׳) פֵּרֵשׁ דְּרַק בְּנֵי קְטוּרָה עַצְמָם — הַו׳ בָּנִים — הָיוּ חַיָּבִים בְּמִילָה, אֲבָל לֹא זַרְעָם אַחֲרֵיהֶם;
