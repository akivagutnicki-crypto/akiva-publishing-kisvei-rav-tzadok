# מִצְוַת עֲשֵׂה א׳ — פֶּרֶק א׳
# בְּסִיַּעְתָּא דִשְׁמַיָּא. פָּרָשַׁת בְּרֵאשִׁית

(א) **מִצְוַת עֲשֵׂה** לִשָּׂא אִשָּׁה וְלִפְרוֹת וְלִרְבּוֹת לְקִיּוּם הַמִּין, שֶׁנֶּאֱמַר: "וְאַתֶּם פְּרוּ וּרְבוּ". וְנֶאֱמַר: "לֹא תֹהוּ בְרָאָהּ, לָשֶׁבֶת יְצָרָהּ". וּמִטַּעַם זֶה מִצְוָה זוֹ גְּדוֹלָה מִכָּל מִצְוֹת שֶׁבַּתּוֹרָה, זוּלַת תַּלְמוּד תּוֹרָה (תּוֹסָפוֹת פֶּרֶק קַמָּא דַּעֲבוֹדָה זָרָה י״ג עַמּוּד א׳, וְגִטִּין מ״א עַמּוּד ב׳). וְכָל מִי שֶׁלֹּא נָשָׂא אִשָּׁה הֲרֵי הוּא שׁוֹפֵךְ דָּמִים, וּמְמַעֵט הַדְּמוּת, וְגוֹרֵם לַשְּׁכִינָה שֶׁתִּסְתַּלֵּק מִיִּשְׂרָאֵל, וְאֵינוֹ קָרוּי אָדָם; וְכֵיוָן שֶׁנָּשָׂא אִשָּׁה עֲוֹנוֹתָיו מִתְפַּקְפְּקִים. וְכָל זְמַן שֶׁלֹּא נָשָׂא שָׁרוּי בְּלֹא שִׂמְחָה, בְּלֹא בְּרָכָה, טוֹבָה, תּוֹרָה, חוֹמָה, שָׁלוֹם (הַכֹּל בְּפֶרֶק הַבָּא עַל יְבִמְתּוֹ). וּבִבְרֵאשִׁית רַבָּה (פָּרָשָׁה י״ז) מוֹסִיף: שָׁרוּי בְּלֹא כַּפָּרָה, בְּלֹא עֵזֶר. וּבְפֶרֶק קַמָּא דִּבְרָכוֹת (דַּף י׳ עַמּוּד א׳) בְּחִזְקִיָּה אָמְרִינַן: "מוֹת" — בָּעוֹלָם הַזֶּה, "תָּמוּת" — בָּעוֹלָם הַבָּא, מִשּׁוּם דְּלָא נְסַבִית. וּבְפֶרֶק עַרְבֵי פְּסָחִים (דַּף קי״ב עַמּוּד ב׳): ז׳ כִּמְנוּדִּין לַשָּׁמַיִם — יְהוּדִי שֶׁאֵין לוֹ אִשָּׁה, וְשֶׁיֵּשׁ לוֹ אִשָּׁה וְאֵין לוֹ בָּנִים. וּבַזֹּהַר חָדָשׁ רוּת, פָּסוּק "וַתִּקְרֶאנָה", אָמְרִינַן דְּאֵין מַכְנִיסִים אוֹתוֹ בִּמְחִיצָתוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא (עַיֵּן שָׁם):

**וְנוֹהֶגֶת** מִצְוָה זוֹ בְּכָל מָקוֹם וּבְכָל זְמַן. וּמוּכָח שָׁם בִּבְרָכוֹת גַּבֵּי חִזְקִיָּה, דַּאֲפִלּוּ יוֹדֵעַ שֶׁיִּהְיוּ בָּנָיו רָעִים וְחַטָּאִים לַה׳ — חַיָּב בְּמִצְוָה זוֹ (וּכְגוֹן הָאֲנוּסִים שֶׁיּוֹדְעִים שֶׁבְּנֵיהֶם יַעַבְרוּ עַל הַמִּצְוֹת, וְכַהֲנֵ״ל). וּמִיהוּ בִּמְקוֹם גְּזֵרַת הַמַּלְכוּת יֵשׁ אוֹמְרִים דְּלָא שַׁיְּכָא לְמִצְוָה, מִשּׁוּם דְּאָמְרִינַן בְּסוֹף פֶּרֶק חֶזְקַת הַבָּתִּים (דַּף ס׳ עַמּוּד ב׳): בְּדִין הָיָה שֶׁנִּגְזוֹר עַל עַצְמֵנוּ מִשֶּׁחָרַב בֵּית הַמִּקְדָּשׁ שֶׁלֹּא נִשָּׂא נָשִׁים וְלֹא נוֹלִיד בָּנִים, וְיִכְלוּ זַרְעוֹ שֶׁל אַבְרָהָם אָבִינוּ מֵאֵלָיו, אֶלָּא שֶׁאֵין רוֹב הַצִּבּוּר יְכוֹלִים לַעֲמוֹד (יֵשׁ לְפָרֵשׁ שֶׁאֵין יְכוֹלִים לְהַעֲמִיד עַל עַצְמָם שֶׁלֹּא יָבוֹאוּ לִידֵי אִסּוּר הַשְׁחָתַת זֶרַע, רַחֲמָנָא לִצְלַן, אוֹ שְׁאָר אִסּוּרִים; וּלְפִי זֶה דַּוְקָא "יָכוֹל לְהַעֲמִיד" בָּעִינַן, וּפָשׁוּט). אֲבָל הַתּוֹסָפוֹת כָּתְבוּ דְּזֶהוּ כְּשֶׁכְּבָר קִיֵּם פְּרִיָּה וּרְבִיָּה הָיָה בְּדִין לִגְזוֹר שֶׁלֹּא יוֹסִיף לְהַרְבּוֹת בָּנִים, אֲבָל כְּשֶׁעֲדַיִן לֹא קִיֵּם — אָסוּר. וְכֵן הַסְכָּמַת הַפּוֹסְקִים כְּהַתּוֹסָפוֹת. וּמִכָּל מָקוֹם אֵין לְבֵית דִּין לָכוֹף לָעוֹבֵר וְאֵינוֹ נוֹשֵׂא. וּלְעִנְיַן לְהוֹכִיחוֹ — מְסֻפְּקַנִי; וְאֶפְשָׁר דְּחַיָּב לְהוֹכִיחוֹ לָעוֹבֵר, וְעַיֵּן בִּמְקוֹמוֹ בְּמִצְוַת תּוֹכָחָה:

**מֵאֵימָתַי** יִתְחַיֵּב בְּמִצְוָה זוֹ? מִשֶּׁנַּעֲשׂוּ לוֹ י״ז שְׁלֵמוֹת, בְּהַתְחָלַת שְׁנַת הַי״ח, עַד שְׁנַת הָעֶשְׂרִים. וְכֵיוָן שֶׁעָבְרוּ עָלָיו כ׳ וְלֹא נָשָׂא, הֲרֵי כָּל שָׁעָה שֶׁאֵינוֹ נוֹשֵׂא עוֹבֵר בַּעֲשֵׂה, וְהַקָּדוֹשׁ בָּרוּךְ הוּא אוֹמֵר עָלָיו "תִּפַּח עַצְמוֹתָיו", וַהֲרֵי הוּא כָּל יָמָיו בְּהִרְהוּר עֲבֵירָה. וְהַמַּקְדִּים לִשָּׂא בֶּן י״ג בְּהַתְחָלַת שְׁנַת
