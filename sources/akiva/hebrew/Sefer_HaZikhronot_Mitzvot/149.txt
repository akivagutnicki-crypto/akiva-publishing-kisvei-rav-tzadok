~ קֹדֶשׁ, וְהַפְּתוּחוֹת לְהַר הַבַּיִת — חֹל. וְנִרְאֶה, דְּכֵן הַמְּחִלּוֹת שֶׁבִּירוּשָׁלַיִם: הַפְּתוּחוֹת לִירוּשָׁלַיִם — כִּירוּשָׁלַיִם, וְלַחוּץ — כְּלַחוּץ; וּפְתוּחוֹת לִשְׁנֵיהֶם — מַה שֶּׁבִּפְנִים כְּבִפְנִים, וְשֶׁבַּחוּץ כְּחוּץ. וְעֹבִי הַחוֹמָה, וְכֵן הַחַלּוֹנוֹת — אִם הֵם נְמוּכִים, הֲרֵי הֵם כְּלִפְנִים (אַף עַל פִּי שֶׁעֹבִי הַשַּׁעַר נִתְבָּאֵר לְעֵיל שֶׁאֵינוֹ כְּלִפְנִים — לֹא דָּמֵי לְהָכָא, דְּשָׁם טַעְמָא אַחֲרִינָא אִיכָּא, כְּמוֹ שֶׁנִּתְבָּאֵר שָׁם); וְאִם גְּבוֹהִים, עַד שֶׁשָּׁוֶה לְגַגִּין וַעֲלִיּוֹת — הֲרֵי דִּינוֹ כְּגַגִּין וַעֲלִיּוֹת, שֶׁלֹּא נִתְקַדְּשׁוּ, כְּמוֹ שֶׁנִּתְבָּאֵר. וְהַנִּכְנָס לַמִּקְדָּשׁ (וְכֵן לִירוּשָׁלַיִם) דֶּרֶךְ גַּגּוֹת וַעֲלִיּוֹת — אַף עַל פִּי שֶׁאֵין דֶּרֶךְ בִּיאָה בְּכָךְ — מִכָּל מָקוֹם מִצְוָה לְשַׁלְּחוֹ. כֵּן נִרְאֶה פָּשׁוּט — דְּלֹא עָדִיף מִכֵּלִים וּשְׁאָרֵי, דְּכָל שֶׁנִּמְצָא טֻמְאָה בַּמִּקְדָּשׁ, וּמְצֹרָע בִּירוּשָׁלַיִם — צָרִיךְ לְשַׁלְּחוֹ:

**וְאֵיזֶהוּ** הַטָּמֵא שֶׁצָּרִיךְ לְשַׁלְּחוֹ? כָּל שֶׁנִּטְמָא בְּאַחַת מֵהַטֻּמְאוֹת שֶׁיִּתְפָּרְשׁוּ בִּמְקוֹמָן, וּכְפִי הַג׳ חִלּוּקִים שֶׁנִּתְבָּאֵר. וְאִם נִטְמָא בְּעִנְיָן שֶׁהוּא טָמֵא מִן הַתּוֹרָה — צָרִיךְ לְשַׁלְּחוֹ מִן הַתּוֹרָה; וְאִם אֵינוֹ טָמֵא אֶלָּא מִדִּבְרֵי סוֹפְרִים — כְּגוֹן שֶׁאָכַל אֳכָלִים טְמֵאִים, וְשָׁתָה מַשְׁקִים טְמֵאִים, וְכַיּוֹצֵא, אוֹ שֶׁנִּטְמָא בְּאַב מֵאֲבוֹת הַטֻּמְאָה שֶׁל דִּבְרֵיהֶם — הֲרֵי הוּא טָעוּן שִׁלּוּחַ מִן הַמִּקְדָּשׁ מִדִּבְרֵיהֶם. (וְיֵשׁ לְעַיֵּן בָּזֶה, בְּעִנְיָן דְּמָצִינוּ לִפְעָמִים דְּנַעֲשֶׂה מִדְּרַבָּנָן דְּאוֹרַיְתָא; וְהַאֲרַכְנוּ בָּזֶה בִּמְקוֹמוֹ, וְאֵין כָּאן מְקוֹמוֹ לְהַאֲרִיךְ. וְיֵשׁ לִי סָפֵק — לְפִי מַה שֶּׁיִּתְבָּאֵר לְקַמָּן, סִימָן שֶׁאַחַר זֶה, לְעִנְיַן מַלְקוֹת וְכָרֵת, דְּאֵינוֹ בט״מ אֶלָּא עַל טֻמְאוֹת שֶׁהַנָּזִיר מְגַלֵּחַ עֲלֵיהֶם, אֲבָל שֶׁאֵין הַנָּזִיר מְגַלֵּחַ עֲלֵיהֶם — אֲפִלּוּ הֵם מִן הַתּוֹרָה — אֵין נִכְרָת, עַיֵּן שָׁם — אִם הוּא הַדִּין לְעִנְיַן חִיּוּב עֲשֵׂה דְּשִׁלּוּחַ, דְּלֵיכָּא עַל זֶה; וְצָרִיךְ עִיּוּן.) וְעַיֵּן בְּתוֹסָפוֹת עֵירוּבִין (ק״ה עַמּוּד א׳), וְעַיֵּן בִּפְסָחִים (צ״ה סוֹף עַמּוּד ב׳), עַיֵּן שָׁם הֵיטֵב:

**וּבְכֵלִים** גַּם כֵּן הַדִּין כֵּן: אִם טֻמְאָתָהּ דִּבְרֵי תּוֹרָה — שִׁלּוּחָהּ דִּבְרֵי תּוֹרָה; וְאִם טֻמְאָתָהּ מִדִּבְרֵי סוֹפְרִים — שִׁלּוּחָהּ מִדִּבְרֵי סוֹפְרִים. וּלְדַעַת הָרַמְבַּ״ם, בִּכְלִי — אֵין טָעוּן שִׁלּוּחַ אֶלָּא כְּשֶׁהוּא אַב הַטֻּמְאָה, וְלֹא כְּשֶׁהוּא וְלַד הַטֻּמְאָה — דְּאָז אֵין טָעוּן שִׁלּוּחַ אֶלָּא מִדְּרַבָּנָן; וְהָרַאֲבַ״ד חוֹלֵק עָלָיו. וְהַכְּלִי שֶׁהָיְתָה טֻמְאָתָהּ מִדִּבְרֵי תּוֹרָה, שֶׁנִּמְצֵאת בָּעֲזָרָה בְּשַׁבָּת, שֶׁאָסוּר לְטַלְטְלָהּ מִשּׁוּם שְׁבוּת — אִם נִמְצֵאת בְּמָקוֹם שֶׁשִּׁלּוּחָהּ הוּא מִן הַתּוֹרָה — מֻתָּר לְהוֹצִיאָהּ; וְהוֹאִיל וְהֻתַּר לְהוֹצִיאָהּ מִמְּקוֹם מְצִיאוּתָהּ, שֶׁשָּׁם הוּא מִן הַתּוֹרָה — רַשַּׁאי גַּם כֵּן לְהוֹצִיאָהּ מִמָּקוֹם שֶׁאָסוּר מִדְּרַבָּנָן, עַד מְקוֹם הַמֻּתָּר (וְשָׁם כּוֹפֶה עָלָיו פְּסַכְתֵּר, כְּדֵי שֶׁלֹּא יִתְטַמְּאוּ בּוֹ כֹּהֲנִים; עַיֵּן בְּמִשְׁנֶה לַמֶּלֶךְ). וַאֲפִלּוּ נָפַל מִמֶּנּוּ (יְרוּשַׁלְמִי) בַּהֲלִיכָתוֹ, בְּמָקוֹם שֶׁאֵין אִסּוּרוֹ אֶלָּא מִדְּרַבָּנָן — רַשַּׁאי לְהַגְבִּיהוֹ וּלְהוֹצִיאוֹ לִמְקוֹם הַמֻּתָּר, הוֹאִיל וּכְבָר נִרְאָה כְּלִי זוֹ לָצֵאת; וְהוֹאִיל וְאִשְׁתְּרֵי — אִשְׁתְּרֵי. אֲבָל אִם
