~ לִשָּׂא אִשָּׁה, שֶׁלֹּא לַעֲמוֹד בְּלֹא אִשָּׁה. וְיֵשׁ חוֹלְקִים וְסוֹבְרִים דְּכֵיוָן דְּזֶה אֵינוֹ אֶלָּא אִסּוּר דְּרַבָּנָן, כְּמוֹ שֶׁנִּתְבָּאֵר, אֵין לִמְכּוֹר סֵפֶר תּוֹרָה בִּשְׁבִיל זֶה. וְהָעִקָּר כְּדֵעָה רִאשׁוֹנָה. מִיהוּ אִם יֵשׁ לוֹ לִשָּׂא אִשָּׁה שֶׁאֵינָהּ בַּת בָּנִים, וְאֵין לוֹ לִשָּׂא בַּת בָּנִים — אֵין לִמְכּוֹר סֵפֶר תּוֹרָה בִּשְׁבִיל לִשָּׂא בַּת בָּנִים, דְּאֵינוֹ אֶלָּא כְּעֵין יִשּׁוּב דֶּרֶךְ אֶרֶץ, כַּנִּזְכָּר לְעֵיל. וְיֵשׁ חוֹלְקִים וְסוֹבְרִים גַּם בִּשְׁבִיל זֶה, הוֹאִיל וְתַקָּנַת חֲכָמִים הוּא, מוֹכְרִים סֵפֶר תּוֹרָה בִּשְׁבִיל כֵּן, וְאַף עַל פִּי שֶׁיֵּשׁ לוֹ מְעַט וְיוּכַל לִשָּׂא שֶׁאֵינָהּ בַּת בָּנִים בְּלִי מְכִירַת סֵפֶר תּוֹרָה. וְנִרְאֶה לִי דְּיִתֵּן לְאַחֵר כָּל מַה שֶּׁיֵּשׁ לוֹ, בְּעִנְיָן שֶׁלֹּא יוּכַל לִשָּׂא גַּם שֶׁאֵינָהּ בַּת בָּנִים, וְיִמְכּוֹר הַסֵּפֶר תּוֹרָה, וּבְכַהַאי גַּוְנָא שַׁפִּיר דָּמֵי (דִּכְהַאי גַּוְנָא כָּתַבְתִּי בְּסִיַּעְתָּא דִשְׁמַיָּא בְּהִלְכוֹת עֲבָדִים, דְּרַשַּׁאי לְהַפְקִיר כָּל מַה שֶּׁיֵּשׁ לוֹ וְלִמְכּוֹר עַצְמוֹ לְעֶבֶד עִבְרִי):

**בְּנֵי נֹחַ** גַּם כֵּן חַיָּבִים בְּמִצְוָה זוֹ (שְׁאִלְתּוֹת דְּרַב אַחַאי גָּאוֹן סִימָן קס״ה). וְהַמַּהַרְשָׁ״א (חֲגִיגָה דַּף ב׳ עַמּוּד ב׳) חוֹלֵק עַל הַתּוֹסָפוֹת שֶׁכָּתְבוּ כְּהַשְּׁאִלְתּוֹת, מִגְּמָרָא דְּסַנְהֶדְרִין (דַּף נ״ט עַמּוּד ב׳). וְעַיֵּן בְּמִשְׁנֶה לַמֶּלֶךְ (פֶּרֶק י׳ מִמְּלָכִים), וּבִשְׁאִילַת שָׁלוֹם לְהַשְּׁאִלְתּוֹת שָׁם, וּבְטוּרֵי אֶבֶן בְּחִדּוּשֵׁי חֲגִיגָה שָׁם, וּבִשְׁבוּת יַעֲקֹב (חֵלֶק ב׳ סִימָן קל״ד). וּלְפִיכָךְ צָרִיךְ לִזָּהֵר בְּעוֹבֵד כּוֹכָבִים שֶׁכְּבָשַׁתּוּ מִלְּמָנְעוֹ מִפְּרִיָּה וּרְבִיָּה (וְכֵן לַהֲבִיאוֹ לִידֵי הַשְׁחָתַת זַרְעוֹ, כְּמוֹ שֶׁיִּתְבָּאֵר דִּבְהָא תַּלְיָא), דְּעוֹבֵר עַל "לִפְנֵי עִוֵּר לֹא תִתֵּן מִכְשֹׁל". וְכֵן הָעֲבָדִים חַיָּבִים, אֲבָל נָשִׁים פְּטוּרוֹת. וְיֵשׁ אוֹמְרִים דְּמִכָּל מָקוֹם מְצֻוּוֹת עַל "לֹא תֹהוּ בְרָאָהּ, לָשֶׁבֶת יְצָרָהּ", וְנַפְקָא מִינָּהּ דִּבְזָכָר אוֹ נְקֵבָה סַגִּי לְהוּ, כֵּיוָן דַּהֲוֵי "שֶׁבֶת" כָּל דְּהוּא. וְהָעִקָּר כְּדֵעָה רִאשׁוֹנָה. מִיהוּ אַף עַל פִּי שֶׁאֵינָהּ מְצֻוָּה, מִכָּל מָקוֹם מִצְוָה קָעָבְדָא כְּשֶׁמִּתְקַדֶּשֶׁת וְנִשֵּׂאת, לְפִי שֶׁבִּלְתָהּ לֹא יוּכַל הָאִישׁ לַעֲשׂוֹת מִצְוָתוֹ; וְגַם כִּי אֶפְשָׁר בְּאַחֶרֶת, מִכָּל מָקוֹם סְתָמֵיהּ בָּעֵי גַּבְרָא אִתְּתָא, וְכָל הַקּוֹדֶמֶת לְסַיְּעוֹ מִצְוָה עָבְדָא (תְּשׁוּבַת הָרַ״ן סִימָן ל״ב, עַיֵּן שָׁם בְּהַגָּהָה, וּבְהָרַ״ן רֵישׁ פֶּרֶק ב׳ דְּקִדּוּשִׁין). וְיֵשׁ אוֹמְרִים שֶׁלֹּא תַּעֲמוֹד בְּלֹא אִישׁ מִפְּנֵי הַחֲשָׁד. וּמִכָּל מָקוֹם כְּשֶׁצְּרִיכָה לְגַדֵּל יְתוֹמִים, וְחוֹשֶׁשֶׁת כְּשֶׁתִּנָּשֵׂא מִשּׁוּם קְטָטָה בֵּינָם וּבֵין בַּעְלָהּ, וְכָל כַּהַאי גַּוְנָא — וַדַּאי דְּרַשָּׁאָה שֶׁלֹּא לְהִנָּשֵׂא. וּמוֹכְרִים סֵפֶר תּוֹרָה לְהַשִּׂיא יְתוֹמָה גַּם כֵּן, וְאַף עַל פִּי שֶׁאֵין הָאִשָּׁה מְצֻוָּה (כֵּן הוּא הַסְכָּמַת הַתַּשְׁבֵּ״ץ בְּחוּט הַמְשֻׁלָּשׁ טוּר ג׳ סִימָן ל״א), וְעַיֵּן בָּאַחֲרוֹנִים שֶׁנֶּחְלְקוּ בָּזֶה. וּפָשׁוּט כִּי כָּל מִי שֶׁרוֹצֶה לְהִטַּפֵּל וְלִזְכּוֹת בְּמִצְוָה זוֹ, לְהַשִּׂיאָהּ אוֹ לְהַשִּׂיאוֹ — רַשַּׁאי לִמְכּוֹר סֵפֶר תּוֹרָה שֶׁלּוֹ, אִם אֵין לוֹ מָעוֹת אוֹ שְׁאָר חֲפָצִים שֶׁבִּיכָלְתּוֹ לְהַשִּׂיאָם; וְכֵן מַשְׁמָע בַּתַּשְׁבֵּ״ץ שָׁם. דִּינֵי אֵיזֶה אִשָּׁה יִקַּח — יִתְבָּאֵר בְּמִצְוַת קִדּוּשִׁין:
