~ י״ד — מִצְוָה מִן הַמֻּבְחָר, שֶׁנִּצּוֹל מֵהִרְהוּרִים (הַכֹּל בְּפֶרֶק קַמָּא דְּקִדּוּשִׁין). וּבְסַנְהֶדְרִין (דַּף ע״ו עַמּוּד ב׳) אָמְרִינַן: הַמַּשִּׂיא בָּנָיו סָמוּךְ לְפִרְקָן — עָלָיו הַכָּתוּב אוֹמֵר "וּפָקַדְתָּ נָוְךָ וְלֹא תֶחֱטָא". וְהַיְנוּ בִּהְיוֹתָם בֶּן י״ג, אֲבָל קֹדֶם לֹא יִשָּׂא, דַּהֲוֵי כִּבְעִילַת זְנוּת, דְּאֵין קִדּוּשִׁין לְקָטָן. וְיֵשׁ אוֹמְרִים דְּ"סָמוּךְ לְפִרְקָן" הַיְנוּ שָׁנָה אוֹ חֲצִי קֹדֶם י״ג גַּם כֵּן אִיכָּא מִצְוָה, וְלָא הֲוֵי כִּבְעִילַת זְנוּת כְּשֶׁאָבִיו מַשִּׂיאוֹ. אֲבָל הַמַּשִּׂיא אִשָּׁה לִבְנוֹ קָטָן שֶׁאֵין סָמוּךְ לְפִרְקוֹ — עָלָיו הַכָּתוּב אוֹמֵר "לְמַעַן סְפוֹת הָרָוָה אֶת הַצְּמֵאָה" וְכוּלֵי, מִפְּנֵי שֶׁהִיא צְמֵאָה לְתַשְׁמִישׁ וְתִזְנֶה עָלָיו; אִם לֹא כְּשֶׁגַּם הִיא קְטַנָּה, אָז שַׁפִּיר דָּמֵי לְדָבָר זֶה אַף קֹדֶם י״ג. וְיֵשׁ לִסְמוֹךְ עַל זֶה בִּשְׁעַת הַדְּחָק, כְּגוֹן שֶׁיֵּשׁ לָחוּשׁ לִגְזֵרַת הַמַּלְכוּת כְּמוֹ שֶׁאֵרַע כֵּן כַּמָּה פְּעָמִים, אוֹ בְּהַאי גַּוְנָא מִשּׁוּם דֹּחַק מָעוֹת. וְיִשָּׂא אִשָּׁה קֹדֶם שֶׁיִּלְמוֹד תּוֹרָה, כְּדֵי שֶׁיִּלְמוֹד אַחַר כָּךְ תּוֹרָה בְּטָהֳרָה. בַּמֶּה דְּבָרִים אֲמוּרִים — לְעָשִׁיר, שֶׁכְּשֶׁיִּשָּׂא לֹא יִטָּרֵד בִּמְזוֹנוֹתָיו; אֲבָל בְּעָנִי, שֶׁאִם יִשָּׂא יִטָּרֵד בִּמְזוֹנוֹת — יִלְמוֹד תּוֹרָה קֹדֶם, כָּל הַתּוֹרָה כֻּלָּהּ, עַד שֶׁיִּהְיֶה בָּקִי בָּהֶן וּבְטַעְמֵיהֶם וּבְדִקְדּוּקֵיהֶם בְּדֶרֶךְ קְצָרָה, וְאַחַר כָּךְ יִשָּׂא אִשָּׁה. וַאֲפִלּוּ עָבְרוּ עָלָיו כ׳ וַעֲדַיִן אֵינוֹ בָּקִי בְּכָל הַתּוֹרָה, וְנַפְשׁוֹ חָשְׁקָה בַּתּוֹרָה וְיָרֵא שֶׁמָּא יִטָּרֵד — רַשַּׁאי לְהִדָּחוֹת מִצְוַת פְּרִיָּה וּרְבִיָּה בִּשְׁבִיל מִצְוַת תַּלְמוּד תּוֹרָה. וּבִלְבַד שֶׁלֹּא יִהְיֶה יִצְרוֹ מִתְגַּבֵּר עָלָיו; אֲבָל אִם יִצְרוֹ מִתְגַּבֵּר עָלָיו, עַד שֶׁנִּמְצָא שֶׁאֵין לִבּוֹ פָּנוּי מֵהִרְהוּרִים — יִשָּׂא אִשָּׁה תְּחִלָּה, וְאַחַר כָּךְ יִלְמוֹד תּוֹרָה בְּטָהֳרָה כְּפִי יְכָלְתּוֹ:

**אֲפִלּוּ** נָשָׂא אִשָּׁה וְאֵין לוֹ בָּנִים (כַּנִּזְכָּר לְעֵיל מִפֶּרֶק עַרְבֵי פְּסָחִים), כְּגוֹן שֶׁנָּשָׂא אִשָּׁה עֲקָרָה, זְקֵנָה, קְטַנָּה, אוֹ שֶׁאֵינָהּ רְאוּיָה לֵילֵד — לֹא קִיֵּם הַמִּצְוָה, אֶלָּא צָרִיךְ לִשָּׂא אִשָּׁה בַּת בָּנִים. וְכֵיוָן שֶׁהוֹלִיד זָכָר וּנְקֵבָה קִיֵּם הַמִּצְוָה, וּבִלְבַד שֶׁלֹּא יִהְיֶה הַבֵּן סָרִיס אוֹ הַנְּקֵבָה אַיְלוֹנִית. אֲבָל אִם הַבֵּן נָשָׂא אַיְלוֹנִית, אוֹ הַבַּת נִשֵּׂאת לְסָרִיס, אוֹ שֶׁלֹּא נִשְּׂאוּ כְּלָל — קִיֵּם הַמִּצְוָה. וְאִם מֵתוּ: אִם הִנִּיחוּ זָכָר וּנְקֵבָה, אֲפִלּוּ הַנְּקֵבָה מֵהַזָּכָר וְהַזָּכָר מֵהַנְּקֵבָה — קִיֵּם פְּרִיָּה וּרְבִיָּה; אֲבָל אִם לֹא הִנִּיחוּ, אוֹ שֶׁהִנִּיחַ אֶחָד זָכָר וּנְקֵבָה וְהַשֵּׁנִי לֹא הִנִּיחַ כְּלוּם — לֹא קִיֵּם הַמִּצְוָה. הָיוּ לוֹ ב׳ זְכָרִים בְּלֹא נְקֵבָה, אוֹ ב׳ נְקֵבוֹת בְּלֹא זָכָר — לֹא קִיֵּם. וְכֵן אִם הָיוּ לוֹ זָכָר וּנְקֵבָה וּמֵתוּ וְהִנִּיחוּ ב׳ זְכָרִים אוֹ ב׳ נְקֵבוֹת, אוֹ שֶׁהָיָה לוֹ זָכָר וּנְקֵבָה וּמֵת הַזָּכָר וְהִנִּיחַ בַּת, אוֹ לְהֵפֶךְ — לֹא קִיֵּם פְּרִיָּה וּרְבִיָּה. וְיֵשׁ חוֹלְקִים וְסוֹבְרִים דִּכְשֶׁהָיוּ לוֹ זָכָר וּנְקֵבָה וּמֵתוּ שְׁנֵיהֶם אוֹ אֶחָד, וְהִנִּיחוּ, וְנִמְצָא לוֹ ב׳ זְכָרִים אוֹ ב׳ נְקֵבוֹת — קִיֵּם פְּרִיָּה וּרְבִיָּה, כֵּיוָן שֶׁבָּאִים מִזָּכָר וּנְקֵבָה. וְדַע דִּמְסֻפְּקַנִי אִם מֵתוּ גַּם אֵשֶׁת הַזָּכָר וּבַעַל
