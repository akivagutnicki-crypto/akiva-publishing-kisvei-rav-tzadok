~ רֹב עַם בְּבֵית הַכְּנֶסֶת — מְבִיתוֹ, וְאֵין צֹרֶךְ כְּלָל — אָסוּר, דְּלֹא הוּתַּר אֲמִירָה לְעוֹבֵד כּוֹכָבִים אֲפִלּוּ בִּדְרַבָּנָן אֶלָּא לְצֹרֶךְ מִצְוָה. וְאִם הָאִיזְמֵל בְּמָקוֹם אֶחָד וְתִינוֹק בְּמָקוֹם אַחֵר, וְצָרִיךְ עוֹבֵד כּוֹכָבִים לְהָבִיא — מֵבִיא אִיזְמֵל אֵצֶל תִּינוֹק, מִפְּנֵי שֶׁקַּל הוּא הֲבָאַת הָאִיזְמֵל מֵהֲבָאַת הַתִּינוֹק (תּוֹסָפוֹת שַׁבָּת ק״ל עַמּוּד א׳). וִיהֵא הַחוֹל לִזְרוֹק בּוֹ הָעָרְלָה מוּכָן מִבְּעוֹד יוֹם. וְיֵשׁ אוֹמְרִים דַּאֲפִלּוּ מוּכָן מִבְּעוֹד יוֹם אָסוּר, אֲבָל לֹא נָהֲגוּ כֵּן. אֲבָל אִיזְמֵל שֶׁל מִילָה, שֶׁמַּקְצֵהוּ שֶׁלֹּא לְהִשְׁתַּמֵּשׁ בּוֹ דָּבָר אַחֵר — אָסוּר לְטַלְטְלוֹ קֹדֶם הַמִּילָה אוֹ אַחַר כָּךְ, אֶלָּא לְצֹרֶךְ הַמִּילָה בִּלְבַד. וְאִם יָרֵא שֶׁמָּא יֹאבַד — יֵשׁ אוֹמְרִים דְּהוּתַּר לְטַלְטְלוֹ לְהַטְמִינוֹ בְּאוֹתָהּ חָצֵר, שֶׁאִם אַתָּה אוֹסְרוֹ — יִמָּנַע מִלִּימּוֹל. וְיֵשׁ אוֹמְרִים דִּכְשֶׁעוֹדוֹ בְּיָדוֹ רַשַּׁאי לְטַלְטְלוֹ לְהַטְמִינוֹ לְכָל מָקוֹם שֶׁיִּרְצֶה. וְאִם הוּא בְּעַצְמוֹ הַפּוֹרֵעַ, שֶׁאֵין יָכוֹל לֶאֱחוֹז הַסַּכִּין בְּיָדוֹ — יוּכַל לִתְּנוֹ לְאַחֵר, וְהָאַחֵר יְטַלְטְלוֹ; אוֹ הוּא עַצְמוֹ יִקָּחֶנּוּ אַחַר כָּךְ מֵהָאַחֵר וִיטַלְטְלוֹ לְהַצְנִיעוֹ. אֲבָל אִם הִנִּיחוֹ עַל גַּבֵּי הַקַּרְקַע, אוֹ עַל מָקוֹם אַחֵר שֶׁלֹּא בְּיַד אָדָם — נִרְאֶה דִּלְפִי זֶה אָסוּר אַף לְטַלְטֵל לְהַצְנִיעוֹ בְּאוֹתוֹ חָצֵר, דְּלָא שַׁיָּךְ שֶׁמָּא יִמָּנַע, דִּבְאֶמֶת יִזָּהֵר לֶאֱחוֹז בְּיָדוֹ אוֹ לִתְּנוֹ בְּיַד אַחֵר, וְשֶׁלֹּא לְהַנִּיחוֹ. מִיהוּ אַף עַל פִּי שֶׁהִנִּיחוֹ — אִם שָׁכַח וַאֲחָזוֹ בְּיָדוֹ — רַשַּׁאי לְטַלְטְלוֹ לְכָל מָקוֹם שֶׁיִּרְצֶה. וְיֵשׁ אוֹמְרִים דִּבְכָל גַּוְנֵי מֻתָּר לְטַלְטְלוֹ, וְאַף קֹדֶם הַמִּילָה, דְּכֵיוָן שֶׁהוּתַּר לְמִקְצַת שַׁבָּת בְּעֵת הַמִּילָה — הוּתַּר לְכָל הַשַּׁבָּת; וְנָכוֹן לְהַחְמִיר. וּלְקַנֵּחַ הַסַּכִּין וּלְרָחֲצוֹ — לְכוּלֵּי עָלְמָא אָסוּר, דְּאֵין מְדִיחִים בְּכֵלִים בְּשַׁבָּת. מִיהוּ אִם יַנִּיחוֹ עַד לְמָחָר, וְיִיבַשׁ הַדָּם עָלָיו, יִתְקַלְקֵל הַסַּכִּין — אֶפְשָׁר דְּשַׁפִּיר דָּמֵי לְרָחֲצוֹ, דִּלְמָא יִמָּנַע וְלֹא יָמוּל, וּכְדִבְרֵי הַיֵּשׁ אוֹמְרִים הַנִּזְכָּר לְעֵיל לְעִנְיַן טִלְטוּל שֶׁמָּא יֹאבַד; וְצָרִיךְ לְעַיֵּן. וְכָל דְּבָרִים אֵלּוּ שָׁוִים בֵּין בְּשַׁבָּת בֵּין בְּיוֹם טוֹב, חוּץ מִדָּבָר שֶׁיֵּשׁ בּוֹ אֹכֶל נֶפֶשׁ — כְּגוֹן שֶׁהָיוּ נוֹהֲגִים בַּיָּמִים קַדְמוֹנִים לִתֵּן עַל הַמִּילָה כַּמּוֹן שָׁחִיק, וְיַיִן וָשֶׁמֶן מְעֹרָבִין — דִּבְשַׁבָּת אָסוּר לִשְׁחוֹק אוֹ לְעָרְבוֹ אֶלָּא עַל יְדֵי שִׁנּוּי; אֲבָל בְּיוֹם טוֹב שַׁפִּיר דָּמֵי, כֵּיוָן דַּחֲזֵי לְאֹכֶל נֶפֶשׁ:

**אֶחָד** בֶּן ז׳ חֳדָשִׁים וְאֶחָד בֶּן ט׳ — דּוֹחֶה שַׁבָּת; וְהוּא שֶׁיָּדוּעַ שֶׁהוּא בֶּן ז׳ אוֹ ט׳ (כְּגוֹן שֶׁפֵּרַשׁ בַּעְלָהּ אַחַר שֶׁבְּעָלָהּ). אֲבָל נוֹלַד לְח׳, אוֹ סְפֵק בֶּן ז׳ סְפֵק בֶּן ח׳: אִם גָּמְרוּ שְׂעָרוֹ וְצִפָּרְנָיו — מָלִים; וְאִם לָאו, אֶלָּא שֶׁהָיוּ שְׂעָרוֹ לָקוּי וְאֵין צִפָּרְנָיו שְׁלֵמִים כִּבְרִיָּתָן — אֵין מָלִין. וְיֵשׁ אוֹמְרִים דִּבְסָפֵק בֶּן ח׳ מָלִין אוֹתוֹ בְּשַׁבָּת, מִמָּה נַפְשָׁךְ: דְּאִם הוּא בֶּן ח׳ — וְהוּא כְּמֵת — הֲרֵי הוּא כִּמְחַתֵּךְ בְּבָשָׂר בְּעָלְמָא. וְכָל זֶה בְּמִילָה בִּזְמַנָּהּ; אֲבָל מִילָה שֶׁלֹּא בִּזְמַנָּהּ — כְּגוֹן שֶׁחָלָה וְנִדְחָה הַמִּילָה, אוֹ גֵּר שֶׁבָּא לְהִתְגַּיֵּר — אֵין מִילָתוֹ דּוֹחָה שַׁבָּת וְלֹא יוֹם טוֹב. וְאִם נוֹלַד בֵּין הַשְּׁמָשׁוֹת, שֶׁדִּינוֹ לְמוּלוֹ מִסָּפֵק בַּיּוֹם הָאַחֲרוֹן, כְּדֵי שֶׁלֹּא יִהְיֶה סָפֵק קֹדֶם
