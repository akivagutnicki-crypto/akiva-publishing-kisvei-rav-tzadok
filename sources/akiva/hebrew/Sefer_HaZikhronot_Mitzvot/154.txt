~ מְנָחוֹת כ״ז עַמּוּד ב׳, דִּבּוּר הַמַּתְחִיל "לְמַעוּטֵי"); אֲבָל מַכִּין אוֹתוֹ מַכַּת מַרְדּוּת (וְדַוְקָא שֶׁנִּכְנַס וְיָצָא תֵּכֶף; אֲבָל אִם שָׁהָה (מִשְׁנֶה לַמֶּלֶךְ) שָׁם כְּשִׁעוּר הִשְׁתַּחֲוָיָה, שֶׁיִּתְבָּאֵר לְקַמָּן — חַיָּב כָּרֵת, אַף עַל פִּי שֶׁנִּכְנַס דֶּרֶךְ גַּגִּין; דְּלֹא עָדִיף מֵאִלּוּ נִטְמָא שָׁם בַּמִּקְדָּשׁ, שֶׁיִּתְבָּאֵר בְּסָמוּךְ). וְכֵן הַדִּין בְּנִכְנָס בְּמִגְדָּל הַפּוֹרֵחַ בָּאֲוִיר — בֵּין שֶׁנִּכְנַס לַמִּקְדָּשׁ בְּמִגְדָּל דֶּרֶךְ גַּגּוֹת, בֵּין שֶׁנִּכְנַס בּוֹ דֶּרֶךְ פְּתָחִים:

**אֶחָד** טָמֵא שֶׁנִּכְנַס לַמִּקְדָּשׁ טָהוֹר, וְאֶחָד טָהוֹר שֶׁנִּכְנַס לַמִּקְדָּשׁ שֶׁיֵּשׁ בּוֹ טֻמְאָה — כְּגוֹן שֶׁהָיָה מֵת תַּחַת אֹהֶל בַּמִּקְדָּשׁ, וְנִכְנַס הוּא תַּחַת הָאֹהֶל — הֲרֵי זֶה חַיָּב כָּרֵת, שֶׁהֲרֵי בִּיאָתוֹ וְטֻמְאָה בָּאִין כְּאַחַת. וְאִם נִכְנַס לַמִּקְדָּשׁ, וְנִטְמָא שָׁם אַחַר שֶׁנִּכְנַס, וְלֹא בָּאוּ הַבִּיאָה וְטֻמְאָה כְּאַחַת — בֵּין שֶׁנִּטְמָא בְּאֹנֶס בֵּין בְּרָצוֹן, בֵּין שׁוֹגֵג בֵּין מֵזִיד (וְלֹא הִתְרוּ בּוֹ קֹדֶם שֶׁטִּמֵּא אֶת עַצְמוֹ; אֲבָל אִם הִתְרוּ בּוֹ שֶׁלֹּא יְטַמֵּא עַצְמוֹ, וְטִמֵּא עַצְמוֹ — לוֹקֶה; כֵּן כָּתְבוּ תּוֹסָפוֹת שְׁבוּעוֹת י״ז עַמּוּד א׳, דִּבּוּר הַמַּתְחִיל "צָרִיךְ"), וְהִתְרוּ בּוֹ שֶׁלֹּא יִשְׁהֶה, וְשָׁהָה מְעַט — פָּטוּר מִכְּלוּם (וְנִרְאֶה, דְּמִצְוַת עֲשֵׂה דְּשִׁלּוּחַ אִיכָּא אַף אַשְּׁהִיָּה מוּעֶטֶת, שֶׁלֹּא יִשְׁהֶה, וּכְמוֹ שֶׁנִּתְבָּאֵר לְעֵיל, סִימָן שֶׁלִּפְנֵי זֶה; וְאַמִּצְוַת עֲשֵׂה — מַכִּין עַד שֶׁתֵּצֵא נַפְשׁוֹ); אֶלָּא אִם כֵּן שָׁהָה הַרְבֵּה — כְּשִׁעוּר הִשְׁתַּחֲוָאָה שֶׁל פִּשּׁוּט יָדַיִם וְרַגְלַיִם, שֶׁשִּׁעֲרוּ חֲכָמִים שִׁעוּרוֹ כְּדֵי לִקְרוֹת פָּסוּק "וַיִּכְרְעוּ אַפַּיִם אַרְצָה עַל הָרִצְפָּה וַיִּשְׁתַּחֲווּ וְהוֹדוֹת לַה׳ כִּי טוֹב כִּי לְעוֹלָם חַסְדּוֹ"; שֶׁאָז, אִם שָׁהָה כְּשִׁעוּר הַזֶּה, בְּשׁוֹגֵג (הֶעְלֵם מִקְדָּשׁ אוֹ הֶעְלֵם טֻמְאָה; וְאִם טִמֵּא עַצְמוֹ בְּמֵזִיד — צָרִיךְ שֶׁלֹּא יָדַע אָז שֶׁהָיָה בַּמִּקְדָּשׁ, דִּבְלָאו הָכֵי הֲוָה לֵיהּ יְדִיעָה מִתְּחִלָּה, וַאֲפִלּוּ שָׁכַח אַחַר כָּךְ, כְּשֶׁשָּׁהָה — אֵינוֹ חַיָּב קָרְבָּן; כֵּן כָּתְבוּ הַתּוֹסָפוֹת. אֲבָל בְּסֵפֶר תּוֹרַת חַיִּים שָׁם כָּתַב, דַּאֲפִלּוּ בִּכְהַאי גַּוְנָא, דַּהֲוָה לֵיהּ יְדִיעָה בַּתְּחִלָּה וְשָׁכַח אַחַר כָּךְ — חַיָּב קָרְבָּן, עַיֵּן שָׁם) — חַיָּב קָרְבָּן; וּבְמֵזִיד — חַיָּב כָּרֵת, וְלוֹקֶה:

**(וְאַף עַל גַּב** דַּהֲוֵי הַתְרָאַת סָפֵק — דְּשֶׁמָּא לֹא יִשְׁהֶה כָּל כָּךְ — כְּמוֹ שֶׁכָּתוּב בְּחִדּוּשֵׁי הָרַשְׁבָּ״א שָׁם — מִכָּל מָקוֹם, הָא קַיְמָא לָן הַתְרָאַת סָפֵק שְׁמָהּ הַתְרָאָה. וְאַף עַל פִּי דְּלֹא עָבֵיד מַעֲשֶׂה — מִכָּל מָקוֹם, תְּחִלַּת הַטֻּמְאָה, שֶׁהָיָה בְּמַעֲשֶׂה — הֲוֵי מַעֲשֶׂה גַּם לַשְּׁהִיָּה. מִיהוּ, כָּתַב בְּמִשְׁנֶה לַמֶּלֶךְ: דְּאִם נִטְמָא בְּאֹנֶס — לֹא חָשִׁיב תְּחִלַּת הַטֻּמְאָה בְּמַעֲשֶׂה, דְּגַם כֵּן הוּא שֶׁלֹּא בְּמַעֲשֶׂה; וּלְכָךְ אֵין לוֹקִין אַף עַל הַשְּׁהִיָּה, דַּהֲוֵי לָאו שֶׁאֵין בּוֹ מַעֲשֶׂה; וְאִם נִטְמָא בְּשׁוֹגֵג — מְסַתְּפֵק אִם מִקְּרֵי מַעֲשֶׂה אִם לֹא, עַיֵּן שָׁם. מִיהוּ, לְפִי מַה שֶּׁכָּתַב הָרַב הַמַּגִּיד (פֶּרֶק י״ג מֵהִלְכוֹת שְׂכִירוּת), דְּכָל לָאו שֶׁיֵּשׁ בּוֹ פְּעָמִים מַעֲשֶׂה — אַף כְּשֶׁאֵין בּוֹ מַעֲשֶׂה לוֹקֶה, עַיֵּן שָׁם; וְעַיֵּן בְּמִשְׁנֶה לַמֶּלֶךְ (פֶּרֶק ד׳ מִמַּלְוֶה), שֶׁהֶאֱרִיךְ בּוֹ, וְאֵין כָּאן מְקוֹמוֹ — אִם כֵּן, לִדְבָרָיו הוּא הַדִּין לְנִדּוֹן דִּידַן — בְּכָל גַּוְנֵי לָקֵי):
