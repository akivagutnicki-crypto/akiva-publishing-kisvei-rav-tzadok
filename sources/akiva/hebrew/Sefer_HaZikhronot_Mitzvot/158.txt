~ וְכֻלָּם לֹא אָמְרוּ עַל הַתְּשׁוּבָה, זוּלַת הַקָּדוֹשׁ בָּרוּךְ הוּא בְּעַצְמוֹ. וְהַיְנוּ, כִּי בְּכֻלָּם הָיָה הַקִּלְקוּל עַל יְדֵי הֶעָוֹן; וּבַמֶּה יְתֻקַּן אוֹתוֹ קִלְקוּל, שֶׁכְּבָר נַעֲשָׂה עַל יְדֵי הָעֲבֵרָה? אֲבָל הַקָּדוֹשׁ בָּרוּךְ הוּא בְּעַצְמוֹ — שֶׁאֶצְלוֹ אֵין שַׁיָּךְ שׁוּם קִלְקוּל, רַק הַחֵטְא בַּמֶּה שֶּׁהָאָדָם עָבַר עַל רְצוֹנוֹ וְצִוּוּיוֹ — בָּזֶה, כֵּיוָן שֶׁנִּתְחָרֵט וְשָׁב, הֲרֵי הוּא מוֹחֵל לוֹ; וְלָכֵן הוּא אוֹמֵר: "יַעֲשֶׂה תְּשׁוּבָה וְיִתְכַּפֵּר". וְלָכֵן הַתְּשׁוּבָה מַגַּעַת עַד כִּסֵּא כְּבוֹדוֹ יִתְבָּרַךְ, וְאֶצְלוֹ מִתְקַבֶּלֶת; וְאַחַר כָּךְ חוֹזֶרֶת בְּכֹחוֹ יִתְבָּרַךְ שֶׁקִּבְּלָהּ, וּמוֹסֶפֶת גַּם כֵּן כֹּחַ בַּפָּמַלְיָא שֶׁל מַעְלָה; וְהָבֵן:

**וְנוֹהֶגֶת** בְּכָל מָקוֹם וּבְכָל זְמַן. וְאַף עַל פִּי שֶׁמִּפְּשׁוּטוֹ שֶׁל מִקְרָא אֶפְשָׁר שֶׁאֵין הַוִּדּוּי אֶלָּא נִגְרָר לְקָרְבָּן — שֶׁהֲרֵי אָמַר אַחַר כָּךְ: "וְהֵשִׁיב אֶת אֲשָׁמוֹ" וְגוֹ׳, "מִלְּבַד אֵיל הַכִּפֻּרִים" (עַיֵּן בְּסֵפֶר הַחִנּוּךְ, הֵבִיא מִסֵּפֶר וַיִּקְרָא; וְלֹא יָדַעְתִּי לָמָּה הֵבִיא מִמֶּרְחָק לַחְמוֹ) — קִבְּלוּ חֲכָמִים (מְכִילְתָּא) מִפִּי הַשְּׁמוּעָה, שֶׁהַוִּדּוּי מִצְוָה בִּפְנֵי עַצְמָהּ, וְאֵינָהּ נִגְרֶרֶת לְקָרְבָּן; וַאֲפִלּוּ כְּשֶׁאֵין מְבִיאִים קָרְבָּן — מִתְוַדִּים. "וְאֵין לִי אֶלָּא בָּאָרֶץ (סִפְרֵי זוּטָא, הוּבָא בְּיַלְקוּט), שֶׁהוּא מְקוֹם קָרְבָּנוֹת וּמְקוֹם הַכַּפָּרָה; בְּחוּץ לָאָרֶץ מִנַּיִן שֶׁמִּתְוַדִּין? תַּלְמוּד לוֹמַר: 'וְהִתְוַדּוּ אֶת עֲוֹנָם וְאֶת עֲוֹן אֲבֹתָם'" — כְּלוֹמַר, עֲוֹן אֲבוֹתָם שֶׁחָטְאוּ וְגֹרְשׁוּ מִן הָאָרֶץ; וְלָמַדְנוּ שֶׁאַף בְּחוּץ לָאָרֶץ מִתְוַדִּין. וְכֵן דָּנִיֵּאל אָמַר בְּחוּץ לָאָרֶץ: "לְךָ ה׳ הַצְּדָקָה, וְלָנוּ בֹּשֶׁת הַפָּנִים" וְגוֹ׳:

**וְנוֹהֶגֶת** בְּכָל אָדָם — בֵּין בִּזְכָרִים בֵּין בִּנְקֵבוֹת — שֶׁנֶּאֱמַר "אִישׁ אוֹ אִשָּׁה". "יָכוֹל שֶׁאֲנִי מוֹצִיא טֻמְטוּם וְאַנְדְּרוֹגִינוֹס (סִפְרֵי פִּסְקָא ב׳)? תַּלְמוּד לוֹמַר: 'וְאָשְׁמָה הַנֶּפֶשׁ הַהִוא' — מִכָּל מָקוֹם". "וְאֵין לִי אֶלָּא בְּנֵי יִשְׂרָאֵל; גֵּרִים מִנַּיִן (סִפְרֵי זוּטָא וּבְיַלְקוּט)? תַּלְמוּד לוֹמַר: 'אִישׁ' — מִכָּל מָקוֹם" (וּבְסִפְרֵי שָׁם, וְהוּבָא גַּם כֵּן בְּיַלְקוּט, אִיתָא: מִ"וְּאָשְׁמָה הַנֶּפֶשׁ הַהִוא" — לְרַבּוֹת גֵּרִים). "אִם כֵּן, לָמָּה נֶאֱמַר 'בְּנֵי יִשְׂרָאֵל'? לְהוֹצִיא הָעוֹבְדֵי כּוֹכָבִים וְהַתּוֹשָׁבִים" (פֵּירוּשׁ: גֵּר תּוֹשָׁב). וּמִכֵּיוָן שֶׁנֶּאֱמַר "וְאָשְׁמָה הַנֶּפֶשׁ הַהִוא" — דְּמַשְׁמַע נֶפֶשׁ מִכָּל מָקוֹם — יָכוֹל אֲפִלּוּ קָטָן בְּמַשְׁמָע (כְּמוֹ "וְאִישׁ כִּי יַכֶּה כָּל נֶפֶשׁ", דְּדָרְשִׁינַן בְּסַנְהֶדְרִין — אֲפִלּוּ קָטָן)? לָמְדוּ (סִפְרֵי) מִפִּי הַקַּבָּלָה — קַל וָחֹמֶר מֵעֲבוֹדָה זָרָה, שֶׁנֶּאֱמַר בּוֹ "אִישׁ", וְאֵין הַקָּטָן מֻזְהָר עָלָיו — קַל וָחֹמֶר לִשְׁאָר מִצְוֹת שֶׁבַּתּוֹרָה, דְּאֵין חַיָּב; וְהוּא הַדִּין שֶׁאֵין חַיָּב בְּמִצְוָה זוֹ דְּוִדּוּי. וְגַם — עַל מַה יִּתְוַדֶּה, כֵּיוָן שֶׁאֵין עָלָיו חִיּוּב שׁוּם מִצְוָה וַעֲבֵרָה?

**מִיהוּ**, כָּתַב הָרַמָ״א בְּאֹרַח חַיִּים (סוֹף סִימָן שמ״ג), דְּמִכָּל מָקוֹם, כְּשֶׁיַּגְדִּיל — טוֹב לַקָּטָן לְקַבֵּל עָלָיו אֵיזֶה דָּבָר לְכַפָּרָה עַל עֲבֵרוֹת שֶׁעָשָׂה בְּקַטְנוּתוֹ. וְעַיֵּן בִּתְשׁוּבוֹת
