**(וְעַיֵּן** בִּתְשׁוּבוֹת צֶמַח צֶדֶק (סִימָן נ׳), כָּתַב לְעִנְיַן אֵשֶׁת אִישׁ שֶׁזִּנְּתָה, דִּתְשׁוּבָתָהּ קָשָׁה — אוּלַי הוֹלִידָה מַמְזֵר; "וְאַף עַל גַּב דְּרֹב בְּעִילוֹת אַחַר הַבַּעַל — זֶהוּ לְעִנְיַן הַוָּלָד; אֲבָל לְעִנְיַן לַעֲשׂוֹת תְּשׁוּבָה, וַדַּאי צְרִיכָה לָחוּשׁ אַף לְמִעוּט, דִּילְמָא נִתְעַבְּרָה מֵהַנּוֹאֵף; וּכְבָר כָּתְבוּ הַפּוֹסְקִים, דִּסְפֵק עֲבֵרָה צָרִיךְ יוֹתֵר תְּשׁוּבָה מִוַּדַּאי עֲבֵרָה" וְכוּלֵי, עַד כָּאן. וְאַחֲרָיו נִמְשְׁכוּ בְּאֵיזֶה תְּשׁוּבוֹת אַחֲרוֹנִים לְמֵימַר, דִּתְשׁוּבָה צָרִיךְ אַף לְמֵיחַשׁ לְמִעוּטָא. וְלִי לֹא נִרְאֶה כֵּן: דְּכֵיוָן דְּלֹא קַיְמָא לָן כְּרַבִּי מֵאִיר, דְּחָיֵישׁ לְמִעוּטָא; וְאִלּוּ הָיְתָה בָּאָה לִשְׁאוֹל לָנוּ בְּדָבָר שֶׁיֵּשׁ בּוֹ רֹב לְהֶתֵּר — הָיִינוּ מוֹרִים לָהּ לְהַתִּיר לַעֲשׂוֹת לְכַתְּחִלָּה, דְּאָזְלִינַן בָּתַר רֹב — אֵיךְ אַחַר כָּךְ נֹאמַר שֶׁיַּעֲשֶׂה תְּשׁוּבָה לָחוּשׁ לְמִעוּט? אֶתְמְהָה! וְלֹא כָּתְבוּ הַפּוֹסְקִים אֶלָּא בִּסְפֵק אִסּוּר, דְּאָסוּר בֶּאֱמֶת — צָרִיךְ תְּשׁוּבָה יוֹתֵר מִוַּדַּאי, וּמִטַּעַם שֶׁנִּתְבָּאֵר; אֲבָל לֹא בְּמִעוּט, דְּלֹא חָיְישִׁינַן לֵיהּ):

**וְעַיֵּן** בְּיַלְקוּט, דָּרֵישׁ "אִישׁ אוֹ אִשָּׁה כִּי יַעֲשׂוּ" — "שֶׁחָשְׁבוּ לַעֲשׂוֹת וְלֹא עָשׂוּ"; וּפָשׁוּט דְּטָעוּת סוֹפֵר, וְצָרִיךְ לוֹמַר: "וְלֹא שֶׁחָשְׁבוּ לַעֲשׂוֹת". וְהַיְנוּ בִּשְׁאָר מִצְוֹת, דְּהָא אָמְרִינַן בְּסוֹף פֶּרֶק קַמָּא דְּקִדּוּשִׁין: "מַחֲשָׁבָה רָעָה אֵין הַקָּדוֹשׁ בָּרוּךְ הוּא מְצָרְפָהּ לְמַעֲשֶׂה" — וְאִם כֵּן, עַל מַה יִּתְוַדֶּה? אֲבָל בַּעֲבוֹדָה זָרָה — דְּאָמְרִינַן הָתָם דְּאַף אַמַּחֲשָׁבָה עוֹבֵר — הוּא הַדִּין נַמִּי דְּצָרִיךְ לִתְשׁוּבָה אַהִרְהוּר וְחִשֵּׁב לַעֲשׂוֹת; וְקָאָמַר, דְּאֵין צָרִיךְ לְהִתְוַדּוֹת בַּפֶּה כְּמוֹ עַל מַעֲשֶׂה, אֶלָּא דַּי בַּלֵּב — דְּמַחֲשָׁבָה מוֹצִיא מִידֵי מַחֲשָׁבָה, וְאֵין מַחֲשָׁבָה מוֹצִיא מִידֵי מַעֲשֶׂה, כְּדֶרֶךְ שֶׁאִיתָא בְּפֶרֶק קַמָּא דְּחֻלִּין (דַּף י״ב), עַיֵּן שָׁם. וְעַיֵּן בְּסֵפֶר כַּד הַקֶּמַח, עַל מַה שֶּׁאִיתָא בְּיוֹמָא "הִרְהוּרֵי עֲבֵרָה קָשִׁין מֵעֲבֵרָה", וּמש״ל (סִימָן א׳, פֶּרֶק ב׳), עַיֵּן שָׁם. (וְעַיֵּן שָׁם עוֹד, בְּסוֹף פֶּרֶק קַמָּא דְּקִדּוּשִׁין, דִּלְעוּלָּא — כָּל שֶׁעָבַר וְשָׁנָה, נֶעֱנָשׁ אַחַר כָּךְ גַּם עַל הַמַּחֲשָׁבָה, עַיֵּן שָׁם; וְעַיֵּן בְּסֵפֶר דַּעַת קְדוֹשִׁים (דְּרוּשׁ א׳), הֶאֱרִיךְ בָּזֶה. מִיהוּ, אֶפְשָׁר — אַף לְעוּלָּא, אִם עָשָׂה תְּשׁוּבָה וַחֲרָטָה וַעֲזִיבָה עַל גּוּף הָעֲבֵרָה — אַחַר כָּךְ אֵין הַמַּחֲשָׁבָה מִצְטָרֶפֶת לְמַעֲשֶׂה; כֵּן נִרְאֶה לִי):

**וְאֵימָתַי** הִיא הַתְּשׁוּבָה? הַיּוֹתֵר רְאוּיָה וּמֻבְחֶרֶת — הִיא תֵּכֶף עֲשׂוֹת הַחֵטְא. וְזֶהוּ דֶּרֶךְ הַתַּלְמִידֵי חֲכָמִים, שֶׁלָּכֵן אָמְרוּ (בְּרָכוֹת י״ט עַמּוּד א׳): "תַּלְמִיד חָכָם שֶׁעָבַר עֲבֵרָה בַּיּוֹם — אַל תְּהַרְהֵר אַחֲרָיו בַּלַּיְלָה, וַדַּאי עָשָׂה תְּשׁוּבָה". כִּי צָרִיךְ הָאָדָם שֶׁיִּרְאֶה עַצְמוֹ לְעוֹלָם (רַמְבַּ״ם רֵישׁ פֶּרֶק ז׳) כְּאִלּוּ הוּא נוֹטֶה לָמוּת, וְשֶׁמָּא יָמוּת בִּשְׁעָתוֹ, וְנִמְצָא עוֹמֵד בְּחֶטְאוֹ; לְפִיכָךְ יָשׁוּב מִיָּד. וְהוּא שֶׁשְּׁלֹמֹה אָמַר בְּחָכְמָתוֹ: "בְּכָל עֵת יִהְיוּ בְגָדֶיךָ לְבָנִים". וְעוֹד (שַׁעֲרֵי תְּשׁוּבָה לְרַבֵּינוּ יוֹנָה, אוֹת ד׳ וְה׳): שֶׁכְּשֶׁיָּשׁוּב מִיָּד — לֹא יוֹסִיף לַחֲטוֹא, כִּי יִזְכּוֹר הַמְּרִירוּת שֶׁהָיָה לוֹ עַכְשָׁיו. וְעוֹד: שֶׁאַחַר כָּךְ, כְּשֶׁיִּשְׁנֶה בַּחֵטְא וְלֹא יָשׁוּב בֵּינְתַיִם — יִהְיֶה נַעֲשֶׂה לוֹ כְּהֶתֵּר,
