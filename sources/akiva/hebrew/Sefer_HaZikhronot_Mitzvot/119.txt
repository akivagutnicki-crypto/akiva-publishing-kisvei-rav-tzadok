~ וַאֲפִלּוּ בְּמָקוֹם שֶׁאֵין צָרִיךְ — רַשַּׁאי לַחְזוֹר וְלוֹמַר הַלֵּל; וְאֵין זֶה בִּכְלַל "הַקּוֹרֵא הַלֵּל בְּכָל יוֹם הֲרֵי זֶה מְגַדֵּף", דְּיוֹם זֶה נִתְחַיֵּב בַּאֲמִירָה. כֵּן נִרְאֶה לִי, דְּלֹא כְּהַמָּגֵן אַבְרָהָם שֶׁלֹּא כָּתַב כֵּן. וּמִצְוַת קְרִיאַת הַלֵּל בַּעֲמִידָה, וּסְמִיכָה אֵינָהּ כַּעֲמִידָה:

**אַחַר כָּךְ** אוֹמֵר קַדִּישׁ שָׁלֵם עִם "תִּתְקַבַּל", וּמוֹצִיאִים סֵפֶר תּוֹרָה, וְקוֹרְאִים בּוֹ ד׳ גַּבְרֵי — לֹא פָּחוֹת וְלֹא יוֹתֵר; וְאֵין מַפְטִירִים בַּנָּבִיא. וְקוֹרֵא כֹּהֵן ג׳ פְּסוּקִים: "וַיְדַבֵּר", "צַו", "וְאָמַרְתָּ"; וְלֵוִי חוֹזֵר וְקוֹרֵא "וְאָמַרְתָּ", וְקוֹרֵא "אֶת הַכֶּבֶשׂ אֶחָד" וַ"עֲשִׂירִית הָאֵיפָה"; וְיִשְׂרָאֵל קוֹרֵא "עוֹלַת תָּמִיד" עַד "וּבְרָאשֵׁי חָדְשֵׁיכֶם"; וְד׳ קוֹרֵא "וּבְרָאשֵׁי חָדְשֵׁיכֶם" עַד סוֹף. וּבְשַׁבָּת מוֹצִיאִים ב׳ סִפְרֵי תוֹרוֹת: בָּרִאשׁוֹן קוֹרְאִים ז׳ בְּסֵדֶר הַיּוֹם, וּבַשֵּׁנִי קוֹרֵא מַפְטִיר "וּבְיוֹם הַשַּׁבָּת" וּ"בְרָאשֵׁי חָדְשֵׁיכֶם" עַד סוֹף; וּמַפְטִיר "הַשָּׁמַיִם כִּסְאִי" — אִם לֹא כְּשֶׁחָל בְּאַרְבַּע פָּרָשִׁיּוֹת, דְּאָז מַפְטִירִין בְּשֶׁל פָּרָשָׁה. וּבְרֹאשׁ חֹדֶשׁ אָב וֶאֱלוּל — מַחֲלֹקֶת אִם דּוֹחֶה הַפְטָרַת "שִׁמְעוּ" וַ"עֲנִיָּה סֹעֲרָה" לְ"הַשָּׁמַיִם כִּסְאִי"; וְהַמִּנְהָג לְהַפְטִיר "הַשָּׁמַיִם כִּסְאִי". וְאֵינוֹ חוֹתֵם בַּבְּרָכָה בְּשֶׁל רֹאשׁ חֹדֶשׁ; וּבָאֶמְצַע — מַחֲלֹקֶת אִם לְהַזְכִּיר, וְהַמִּנְהָג שֶׁלֹּא לְהַזְכִּיר. וּבְשַׁבָּת שֶׁמָּחֳרָתוֹ רֹאשׁ חֹדֶשׁ — מַפְטִיר "וַיֹּאמֶר לוֹ יְהוֹנָתָן מָחָר חֹדֶשׁ"; וְאֵינוֹ דּוֹחֶה לַ"עֲנִיָּה סֹעֲרָה", וְאֵין צָרִיךְ לוֹמַר לְד׳ פָּרָשִׁיּוֹת. וְאִם רֹאשׁ חֹדֶשׁ בְּשַׁבָּת וּבְיוֹם א׳ — מִנְהַג הַסְּפָרַדִּים לוֹמַר קֹדֶם "הַשָּׁמַיִם כִּסְאִי" וְאַחֲרָיו פָּסוּק רִאשׁוֹן וְאַחֲרוֹן מֵהַפְטָרַת "מָחָר חֹדֶשׁ"; וּבִמְדִינוֹתֵינוּ לֹא נָהֲגוּ אֶלָּא "הַשָּׁמַיִם כִּסְאִי", שֶׁאֵין מַפְסִיק מִנָּבִיא לְנָבִיא, אִם לֹא בְּהַפְטָרָה בְּאוֹתוֹ נָבִיא — כְּגוֹן בַּחֲתֻנָּה, שֶׁאוֹמֵר פָּסוּק רִאשׁוֹן וְאַחֲרוֹן מֵהַפְטָרַת "שׂוֹשׂ אָשִׂישׂ" אַחַר "הַשָּׁמַיִם כִּסְאִי":

**אַחַר כָּךְ** אוֹמֵר חֲצִי קַדִּישׁ, "אַשְׁרֵי" וּ"בָא לְצִיּוֹן גּוֹאֵל", וְאֵין אוֹמְרִים "לַמְנַצֵּחַ" (וּבְשַׁבָּת — אֵין מַזְכִּירִים נְשָׁמוֹת, וְאֵין אוֹמְרִים "אַב הָרַחֲמִים"). וּמַחֲזִיר סֵפֶר תּוֹרָה לִמְקוֹמוֹ, וְאוֹמֵר חֲצִי קַדִּישׁ, וְחוֹלֵץ תְּפִלִּין — וְאִם אֵין שָׁהוּת, יְזִיזֵם מִמְּקוֹמָם; וְעוֹמְדִים לְהִתְפַּלֵּל מוּסָף, וְאוֹמְרִים "וּמִנְחָתָם וְנִסְכֵּיהֶם" — דְּלֹא כְּרַשַׁ״ל, שֶׁמְּדַלֵּג "וְנִסְכֵּיהֶם". וּבְנֻסַּח "זִכָּרוֹן לְכֻלָּם יִהְיוּ" — יֵשׁ אוֹמְרִים הָכֵי, וְיֵשׁ אוֹמְרִים "לְכֻלָּנוּ"; וְאֵין לְשַׁנּוֹת שׁוּם מִנְהָג. וְנָהֲגוּ לוֹמַר "אֱלֹקֵינוּ וֵאלֹקֵי אֲבוֹתֵינוּ, חַדֵּשׁ" וְכוּלֵי, וְאוֹמֵר י״ב לְשׁוֹנוֹת, כְּנֶגֶד י״ב חֳדָשִׁים; וּבִשְׁנַת עִבּוּר — עַד אַחַר חֹדֶשׁ הָעִבּוּר — מוֹסִיף לוֹמַר "וּלְכַפָּרַת פָּשַׁע", נֶגֶד הַחֹדֶשׁ הַי״ג. וְחוֹתֵם: "בָּרוּךְ אַתָּה ה׳ מְקַדֵּשׁ יִשְׂרָאֵל וְרָאשֵׁי חֳדָשִׁים"; וּבְשַׁבָּת — כּוֹלֵל בַּבְּרָכָה דְּשַׁבָּת וְרֹאשׁ חֹדֶשׁ, וְחוֹתֵם "מְקַדֵּשׁ הַשַּׁבָּת וְיִשְׂרָאֵל וְרָאשֵׁי חֳדָשִׁים". וְאִם שָׁכַח "רֹאשׁ חֹדֶשׁ" — כֵּיוָן שֶׁהִזְכִּיר "יִשְׂרָאֵל", שַׁפִּיר דָּמֵי; וְאִם לֹא הִזְכִּיר נַמִּי "יִשְׂרָאֵל" — אִם הִזְכִּיר בָּאֶמְצַע "וּמוּסְפֵי רֹאשׁ חֹדֶשׁ"; וְאִם לֹא הִזְכִּיר — נִרְאֶה דְּחוֹזֵר לָרֹאשׁ. וְאִם שָׁכַח וְהִתְפַּלֵּל "יוֹצֵר" בִּמְקוֹם מוּסָף — אַף עַל פִּי שֶׁהִזְכִּיר "יַעֲלֶה וְיָבוֹא" בָּעֲבוֹדָה — לֹא יָצָא. וְכֵן בְּהִתְפַּלֵּל
