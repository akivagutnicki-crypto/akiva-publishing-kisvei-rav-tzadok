**וְכָתַב** בִּכְנֶסֶת הַגְּדוֹלָה לָחוּשׁ לְדֵעָה זוֹ, וּלְבָרֵךְ עַל חַמָּה בִּתְקוּפָתָהּ בְּלִי שֵׁם וּמַלְכוּת; וְכָל הָאַחֲרוֹנִים הִשִּׂיגוּ עָלָיו, וְעָשׂוּ מַעֲשֶׂה לְבָרֵךְ בְּשֵׁם וּמַלְכוּת. וְלַעֲנִיּוּת דַּעְתִּי — צָרִיךְ עִיּוּן בְּלָאו הָכֵי בָּזֶה: דְּהָא דִּבְכ״ח שָׁנָה חוֹזֶרֶת חַמָּה לִתְקוּפָתָהּ שֶׁהָיְתָה בִּתְחִלַּת הַבְּרִיאָה, בְּלֵיל ד׳ — זֶהוּ לִתְקוּפַת שְׁמוּאֵל, כְּמוֹ שֶׁנִּתְבָּאֵר (לְמַעְלָה פֶּרֶק ב׳); אֲבָל לִתְקוּפַת רַב אַדָּא נִתְבָּאֵר שָׁם שֶׁאֵין כָּאן מַחֲזוֹר גָּדוֹל שֶׁל כ״ח שָׁנָה כְּלָל, אֶלָּא מַחֲזוֹר שֶׁל י״ט שָׁנָה, שֶׁבְּכָל תְּחִלַּת מַחֲזוֹר הַתְּקוּפָה קוֹדֶמֶת לְמוֹלְדָהּ ט׳ שָׁעוֹת וְתרמ״ב חֲלָקִים; וּלְעוֹלָם לֹא תַּחְזוֹר הַתְּקוּפָה לִמְקוֹמָהּ שֶׁהָיְתָה בִּתְחִלַּת הַבְּרִיאָה, אֶלָּא כְּשֶׁיַּחְזְרוּ מוֹלָדוֹת לִמְקוֹמָן, וּבְאוֹתָם חֳדָשִׁים (כִּי יִתָּכֵן שֶׁיַּחְזְרוּ לִמְקוֹמָם, אֲבָל לֹא לְאוֹתָם הַחֳדָשִׁים) — שֶׁיָּשׁוּב מוֹלַד בהר״ד לְחֹדֶשׁ תִּשְׁרֵי; אָז תְּקוּפַת נִיסָן שֶׁל שָׁנָה הַהוּא יָשׁוּב לִמְקוֹמוֹ שֶׁהָיָה בְּעֵת הַבְּרִיאָה. וְזֶה לֹא יִהְיֶה אֶלָּא אַחַר כַּמָּה וְכַמָּה אֲלָפִים שָׁנָה, שֶׁאֵין יְמֵי עֲמִידַת עוֹלָם הַזֶּה מַסְפִּיקִים. וְצָרִיךְ לוֹמַר דְּהַהִיא בָּרַיְתָא אַתְיָא כִּשְׁמוּאֵל. וּלְפִי זֶה, קָשֶׁה לְמַאי דְּפָסְקוּ הָרַמְבַּ״ם וּשְׁאָר פּוֹסְקִים — זָכַרְנוּ (לְעֵיל פֶּרֶק ב׳) — דְּכִתְקוּפַת רַב אַדָּא הָעִקָּר; אִם כֵּן, מַאי טַעְמָא פָּסְקוּ הָכָא לְבָרֵךְ? וַהֲרֵי לְרַב אַדָּא הֲוֵי בְּרָכָה לְבַטָּלָה! וְשׁוּב מָצָאתִי בְּמַשְׂאַת בִּנְיָמִין (סִימָן ק״א) הִתְעוֹרֵר בָּזֶה; וְעַיֵּן שָׁם מַה שֶּׁכָּתַב, דְּלָאו קֻשְׁיָא הוּא, דִּלְעִנְיַן הַבְּרָכָה סָמְכִינַן אַדִּשְׁמוּאֵל. וַאֲנָא בַּעֲנִיּוּתִי לֹא יָדַעְתִּי מַדּוּעַ סָמְכִינַן אַדִּשְׁמוּאֵל, וְלוֹמַר בְּרָכָה לְבַטָּלָה — דִּלְדַעַת פּוֹסְקִים הוּא אִסּוּר דְּאוֹרַיְתָא, לָאו דְּ"לֹא תִשָּׂא", שֶׁעָלָיו נִזְדַּעְזַע וְכוּלֵי:

**וְלָכֵן**, לַעֲנִיּוּת דַּעְתִּי, עָדִיף טְפֵי לְבָרֵךְ בְּלִי שֵׁם וּמַלְכוּת; וְנִרְאֶה לִי דִּיהַרְהֵר בְּלִבּוֹ הַשֵּׁם וּמַלְכוּת, וְשַׁפִּיר דָּמֵי. וּבָרוּךְ ה׳, בְּיָמֵינוּ — בִּתְקוּפַת נִיסָן שְׁנַת תר״א, בְּיוֹם ז׳ שֶׁל פֶּסַח, שֶׁהוּא תְּחִלַּת מַחֲזוֹר גָּדוֹל — לֹא הֱבִיאָנוּ יִתְבָּרַךְ שְׁמוֹ לִידֵי חֲשַׁשׁ בְּרָכָה לְבַטָּלָה; וְהָיָה יוֹם מְעֻנָּן עַד אַחַר חֲצוֹת, וְאָז בֵּרַכְנוּ בְּלִי שֵׁם וּמַלְכוּת, דְּהוּא אַלִּיבָּא דְּכֻלֵּי עָלְמָא. וְעַיֵּן עוֹד שָׁם בְּמַשְׂאַת בִּנְיָמִין, מַה שֶּׁכָּתַב בְּטַעַם דְּאַתְּקוּפַת רַב אַדָּא אֵין לְבָרֵךְ; וְאֵין דְּבָרָיו מוּבָנִים לִי, דִּתְחִלַּת הַתְּקוּפָה — בַּמֶּה הַחַמָּה חוֹזֶרֶת לִנְקֻדָּתָהּ? אִם רוֹצֶה לוֹמַר בְּגַלְגַּל הַמַּזָּלוֹת — הָא זֶהוּ בְּכָל שָׁנָה; וְאִם רוֹצֶה לוֹמַר בְּאוֹפַן הַמִּישׁוֹר — הָא זֶה לֹא יִהְיֶה אֶלָּא לְאַחַר שֶׁיַּקִּיף גַּלְגַּל הַמַּזָּלוֹת כֻּלּוֹ, וְזֶהוּ בְּכ״ד אֶלֶף שָׁנָה. וַהֲרֵי אֵין הַמַּחֲזוֹר גָּדוֹל שֶׁל כ״ח שָׁנָה אֶלָּא שֶׁאָז חוֹזֶרֶת הַחַמָּה לְאוֹתוֹ מָקוֹם בְּגַלְגַּל הַמַּזָּלוֹת — שֶׁהוּא בְּכָל שָׁנָה — בְּאוֹתוֹ יוֹם שֶׁהָיָה בִּתְחִלַּת הַבְּרִיאָה; וְזֶה אֵינוֹ כְּלָל לִתְקוּפַת רַב אַדָּא, בְּמַחֲזוֹר שֶׁל י״ט שָׁנָה, אֶלָּא שֶׁאָז (מִשְׁתַּוִּים שְׁנוֹת הַלְּבָנָה וְהַחַמָּה; וְזֶהוּ גַּם לִשְׁמוּאֵל, בְּקֵרוּב) הַתְּקוּפָה רְחוֹקָה מֵהַמּוֹלָד ט׳ תרמ״ב, לֹא זוּלַת. וְלֹא הֵבַנְתִּי עִנְיַן חֲזָרַת עַל נְקֻדָּתָן שֶׁזָּכַר הַמַּשְׂאַת בִּנְיָמִין שָׁם, יְעַיֵּן שָׁם הֵיטֵב:
