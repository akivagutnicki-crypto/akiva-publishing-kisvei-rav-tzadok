~ בְּעַצְמוֹ. וְעַל כָּל פָּנִים, לֹא יַקְדִּים קֹדֶם הָנֵץ הַחַמָּה, אֶלָּא כְּשֶׁיִּרְאֶה הַשֶּׁמֶשׁ כֻּלָּהּ לְעֵינָיו. וְאִם לֹא בֵּרַךְ עַד ג׳ שָׁעוֹת עַל הַיּוֹם — דְּהַיְנוּ עַד רֶבַע הַיּוֹם — שׁוּב לֹא יְבָרֵךְ (הַמַּעֲדַנֵּי יוֹם טוֹב כָּתַב עַל זֶה: "וְלֹא יָדַעְתִּי מְנָא לֵיהּ". וְנִרְאֶה לִי, דִּלְשׁוֹן "בַּבֹּקֶר" שֶׁכָּתְבוּ תַּלְמִידֵי רַבֵּינוּ יוֹנָה וְהַפּוֹסְקִים — דַּיֵּיק לֵיהּ; דִּלְשׁוֹן "בַּבֹּקֶר" מַשְׁמַע בְּכַמָּה מְקוֹמוֹת דְּהַיְנוּ עַד ג׳ שָׁעוֹת עַל הַיּוֹם — עַיֵּן בְּפֶרֶק קַמָּא דִּבְרָכוֹת וּשְׁאָר מְקוֹמוֹת) בְּשֵׁם וּמַלְכוּת. וְיֵשׁ אוֹמְרִים, דְּעַד חֲצִי הַיּוֹם מְבָרֵךְ בְּשֵׁם; וְאַחַר חֲצִי הַיּוֹם — אִם הֵזִיד, אוֹ שָׁכַח, אוֹ שֶׁכִּסּוּהָ עָבִים, וְלֹא בֵּרַךְ — שׁוּב לֹא יְבָרֵךְ, אֶלָּא "בָּרוּךְ עוֹשֶׂה" וְכוּלֵי, בְּלֹא שֵׁם וּמַלְכוּת; וְכֵן עִקָּר. וּפָשׁוּט, דְּאַף בְּחָל בְּיוֹם טוֹב — יְבָרֵךְ, דְּלֹא דָּמֵי לְקִדּוּשׁ לְבָנָה. וְכֵן נַמִּי, אֵין צָרִיךְ לְבָרְכָהּ מְעֻמָּד דַּוְקָא; וּמְבָרֵךְ בְּבֵיתוֹ, כָּל שֶׁרוֹאֶה דֶּרֶךְ חַלּוֹן וּפֶתַח:

**וּמִסְתַּבְּרָא**, דְּכָל שֶׁרוֹאֶה גּוּף הַחַמָּה — אַף עַל פִּי שֶׁרוֹאֶה אוֹתָהּ דֶּרֶךְ עֲנָנִים הַסּוֹבְבִים עָלֶיהָ, אִם אֵינָם עֲנָנִים עָבִים שֶׁמַּאֲפִילִים וּמוֹנְעִים רְאִיָּתָהּ, אֶלָּא נִרְאֵית דֶּרֶךְ הָעֲנָנִים, אַף עַל פִּי שֶׁאֵינֶנָּה מַבְהֶקֶת כָּל כָּךְ — מִכָּל מָקוֹם מְבָרֵךְ; שֶׁהֲרֵי בְּלָאו הָכֵי נַמִּי מַאֲהִילִין עַל גּוּף הַחַמָּה ב׳ גַּלְגַּלִּים שֶׁתַּחְתֶּיהָ, וְדֶרֶךְ אוֹתָם הַגַּלְגַּלִּים הִיא נִרְאָה; וּבְוַדַּאי, אִלּוּ לֹא הָיוּ הֵם מָסָךְ הַמַּבְדִּיל — הָיְתָה מַבְהֶקֶת עוֹד יוֹתֵר; אֶלָּא דְּכֵיוָן שֶׁנִּרְאָה גּוּף הַחַמָּה — שַׁפִּיר דָּמֵי, וּמְבָרְכִינַן. וְאֶפְשָׁר דְּאֵין צָרִיךְ בָּזֶה שֶׁיִּהְיֶה נִכָּר זְרִיחָתָהּ עַל גַּבֵּי קַרְקַע, לִהְיוֹת לוֹ צֵל; וְצָרִיךְ לְעַיֵּן בָּזֶה:

**וְיֵשׁ** חוֹלְקִים עַל זֶה, וְסוֹבְרִים דְּאֵין צָרִיךְ לְבָרֵךְ עַל זֶה כְּלָל; וּפֵרְשׁוּ "הָרוֹאֶה חַמָּה בִּתְקוּפָתָהּ" דִּמְבָרֵךְ — הַיְנוּ כְּשֶׁהָיוּ ג׳ יָמִים מְעֻנָּנִים בִּימוֹת הַגְּשָׁמִים, וְלֹא נִרְאוּ בָּהֶם חַמָּה וּלְבָנָה, כּוֹכָבִים וּמַזָּלוֹת — בְּעֵת שֶׁיִּתְרָאוּ, צָרִיךְ לְבָרֵךְ עֲלֵיהֶם "עוֹשֶׂה בְרֵאשִׁית" (וּבַלְּבוּשׁ הֵבִיא גַּם כֵּן דַּעַת יֵשׁ אוֹמְרִים אֵלּוּ. וּלְפִי עֲנִיּוּת דַּעְתִּי, נִרְאֶה דְּהֶעָרוּךְ לֹא פֵּרֵשׁ כֵּן אֶלָּא דִּבְרֵי הַגְּמָרָא; אֲבָל לַהֲלָכָה, הָאִידָּנָא אֵין צָרִיךְ לְבָרֵךְ גַּם עַל זֶה — לְמַאי דְּמַסִּיק רַפְרָם בַּר פַּפָּא, דְּמִיּוֹם שֶׁחָרַב בֵּית הַמִּקְדָּשׁ לֹא נִרְאֵית רָקִיעַ בְּטָהֳרָתָהּ; אִם כֵּן, תָּמִיד הִיא מְעֻנָּן, וְאֵין צָרִיךְ לְבָרֵךְ עַכְשָׁיו. וְעַיֵּן בְּאֵלִיָּה רַבָּה (ק״ד), שֶׁכָּתַב דְּהַבַּהֲ״ג וְהָרַאֲבַ״ן וְהָאֲגֻדָּה פָּסְקוּ כְּרִיבַ״ל, דְּלֹא כְּרַפְרָם בַּר פַּפָּא. מִיהוּ, לְדִידַן דְּקַיְמָא לָן כְּרַפְרָם בַּר פַּפָּא — הָכֵי נַמִּי בְּהָא אֵין צָרִיךְ לְבָרֵךְ. וְלָכֵן נִרְאֶה לִי, דְּאֵין לְדַקְדֵּק אַחַר זֶה; וְאִם נִזְדַּמֵּן שֶׁיָּדַע שֶׁג׳ יָמִים לֹא נִרְאוּ חַמָּה וּלְבָנָה, כּוֹכָבִים וּמַזָּלוֹת, וְאַחַר כָּךְ נִרְאוּ — יְבָרֵךְ בְּלֹא שֵׁם וּמַלְכוּת, דְּהוּא סְפֵקָא דִּפְלֻגְתָּא, כַּנִּזְכָּר לְעֵיל). אֲבָל זוּלַת זֶה — אֵין צָרִיךְ לְבָרֵךְ:
