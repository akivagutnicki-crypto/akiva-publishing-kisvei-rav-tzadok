~ (ה) כַּנִּזְכָּר לְעֵיל, וּשְׁלֵמָה — וְנִיסָן וּפֶסַח בְּג׳; הַסִּימָן: הש״ג. (ו) בְּחוּלוֹ בְּז׳, וְשָׁנָה חֲסֵרָה — וְנִיסָן וּפֶסַח אָז גַּם כֵּן בְּג׳; הַסִּימָן: זח״ג. (ז) כַּנִּזְכָּר לְעֵיל, וּשְׁלֵמָה — וְנִיסָן וּפֶסַח בְּה׳; הַסִּימָן: זש״ה. כָּל אֵלּוּ י״ד סִדְרֵי קְבִיעוּת אֶפְשָׁרִים לְהִמָּצֵא. וְדַע, כִּי כָּל הַדְּבָרִים שֶׁנִּתְבָּאֲרוּ עַד הֵנָּה אֵינָם אֶלָּא לְפִי חֶשְׁבּוֹן זֶה, לֹא בְּעֵת שֶׁהָיוּ קוֹבְעִים עַל פִּי הָרְאִיָּה:

**יוֹם** שְׁלֹשִׁים עוֹשִׂים אוֹתוֹ לְעוֹלָם רֹאשׁ חֹדֶשׁ, לְפִי חֶשְׁבּוֹן זֶה — מִפְּנֵי שֶׁהוּא לְרֹב הַפְּעָמִים יוֹם הַמּוֹלָד. וְאִם הָיָה הַחֹדֶשׁ שֶׁעָבַר חָסֵר — יִהְיֶה יוֹם ל׳ רֹאשׁ חֹדֶשׁ הַבָּא; וְאִם מָלֵא — יִהְיֶה יוֹם ל׳ מִכְּלַל הַחֹדֶשׁ שֶׁעָבַר, לְתַשְׁלוּם מִלּוּאוֹ (וְכוֹתְבִים בּוֹ בְּיוֹם רֹאשׁ חֹדֶשׁ פְּלוֹנִי, שֶׁהוּא ל׳ לְחֹדֶשׁ פְּלוֹנִי שֶׁעָבַר), וְיִהְיֶה יוֹם ל״א רֹאשׁ חֹדֶשׁ הַבָּא (וְכוֹתְבִים בּוֹ בְּרֹאשׁ חֹדֶשׁ פְּלוֹנִי), וּמִמֶּנּוּ הוּא הַמִּנְיָן, וְהוּא יוֹם הַקְּבִיעָה:

(**וּלְפִי** עֲנִיּוּת דַּעְתִּי, הַטַּעַם לְב׳ יָמִים הוּא מִשּׁוּם סָפֵק, וְכִדְאִיתָא בְּרֹאשׁ הַשָּׁנָה (כ״ג עַמּוּד א׳) — מִשּׁוּם בִּטּוּל מְלָאכָה לָעָם ב׳ יָמִים, עַיֵּן שָׁם בַּתּוֹסָפוֹת, דִּבּוּר הַמַּתְחִיל "מִשּׁוּם" — דְּקָאֵי אַכָּל רֹאשׁ חֹדֶשׁ. וְאַף עַל גַּב דְּהוּא רַק מִדְּרַבָּנָן — צָרִיךְ לוֹמַר: כֵּיוָן דְּגוּף רֹאשׁ חֹדֶשׁ דְּאוֹרַיְתָא, כְּמוֹ שֶׁאִיתָא (שָׁם י״ט עַמּוּד א׳), עַיֵּן שָׁם; וְעַיֵּן מַה שֶּׁכָּתַבְתִּי לְעֵיל, פֶּרֶק א׳, בַּחֲנֻכָּה וּפוּרִים. וְהָא דִּבְחָסֵר לֹא עָבְדִינַן ב׳ יָמִים — הַיְנוּ מִשּׁוּם דְּאִם כֵּן לֹא נֵדַע אֵימָתַי רֹאשׁ חֹדֶשׁ חָסֵר, וְאֵימָתַי מָלֵא, כֵּיוָן דְּתָמִיד ב׳ יָמִים. (וְאָז, כְּשֶׁקִּדְּשׁוּ עַל פִּי הָרְאִיָּה — אֶפְשָׁר בֶּאֱמֶת תָּמִיד עָשׂוּ ב׳ יָמִים כְּשֶׁלֹּא יָדְעוּ; מַה שֶּׁאֵין כֵּן עַכְשָׁיו, בָּזֶה לֹא הֶחֱזִיקוּ בְּמִנְהַג אֲבוֹתֵיהֶם.) וּבָחֲרוּ בְּמָלֵא לַעֲשׂוֹתוֹ ב׳ יָמִים, דְּכֵן הָיָה כְּשֶׁהִשִּׂיאוּ מַשּׂוּאוֹת, כִּדְאִיתָא בְּרֹאשׁ הַשָּׁנָה (כ״ג עַמּוּד א׳), עַיֵּן שָׁם בַּתּוֹסָפוֹת, דִּבּוּר הַמַּתְחִיל "וְלֵיעַבֵּיד". וְעוֹד, לְפִי דַּעַת הָרַמְבַּ״ם — דִּבְכָל חֹדֶשׁ לֹא קִבְּלוּ עֵדוּת מִן הַמִּנְחָה וּלְמַעְלָה, מִשּׁוּם קָרְבַּן מוּסָף; אִי נַמִּי לְאַחַר תַּקָּנַת רַבִּי יוֹחָנָן בֶּן זַכַּאי נַמִּי — מִכָּל מָקוֹם, דָּמֵי כָּל רֹאשׁ חֹדֶשׁ לְרֹאשׁ הַשָּׁנָה, דְּגַם בִּירוּשָׁלַיִם עַצְמָהּ עָבְדוּ פְּעָמִים ב׳ יָמִים; וּלְכָךְ תִּקְּנוּ בּוֹ ב׳ יָמִים, וְלֹא בְּחָסֵר; וּלְכָךְ גַּם בְּאֶרֶץ יִשְׂרָאֵל עוֹשִׂין ב׳ יָמִים. וְשָׁדִינַן הַסָּפֵק מִלְּפָנָיו וְלֹא לְאַחֲרָיו, כְּבְכָל סְפֵיקָא — עַיֵּן בְּרֹאשׁ הַשָּׁנָה (כ׳ עַמּוּד ב׳): דְּכָאן הוּא הַסָּפֵק לְפָנָיו; מַה שֶּׁאֵין כֵּן שָׁם, דֶּאֱלוּל וַאֲדָר הַסָּמוּךְ לְנִיסָן עָבְדֵי חָסֵר — הַסָּפֵק לְאַחֲרָיו, עַיֵּן שָׁם):

**וּלְפִיכָךְ** עוֹשִׂין בְּחֶשְׁבּוֹן זֶה רֹאשׁ חֹדֶשׁ תָּמִיד לְעוֹלָם יוֹם אֶחָד אִם חֹדֶשׁ שֶׁלְּפָנָיו חָסֵר, וּב׳ יָמִים אִם שֶׁלְּפָנָיו מָלֵא. וְנִמְצָא: אֵין בֵּין יוֹם רֹאשׁ חֹדֶשׁ אֶחָד לַחֲבֵירוֹ אֶלָּא יוֹם אֶחָד בִּימֵי הַשָּׁבוּעַ בִּלְבַד לְעוֹלָם — שֶׁבֵּין בְּמָלֵא, בֵּין
