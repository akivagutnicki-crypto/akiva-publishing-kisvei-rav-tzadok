~ לַצִּיר הַדְּרוֹמִי (וְקַו הַקָּצָר שֶׁבִּנְקֻדַּת א׳ — הוּא צִיר הַדְּרוֹמִי, וְשֶׁבִּנְקֻדַּת ב׳ — הַצְּפוֹנִי). וְזֶה עִגּוּל מְדֻמֶּה מִצָּפוֹן לְדָרוֹם, וְאֵינוֹ גַּלְגַּל הַסּוֹבֵב, שֶׁהוּא מִמִּזְרָח לְמַעֲרָב; אֲבָל עִגּוּל הָרָקִיעַ נְצַיְּרֵהוּ לְצָרְכֵּנוּ. (וְעַל עִגּוּל כָּזֶה, הַמַּתְחִיל מֵאֹפֶק הַדְּרוֹמִי, וְעוֹלֶה מִמַּעַל לְרֹאשׁוֹ, וְחוֹזֵר וְיוֹרֵד לָאֹפֶק הַצְּפוֹנִי, וּמִמֶּנּוּ מִתַּחַת לַכַּדּוּר, וְתַחַת רַגְלָיו מַמָּשׁ, עַד שֶׁחוֹזֵר לָאֹפֶק הַדְּרוֹמִי, וּפוֹגֵעַ בְּב׳ נְקֻדּוֹת בְּקַו הַשָּׁוֶה, לְמַעְלָה וּלְמַטָּה — הוּא הַנִּקְרָא אֵצֶל הַתּוֹכְנִים "אֹפֶק חֲצִי הַיּוֹם", כִּי בְּהֶמְשֵׁךְ אוֹפַן זֶה יִהְיֶה תָּמִיד חֲצִי הַיּוֹם לָאָדָם הַהוּא; וְהוּא מוּבָן בְּצִיּוּרֵנוּ). וְקַו ג׳ ד׳ — הוּא גַּלְגַּל הַמִּישׁוֹר; וּלְפִי שֶׁאִי אֶפְשָׁר לְצַיֵּר עַל שֶׁטַח הַנְּיָר עִגּוּל תּוֹךְ הָעִגּוּל הַפְּכִּי — הִצַּגְנוּ תְּמוּרָתוֹ קַו, וּמוֹרֶה עַל עִגּוּל הֶפְכּוֹ מִמִּזְרָח לְמַעֲרָב, וְקָבוּעַ מַמָּשׁ בְּאֶמְצַע עִגּוּל זֶה — מִמֶּנּוּ לְצַד דְּרוֹמִי כְּמִמֶּנּוּ לִצְפוֹנִי. וְקַו ה׳ ו׳ — הוּא גַּלְגַּל הַמַּזָּלוֹת וְהַחַמָּה, וְאֵינֶנּוּ מֻקָּף מִגַּלְגַּל הַיּוֹמִי, אֲבָל נוֹטֶה מִמֶּנּוּ לְצָפוֹן וּלְדָרוֹם, וּפוֹגֵשׁ בּוֹ בִּנְקֻדָּה אַחַת לְמַעְלָה — וְהִיא נְקֻדַּת ח׳ — וְכֵן בִּנְקֻדָּה א׳ לְמַטָּה. וְהָעִגּוּלִים הַקְּטַנִּים שֶׁבְּתוֹכוֹ — הִיא הַחַמָּה בְּתַהֲלוּכוֹתֶיהָ:

**וְהִנֵּה** בַּתְּחִלָּה, כְּשֶׁהַחַמָּה תְּחִלַּת מַעֲרָבִית צְפוֹנִית בִּנְקֻדַּת ו׳ — וְגַלְגַּל הַיּוֹמִי יַנְחֵהוּ מִשָּׁם, וִיסַבְּבוֹ מִמִּזְרָח לְמַעֲרָב דֶּרֶךְ מִתַּחַת לַכַּדּוּר, עַד שֶׁיַּגִּיעַ לְאוֹתוֹ קַו לְמַעְלָה בִּקְצֵה הַמִּזְרָח, וִיסַבְּבוֹ עַד שֶׁיָּשׁוּב לִנְקֻדַּת ח׳ הַגַּלְגַּל הַחַמָּה; וְשָׁם הַחַמָּה כְּבָר עָלְתָה בַּיּוֹם הַהוּא בְּמִקְצָת; וְיָשׁוּב לְגַלְגְּלָהּ דֶּרֶךְ קַו ל׳ מִתַּחַת לַכַּדּוּר, וְיָבוֹא לְמַעְלָה לְקַו כ׳; וְכֵן יָשׁוּב לְגַלְגְּלָהּ, וְעָלְתָה הַחַמָּה עוֹד; וְכֵן יָשׁוּב לְסַבְּבָהּ דֶּרֶךְ קַוֵּי כ׳ י׳ ט׳, עַד שֶׁיַּגִּיעַ לְמַזַּל תְּאוֹמִים — וְהִיא מְקוֹם הַפְּגִישָׁה בְּקַו הַשָּׁוֶה — וְשָׁם יִהְיֶה עִגּוּל סְבִיבוֹ דֶּרֶךְ קַו הַשָּׁוֶה. וּמִשָּׁם יָשׁוּב הַחַמָּה לַעֲלוֹת מֵעַל קַו הַשָּׁוֶה, וִיהִי סִבּוּבֵי גַּלְגַּל הַיּוֹמִי לוֹ דֶּרֶךְ קַוֵּי פ׳ ע׳ ס׳ נ׳ מ׳, וְיַעֲשֶׂה בּוֹ עִגּוּלִים שְׁלֵמִים. וְאַחַר כָּךְ יָשׁוּב הַחַמָּה לַאֲחוֹרֵי הַכַּדּוּר, וְיָשׁוּב הַגַּלְגַּל לְסַבֵּב עִמָּהּ דֶּרֶךְ קַוֵּי מ׳ נ׳ ס׳ ע׳ פ׳, וְאַחַר כָּךְ דֶּרֶךְ קַו ג׳ ד׳, שֶׁהוּא קַו הַשָּׁוֶה, וְאָז הַחַמָּה בְּמַזַּל טָלֶה; וְאַחַר כָּךְ יִסּוֹג דֶּרֶךְ קַוֵּי ט׳ י׳ כ׳ ל׳, עַד שֶׁיָּשׁוּב לִמְקוֹמוֹ. וְהַקַּוִּים — הֵן הֵן הָעִגּוּלִים וּמְסִבָּה הַיּוֹמִית שֶׁזָּכַרְנוּ. וְהֶחָפֵץ יוּכַל לִרְאוֹת הַדָּבָר בַּמּוּחָשׁ, בַּעֲשׂוֹתוֹ כְּתַבְנִית הַנִּזְכָּר בַּחֲשׁוּקִים עֲגֻלִּים:

**וּמִזֶּה** נַשְׂכִּיל לְהָבִין מַה שֶּׁאֵין קֶצֶב הַיָּמִים שָׁוֶה; אֲבָל יֵשׁ מְדִינוֹת שֶׁאֶצְלָם תָּמִיד הַיּוֹם וְהַלַּיְלָה שָׁוֶה; וּבִמְדִינוֹת יִתְחַלְּפוּ בָּהֶם — פְּעָמִים הַיּוֹם גָּדוֹל, וּפְעָמִים הַלַּיְלָה, וּפְעָמִים שְׁנֵיהֶם שָׁוִים, וְכֵן הוּא מְדִינוֹתֵינוּ, וְכֵן רֹב
