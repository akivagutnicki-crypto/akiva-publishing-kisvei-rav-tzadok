~ בְּחָסֵר, אֵין מֵרֹאשׁ חֹדֶשׁ לַחֲבֵירוֹ אֶלָּא כ״ט יוֹם. וְנִמְצְאוּ יְמֵי רֹאשׁ חֹדֶשׁ מִתְגַּלְגְּלִים עַל יְמֵי הַשָּׁבוּעַ, וְחוֹזְרִים חֲלִילָה. כֵּיצַד? הֲרֵי שֶׁהָיָה רֹאשׁ חֹדֶשׁ נִיסָן בְּיוֹם א׳ — וְנִיסָן לְעוֹלָם מָלֵא — רֹאשׁ חֹדֶשׁ אִיָּר בְּיוֹם ב׳ וּבְיוֹם ג׳, וְסִיוָן בְּד׳, וְתַמּוּז בְּה׳ וּבְו׳, וְאָב בְּו׳, וֶאֱלוּל בְּא׳ וּבְב׳, וְתִשְׁרֵי בְּג׳, וְחֶשְׁוָן בְּד׳ וּבְה׳; וְכֵן כִּסְלֵו — אִם חֶשְׁוָן מָלֵא, הוּא ב׳ יָמִים, וְאִם חָסֵר — יוֹם אֶחָד; וְכֵן טֵבֵת; וְכֻלָּם חוֹזְרִים חֲלִילָה:

**וְנִמְשָׁךְ** עוֹד, כִּי בִּהְיוֹת "לֹא אד״ו רֹאשׁ" חֹדֶשׁ תִּשְׁרֵי — הִנֵּה לֹא יִהְיֶה: אֱלוּל (יוֹם ב׳ דְּרֹאשׁ חֹדֶשׁ) — זג״ה; וְלֹא אָב — הא״ג; וְלֹא תַּמּוּז — דו״ב; סִיוָן — בה״ז; אִיָּר — אד״ו; נִיסָן — וב״ד; אֲדָר — הא״ג (וַאֲדָר רִאשׁוֹן בִּשְׁנַת עִבּוּר, שֶׁהוּא תָּמִיד מָלֵא — אֵינוֹ חָל בִּימֵי גו״א); שְׁבָט — גו״א (וּבִשְׁנַת עִבּוּר — אד״ו); טֵבֵת — בה״ז (וּבְעִבּוּר — זג״ה); כִּסְלֵו — אִם חָסֵר, אד״ו, וְאִם מָלֵא — לֹא זג״ה (וּבְעִבּוּר — לֹא וב״ד אִם חָסֵר, וְאִם מָלֵא — לֹא הא״ג); חֶשְׁוָן — לֹא גו״א, מִטַּעַם רֹאשׁ הַשָּׁנָה הָעָבָר. (וְנִמְשָׁךְ מִזֶּה עוֹד פְּרָטִים לִשְׁאָר הֶחֳדָשִׁים, אֲשֶׁר עַד כֹּה לֹא דִּבַּרְנוּ אֶלָּא מֵרֹאשׁ הַשָּׁנָה שֶׁלְּאַחֲרֵיהֶם; וְלֹא רָצִיתִי לְהַאֲרִיךְ, כִּי מוּבָן בְּקַל לְכָל מְעַיֵּן.) וּמִטַּעַם רֹאשׁ הַשָּׁנָה דִּלְהַבָּא — הַדָּבָר תָּלוּי בְּסִדְרֵי הַשָּׁנָה, חֲסֵרָה אוֹ מְלֵאָה אוֹ כְּסִדְרָן, וְאִם שְׁנַת פְּשׁוּטָה וּמְעֻבֶּרֶת; וּמוּבָן בְּקַל:

**אַרְבַּע** עִתִּים וּתְקוּפוֹת בַּשָּׁנָה, וְהֵם: קֹר וָחֹם, קַיִץ וָחֹרֶף, אוֹ זֶרַע — אָבִיב, קָצִיר — אָסִיף. וְחַג הַסֻּכּוֹת בִּזְמַן הָאָסִיף, כְּמוֹ שֶׁנֶּאֱמַר: "חַג הָאָסִיף"; וּפֶסַח בִּזְמַן הָאָבִיב. וְשִׁנּוּיֵי הָעִתִּים וְהַתְּקוּפוֹת שֶׁאָמַרְנוּ — הוּא מִסִּבּוֹת תַּהֲלוּכוֹת הַחַמָּה בָּרָקִיעַ. וּכְבָר אָמַרְנוּ: שִׁימוֹת שְׁנַת הַחַמָּה, לְדַעַת שְׁמוּאֵל — שס״ה יוֹם וּרְבִיעַ; אִם כֵּן, בְּהֶחְלֵקָה לְד׳ חֲלָקִים שָׁוִים לְאַרְבַּעַת הַתְּקוּפוֹת — יַגִּיעַ צ״א יוֹם וְז׳ שָׁעוֹת וּמֶחֱצָה לְכָל תְּקוּפָה. וּשְׁמוֹת כָּל תְּקוּפָה — לְפִי הֶחֳדָשִׁים: עֵת הָאָבִיב נִקְרֵאת "תְּקוּפַת נִיסָן", לְפִי שֶׁתְּקוּפָה זוֹ חָלָה תָּמִיד, אוֹ עַל פִּי רֹב, בְּחֹדֶשׁ נִיסָן; וְשֶׁלְּאַחֲרֶיהָ — "תְּקוּפַת תַּמּוּז"; וְאַחֲרֶיהָ — "תְּקוּפַת תִּשְׁרֵי", וְהִיא הָאָסִיף; וְאַחֲרֶיהָ — "תְּקוּפַת טֵבֵת". וְהִנֵּה, הַצ״א יוֹם כְּלוּלִים בְּי״ג שָׁבוּעוֹת שְׁלֵמוֹת; וְנִמְצָא: אֵין בֵּין תְּקוּפָה לַחֲבֶרְתָּהּ בִּימֵי הַשָּׁבוּעַ אֶלָּא ז׳ שָׁעוֹת וּמֶחֱצָה:

**וְלִהְיוֹת** לַחֳדָשִׁים וְלַתְּקוּפוֹת הַתְחָלַת הַשָּׁנָה מִנִּיסָן — כְּמוֹ שֶׁנֶּאֱמַר: "רִאשׁוֹן הוּא לָכֶם לְחָדְשֵׁי הַשָּׁנָה" — הִנֵּה רָאוּי לָנוּ לָדַעַת תְּקוּפַת נִיסָן רִאשׁוֹנָה שֶׁל בְּרִיאַת הָעוֹלָם אֵימָתַי הָיְתָה. וּכְבָר בֵּאַרְנוּ שֶׁאָנוּ מְחַשְּׁבִים
