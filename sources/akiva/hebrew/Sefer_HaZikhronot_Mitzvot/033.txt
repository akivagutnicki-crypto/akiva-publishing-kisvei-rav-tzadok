~ נִקּוּרוֹ, וַעֲדַיִן לֹא הִשְׁלִים נִקּוּרוֹ, וְאָסוּר. וְאַף עַל גַּב דְּמִסְּתָמָא לֹא חָיְישִׁינַן לָזֶה, כְּמוֹ שֶׁנִּתְבָּאֵר — כָּאן, דְּהָלַךְ לוֹ תֵּכֶף, יֵשׁ לָחוּשׁ לָזֶה (וְנִסְתַּלֵּק קֻשְׁיַת "הַכְּרֵתִי וּפְלֵתִי"). וַאֲפִלּוּ הֵעִיד עָלָיו עֵד אֶחָד שֶׁהִשְׁלִים נִקּוּרוֹ — אֵינוֹ נֶאֱמָן, דַּהֲרֵי כָּאן אִתְחֲזַק אִסּוּרָא, וְאֵינוֹ בְּיָדוֹ, כֵּיוָן שֶׁאֵין כָּאן מְנַקֵּר אַחֵר; וַאֲפִלּוּ בַּעַל הַיָּרֵךְ עַצְמוֹ — נִרְאֶה דְּאֵינוֹ נֶאֱמָן. וְיֵשׁ חוֹלְקִים וְאוֹמְרִים דְּאֵין זוֹ אִתְחֲזַק אִסּוּרָא, דְּהַיָּרֵךְ לֹא נֶחְזַק בְּאִסּוּר כְּלָל, אֶלָּא הַגִּיד הוּא אָסוּר, וְהוּא מְחֻבָּר לַיָּרֵךְ; וּלְפִיכָךְ נֶאֱמָן עֵד אֶחָד בָּזֶה, אַף עַל פִּי שֶׁאֵינוֹ בְּיָדוֹ לְתַקֵּן; וְהָעִקָּר כְּדֵעָה רִאשׁוֹנָה. מִיהוּ עַכְשָׁיו, דְּאֵין הַמְנַקְּרִים בְּנֵי תּוֹרָה כָּל כָּךְ, וּמַחְמִירִין וּמְקִילִין בְּחָמוּר — רָאוּי שֶׁלֹּא לֶאֱכוֹל בָּשָׂר אֶלָּא מִמְּנַקֵּר יָדוּעַ לוֹ לְבָקִי גָּדוֹל וּמֻמְחֶה, וְאָדָם כָּשֵׁר. וְהַמַּהַרְשַׁ״ל מֵעוֹלָם לֹא אָכַל בָּשָׂר מְנֻקָּר אֶלָּא אִם כֵּן הֶרְאוּ לִמְנַקֵּר אַחֵר שֶׁיִּבְדּוֹק אִם מְנֻקָּר. וְהַגָּאוֹן הרר״י לֹא אָכַל בְּשַׂר אֲחוֹרַיִם וְיָרֵךְ אֶלָּא אִם כֵּן נִקְּרוֹ בְּעַצְמוֹ. וְרָאוּי לְכָל מְדַקְדֵּק בְּמַעֲשָׂיו לִזָּהֵר בָּזֶה. וְאִם נִמְצָא מִטַּבָּח אֶחָד שֶׁנִּיקֵּר שֶׁלֹּא נִיקֵּר יָפֶה — אִם מַה שֶּׁמְּצָאוֹ הוּא מִן הַקְּנוּקְנוֹת, אוֹ מֵהַחִיצוֹן, אוֹ שֶׁלֹּא גָּמַם וְחִטֵּט לַפְּנִימִי לְמַעְלָה וּלְמַטָּה מִכַּף, וְכָל שֶׁכֵּן מִשּׁוּמָנוֹ — כֵּיוָן דְּלֹא עָבַר עַל אִסּוּר תּוֹרָה, אֵין מַעֲבִירִין אוֹתוֹ (עַיֵּן לְעֵיל סוֹף סִימָן ס״ד, וּבְהָהֵ״מ); אֲבָל מְלַמְּדִין אוֹתוֹ, וּמַזְהִירִין אוֹתוֹ שֶׁלֹּא יְזַלְזֵל בְּאִסּוּרִין. מִיהוּ אִם אַחַר הַלִּמּוּד וְהָאַזְהָרָה נִמְצָא שׁוּב — מַעֲבִירִין אוֹתוֹ. אֲבָל אִם נִמְצָא מֵעִקַּר הַגִּיד הָאָסוּר מִן הַתּוֹרָה — מַעֲבִירִין אוֹתוֹ. וְאִם נִמְצָא מִמֶּנּוּ כְּזַיִת בְּמָקוֹם אֶחָד (דְּאַף בְּחָלָב פָּלִיג הָרַשְׁבָּ״א, וְהָכָא יֵשׁ לוֹמַר כּוּלֵּי עָלְמָא מוֹדוּ, כֵּיוָן דְּאֵין לוֹקִין בִּכְזַיִת בִּכְדֵי אֲכִילַת פְּרָס, כְּמוֹ שֶׁנִּתְבָּאֵר לְעֵיל), אוֹ הַגִּיד דְּמִן הַתּוֹרָה שָׁלֵם — הֲרֵי זֶה לוֹקֶה מַכַּת מַרְדּוּת, וּמַעֲבִירִין אוֹתוֹ. וַחֲזָרָתוֹ לְכַשְׁרוּתוֹ וּתְשׁוּבָתוֹ — הַכֹּל לְפִי רְאוּת עֵינֵי הַדַּיָּן, אִם עָשָׂה בְּשׁוֹגֵג אוֹ בְּמֵזִיד, וּלְפִי מַה שֶּׁהוּא אָדָם. וּבְמָקוֹם אַחֵר יִתְבָּאֵר עוֹד מִזֶּה:

**נִתְעָרֵב** גִּיד הַנָּשֶׁה בַּאֲחֵרִים — אֲפִלּוּ בְּאֶלֶף אֵינוֹ בָּטֵל, מִפְּנֵי שֶׁגִּיד הַנָּשֶׁה בְּרִיָּה הוּא, וּבְרִיָּה אֵינָהּ בְּטֵלָה. וְעִקָּרוֹ שֶׁל גִּיד הַנָּשֶׁה אֵינוֹ אֶלָּא הַפְּנִימִי, וְשֶׁעַל הַכַּף בִּלְבַד, כְּשִׁעוּר ד׳ אֶצְבָּעוֹת בְּשׁוֹר וּב׳ אֶצְבָּעוֹת בְּכֶבֶשׂ, וּפָחוֹת מֵרֹחַב אֲגוּדָל בִּגְדִי וְטָלֶה; אַף שֶׁנֶּחְתַּךְ כָּל הַשְּׁאָר, וְזֶה נִשְׁאַר שָׁלֵם — הֲוֵי בְּרִיָּה. מִיהוּ לְדַעַת פּוֹסְקִים דְּכָל גִּיד הַפְּנִימִי דְּאוֹרַיְתָא — צָרִיךְ כָּל גִּיד פְּנִימִי; אֲבָל כְּבָר נִתְבָּאֵר דְּלָא קַיְימָא לַן כֵּן. בַּמֶּה דְּבָרִים אֲמוּרִים — כְּשֶׁאֵינוֹ מַכִּירוֹ, לַגִּיד לִיטְּלוֹ מִשָּׁם; אֲבָל אִם מַכִּירוֹ וּנְטָלוֹ — אֲפִלּוּ נִתְעָרֵב רַק אֶחָד בְּאֶחָד, אֵין אוֹסֵר תַּעֲרוֹבְתּוֹ בִּפְלִיטָתוֹ, לְפִי שֶׁהַגִּיד הוּא כְּעֵץ שֶׁאֵין בּוֹ טַעַם. וַאֲפִלּוּ אֵין מַכִּירוֹ, וְהַכֹּל אָסוּר — מִכָּל מָקוֹם הָרוֹטֶב מֻתָּר, לְפִי שֶׁאֵין הַגִּיד אוֹסֵר בְּנוֹתֵן טַעַם; וְרַק גִּיד עַצְמוֹ אֵין
