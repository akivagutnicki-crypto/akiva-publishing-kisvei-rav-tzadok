~ י״ב חָדְשֵׁי הַלְּבָנָה — הֵם שנ״ד יָמִים, ח׳ שָׁעוֹת, תתע״ו חֲלָקִים, כְּמוֹ שֶׁיִּתְבָּאֵר. וְאִם כֵּן יְתֵירָה שְׁנַת הַחַמָּה עַל שֶׁל לְבָנָה קָרוֹב לְי״א יוֹם. לְפִיכָךְ, כְּשֶׁיִּתְקַבֵּץ מִן הַתּוֹסֶפֶת הַזֶּה כְּמוֹ ל׳ יוֹם, פָּחוֹת מְעַט אוֹ יֶתֶר מְעַט — מוֹסִיפִין חֹדֶשׁ אֶחָד, וְעוֹשִׂין אוֹתָהּ הַשָּׁנָה י״ג חֹדֶשׁ, וְהִיא נִקְרֵאת "שָׁנָה מְעֻבֶּרֶת"; שֶׁאִי אֶפְשָׁר לִהְיוֹת הַשָּׁנָה י״ב חֹדֶשׁ וְכָךְ וְכָךְ יָמִים, שֶׁנֶּאֱמַר: "לְחָדְשֵׁי הַשָּׁנָה" — חֳדָשִׁים אַתָּה מוֹנֶה לַשָּׁנָה, לֹא יָמִים:

**וְהַלְּבָנָה** — יָדוּעַ מֵעִנְיָנֶיהָ הֱיוֹתָהּ תָּמִיד מִתְחַסֶּרֶת וְחוֹזֶרֶת וּמִתְמַלֵּאת; וְזֶהוּ הַנִּקְרָא "חֹדֶשׁ" — כְּשֶׁהַלְּבָנָה חוֹזֶרֶת וּמִתְחַדֶּשֶׁת. וְהִיא נִסְתֶּרֶת בְּכָל חֹדֶשׁ בְּסוֹף חֶסְרוֹנָהּ קֹדֶם שֶׁתַּחֲזוֹר וְתִתְמַלֵּא, וְאֵינָהּ נִרְאֵית כְּלָל כְּמוֹ ב׳ יָמִים, אוֹ פָּחוֹת אוֹ יֶתֶר מְעַט: כְּמוֹ יוֹם אֶחָד קֹדֶם שֶׁתִּדָּבֵק בַּשֶּׁמֶשׁ בְּסוֹף הַחֹדֶשׁ, וּכְמוֹ יוֹם אֶחָד אַחַר שֶׁתִּדָּבֵק בַּשֶּׁמֶשׁ וְתֵרָאֶה בַּמַּעֲרָב בָּעֶרֶב (וְעַיֵּן בְּרֹאשׁ הַשָּׁנָה כ׳ עַמּוּד ב׳, דִּלְפָחוֹת מִכ״ד שָׁעָה אִי אֶפְשָׁר שֶׁלֹּא תִּתְכַּסֶּה וְתִסָּתֵר). (וְעַיֵּן בַּתַּשְׁבֵּ״ץ חֵלֶק א׳ סִימָן קכ״ה, וְחֵלֶק ג׳ סִימָן רס״ו, יִשּׁוּב סוּגְיָא זוֹ לְהָרַמְבַּ״ם, עַיֵּן שָׁם; וְעַיֵּן בְּדִבְרֵי הַמְּפָרֵשׁ לְקַמָּן רפ״ז, וּבְסֵפֶר יְשׁוּעָה בְּיִשְׂרָאֵל שָׁם בְּאֹרֶךְ וּבְרֹחַב מִזֶּה.) כִּי שִׁשָּׁה שָׁעוֹת נִתְכַּסֵּית הַיְשָׁנָה, וְלֹא תֵּרָאֶה שִׁשָּׁה שָׁעוֹת הַחֲדָשָׁה. וּכְשֶׁעוֹלָה הַחֲדָשָׁה — אַחַר שִׁשָּׁה שָׁעוֹת עוֹלָה בְּקֶרֶן דְּרוֹמִית מַעֲרָבִית, וְרוֹאִין אוֹתָהּ בְּנֵי מַעֲרָב; אֲבָל בְּנֵי מִזְרָח לֹא יִרְאוּהָ אֶלָּא אַחַר י״ב שָׁעוֹת, שֶׁתָּבוֹא לַמִּזְרָח. וְכֵן כְּשֶׁמִּתְכַּסָּה הַיְשָׁנָה, וְתִכָּסֶה בְּקֶרֶן דְּרוֹמִית מִזְרָחִית — וְיִרְאוּהָ בְּנֵי מִזְרָח קֹדֶם כִּסּוּיָהּ, אֲבָל בְּנֵי מַעֲרָב לֹא יִרְאוּהָ אֶלָּא י״ב שָׁעוֹת קֹדֶם הַכִּסּוּי שֶׁנִּרְאָה לִבְנֵי מִזְרָח. וְנִמְצָא לְכָל שִׁעוּר כ״ד שָׁעָה, עַל כָּל פָּנִים, נִתְכַּסֵּית לִבְנֵי מַעֲרָב — י״ח שְׁעוֹת הַיְשָׁנָה וְשִׁשָּׁה שְׁעוֹת הַחֲדָשָׁה; וְלִבְנֵי מִזְרָח — י״ח שְׁעוֹת הַחֲדָשָׁה וְשִׁשָּׁה שְׁעוֹת הַיְשָׁנָה, כַּנִּזְכָּר לְעֵיל. וְעַיֵּן שָׁם בְּרַשִׁ״י, פֵּרֵשׁ הַטַּעַם לְקָטְנוּ הַלְּבָנָה; אֲבָל דַּעַת הָרַמְבַּ״ם — מִשּׁוּם דְּבִיקָה בַּשֶּׁמֶשׁ, נְכוֹנִים, וְנִיחָא לִדְבָרָיו דִּבְרֵי רַבּוֹתָיו דְּרַשִׁ״י (כ״ה רֵישׁ עַמּוּד א׳, דִּבּוּר הַמַּתְחִיל "עֵדֵי שֶׁקֶר"), עַיֵּן שָׁם. וְעַיֵּן (כ״ג עַמּוּד ב׳) רַשִׁ״י דִּבּוּר הַמַּתְחִיל "כֵּיצַד"; וְקֻשְׁיַת הַתּוֹסָפוֹת שָׁם (כ״ד עַמּוּד ב׳, דִּבּוּר הַמַּתְחִיל "רְאִינוּהוּ") אֵינָהּ קֻשְׁיָא לְפִי עֲנִיּוּת דַּעְתִּי, אִם אָנַן קַיְימָא לַן כְּרַבִּי יוֹחָנָן בֶּן נוּרִי, וְלֹא כְּרַבָּן גַּמְלִיאֵל. וְדִבְרֵי הַתּוֹסָפוֹת שָׁם בְּתֵרוּצָם — צָרִיךְ עִיּוּן לַעֲנִיּוּת דַּעְתִּי, דְּמַשְׁמַע דְּמַיְירֵי מִשְּׁעוֹת הַשָּׁווֹת, שֶׁהֵם כ״ד בְּמֵעֵת לְעֵת, לֹא מְזַמַּנִּיּוֹת, עַיֵּן שָׁם. מִיהוּ לְהָרַמְבַּ״ם שָׁם, בְּפֵרוּשׁ הַמִּשְׁנָיוֹת, דֶּרֶךְ אַחֵר — בְּפֵרוּשׁ הַמִּשְׁנָיוֹת שָׁם (עַיֵּן שָׁם). וּבְלֵיל שֶׁתֵּרָאֶה בַּמַּעֲרָב בָּעֶרֶב אַחַר שֶׁנִּסְתְּרָה — הוּא תְּחִלַּת הַחֹדֶשׁ, וּמוֹנִים מֵאוֹתוֹ הַיּוֹם כ״ט יוֹם. וְאִם יֵרָאֶה הַיָּרֵחַ לֵיל ל׳ — יִהְיֶה יוֹם ל׳ רֹאשׁ חֹדֶשׁ; וְאִם לֹא יֵרָאֶה — יִהְיֶה יוֹם ל״א רֹאשׁ חֹדֶשׁ, וִיהֵא יוֹם ל׳ מֵחֹדֶשׁ שֶׁעָבַר. וְלָמָּה כֵּן? לְפִי שֶׁאֵין חֹדֶשׁ שֶׁל לְבָנָה, מִיּוֹם שֶׁמִּתְחַדֶּשֶׁת פַּעַם אַחַת עַד שֶׁתַּחֲזוֹר
