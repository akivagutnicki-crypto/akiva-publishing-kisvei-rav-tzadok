**וְלָמַדְנוּ** מֵהַיְרוּשַׁלְמִי, דִּבְנִיסָן וְתִשְׁרֵי מְאַיְּמִין לְקַדֵּשׁ וְלֹא לְעַבֵּר (וְהַיְנוּ מִשּׁוּם דְּרֹב אֱלוּל וַאֲדָר אֵין מְעֻבָּר, כְּמוֹ שֶׁכָּתַבְתִּי לְעֵיל). וְהִנֵּה תַּלְמוּדִין קָאָמַר דִּפְלִיגֵי; וְצָרִיךְ לוֹמַר דְּלֹא מַשְׁמַע לְתַלְמוּדִין לְמֵימַר דְּרַב דִּימִי מִנְּהַרְדְּעָא — דְּהוּא כְּר״ב דִּבִירוּשַׁלְמִי — מַיְירֵי רַק בְּנִיסָן וְתִשְׁרֵי; וְלִדְיָעָה קַמַּיְיתָא, דְּהוּא כְּרָבָא וְרַבִּי יְהוּדָה נְשִׂיאָה — אִם כֵּן וַדַּאי נַמִּי אִי אֶפְשָׁר לוֹמַר דְּהָא דִּמְאַיְּמִין לְקַדְּשׁוֹ — הוּא רַק בְּנִיסָן וְתִשְׁרֵי, כְּמוֹ דְּלֹא מוֹקֵי דִּבְרֵי רַב דִּימִי בְּכַהַאי גַּוְנָא. אֲבָל הָא דִּלְעַבְּרוֹ מְאַיְּמִין וּלְצֹרֶךְ — שַׁפִּיר אִיכָּא לְאוֹקְמֵי רַק בִּשְׁאָר חֳדָשִׁים; וְזֶה אֵין דֹּחַק כְּלָל, וּכְסְבָרַת הַיְרוּשַׁלְמִי הַנִּזְכָּר לְעֵיל, וְכָךְ יִרְאֶה לוֹ לְהָרַמְבַּ״ם. וּבֶאֱמֶת אֵין לָנוּ לַחְפּוֹשׂ הוֹכָחוֹת עַל זֶה בַּתַּלְמוּד, כִּי יָדוּעַ מַה שֶּׁכָּתַב הָרַמְבַּ״ם: "יִרְאֶה לִי" — מִסְתַּבְּרָא לֵיהּ מִדְּנַפְשֵׁיהּ הָכֵי. וְצָרִיךְ לְעַיֵּן: אִם רַק לְעִנְיַן לְאַיֵּם כָּתַב כֵּן, אֲבָל לְצֹרֶךְ מְעַבְּרִין אַף בְּנִיסָן וְתִשְׁרֵי — כְּדְעוּלָּא, דְּעִבְּרוּהָ לֶאֱלוּל לְצֹרֶךְ; אוֹ אַף לְצֹרֶךְ כֵּן, וְכֵן מַשְׁמַע בַּגְּמָרָא דִּלְצֹרֶךְ וּמְאַיְּמִין מַמָּשׁ כְּהָדָדֵי, וְאִם כֵּן וַדַּאי פָּלִיג אַדְּעוּלָּא, וְכִדְאִיתָא כֵּן בִּירוּשַׁלְמִי לְר״ז. וַהֲרֵי גַּם רַב דִּימִי פָּלִיג אַדְּעוּלָּא; וְצָרִיךְ עִיּוּן בָּזֶה. וְעַיֵּן בְּלֶחֶם מִשְׁנֶה מַה שֶּׁכָּתַב בְּדִבְרֵי הָרַמְבַּ״ם, וְיֵשׁ לְהָשִׁיב הַרְבֵּה עַל דְּבָרָיו; וּבְיִחוּד יִשּׁוּב הֲלָכָה י״ט, בְּלְקַדֵּשׁ — דְּלָא נָחֵית כְּלָל לְבָאֲרָהּ, וְאֵין צֹרֶךְ; דְּבְעַצְמוֹ כָּתַב שֶׁדְּבָרָיו דְּחוּקִים, עַיֵּן שָׁם. וּמַה שֶּׁכָּתַבְתִּי נִרְאֶה לִי מְרֻוָּח בְּדִבְרֵי הָרַמְבַּ״ם, בְּסִיַּעְתָּא דִשְׁמַיָּא:

**מִיהוּ** בְּחִדּוּשֵׁי הָרַשְׁבָּ״א, פֶּרֶק "רָאוּהוּ בֵּית דִּין" (כ״ה עַמּוּד ב׳), דִּבּוּר הַמַּתְחִיל "וְאֵימָא הָכֵי נַמִּי", כָּתַב: דְּהֵיכִי דְּאַפְסְקֵיהּ לַיְלָה, שֶׁהוּא זְמַן שֶׁאֵין רָאוּי לְקִדּוּשׁ — שׁוּב אֵין יָכוֹל לְקַדְּשׁוֹ לְאַחַר זְמַן, עַיֵּן שָׁם. וְנִרְאֶה דִּמְפָרֵשׁ "מְאַיְּמִין" וְכוּלֵי כְּפֵרוּשׁ רַשִׁ״י — דְּהַיְנוּ לְצֹרֶךְ; וּבְהֻצְרְכוּ נַמִּי מְאַיְּמִין; וְאִי אִיְּמוּ וְלֹא הוֹעִיל — אֲפִלּוּ לְצֹרֶךְ אֵין מְעַבְּרִין:

**וְרָאִיתִי** בְּסֵפֶר טוּרֵי אֶבֶן (כ׳ עַמּוּד ב׳), דִּבּוּר הַמַּתְחִיל "אָמַר לֵיהּ לֹא שָׁמַעְתִּי", כָּתַב גַּם כֵּן: דִּבְרֵי הָרַמְבַּ״ם תְּמוּהִים, דְּוַדַּאי אֵין קִדּוּשׁ לְמַפְרֵעַ וְכוּלֵי, עַיֵּן שָׁם. גַּם בְּסֵפֶר קָרְבַּן הָעֵדָה, בְּפֵרוּשׁוֹ לַיְרוּשַׁלְמִי (סוֹף פֶּרֶק קַמָּא הֲלָכָה ח׳), דִּבּוּר הַמַּתְחִיל "וְלֹא לְמַפְרֵעַ", כָּתַב: "דְּדָבָר תֵּימַהּ שֶׁאָכְלוּ בְּיוֹם הַכִּפּוּרִים אִם יְקַדְּשׁוּ לְמַפְרֵעַ"; אֲבָל הֵבִיא רְאָיָה גְּדוֹלָה לְהָרַמְבַּ״ם מִסּוּגְיָא דִּירוּשַׁלְמִי שָׁם, עַיֵּן שָׁם. וְהִנֵּה זוֹ אֵין תְּמִיהָה — דְּאָכְלוּ בְּיוֹם הַכִּפּוּרִים; דְּמוּטָב יְקַלְקֵל יוֹם הַכִּפּוּרִים, וְלֹא יְקַלְקְלוּ מוֹעֲדוֹת, כִּדְאָמְרִינַן כַּהַאי גַּוְנָא בַּגְּמָרָא (י״ט עַמּוּד ב׳) גַּבֵּי רֹאשׁ הַשָּׁנָה. גַּם מַה שֶּׁכָּתַב עוֹד — יֵשׁ לְפַקְפֵּק; כְּבָר בֵּרַרְתִּי לְיַשֵּׁב בְּסִיַּעְתָּא דִשְׁמַיָּא. אֲבָל רְאָיָתוֹ הַגְּדוֹלָה מֵהַיְרוּשַׁלְמִי — הִנֵּה הוּא לְפִי פֵּרוּשׁוֹ שָׁם; אֲבָל לְפִי עֲנִיּוּת דַּעְתִּי, פֵּרוּשׁוֹ
