~ אִי נַמִּי קָטָן שֶׁבְּז״ר, עַיֵּן שָׁם; אֲבָל הַאי — בַּר דַּעַת.) אֲבָל בְּטֻמְאַת מֵת וּמְצֹרָע וּשְׁאָר טֻמְאוֹת — שָׁוֶה טֻמְטוּם וְאַנְדְּרוֹגִינוֹס לְכָל אָדָם. "אֵין לִי (סִפְרֵי) אֶלָּא שֶׁיָּכוֹל לְהִשְׁתַּלֵּחַ; שֶׁאֵין יָכוֹל לְהִשְׁתַּלֵּחַ — כְּגוֹן חוֹלֶה וְזָקֵן שֶׁאֵין יְכוֹלִים לֵילֵךְ — וְכֵן הַכֵּלִים, מִנַּיִן שֶׁצָּרִיךְ לְשַׁלְּחָם? תַּלְמוּד לוֹמַר: 'אֶל מִחוּץ לַמַּחֲנֶה תְּשַׁלְּחוּם'" — דַּהֲוָה לֵיהּ לוֹמַר "תְּשַׁלַּח", וּכְתִיב "תְּשַׁלְּחוּם" — וָא״ו וּמֵ״ם יְתֵרִים, לְרַבּוֹת הַכֹּל שֶׁצָּרִיךְ שִׁלּוּחַ. (וְעַיֵּן שָׁם עוֹד לִמּוּדִים; וְעַיֵּן בְּנִדָּה שָׁם לִמּוּד אַחֵר. וְשָׁם אָמַר רַבִּי יְהוּדָה, דְּאֵינוֹ חַיָּב בְּשִׁלּוּחַ בִּכְלִי חֶרֶס, שֶׁאֵין לוֹ טָהֳרָה בְּמִקְוֶה, אֶלָּא בִּכְלֵי מַתָּכוֹת וְשֶׁטֶף, שֶׁיֵּשׁ לָהֶם טָהֳרָה בְּמִקְוֶה. וְעַיֵּן שָׁם בְּתוֹסָפוֹת, וּבַגְּמָרָא דְּשִׁלְהֵי עֵירוּבִין; וּמַשְׁמַע הָתָם בַּגְּמָרָא, מִדְּפָרֵיךְ מִינֵּיהּ — דְּהִלְכְתָא כְּוָתֵיהּ; וּמַשְׁמַע דְּאַף חִיּוּב עֲשֵׂה אֵין כָּאן. וְהָרַמְבַּ״ם לֹא זְכָרוֹ — נִרְאֶה דְּלֹא פָּסַק כֵּן, וְלֹא יָדַעְתִּי טַעְמוֹ):

**וְכֵן** הַגְּדוֹלִים מֻזְהָרִים עַל הַקְּטַנִּים טְמֵאִים לְשַׁלְּחָם מִן הַמִּקְדָּשׁ (ז״ר), שֶׁהֲרֵי לֹא גְּרִיעֵי מִכֵּלִים, דִּמְחֻיָּב לְשַׁלְּחָם; וּלְפִי זֶה פָּשׁוּט, דְּהוּא הַדִּין עַל קָטָן, חֵרֵשׁ וְשׁוֹטֶה — דְּצָרִיךְ לְשַׁלְּחוֹ מֵהַאי טַעְמָא. וְשֶׁרֶץ — יֵשׁ אוֹמְרִים (רַבֵּינוּ חֲנַנְאֵל בְּתוֹסָפוֹת עֵירוּבִין ק״ג עַמּוּד ב׳, דְּגָרֵיס "דְּכֻלֵּי עָלְמָא פָּטוּר") דְּאֵין מֻזְהָר עָלָיו מִן הַתּוֹרָה לְשַׁלְּחוֹ, לְפִי שֶׁאֵין לוֹ טָהֳרָה בְּמִקְוֶה; וְיֵשׁ חוֹלְקִים (רַשִׁ״י וְתוֹסָפוֹת וְהָרַמְבַּ״ם), וְכֵן עִקָּר:

**וְאֵין** הַשִּׁלּוּחַ שָׁוֶה בְּכָל הַטְּמֵאִים. לְפִי שֶׁג׳ פְּעָמִים נֶאֱמַר בְּמִקְרָא זֶה "מַחֲנֶה", וְלָמְדוּ חֲכָמִים (פֶּרֶק ז׳ מֵהִלְכוֹת בֵּית הַבְּחִירָה) שֶׁהוּא נֶגֶד שָׁלֹשׁ מַחֲנוֹת שֶׁהָיוּ בַּמִּדְבָּר: מַחֲנֵה יִשְׂרָאֵל — וְהוּא ד׳ מַחֲנוֹת הַדְּגָלִים; וּמַחֲנֵה לְוִיָּה — שֶׁנֶּאֱמַר בָּהּ "סָבִיב לַמִּשְׁכָּן יַחֲנוּ"; וּמַחֲנֵה שְׁכִינָה — וְהוּא מִפֶּתַח חֲצַר אֹהֶל מוֹעֵד וְלִפְנִים. וְהָיָה כֵּן כְּנֶגְדָּן לְדוֹרוֹת בְּבֵית הַמִּקְדָּשׁ: מִפֶּתַח יְרוּשָׁלַיִם — וְהַפֶּתַח כְּלַחוּץ (פְּסָחִים פ״ה) — עַד הַר הַבַּיִת — כְּמַחֲנֵה יִשְׂרָאֵל; וּמִשֶּׁקִּדֵּשׁ יְהוֹשֻׁעַ אֶת אֶרֶץ יִשְׂרָאֵל — הוֹסִיף לְמַחֲנֵה יִשְׂרָאֵל כָּל הָעֲיָרוֹת הַמֻּקָּפוֹת חוֹמָה, שֶׁיִּהְיוּ כִּירוּשָׁלַיִם; וְאֵיזֶה הֵם עֲיָרוֹת מֻקָּפוֹת חוֹמָה וּפְרָטֵיהֶם — יִתְבָּאֲרוּ בְּדִין בָּתֵּי עָרֵי חוֹמָה. (וְיֵשׁ לְעַיֵּן, מִמַּה שֶּׁכָּתַב הָרַמְבַּ״ם שֶׁאֵין לוֹקֶה — אוּלַי אֵין חִיּוּב עֲשֵׂה גַּם כֵּן עַל כָּל עֲיָרוֹת מֻקָּפוֹת חוֹמָה, אֶלָּא אִסּוּר דְּרַבָּנָן בִּלְבַד.) וּמִפֶּתַח הַר הַבַּיִת — וְהַפֶּתַח כְּלַחוּץ (יְרוּשַׁלְמִי פֶּרֶק ד׳ דִּפְסָחִים, עַיֵּן שָׁם הַטַּעַם) — עַד פֶּתַח הָעֲזָרָה, שֶׁהוּא שַׁעַר נִיקָנוֹר — מַחֲנֵה לְוִיָּה (וְעֹבִי הַשַּׁעַר עַצְמוֹ כְּלַחוּץ — מַתְנִיתִין פֶּרֶק קַמָּא דְּסוֹטָה, עַיֵּן שָׁם בְּרַשִׁ״י, וּבְתוֹסָפוֹת פֶּרֶק קַמָּא דִּיבָמוֹת, וּבִפְסָחִים דַּף פ״ה, וּשְׁאָר דּוּכְתֵּי). וּמִפֶּתַח הָעֲזָרָה וְלִפְנִים — מַחֲנֵה שְׁכִינָה:
