~ בְּמָגֵן אַבְרָהָם, בְּשֵׁם מַגִּיד: דַּוְקָא אִם מְחַדְּשִׁים בְּמוֹצָאֵי שַׁבָּת — יִהְיֶה הַצְלָחָה בְּאוֹתוֹ חֹדֶשׁ). לָכֵן רָאוּי לְכָל אָדָם לְדַקְדֵּק אַחַר זֶה, לְפַנּוֹת מִטְּרָדוֹתָיו לֵילֵךְ לְחַדְּשָׁהּ וּלְהַקְבִּיל פְּנֵי הַשְּׁכִינָה — אִם לֹא שֶׁטָּרוּד בִּדְבַר מִצְוָה, דְּעוֹסֵק בְּמִצְוָה פָּטוּר מִן הַמִּצְוָה; וְכָל שֶׁכֵּן כְּשֶׁעָסוּק בְּהַכְנָסַת אוֹרְחִים, דִּגְדוֹלָה מֵהַקְבָּלַת פְּנֵי הַשְּׁכִינָה, כִּדְאִיתָא בְּרֵישׁ פֶּרֶק "מַפְנִין". וְיִלְבַּשׁ בְּגָדִים נָאִים, כְּהוֹלֵךְ לְהַקְבִּיל פְּנֵי שְׁכִינָה. וְכֵן יֵצֵא לַחוּץ, וְלֹא יְקַדֵּשׁ בְּבֵיתוֹ — מֵהַאי טַעְמָא. וּמִיהוּ, אִם לֹא יוּכַל לָצֵאת לַחוּץ — כְּגוֹן שֶׁחוֹשֵׁשׁ אֵיזֶה מֵחוּשׁ, אוֹ שֶׁדָּר בֵּין הָעוֹבְדֵי כּוֹכָבִים, אוֹ שֶׁהוּא בְּחֶבְרַת שִׂמְחָה, וּכְהַאי גַּוְנָא, שֶׁאֵין יָכוֹל לָצֵאת; וְכָל שֶׁכֵּן כְּשֶׁבַּחוּץ הַמְּבוֹאוֹת מְטֻנָּפוֹת — יְקַדֵּשׁ בְּבֵיתוֹ, דֶּרֶךְ חַלּוֹן פָּתוּחַ אוֹ פֶּתַח. וְאִם הוּא בִּימֵי הַסְּתָיו, שֶׁאֵינוֹ רוֹצֶה לִפְתּוֹחַ הַחַלּוֹן מִפְּנֵי הַקֹּר — רַשַּׁאי לְקַדְּשָׁהּ דֶּרֶךְ חַלּוֹן הַסָּגוּר, כֵּיוָן שֶׁרוֹאֶה אוֹתָהּ דֶּרֶךְ הַזְּכוּכִית; דְּכָל שֶׁהַלְּבָנָה נִרְאֵית בְּעִנְיָן שֶׁיּוּכַל לְהַכִּיר כָּל דָּבָר שֶׁאֶפְשָׁר לְהַכִּיר לְאוֹרָהּ — שַׁפִּיר דָּמֵי, אַף שֶׁרוֹאֶה דֶּרֶךְ מָסָךְ מַבְדִּיל; וַאֲפִלּוּ בַּחוּץ — שַׁפִּיר דָּמֵי (אֶלָּא דִּלְכַתְּחִלָּה אֵין לַעֲשׂוֹת כֵּן). וְלָכֵן רַשַּׁאי לְקַדְּשָׁהּ כְּשֶׁעֲנָנִים דַּקִּים וּקְלוּשִׁים סוֹבְבִים עָלֶיהָ, כֵּיוָן שֶׁאוֹרָהּ נִרְאֵית; וְאַף עַל פִּי שֶׁאִלּוּ לֹא הָיוּ הֶעָבִים — הָיְתָה אוֹרָהּ מַבְהֶקֶת יוֹתֵר; כֵּיוָן דְּמִכָּל מָקוֹם זְרִיחָתָהּ נִכֶּרֶת עַל גַּבֵּי קַרְקַע, וְיֵשׁ לָהּ צֵל. אֲבָל עָבִים עָבִים — אַף עַל פִּי שֶׁגּוּף הַלְּבָנָה נִרְאֵית, אֶלָּא שֶׁאֵין נִרְאֵית זְרִיחָתָהּ — וְכֵן שְׁאָר מָסָךְ מַבְדִּיל כְּהַאי גַּוְנָא — לֹא יְקַדֵּשׁ; אֶלָּא אִם כֵּן כְּבָר הִתְחִיל לְבָרֵךְ קֹדֶם שֶׁיִּהְיֶה הַמָּסָךְ — דְּאָז, כֵּיוָן שֶׁהִתְחִיל, גּוֹמֵר. וּלְכַתְּחִלָּה, אִם רוֹאֶה שֶׁיִּתְכַּסֶּה וְיִהְיֶה מָסָךְ בְּאֶמְצַע בִּרְכָתוֹ — לֹא יַתְחִיל לְבָרֵךְ כְּלָל; אִם לֹא שֶׁכְּבָר הוּא בָּעֵת אַחֲרוֹנָה שֶׁל זְמַן קִדּוּשׁ — דְּאָז נִרְאֶה דִּיקַדֵּשׁ בְּלִי שֵׁם וּמַלְכוּת:

**מֵאֵימָתַי** מְבָרְכִין עַל הַיָּרֵחַ? מִשֶּׁיַּעַבְרוּ עָלָיו ז׳ יָמִים (נִרְאֶה: מֵעֵת לְעֵת, מֵעֵת הַמּוֹלָד). וְיֵשׁ אוֹמְרִים: מִשֶּׁיַּעַבְרוּ עָלָיו ג׳ יָמִים — רַשַּׁאי לְבָרֵךְ. וְיֵשׁ אוֹמְרִים: אֲפִלּוּ בְּיוֹם א׳, אִם נִרְאֶה אוֹרָהּ יָפֶה — רַשַּׁאי. וְעַד אֵימָתַי מְבָרְכִים? יֵשׁ אוֹמְרִים: עַד יוֹם ט״ז מִיּוֹם הַמּוֹלָד — וְלֹא מֵעֵת לְעֵת מֵעֵת הַמּוֹלָד, אֶלָּא מִיּוֹמוֹ — וְלֹא ט״ז בִּכְלָל. וְיֵשׁ אוֹמְרִים: עַד יוֹם ט״ז, מֵעֵת לְעֵת מֵרֶגַע הַמּוֹלָד. וְיֵשׁ אוֹמְרִים, דְּאֵין לְקַדֵּשׁ אֶלָּא עַד חֲצִי כ״ט י״ב תשצ״ג מֵרֶגַע הַמּוֹלָד. וְיֵשׁ בָּזֶה לְהַחְמִיר עַל דֵּעָה אַחֲרוֹנָה, וּלְהָקֵל — וּלְהַחְמִיר — עַל דֵּעָה רִאשׁוֹנָה; וְכֵן עִקָּר:

**מִיהוּ**, אִם הָיָה לִקּוּי לְבָנָה בְּאוֹתוֹ חֹדֶשׁ (נִרְאֶה לִי — דַּוְקָא בְּאוֹתוֹ מָקוֹם וּמְדִינָה), מִזְּמַן הַלִּקּוּי וְאֵילָךְ אֵין מְבָרְכִים עָלֶיהָ, כִּי עֵת הַלִּקּוּי הוּא מַמָּשׁ אֶמְצַע הַחֹדֶשׁ. פֵּירוּשׁ: כִּי סִבַּת הַלִּקּוּי — כִּי הַלְּבָנָה אֵין לָהּ אוֹר מֵעַצְמָהּ, אֶלָּא מַה שֶּׁמְּקַבֶּלֶת מֵאוֹר הַשֶּׁמֶשׁ; וּכְשֶׁהִיא מַמָּשׁ נֶגְדִּית לְהַשֶּׁמֶשׁ —
