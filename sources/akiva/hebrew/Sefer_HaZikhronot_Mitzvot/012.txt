~ יָבוֹא לִידֵי טֻמְאָה בַּלַּיְלָה (עֲבוֹדָה זָרָה כ׳ עַמּוּד ב׳), שֶׁהָאָדָם רוֹאֶה בַּלַּיְלָה מֵעִנְיָנִים שֶׁמְּהַרְהֵר בַּיּוֹם, כִּדְאִיתָא פֶּרֶק בַּתְרָא דִּבְרָכוֹת. וְאָז יֵעָנֵשׁ, כֵּיוָן שֶׁתָּבוֹא הַמַּחֲשָׁבָה לִידֵי מַעֲשֶׂה, וְעָשְׂתָה הַמַּחֲשָׁבָה פֵּרוֹת — נֶעֱנָשׁ גַּם עַל הַמַּחֲשָׁבָה, כִּדְאִיתָא סוֹף פֶּרֶק קַמָּא דְּקִדּוּשִׁין. וְאִם יָבוֹא לִידֵי הִרְהוּר — יַסִּיעַ לִבּוֹ לְדִבְרֵי תּוֹרָה, שֶׁהִיא "אַיֶּלֶת אֲהָבִים וְיַעֲלַת חֵן", וּמַצֶּלֶת מִיֵּצֶר הָרָע:

**לְפִיכָךְ** צָרִיךְ הָאָדָם לִזָּהֵר מִכָּל דָּבָר שֶׁמֵּבִיא לִידֵי הִרְהוּר, וְלֹא יְדַבֵּר דִּבּוּרִים הַמְּבִיאִים לִידֵי הִרְהוּר (מִלְּבַד שֶׁיֵּשׁ בָּזֶה אִסּוּר דְּנִבּוּל פֶּה, שֶׁיִּתְבָּאֵר בִּמְקוֹמוֹ). וְכֵן לֹא יִסְתַּכֵּל בְּאִשָּׁה, אוֹ בְּבִגְדֵי צִבְעוֹנִים שֶׁל אִשָּׁה שֶׁמַּכִּירָהּ, מִפְּנֵי שֶׁמֵּבִיא לְהִרְהוּר (מִלְּבַד שֶׁיֵּשׁ בָּזֶה אִסּוּר שֶׁיָּבוֹא לִזְנוּת, וְיִתְבָּאֵר בִּפְרָטוּת בִּמְקוֹמוֹ). וְאָסוּר לוֹ לְהִסְתַּכֵּל שִׁמּוּשׁ אִישׁ עִם אִשְׁתּוֹ (כְּגוֹן עוֹבְדֵי כּוֹכָבִים שֶׁאֵינָם מַקְפִּידִים). וַאֲפִלּוּ בַּבְּהֵמָה, חַיָּה וָעוֹף כְּשֶׁנִּזְקָקִין זֶה לָזֶה — לֹא יִסְתַּכֵּל בָּהֶם, אֶלָּא לְצַד אַחֵר, אוֹ יַעֲצִים עֵינָיו (כֵּן מַשְׁמָע בְּרַשִׁ״י) תֵּכֶף שֶׁנִּרְאוּ לְנֶגֶד עֵינָיו. (וְאֶפְשָׁר בִּבְרוּאִים הַקְּטַנִּים — זְבוּבִים וְכַיּוֹצֵא בּוֹ — שֶׁאֵין נִרְאֶה תַּשְׁמִישָׁם לָעַיִן לְהֶדְיָא, לֵיכָּא כָּל כָּךְ הִרְהוּר, וְרַשַּׁאי לְהִסְתַּכֵּל. וְנִרְאֶה הַכֹּל לְפִי מַה שֶּׁהוּא יוֹדֵעַ בְּנַפְשׁוֹ אִם יָבוֹא לְהִרְהוּר.) וּמֻתָּר לְמַרְבִּיעֵי בְּהֵמָה לְהַכְנִיס כְּמִכְחוֹל בַּשְּׁפוֹפֶרֶת, מִפְּנֵי שֶׁהֵם עֲסוּקִים בִּמְלַאכְתָּם לֹא יָבוֹאוּ לְהִרְהוּר. וְכֵן אוֹתָם הַלּוֹמְדִים חָכְמַת הַטִּבְעִיּוֹת, וּצְרִיכִים לֵידַע תַּשְׁמִישָׁן שֶׁל כָּל בְּרִיָּה כֵּיצַד — נִרְאֶה נַמִּי דְּשַׁפִּיר דָּמֵי לְהִסְתַּכֵּל, דַּעֲסוּקִים בְּחָכְמָתָם. וְלֹא יִסְתַּכֵּל בְּאִשְׁתּוֹ בִּמְקוֹם הַטִּנּוֹפֶת, שֶׁיָּבוֹא לְהִרְהוּר וּמְגָרֵי יֵצֶר הָרָע אַנַּפְשֵׁיהּ בְּהוֹצָאַת זֶרַע לְבַטָּלָה, אִם לֹא יוּכַל לְשַׁמֵּשׁ עִמָּהּ תֵּכֶף (כְּגוֹן שֶׁיָּבוֹאוּ בְּנֵי אָדָם, אוֹ שֶׁהוּא בַּיּוֹם אוֹ לְאוֹר הַנֵּר, וְכַהַאי גַּוְנָא), וּמִלְּבַד שֶׁיֵּשׁ בָּזֶה עוֹד אִסּוּרִים — יִתְבָּאֲרוּ בִּמְקוֹמָם. וְכֵן אָסוּר לוֹ לֵילֵךְ בַּמֶּרְחָץ, בְּמָקוֹם שֶׁהוֹלְכִים בַּמֶּרְחָץ בְּלֹא מִכְנָסַיִם, עִם אָבִיו וַחֲמִיו וּבַעַל אִמּוֹ וּבַעַל אֲחוֹתוֹ; וַאֲפִלּוּ הוּא קָדַם לָהֶם — צָרִיךְ לָצֵאת; וַאֲפִלּוּ מִפְּנֵי כְּבוֹד אָבִיו, שֶׁאָבִיו צָרִיךְ לוֹ שֶׁיְּשַׁמְּשׁוֹ בַּמֶּרְחָץ — אָסוּר. וּלְפִיכָךְ לֹא יִקְרָא הָאָב בְּנוֹ שֶׁיֵּלֵךְ עִמּוֹ לַמֶּרְחָץ, אִם לֹא שֶׁהוּא קָטָן, שֶׁעֲדַיִן שֶׁלֹּא יְהַרְהֵר. וְצָרִיךְ עִיּוּן עַל מַה סָּמְכוּ עַכְשָׁיו. אוּלַי לְפִי שֶׁבִּזְמַנֵּיהֶם הָיוּ הַמֶּרְחֲצָאוֹת רַק לְהָזִיעַ, וְיוֹשְׁבִים בָּטֵל — יֵשׁ לָחוּשׁ לְהִרְהוּר; אֲבָל עַכְשָׁיו, שֶׁבַּמֶּרְחֲצָאוֹת רוֹחֲצִים וּמְשַׁפְשְׁפִים עַצְמָם בִּכְלֵי לְהָזִיעַ, וְאֵין יוֹשְׁבִים בָּטֵל — אִם כֵּן יֵשׁ לוֹמַר בִּמְלַאכְתָּם טְרוּדִים, וְאֵין שַׁיָּךְ הִרְהוּר. וּמִכָּל מָקוֹם הַדָּבָר קָשֶׁה לְהָקֵל בְּדִינָא דִּגְמָרָא, וּבְשֶׁגַּם נִמְצָאִים כַּמָּה שֶׁבֵּינְתַיִם יוֹשְׁבִים בָּטֵל. אֲבָל בְּעֶרְוַת חֲבֵרוֹ — זוּלַת הָנֵי, וְכָל שֶׁכֵּן בְּעֶרְוַת עַצְמוֹ — רַשַּׁאי לְהִסְתַּכֵּל. וּמִכָּל מָקוֹם אִם יוֹדֵעַ בְּעַצְמוֹ שֶׁיָּבוֹא עַל יְדֵי זֶה לִידֵי הִרְהוּר — יִגְדּוֹר לְעַצְמוֹ שֶׁלֹּא לְהִסְתַּכֵּל, וְכֵן יִגְדּוֹר בְּכָל כַּהַאי גַּוְנָא, כְּפִי שֶׁמַּרְגִּישׁ בּוֹ:
