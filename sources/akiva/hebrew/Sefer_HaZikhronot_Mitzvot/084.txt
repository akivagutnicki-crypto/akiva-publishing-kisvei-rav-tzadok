~ שֶׁהוּא מָלֵא, וְאִי אֶפְשָׁר לְשַׁנּוֹתוֹ; לְפִיכָךְ תִּקְּנוּ בַּחֹדֶשׁ הַמָּלֵא שֶׁאַחֲרָיו — וְהוּא כִּסְלֵו — לַעֲשׂוֹתוֹ חָסֵר כְּשֶׁמִּצְטָרְכִים, וְנִמְצָא שָׁנָה חֲסֵרָה יוֹם אֶחָד. לְפִיכָךְ בָּאָה הַשָּׁנָה הַמְּלֵאָה פְּעָמִים הַרְבֵּה, כְּדֵי לְהַשְׁלִים גַּם יוֹם זֶה הֶחָסֵר:

**וְנִמְצָאנוּ** לְמֵדִים, שֶׁג׳ שָׁנִים יֵשׁ: שָׁנָה חֲסֵרָה — וְהִיא שֶׁל שנ״ג יוֹם, שֶׁהֵם נ׳ שָׁבוּעוֹת וְג׳ יוֹם, וּבָהּ מַרְחֶשְׁוָן וְכִסְלֵו שְׁנֵיהֶם חֲסֵרִים; וּשְׁנַת כְּסִדְרָן — וְהִיא שנ״ד יוֹם, נ׳ שָׁבוּעוֹת וְד׳ יָמִים, וּבָהּ הֶחֳדָשִׁים כֻּלָּם כְּסִדְרָן, אֶחָד חָסֵר וְאֶחָד מָלֵא; וְשָׁנָה מְלֵאָה — וְהִיא שֶׁל שנ״ה יוֹם, וְהוּא נ׳ שָׁבוּעוֹת וְה׳ יָמִים, וּמַרְחֶשְׁוָן וְכִסְלֵו שֶׁלָּהּ שְׁנֵיהֶם מְלֵאִים. וְאֵין הַשָּׁנָה יְתֵירָה עַל שנ״ה יוֹם, וְלֹא פְּחוּתָה מִן שנ״ג יוֹם — וּכְמוֹ שֶׁנִּתְבָּאֵר לְעֵיל בְּמִשְׁפְּטֵי הַדְּחִיּוֹת — שֶׁאֵין עוֹשִׂין בַּשָּׁנָה יוֹתֵר מִז׳ חֳדָשִׁים חֲסֵרִים אוֹ ז׳ מְלֵאִים. וְכֵן אַתָּה דָּן לְשָׁנָה מְעֻבֶּרֶת: שֶׁאֲדָר הָרִאשׁוֹן עוֹשִׂין אוֹתוֹ מָלֵא, וְהַשֵּׁנִי חָסֵר; לְפִי שֶׁאֲדָר הַסָּמוּךְ לְנִיסָן תָּמִיד חָסֵר, וְחֹדֶשׁ הָעִבּוּר — רָצוּ לְמַלְּאוֹת; לְפִיכָךְ מִלְּאוּ לָרִאשׁוֹן. וְנִמְצָא: אִם הַשָּׁנָה כְּסִדְרָן — הוּא שפ״ד יוֹם, שֶׁהֵם נ״ד שָׁבוּעוֹת וְשִׁשָּׁה יָמִים; וְאִם הִיא חֲסֵרָה, וְחֶשְׁוָן וְכִסְלֵו שְׁנֵיהֶם חֲסֵרִים — הִיא שפ״ג יוֹם, שֶׁהֵם נ״ד שָׁבוּעוֹת, ה׳ יוֹם; וְאִם מְלֵאָה, וּשְׁנֵיהֶם מְלֵאִים — הִיא שפ״ה יוֹם, וְהֵם נ״ה שָׁבוּעוֹת שְׁלֵמוֹת:

**וּלְפִי** מַה שֶּׁנִּתְבָּאֵר — דֶּרֶךְ הַיְדִיעָה סִדְרֵי אֵיזוֹ שָׁנָה, אִם הֶחֳדָשִׁים כְּסִדְרָן, אוֹ מְלֵאִים, אוֹ חֲסֵרִים — נָקֵל הוּא: כְּשֶׁנַּחְשׁוֹב רֹאשׁ הַשָּׁנָה הַנִּרְצֵית, בְּאֵיזֶה יוֹם נִקְבָּע, לְפִי סֵדֶר הַקְּבִיעוּת וּמִשְׁפְּטֵי הַדְּחִיּוֹת שֶׁבֵּאַרְנוּ; וְכֵן רֹאשׁ הַשָּׁנָה שֶׁלְּאַחֲרֶיהָ — בְּאֵיזֶה יוֹם הוּא, לְפִי הַקְּבִיעוּת הַנִּזְכָּר; וְנַחְשׁוֹב מִנְיַן הַיָּמִים שֶׁבֵּינְתַיִם, מִתְּחִלַּת יוֹם קְבִיעָה הָרִאשׁוֹן עַד תְּחִלַּת יוֹם קְבִיעָה הָאַחֲרוֹן, בִּימֵי הַשָּׁבוּעַ: אִם נִמְצָא בֵּינֵיהֶם ג׳ יָמִים — הַשָּׁנָה חֲסֵרָה, וְהִיא שנ״ג יוֹם, שֶׁנִּשְׁאַר ג׳ יָמִים יִתְרוֹן הַשָּׁבוּעוֹת; וְאִם בֵּינְתַיִם ד׳ יָמִים — הַשָּׁנָה כְּסִדְרָן; וְאִם ה׳ — הֲרֵי הִיא מְלֵאָה:

**וְכָל** זֶה אִם הִיא פְּשׁוּטָה; אֲבָל אִם הִיא מְעֻבֶּרֶת — נַחְשׁוֹב מִנְיַן הַיָּמִים שֶׁבֵּינְתַיִם: אִם ה׳ יָמִים — הִיא חֲסֵרָה, וְיָמֶיהָ שפ״ג; וְאִם ו׳ — כְּסִדְרָן; וְאִם ז׳ — מְלֵאָה. וְהַדְּבָרִים מְבֹאָרִים עִם מַה שֶּׁקָּדַם. וְיֵשׁ סִימָנִים שֶׁנִּסְמוֹךְ עֲלֵיהֶם, כְּדֵי שֶׁלֹּא נִטְעֶה בְּחֶשְׁבּוֹן סִדְרֵי הַשָּׁנָה; וְהֵן בְּנוּיִים עַל עִקְּרֵי הַקְּבִיעוּת וְהַחֶשְׁבּוֹן שֶׁבֵּאַרְנוּם. וְאֵלּוּ הֵן: כָּל רֹאשׁ הַשָּׁנָה שֶׁיִּהְיֶה רֹאשׁ הַשָּׁנָה שֶׁלָּהּ בְּג׳ — תִּהְיֶה לְעוֹלָם כְּסִדְרָן, בֵּין פְּשׁוּטָה בֵּין מְעֻבֶּרֶת. כִּי אִם הִיא פְּשׁוּטָה — הִנֵּה לֹא
