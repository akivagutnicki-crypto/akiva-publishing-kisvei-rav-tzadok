**וּמִיהוּ**, אַף עַל גַּב דְּהֻקְּשׁוּ לְמוֹעֵד — לֹא לְכָל מִילֵּי; וּמֻתָּר בַּעֲשִׂיַּת מְלָאכָה, אִם לֹא בִּמְלָאכָה כְּבֵדָה, דְּיֵשׁ אוֹסְרִים; וְגַם בָּזֶה אֵין הֲלָכָה כֵּן. וּמִכָּל מָקוֹם, לֹא יוּכַל לִכְפּוֹת מְשָׁרְתָיו לַעֲשׂוֹת לוֹ מְלָאכָה כְּבֵדָה, אִם רְצוֹנָם לְהַחְמִיר עַל עַצְמָם. וְהַנָּשִׁים נָהֲגוּ לְהִמָּנַע מֵעֲשִׂיַּת מְלָאכָה בְּב׳ יָמִים — שֶׁלְּפִי שֶׁלֹּא פֵּרְקוּ נִזְמֵיהֶם לָעֵגֶל, נִתַּן לָהֶם רֹאשׁ חֹדֶשׁ לְיוֹם טוֹב. וּבִקְצָת מְקוֹמוֹת נָהֲגוּ לְהִמָּנַע רַק מִמְּלָאכוֹת לְהִשְׂתַּכֵּר; אֲבָל לִתְפּוֹר וּלְתַקֵּן בְּבֵיתָם — שַׁפִּיר דָּמֵי, דְּמֵעִקָּרָא הָכֵי קַבִּילוּ עֲלַיְיהוּ. וְדַוְקָא שֶׁהִתְנוּ בְּפֵירוּשׁ (אוֹ שֶׁהוּא מִנְהָג פָּשׁוּט כָּךְ, דְּוַדַּאי אַדַּעְתָּא דְּמִנְהָגָא עָבְדָא); אֲבָל בִּסְתָם — אֵין לְהָקֵל לָהֶם כְּלָל. וַאֲפִלּוּ בְּאַתְרָא דִּנְהִיגֵי לְהָקֵל — מִכָּל מָקוֹם אֵין יָכוֹל הַבַּעַל לָכוֹף לַעֲשׂוֹת, אִם תִּרְצֶה לְהַחְמִיר עַל עַצְמָהּ. וְנִרְאֶה פָּשׁוּט, דִּדְבַר הָאָבֵד שַׁפִּיר דָּמֵי, דְּלֹא עָדִיף מֵחוֹל הַמּוֹעֵד; וְכֵן שְׁאָר מִילֵּי דִּשְׁרֵי בְּחוֹל הַמּוֹעֵד — הָכֵי נַמִּי שְׁרֵי וַדַּאי:

**סֵדֶר** תְּפִלַּת רֹאשׁ חֹדֶשׁ: מוֹסִיף בִּתְפִלַּת שְׁמוֹנֶה עֶשְׂרֵה דְּשַׁחֲרִית וְעַרְבִית וּמִנְחָה "יַעֲלֶה וְיָבוֹא" בִּ"רְצֵה". וְאִם שָׁכַח אוֹ הֵזִיד וְלֹא אָמַר בְּעַרְבִית — בֵּין בְּא׳ דְּרֹאשׁ חֹדֶשׁ בֵּין בְּב׳ — אִם נִזְכַּר קֹדֶם שֶׁאָמַר הַשֵּׁם מִבִּרְכַּת "רְצֵה" — חוֹזֵר לִ"רְצֵה"; וְאִם נִזְכַּר אַחַר שֶׁאָמַר הַשֵּׁם — אֲפִלּוּ לֹא הִתְחִיל עֲדַיִן "מוֹדִים" — מִתְפַּלֵּל כְּדַרְכּוֹ, וְאֵינוֹ חוֹזֵר, מִפְּנֵי שֶׁאֵין מְקַדְּשִׁין הַחֹדֶשׁ בַּלַּיְלָה. וּמַשְׁכַּחַתְּ לָהּ שֶׁיַּחְזוֹר אִם שָׁכַח בַּלַּיְלָה — כְּגוֹן שֶׁשָּׁכַח מִנְחָה דְּעֶרֶב רֹאשׁ חֹדֶשׁ, דְּמִתְפַּלֵּל בַּלַּיְלָה ב׳ שֶׁל רֹאשׁ חֹדֶשׁ, וְשָׁכַח בָּרִאשׁוֹנָה — דְּאָז חוֹזֵר, דְּצָרִיךְ לְהַקְדִּים חוֹבַת הַיּוֹם; וְכֵיוָן דְּאָמַר בְּב׳ "יַעֲלֶה וְיָבוֹא" וְלֹא בְּא׳ — הָוְיָא ב׳ חוֹבַת הַיּוֹם. אֲבָל אִם שָׁכַח בִּשְׁנֵיהֶם, אוֹ בַּשְּׁנִיָּה — אֵינוֹ חוֹזֵר:

**וּבְשַׁחֲרִית** וּבְמִנְחָה — אִם נִזְכַּר קֹדֶם שֶׁהִתְחִיל "מוֹדִים" — אוֹמֵר בְּמָקוֹם שֶׁנִּזְכַּר, בְּלֹא חֲתִימָה. וְאִם נִזְכַּר אַחַר שֶׁהִתְחִיל "מוֹדִים", קֹדֶם שֶׁסִּיֵּם תְּפִלָּתוֹ אוֹ תַּחֲנוּנִים שֶׁאַחַר תְּפִלָּתוֹ — חוֹזֵר לִ"רְצֵה". וְאִם נִזְכַּר לְאַחַר שֶׁסִּיֵּם תְּפִלָּתוֹ וְתַחֲנוּנִים שֶׁלְּאַחַר הַתְּפִלָּה — דְּהַיְנוּ שֶׁכְּבָר אָמַר "יִהְיוּ לְרָצוֹן אִמְרֵי פִי" (בַּתְרָאָה, לְאוֹתָן הַנּוֹהֲגִים לְאָמְרוֹ ב׳ פְּעָמִים — אַחַת קֹדֶם "אֱלֹקַי נְצוֹר" וְאַחַת אַחֲרָיו), אַף עַל פִּי שֶׁעֲדַיִן לֹא עָקַר רַגְלָיו וְלֹא אָמַר "עוֹשֶׂה שָׁלוֹם" — חוֹזֵר לְרֹאשׁ הַתְּפִלָּה, אִם מִתְפַּלֵּל בְּיָחִיד; וְאִם בְּצִבּוּר — יַטֶּה אָזְנוֹ וִיכַוֵּן לִתְפִלַּת הַשְּׁלִיחַ צִבּוּר כֻּלָּהּ. וְאִם נִזְכַּר אַחַר שֶׁכְּבָר הִתְפַּלֵּל מוּסָף — אֵין צָרִיךְ לַחְזוֹר לְהִתְפַּלֵּל שַׁחֲרִית, דִּכְבָר הִזְכִּיר שֶׁל רֹאשׁ חֹדֶשׁ בְּמוּסָף. וְלָכֵן, שְׁלִיחַ צִבּוּר — אִם נִזְכַּר אַחַר שֶׁהִשְׁלִים תְּפִלָּתוֹ — אֵין לְהַטְרִיחַ צִבּוּרָא שֶׁיַּחְזוֹר לָרֹאשׁ, אֶלָּא סוֹמֵךְ לְכַתְּחִלָּה; דְּכָל הֵיכִי דְּאִי אֶפְשָׁר — כְּדִיעֲבַד דָּמֵי. וּמַשְׁכַּחַתְּ לָהּ גַּם בְּשַׁחֲרִית שֶׁלֹּא יַחְזוֹר — אִם שָׁכַח לְהִתְפַּלֵּל עַרְבִית, דְּמִתְפַּלֵּל בְּשַׁחֲרִית ב׳, וְשָׁכַח בַּשְּׁנִיָּה — דְּאֵין חוֹזֵר; כֵּן כָּתַב בִּכְנֶסֶת הַגְּדוֹלָה. אֲבָל בְּמָגֵן אַבְרָהָם (סִימָן ק״ח, סְעִיף קָטָן
