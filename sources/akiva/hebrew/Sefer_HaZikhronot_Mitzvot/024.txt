~ הַמִּילָה רוֹחֲצִין, כְּמוֹ שֶׁכָּתוּב, לְפִי שֶׁאֵין הָרְחִיצָה עִקָּרִית כָּל כָּךְ; וְכֵן בַּיּוֹם ג׳ לַמִּילָה, כְּשֶׁרוֹאִים צֹרֶךְ לְרָחֲצוֹ — מְכִינִים חַמִּין מִבְּעוֹד יוֹם וְרוֹחֲצִים בְּשַׁבָּת. וּלְפִיכָךְ אֵין לָחוּשׁ לָמוּל מִילָה שֶׁלֹּא בִּזְמַנָּהּ בַּיּוֹם ה׳ אוֹ ו׳; וְאַף עַל פִּי שֶׁיֵּשׁ אוֹמְרִים שֶׁבִּזְמַנִּים קַדְמוֹנִים, שֶׁהָיוּ נוֹהֲגִים בִּרְחִיצָה יוֹם ג׳, הָיָה אָסוּר דְּיִצְטָרְכוּ לְחַלֵּל שַׁבָּת — הָאִידָּנָא שַׁפִּיר דָּמֵי. אֲבָל יִזָּהֲרוּ שֶׁלֹּא יִשְׁרוּ סְדִין בַּמַּיִם כְּשֶׁרוֹחֲצִים אוֹתוֹ, שֶׁשְּׁרִיַּת בֶּגֶד זֶהוּ כִּבּוּסוֹ; אוֹ יָשִׂים נָכְרִי הַסְּדִין לַמַּיִם, דְּכֵיוָן דְּאֵין מִתְכַּוֵּן לְכַבֵּס — שַׁפִּיר דָּמֵי עַל יְדֵי נָכְרִי. אֲבָל לֹא יֹאמַר לְנָכְרִי, אִם נִשְׁפְּכוּ הַחַמִּין, לְהָחֵם אֲחֵרִים בִּשְׁבִילוֹ. וְכָל זֶה בִּסְתָם; אֲבָל אִם רוֹאִים שֶׁיֵּשׁ לָחוּשׁ לְסַכָּנָה אִם לֹא יִרְחֲצוּ אוֹתוֹ אַחַר הַמִּילָה — וַדַּאי מֻתָּר לְרָחֲצוֹ וּלְחַלֵּל עָלָיו הַשַּׁבָּת. וְאִם רוֹאִים זֶה קֹדֶם הַמִּילָה, שֶׁאִם יָמוּלוּ אוֹתוֹ יִצְטָרְכוּ לְחַלֵּל הַשַּׁבָּת אַחַר כָּךְ — אֵין רַשָּׁאִים לְמוּלוֹ. וְנִרְאֶה דְּתָלוּי בִּפְלֻגְתָּא הַנִּזְכָּר לְעֵיל בַּסַּמָּנִים, עַיֵּן שָׁם. וְכֵן אִם רוֹאִים בְּיֶלֶד זֶה שֶׁיִּצְטָרְכוּ לְרָחֲצוֹ בַּיּוֹם ג׳, וְהוּא שֶׁלֹּא בִּזְמַנּוֹ — אָסוּר לְמוּלוֹ בַּיּוֹם ה׳ אוֹ ו׳, לְדַעַת יֵשׁ אוֹמְרִים, כַּנִּזְכָּר לְעֵיל:

**אָדָם** שֶׁלֹּא מָל, אוֹ לֹא פָּרַע מֵעוֹלָם — לֹא יָמוּל בְּשַׁבָּת, הֵיכִי דְּאֶפְשָׁר בְּאַחֵר, שֶׁמָּא יְקַלְקֵל וְנִמְצָא מְחַלֵּל שַׁבָּת. וְאִם כְּבָר מָל פַּעַם אַחַת — מֻתָּר. וְהֵיכִי דְּלֵיתָא לְאַחֵר לָמוּל — שַׁפִּיר דָּמֵי לָמוּל אַף זֶה שֶׁלֹּא מָל. וַאֲפִלּוּ הָאָב מָל אֶת בְּנוֹ בְּשַׁבָּת — וְאֵינוֹ חָשׁוּב כִּמְתַקֵּן; וּלְדִבְרֵי הָאוֹמֵר דְּאֵין הָאָב רַשַּׁאי לַעֲשׂוֹת שָׁלִיחַ לָמוּל — אָז הוּא עָדִיף, וַאֲפִלּוּ לֹא מָל מֵעוֹלָם רַשַּׁאי לָמוּל, דַּהֲוֵי כְּלֹא אֶפְשָׁר לְמִצְוָתָא דִּידֵיהּ. אֲבָל לְדִידַן, דְּרַשַּׁאי לַעֲשׂוֹת שָׁלִיחַ — אַל יָמוּל מִשּׁוּם מִצְוָה בּוֹ יוֹתֵר מִבִּשְׁלוּחוֹ, בִּלְבַד בַּפַּעַם רִאשׁוֹן בְּשַׁבָּת. וְיֵשׁ אוֹמְרִים שֶׁלֹּא יְמוּלוֹ שְׁנֵי מוֹהֲלִים — אֶחָד חוֹתֵךְ וְאֶחָד פּוֹרֵעַ — בְּשַׁבָּת; וְיֵשׁ חוֹלְקִים, וְכֵן עִקָּר. וּמִכָּל מָקוֹם נָכוֹן לְהַחְמִיר לְכַתְּחִלָּה לְעַצְמוֹ, אֲבָל לַאֲחֵרִים אֵין לִמְחוֹת אַף לְכַתְּחִלָּה. וְאִם יֵשׁ אֶחָד שֶׁכְּבָר מָל פַּעַם אַחַת, וְאֵין כָּאן מִי שֶׁכְּבָר פָּרַע פַּעַם אַחַת כְּלָל — נִרְאֶה לִי דְּיִתְּנוּ גַּם הַפְּרִיעָה לָזֶה שֶׁיּוֹדֵעַ לָמוּל, דְּעַל כָּל פָּנִים אִתְחַזַּק בְּמִקְצָת שֶׁהוּא מִמַּאְדִּים לִהְיוֹת אֻמָּנָא, וְלֹא בָּעִית מִדְּמָא. וַאֲפִלּוּ אֵין שָׁם כְּלָל לֹא מִי שֶׁמָּל פַּעַם אַחַת וְלֹא מִי שֶׁפָּרַע — נִרְאֶה דְּיִתְּנוּ לְאֶחָד הַמִּילָה וְהַפְּרִיעָה, וְלֹא יַחְלְקוּ; דְּזֶה שֶׁמָּל, כֵּיוָן שֶׁמָּל — הֲרֵי הֻחְזַק לַמִּילָה לִהְיוֹת אֻמָּן, וְעָדִיף גַּם כֵּן לִפְרוֹעַ מֵאַחֵר, כְּמוֹ שֶׁנִּתְבָּאֵר:

**שְׁנֵי** תִּינוֹקוֹת הַבָּאִים לִימּוֹל בְּיוֹם אֶחָד — יָמוּל מִקֹּדֶם זֶה שֶׁנּוֹלַד קֹדֶם. וְאִם הָאֶחָד בִּזְמַנּוֹ וְהַשֵּׁנִי שֶׁלֹּא בִּזְמַנּוֹ — יַקְדִּים אוֹתוֹ שֶׁבִּזְמַנּוֹ, אַף דְּזֶה מֻקְדָּם לְהוֹלָדָה. וְאִם הָאֶחָד דְּרַבָּנָן, כְּגוֹן שֶׁנִּמְשְׁכָה עָרְלָתוֹ, אוֹ סָפֵק, כְּגוֹן אַנְדְּרוֹגִינוֹס,
