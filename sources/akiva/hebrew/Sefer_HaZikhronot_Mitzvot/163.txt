~ מִתְחָרֵט מֵעִקָּרוֹ עַל עֲשִׂיַּת אוֹתָהּ מַעֲשִׂים (אִם הֵם עֲבֵרַת הַלָּאוִין; וּבְבִטּוּל מִצְוַת עֲשֵׂה — יִתְחָרֵט עַל מְנִיעַת אוֹתוֹ מַעֲשֶׂה הַטּוֹב). וְזֶהוּ בַּעֲבֵרוֹת שֶׁבֵּין אָדָם לַמָּקוֹם — דַּי עַל הֶעָבָר בַּחֲרָטָה לְבַד. אֲבָל עֲבֵרוֹת שֶׁבֵּין אָדָם לַחֲבֵרוֹ — כְּגוֹן שֶׁגָּזַל אוֹ גָּנַב מֵחֲבֵרוֹ, אוֹ שֶׁחֲבָלוֹ אוֹ קִלְּלוֹ, אוֹ סִפֵּר עָלָיו לָשׁוֹן הָרָע, וְהַדּוֹמֶה — אֵין הַחֲרָטָה מוֹעֶלֶת לוֹ עַל הֶעָבָר עַד שֶׁיְּרַצֶּה אֶת חֲבֵרוֹ, וִיתַקֵּן אֶת אֲשֶׁר עִוֵּת, הוֹאִיל וּבְיָדוֹ לְתַקֵּן; שֶׁזֶּהוּ עִקַּר הַחֲרָטָה — שֶׁאֵינוֹ רוֹצֶה בְּקִיּוּמוֹ שֶׁל דָּבָר; וְאַחֲרֵי שֶׁבְּיָדוֹ לְתַקֵּן שֶׁלֹּא יִהְיֶה הַדָּבָר הַהוּא מְקֻיָּם — צָרִיךְ לְתַקֵּן. וּלְפִיכָךְ, אִם גָּזַל אוֹ גָּנַב — צָרִיךְ לְהַחֲזִיר לְבַעַל הַמָּמוֹן מָמוֹן שֶׁחַיָּב לוֹ, וּכְמוֹ שֶׁיִּתְבָּאֵר בִּמְקוֹמוֹ בְּמִצְוַת הֲשָׁבַת הַגְּזֵלָה. וְאַף עַל פִּי שֶׁהֶחֱזִיר לוֹ מָמוֹנוֹ — צָרִיךְ לְפַיְּסוֹ וּלְבַקֵּשׁ מִמֶּנּוּ שֶׁיִּמְחוֹל לוֹ. וְכֵן בַּחֲבָלָה וּקְלָלָה וְכָל כַּיּוֹצֵא בּוֹ — אֲפִלּוּ לֹא הִקְנִיטוֹ אֶלָּא בִּדְבָרִים (וְצָרִיךְ לְפָרֵט לוֹ הַחֵטְא, בַּמֶּה הִקְנִיטוֹ — אִם לֹא כְּשֶׁמִּתְבַּיֵּשׁ בּוֹ). וְאִם אֵינוֹ מִתְפַּיֵּס בָּרִאשׁוֹנָה — מֵבִיא לוֹ שׁוּרָה שֶׁל בְּנֵי אָדָם מֵרֵעָיו, וּפוֹגְעִין בּוֹ וּמְבַקְשִׁים מִמֶּנּוּ; לֹא נִתְרַצָּה לָהֶם — מֵבִיא לוֹ שְׁנִיָּה וּשְׁלִישִׁית:

יוֹתֵר לֹא נִמְצָא לְעֵת עַתָּה בִּכְתַב יַד קָדְשׁוֹ מִזֶּה.
