~ שֶׁבִּזְמַנֵּנוּ, יְעֻיַּן שָׁם הֵיטֵב; וּבְסֵפֶר מְגִלַּת אֶסְתֵּר (אוֹת ח׳), מַה שֶּׁיִּשֵּׁב דִּבְרֵי הָרַמְבַּ״ם — וּדְבָרָיו תְּמוּהִים בְּוַדַּאי, וְלֹא יָרַד לְסוֹף דַּעַת הָרַמְבַּ״ם וְהָרַמְבַּ״ן. וְאִלּוּ לָזֶה יְכַוֵּן הָרַמְבַּ״ם הֲלָכָה לְמֹשֶׁה מִסִּינַי, דִּצְרִיךְ סַנְהֶדְרִין — כָּל שֶׁכֵּן דְּתִקְשֶׁה לֵיהּ "מְנָא לֵיהּ". וְעוֹד, שֶׁהֲרֵי כָּתוּב (וְכֵן בְּרז״ל, עַיֵּן שָׁם בְּרַמְבַּ״ן וּמְגִלַּת אֶסְתֵּר) רֶמֶז לָזֶה בַּכָּתוּב, דְּ"לָכֶם". וּכְבָר כָּתַב הָרַמְבַּ״ם עַצְמוֹ בַּהַקְדָּמָה לְפֵרוּשׁ הַמִּשְׁנָיוֹת, דַּהֲלָכָה לְמֹשֶׁה מִסִּינַי אֵין לוֹ רֶמֶז בַּקְּרָא, עַיֵּן שָׁם. אֲבָל כַּוָּנָתוֹ לַהֲלָכָה לְמֹשֶׁה מִסִּינַי דְּחֶשְׁבּוֹן, עַיֵּן שָׁם, וּפָשׁוּט:

**וּבִימֵי** חַכְמֵי הַמִּשְׁנָה, וְכֵן בִּימֵי חַכְמֵי הַגְּמָרָא, עַד יְמֵי אַבָּיֵי וְרָבָא — הָיוּ עֲדַיִן סְמוּכִים בְּאֶרֶץ יִשְׂרָאֵל, וְעַל קְבִיעוּת אֶרֶץ יִשְׂרָאֵל, עַל פִּי הָרְאִיָּה, הָיוּ סוֹמְכִין. וּמֵאֵימָתַי הִתְחִילוּ כָּל יִשְׂרָאֵל לַחְשׁוֹב בְּחֶשְׁבּוֹן זֶה? מִסּוֹף חַכְמֵי הַגְּמָרָא, כְּעֵת שֶׁחָרְבָה אֶרֶץ יִשְׂרָאֵל, וְלֹא נִשְׁאַר שָׁם בֵּית דִּין קָבוּעַ — הִתְחִילוּ לַחְשׁוֹב כָּךְ. וְזֶה שֶׁאָנוּ מְחַשְּׁבִין, כָּל אֶחָד וְאֶחָד בְּעִירוֹ, וְאוֹמְרִין שֶׁרֹאשׁ חֹדֶשׁ יוֹם פְּלוֹנִי, וְיוֹם טוֹב יוֹם פְּלוֹנִי — לֹא בְּחֶשְׁבּוֹן שֶׁלָּנוּ אָנוּ קוֹבְעִים, וְלֹא עָלָיו אָנוּ סוֹמְכִין; שֶׁאֵין מְעַבְּרִין שָׁנִים וְקוֹבְעִין חֳדָשִׁים בְּחוּץ לָאָרֶץ, וְאֵין אָנוּ סוֹמְכִין אֶלָּא עַל חֶשְׁבּוֹן שֶׁלִּבְנֵי אֶרֶץ יִשְׂרָאֵל וּקְבִיעָתָם. וְזֶהוּ שֶׁאָנוּ מְחַשְּׁבִין — לְגַלּוֹת הַדָּבָר בִּלְבַד הוּא: כֵּיוָן שֶׁאָנוּ יוֹדְעִים שֶׁעַל חֶשְׁבּוֹן זֶה הֵם סוֹמְכִים, אָנוּ מְחַשְּׁבִין לֵידַע יוֹם שֶׁקָּבְעוּ בּוֹ בְּנֵי אֶרֶץ יִשְׂרָאֵל אֵיזֶה יוֹם הוּא; וּקְבִיעוּת בְּנֵי אֶרֶץ יִשְׂרָאֵל — אוֹתוֹ הוּא שֶׁיִּהְיֶה רֹאשׁ חֹדֶשׁ אוֹ יוֹם טוֹב, לֹא מִפְּנֵי הַחֶשְׁבּוֹן שֶׁאָנוּ מְחַשְּׁבִין:

(**זֶהוּ** דַּעַת הָרַמְבַּ״ם. וְנִרְאֶה מִדְּבָרָיו, דְּגַם חֶשְׁבּוֹן זֶה דַּוְקָא בְּאֶרֶץ יִשְׂרָאֵל הוּא; וְכֵן כָּתַב בְּבֵאוּר בְּסֵפֶר הַמִּצְוֹת שֶׁלּוֹ (מִצְוַת עֲשֵׂה קנ״ג), וְזֶה לְשׁוֹנוֹ: "אֵלּוּ הִנְחַנוּ דֶּרֶךְ מָשָׁל שֶׁבְּנֵי אֶרֶץ יִשְׂרָאֵל יֵעָדְרוּ מֵאֶרֶץ יִשְׂרָאֵל — חָלִילָה לָאֵל מֵעֲשׂוֹת זֹאת, כִּי הוּא הִבְטִיחַ שֶׁלֹּא יִמְחֶה אוֹתָהּ הָאֻמָּה מִכֹּל וָכֹל — וְלֹא יִהְיֶה שָׁם בֵּית דִּין, וְלֹא יִהְיֶה בְּחוּץ לָאָרֶץ בֵּית דִּין שֶׁנִּסְמַךְ בְּאֶרֶץ יִשְׂרָאֵל — הִנֵּה חֶשְׁבּוֹנֵנוּ זֶה לֹא יוֹעִילֵנוּ כְּלוּם בְּשׁוּם פָּנִים, לְפִי שֶׁאֵין לָנוּ לַחְשׁוֹב חֳדָשִׁים וּלְעַבֵּר שָׁנִים בְּחוּץ לָאָרֶץ אֶלָּא בַּתְּנָאִים הַנִּזְכָּרִים", עַד כָּאן לְשׁוֹנוֹ. וּלְפִי עֲנִיּוּת דַּעְתִּי לֹא הֵבַנְתִּי: וְאָנָה מָצָא זֶה, שֶׁהִבְטִיחַ הָאֵל לְבַל יֵעָדְרוּ בְּנֵי יִשְׂרָאֵל מֵאֶרֶץ יִשְׂרָאֵל? וְכַדּוֹמֶה שֶׁבְּסִפְרֵי הַקּוֹרוֹת נִזְכָּר כַּמָּה פְּעָמִים שֶׁלֹּא הָיָה שׁוּם יְהוּדִי בְּאֶרֶץ יִשְׂרָאֵל. (וּמנ״ב שָׁנָה שֶׁלֹּא עָבַר רֶגֶל בְּאֶרֶץ יִשְׂרָאֵל, וְקָבְעוּ חֳדָשִׁים בְּחוּץ לָאָרֶץ — שֶׁזָּכַר הָרַמְבַּ״ן — לֹא קָשֶׁה, דְּאָז הָיוּ סְמוּכִים בְּחוּץ לָאָרֶץ.) וְתוּ: אַף דְּחֶשְׁבּוֹן דָּמֵי לִרְאִיָּה, דִּבְעֵי דַּוְקָא אֶרֶץ יִשְׂרָאֵל — מִכָּל מָקוֹם, הָא גַּם בְּחוּץ לָאָרֶץ קוֹבְעִין אַף בִּזְמַן הָרְאִיָּה כְּשֶׁלֹּא הִנִּיחַ כְּמוֹתוֹ בְּאֶרֶץ יִשְׂרָאֵל; אִם כֵּן, הָכֵי נַמִּי, מַדּוּעַ יְעַדִּיף חֶשְׁבּוֹן זֶה? וַהֲרֵי גַּם בְּאֶרֶץ יִשְׂרָאֵל אֵין הַבֵּית דִּין סְמוּכִין. וְעַיֵּן בְּרַמְבַּ״ן — הֶאֱרִיךְ גַּם כֵּן לְהַקְשׁוֹת
