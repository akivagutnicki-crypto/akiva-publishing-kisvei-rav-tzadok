~ עַל הַשָּׁווֹת). וְהַשָּׁעָה מְחֻלֶּקֶת לְאֶלֶף וּפ׳ חֲלָקִים — לְפִי שֶׁמִּסְפָּר זֶה יִתָּכֵן לְחַלְּקוֹ, כְּשֶׁיַּגִּיעֵנוּ הָעִנְיָן, לְמֶחֱצָה, וְלִשְׁלִישׁ, וְלִרְבִיעַ, וּלְחֹמֶשׁ, וּלְשִׁשִּׁית, וְלִשְׁמִינִית, וְלִתְשִׁיעִית, וְלַעֲשִׂירִית, וְלִשְׁנֵים עָשָׂר, וְלִשְׁאָר חֲלָקִים — בִּשְׁלֵמִים, בְּלִי שְׁבָרִים:

**וְהַחֹדֶשׁ** — וְהוּא מִשֶּׁיִּתְקַבֵּץ הַיָּרֵחַ וְהַחַמָּה עַד שֶׁיָּשׁוּבוּ לְהִתְקַבֵּץ פַּעַם שֵׁנִית — לְפִי חֶשְׁבּוֹן זֶה הוּא נֶחְשָׁב תָּמִיד כ״ט יוֹם, וְי״ב שָׁעוֹת, וְתשצ״ג חֲלָקִים, שֶׁהֵם ד׳ שָׁבוּעוֹת וְיוֹם אֶחָד, וְי״ב שָׁעוֹת וְתשצ״ג חֲלָקִים. וְזֶהוּ הַזְּמַן שֶׁבֵּין כָּל מוֹלָד וּמוֹלָד, וְזֶהוּ חֲדָשָׁהּ שֶׁל לְבָנָה. וּשְׁנָה שֶׁל לְבָנָה, אִם תִּהְיֶה י״ב חֹדֶשׁ מֵחֳדָשִׁים אֵלּוּ — כְּלָלָהּ י״ב פְּעָמִים כ״ט י״ב תשצ״ג, שֶׁהֵם שנ״ד יָמִים, ח׳ שָׁעוֹת, תתע״ו חֲלָקִים, שֶׁהֵם נ׳ שָׁבוּעוֹת, ד׳ ח׳ תתע״ו. וְאִם הִיא מִי״ג חֹדֶשׁ — וְהִיא הַשָּׁנָה מְעֻבֶּרֶת — כְּלָלָהּ שפ״ג יָמִים, כ״א שָׁעוֹת, תקפ״ט חֲלָקִים, שֶׁהֵם נ״ד שָׁבוּעוֹת, ה׳ יָמִים, כ״א תקפ״ט. וּשְׁנַת הַחַמָּה, לְפִי דַּעַת שְׁמוּאֵל יַרְחִינָאָה עָלָיו הַשָּׁלוֹם, וּכְדַעַת אֵיזֶה חַכְמֵי הָעַמִּים — הִיא שס״ה יָמִים וְשִׁשָּׁה שָׁעוֹת, שֶׁהִיא רְבִיעַ הַיּוֹם (אֲבָל דַּעַת זוּלָתוֹ אֵינוֹ כֵּן, כְּמוֹ שֶׁיִּתְבָּאֵר לְהַלָּן). נִמְצָא יִתְרוֹן שְׁנַת הַחַמָּה עַל שְׁנַת הַלְּבָנָה הַפְּשׁוּטָה — עֲשָׂרָה יָמִים, כ״א שָׁעוֹת, ר״ד חֲלָקִים:

**וְכָל** י״ט שָׁנָה, אִם נַעֲשָׂה ז׳ שָׁנִים מֵהֶם מְעֻבָּרוֹת — יִשְׁתַּוּוּ עִם שְׁנוֹת הַחַמָּה, וְלֹא יִשָּׁאֵר יִתְרוֹן אֶלָּא שָׁעָה אַחַת וְתפ״ה חֲלָקִים. לְפִי שֶׁי״ט פְּעָמִים י׳ כ״א ר״ד — שֶׁהוּא יִתְרוֹן שְׁנַת הַחַמָּה עַל שְׁנַת הַלְּבָנָה — עוֹלֶה ר״ו יוֹם, י״ח שָׁעוֹת, תרל״ו חֲלָקִים; וְחָדְשֵׁי הָעִבּוּר — שֶׁהֵם ז׳ פְּעָמִים כ״ט י״ב תשצ״ג — כְּלָלוֹ ר״ו יָמִים, י״ז שָׁעוֹת, קנ״א חֲלָקִים; וְנִמְצָא בֵּינֵיהֶם רַק שָׁעָה וְתפ״ה חֲלָקִים, כְּמוֹ שֶׁאָמַרְנוּ. נִמְצָא, בְּי״ט שָׁנִים — כְּאִלּוּ שֶׁיִּהְיוּ י״ב מֵהֶם פְּשׁוּטוֹת וְהַז׳ מְעֻבָּרוֹת — כֻּלָּם חָדְשֵׁי הַלְּבָנָה, וְהַשָּׁנִים שְׁנֵי הַחַמָּה. וּלְפִי הַחֶשְׁבּוֹן קָבְעוּ שֶׁיִּהְיוּ הַשָּׁנִים הַמְּעֻבָּרוֹת: הַשָּׁנָה הַשְּׁלִישִׁית מִן הַי״ט, וְהַשִּׁשִּׁית, וְהַשְּׁמִינִית, וְהַי״א, וְהַי״ד, וְהַי״ז, וְהַי״ט; וְסִימָנָם: גו״ח אדז״ט. שֶׁבְּסֵדֶר זֶה יִהְיֶה תָּמִיד חֹדֶשׁ הָעִבּוּר כְּשֶׁיִּתְקַבֵּץ שִׁעוּר קָרוֹב מִל׳ יוֹם:

**וְנִקְרָאוּ** כָּל סֵדֶר י״ט שָׁנִים אֵלּוּ "מַחֲזוֹר", לְפִי שֶׁבָּהֶם חוֹזְרוֹת חֲלִילָה שְׁנוֹת הַלְּבָנָה לְהִשְׁתַּווֹת עִם שְׁנוֹת הַחַמָּה. וּלְפִי שֶׁיִּתְבָּאֵר לְהַלָּן, כִּי יֵשׁ עוֹד מַחֲזוֹר לַחַמָּה — וְהוּא מִכ״ח שָׁנָה — נִקְרָא לָזֶה שֶׁל י״ט שָׁנָה "מַחֲזוֹר קָטָן":
