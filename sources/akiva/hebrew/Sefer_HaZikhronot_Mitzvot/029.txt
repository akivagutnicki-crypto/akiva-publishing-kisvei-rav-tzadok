~ דְּמִילָה. וְיֵשׁ מְקוֹמוֹת נָהֲגוּ שֶׁלֹּא לוֹמַר תַּחֲנוּן אַף בְּבֵית הַכְּנֶסֶת אַחֵר; וּבִדְבָרִים כָּאֵלּוּ — נַהֲרָא נַהֲרָא וּפַשְׁטֵיהּ. וְנוֹהֲגִין לְהַדֵּר לִהְיוֹת סַנְדָּק לִתְפּוֹשׂ הַתִּינוֹק לַמּוֹהֲלוֹ; וִיהַדֵּר גַּם כֵּן הָאָב אַחַר סַנְדָּק הָגוּן, שֶׁיְּהֵא רָאוּי לֵישֵׁב בְּקַתֶּדְרָא אֵצֶל אֵלִיָּהוּ מַלְאַךְ הַבְּרִית. וְיָפֶה כֹּחַ הַסַּנְדָּק מִכֹּחַ הַמּוֹהֵל, לְהַקְדִּימוֹ לִקְרִיאַת הַתּוֹרָה, דְּכָל סַנְדָּק הֲוֵי כְּמַקְטִיר קְטֹרֶת. וְלָכֵן נוֹהֲגִים שֶׁלֹּא לִתֵּן ב׳ יְלָדִים לְסַנְדָּק אֶחָד, אִם הֵם שֶׁל אָב אֶחָד, כִּדְאָמְרִינַן גַּבֵּי קְטֹרֶת: "חֲדָשִׁים לִקְטֹרֶת — בֹּאוּ וְהָפִיסוּ". וּבִמְקוֹמוֹת אֲחֵרִים נָהֲגוּ אַף שֶׁלֹּא שֶׁל אָב אֶחָד; וַאֲחֵרִים נָהֲגוּ רַק בְּשָׁנָה אַחַת; וְהַכֹּל כְּמִנְהַג הַמְּדִינָה. וּבִמְדִינָתֵנוּ לֹא נָהֲגוּ כְּלָל לְדַקְדֵּק בָּזֶה; וַאֲפִלּוּ בְּאַתְרָא דִּנְהִיגֵי — אִם בָּא הָאָב לַעֲשׂוֹתוֹ סַנְדָּק — אֵין בְּכָךְ כְּלוּם, וְלֵית לַן בָּהּ. וְנָהֲגוּ לִהְיוֹת הַסַּנְדָּק גַּם כֵּן לוֹקֵחַ הַיֶּלֶד מֵהַיּוֹלְדוֹת לַהֲבִיאוֹ לַכִּסֵּא שֶׁמָּלִין; וְעַכְשָׁיו נָהֲגוּ לְכַבֵּד זֶה לְאָדָם (אַחֵר), וְגַם בֵּינְתַיִם מְכַבְּדִים כַּמָּה בְּנֵי אָדָם בִּנְשִׂיאוּת הַתִּינוֹק, וְנָכוֹן הוּא לְזַכּוֹת רֹב עַם בַּמִּצְוָה. וְאֵין לְאִשָּׁה לִהְיוֹת מְבִיאָה הַתִּינוֹק לַכִּסֵּא שֶׁמָּלִין, בְּמָקוֹם שֶׁאֶפְשָׁר בְּאִישׁ, דַּהֲוֵי פְּרִיצוּת; וּמִכָּל מָקוֹם עוֹזֶרֶת לְבַעְלָהּ, שֶׁמְּבִיאַתּוּ עַד פֶּתַח בֵּית הַכְּנֶסֶת, וְשָׁם לוֹקְחוֹ הָאִישׁ מִמֶּנָּה וּמְבִיאוֹ לַכִּסֵּא. וְכֵן נָכוֹן שֶׁלֹּא יִכָּנֵס הָאִישׁ בֵּין הַנָּשִׁים אֶל הַיּוֹלֶדֶת לְקַחְתּוֹ:

**נוֹהֲגִים** לַעֲשׂוֹת סְעוּדָה בְּיוֹם הַמִּילָה; וְנָהֲגוּ לִקַּח מִנְיָן לִסְעוּדַת מִילָה. וּמֵעִקְּרֵי סְעוּדַת מִצְוָה; וְכָל מִי שֶׁאֵינוֹ הוֹלֵךְ לִסְעוּדַת מִצְוָה כְּשֶׁמְּזַמְּנִים אוֹתוֹ — הֲוֵי כִּמְנֻדֶּה לַשָּׁמַיִם. וּלְפִיכָךְ יִזָּהֵר מִלְּבַקֵּשׁ עַל הַסְּעוּדָה אָדָם שֶׁיּוֹדֵעַ בּוֹ שֶׁלֹּא יֵלֵךְ, שֶׁלֹּא יְבִיאוֹ לִידֵי מִכְשׁוֹל (וּבְלָאו הָכֵי אִיכָּא גְּנֵבַת דַּעַת). מִיהוּ אִם נִמְצְאוּ שָׁם בְּנֵי אָדָם שֶׁאֵין מְהֻגָּנִים — רַשָּׁאִים לִמְנוֹעַ מִלֵּילֵךְ. וְנִרְאֶה הוּא הַדִּין אִם הַבַּעַל הַבַּיִת רַע עַיִן — יְקַיֵּים הַכָּתוּב "אַל תִּלְחַם לֶחֶם רַע עָיִן"; כָּל שֶׁכֵּן אִם הוּא אוֹיְבוֹ — יִזָּהֵר מִלְּהִתְקַיֵּם בּוֹ "אִם רָעֵב שׂוֹנַאֲךָ" וְכוּלֵי. וְנִרְאֶה נַמִּי דְּכָל זֶה דַּוְקָא כְּשֶׁאֵין עוֹסֵק בַּתּוֹרָה אָז; אֲבָל עוֹסֵק בַּתּוֹרָה — לֹא יְהֵא אֶלָּא מִצְוָה גְּדוֹלָה, הָעוֹסֵק בַּתּוֹרָה פָּטוּר מִן הַמִּצְוָה שֶׁאֶפְשָׁר לַעֲשׂוֹתָהּ עַל יְדֵי אֲחֵרִים; וְכֵן אִם עוֹסֵק בַּעֲשִׂיַּת אֵיזוֹ מִצְוָה — רַשַּׁאי לִמָּנַע. וְאוֹכְלִים בָּשָׂר וְשׁוֹתִים יַיִן בִּסְעוּדָה זוֹ, אֲפִלּוּ בְּשָׁבוּעַ שֶׁחָל ט׳ בְּאָב, וּכְמוֹ שֶׁיִּתְבָּאֵר בִּמְקוֹמוֹ. וּזְמַן הַסְּעוּדָה עִקָּרָהּ בַּיּוֹם; אֲבָל אִם הָיָה תַּעֲנִית צִבּוּר — יַעֲשֶׂנָּה בַּלַּיְלָה אוֹ לְמָחֳרַת. וְאִם הוּא סָמוּךְ לִזְמַן מִנְחָה — יִתְפַּלֵּל מִקֹּדֶם, כְּמוֹ שֶׁיִּתְבָּאֵר בִּמְקוֹמוֹ; וְאֵין נְפִילַת אַפַּיִם בְּמִנְחָה בִּשְׁבִיל הַסְּעוּדָה, וְתַלְיָא בְּמִנְהָגָא. וְדִינֵי אָבֵל בִּסְעוּדָה הַזּוֹ יִתְבָּאֵר בִּמְקוֹמוֹ. נוֹהֲגִים עוֹד לַעֲשׂוֹת סְעוּדָה בְּלֵיל שַׁבָּת שֶׁקֹּדֶם הַמִּילָה, וְנָהֲגוּ בִּטְעִימָה בִּלְבַד מִמִּינֵי מִגְדִים, לְפִי שֶׁאָז כָּל אֶחָד אוֹכֵל סְעוּדָה
