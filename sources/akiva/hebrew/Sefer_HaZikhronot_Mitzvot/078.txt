**וְהַמַּחֲזוֹר** — שִׁשָּׁה אֲלָפִים, תתקל״ט יָמִים, ט״ז שָׁעוֹת, תקצ״ה חֲלָקִים, שֶׁהֵם תתקצ״א שָׁבוּעוֹת, ב׳ יָמִים, ט״ז תקצ״ה. וְאִלּוּ הָיוּ יְמֵי הַחֹדֶשׁ, אוֹ הַשָּׁנָה, אוֹ הַמַּחֲזוֹר — כֻּלָּם בְּשָׁבוּעוֹת שְׁלֵמוֹת, הָיָה הַמּוֹלָד חוֹזֵר חֲלִילָה: שֶׁאִם הָיָה הַמּוֹלָד הָרִאשׁוֹן בִּתְחִלַּת הַשָּׁבוּעַ אוֹ בְּאֶמְצָעָהּ — כֵּן יִהְיֶה הַשֵּׁנִי, וְכֵן הַשְּׁלִישִׁי, וְכֵן כֻּלָּם. אֲבָל לְפִי שֶׁיֵּשׁ יִתְרוֹן עַל הַשָּׁבוּעוֹת, כְּמוֹ שֶׁנִּתְבָּאֵר — הִנֵּה יִתְרַחֵק מוֹלָד אֶחָד מֵחֲבֵירוֹ עִם הַיִּתְרוֹן הַהוּא. וְלָכֵן, אִם יָדַעְנוּ מוֹלַד הַלְּבָנָה בְּחֹדֶשׁ אֶחָד — נוּכַל לָדַעַת גַּם כֵּן בַּחֹדֶשׁ שֶׁאַחֲרָיו, בְּהוֹסִיף עָלָיו א׳ י״ב תשצ״ג, שֶׁהוּא יִתְרוֹן הַשָּׁבוּעוֹת מֵחֹדֶשׁ אֶחָד. וְכֵן בְּחֹדֶשׁ אַחֵר רָחוֹק מִמֶּנּוּ — נוֹסִיף כָּךְ פְּעָמִים א׳ י״ב תשצ״ג כְּמִסְפַּר חֳדָשִׁים אֲשֶׁר רָחַק חֹדֶשׁ הַנִּרְצֶה מֵחֹדֶשׁ הַנּוֹדָע; וְנַעֲשֶׂה מְקֻבֶּצֶת כָּל אֶלֶף וּפ׳ חֲלָקִים — שָׁעָה, וּמְקֻבֶּצֶת כָּל כ״ד שָׁעָה — יוֹם, וּקְבוּצַת כָּל ז׳ יָמִים — נַשְׁלִיךְ; וְהַמּוֹתָר הוּא יִהְיֶה יוֹם וְרֶגַע הַמּוֹלָד שֶׁל אוֹתוֹ הַחֹדֶשׁ בִּימֵי הַשָּׁבוּעַ:

**כֵּיצַד?** הֲרֵי שֶׁיָּדַעְתָּ מוֹלַד חֶשְׁוָן שֶׁל אֵיזוֹ שָׁנָה, שֶׁהָיָה ג׳ יָמִים, ד׳ שָׁעוֹת, תק״מ חֲלָקִים — עַל יוֹם ד׳; וְתִבְקַשׁ לֵידַע מוֹלַד טֵבֵת שֶׁל אוֹתָהּ שָׁנָה: תּוֹסִיף ב׳ פְּעָמִים א׳ י״ב תשצ״ג עַל ג׳ ד׳ תק״מ, וִיהִי קְבוּצָתָם: ה׳, כ״ח, ב״א וקכ״ו. וְאַחֲרֵי הַשְׁווֹת הַחֲלָקִים — תתר״ף לְשָׁעָה, וְהַכ״ד שָׁעוֹת לְיוֹם — יִהְיֶה: ו׳, ה׳, אֶלֶף מ״ו. וְזֶהוּ מוֹלַד טֵבֵת הַנִּרְצֶה; וְכֵן כָּל כַּיּוֹצֵא בָּזֶה. וְכֵן אִם תֹּאבֶה לָדַעַת מוֹלַד אוֹתוֹ חֹדֶשׁ בַּשָּׁנָה הַבָּאָה — תּוֹסִיף עַל אוֹתוֹ מוֹלָד הַיָּדוּעַ לְךָ: אִם הִיא שָׁנָה פְּשׁוּטָה — ד׳ ח׳ תתע״ו; וְאִם מְעֻבֶּרֶת — ה׳ כ״א תקפ״ט; שֶׁאֵלֶּה הֵם הַיִּתְרוֹנוֹת, כְּמוֹ שֶׁנִּתְבָּאֵר. וּקְבוּצָתָם — אַחֲרֵי שֶׁנַּעֲשָׂה מֵחֲלָקִים שָׁעוֹת, וּמִשָּׁעוֹת יָמִים, וּמִיָּמִים שָׁבוּעוֹת, כְּמוֹ שֶׁאָמַרְנוּ — הוּא הַמּוֹלָד הַנִּרְצֶה:

**וְאִם** נַחְפּוֹץ לָדַעַת מוֹלַד אוֹתוֹ חֹדֶשׁ אַחַר י״ט שָׁנָה — שֶׁהוּא מַחֲזוֹר קָטָן — תּוֹסִיף עָלָיו יִתְרוֹן הַמַּחֲזוֹר, שֶׁהוּא ב׳ ט״ז תקצ״ה, כְּמוֹ שֶׁאָמַרְנוּ, וְיֵצֵא לְךָ מוֹלַד אוֹתוֹ חֹדֶשׁ בַּמַּחֲזוֹר שֶׁלְּאַחֲרָיו. וְכֵן נָדִין מִזֶּה: שֶׁאִם יָדַעְנוּ מוֹלַד חֹדֶשׁ אֶחָד, וְנֹאבֶה לֵידַע מוֹלַד חֹדֶשׁ הַקּוֹדְמוֹ — לִיגְרַע מִזֶּה א׳ י״ב תשצ״ג, וְהַנִּשְׁאָר הוּא מוֹלַד הַקּוֹדְמוֹ. כֵּיצַד? בְּדִמְיוֹן, שֶׁזָּכַרְנוּ שֶׁיָּדַעְנוּ מוֹלַד חֶשְׁוָן ג׳ ד׳ תק״מ, וְנַחְפּוֹץ לֵידַע מוֹלַד תִּשְׁרֵי שֶׁל אוֹתָהּ שָׁנָה — נִגְרַע מִמֶּנּוּ א׳ י״ב תשצ״ג. וּלְפִי שֶׁלֹּא נוּכַל לִגְרוֹעַ הַחֲלָקִים תשצ״ג מִן תק״מ — לָכֵן נִקַּח שָׁעָה אַחַת מִן הַד׳ שָׁעוֹת, וְנַעֲשֶׂנָּה חֲלָקִים אֶלֶף פ׳, וּנְחַבְּרֵם אֶל הַתק״מ, וְיִהְיוּ אֶלֶף תר״כ; וְאָז נִגְרַע מֵהֶם תשצ״ג, וְיִשָּׁאֵר תתכ״ז. וְנָבוֹא לִגְרוֹעַ הַשָּׁעוֹת י״ב
