~ כְּגוֹן שֶׁאַחַר כָּךְ לֹא יִמָּצֵא חָכָם, אוֹ שֶׁיְּרֵאִים מִגְּזֵרַת הַמַּלְכוּת. אֲבָל שֶׁלֹּא בִּשְׁעַת הַדְּחָק — אֵין מוֹדִיעִים שֶׁהִיא מְעֻבֶּרֶת אֶלָּא בַּאֲדָר: הֵם אוֹמְרִים "שָׁנָה זוֹ מְעֻבֶּרֶת", וְחֹדֶשׁ הַבָּא אֵינוֹ נִיסָן אֶלָּא אֲדָר שֵׁנִי. וְעַיֵּן בַּמְּפָרֵשׁ הַטַּעַם, וּדְבָרָיו צָרִיךְ עִיּוּן; וְעַיֵּן בַּלְּבוּשׁ. וְלִי נִרְאֶה פָּשׁוּט הַטַּעַם, כְּמוֹ שֶׁכָּתַב רַשִׁ״י (י״ב עַמּוּד א׳) עַל "לִפְנֵי רֹאשׁ הַשָּׁנָה": דִּשְׁמָּא יִשְׁכְּחוּ בְּנֵי גוֹלָה; הָכֵי נַמִּי, לְאַחַר רֹאשׁ הַשָּׁנָה — לְכַתְּחִלָּה צָרִיךְ בִּזְמַנּוֹ, שֶׁלֹּא יִשְׁכָּחוּ, עַיֵּן שָׁם, וּפָשׁוּט. וַאֲפִלּוּ עִבְּרוּ וְאָמְרוּ לִפְנֵי רֹאשׁ הַשָּׁנָה "שָׁנָה זוֹ מְעֻבֶּרֶת" — אֵינָהּ מְעֻבֶּרֶת בַּאֲמִירָה זוֹ, וּצְרִיכִים לַחֲזוֹר וְלוֹמַר לְאַחַר רֹאשׁ הַשָּׁנָה, וּלְכַתְּחִלָּה בַּאֲדָר, כְּמוֹ שֶׁנִּתְבָּאֵר:

(**וְדַע**, דְּאִיתָא בַּגְּמָרָא (י״ב עַמּוּד א׳): אֵין מְעַבְּרִין לֹא מִשָּׁנָה לַחֲבֶרְתָּהּ, וְלֹא שָׁלֹשׁ שָׁנִים זֶה אַחַר זֶה; וְהָא דְּרַבִּי עֲקִיבָא — בֵּית דִּין יָשְׁבוּ, וְקָבְעוּ אַחַת אַחַת בִּזְמַנָּהּ. וּפֵרֵשׁ רַשִׁ״י: "לֹא מִשָּׁנָה לַחֲבֶרְתָּהּ" — שֶׁכְּשֶׁאֵין שָׁנָה זוֹ צְרִיכָה עִבּוּר, אַף עַל פִּי שֶׁשָּׁנָה הַבָּאָה צְרִיכָה עִבּוּר — אֵין מְעַבְּרִין שָׁנָה זוֹ מִפְּנֵי שָׁנָה הַבָּאָה. אֲבָל תּוֹסָפוֹת פֵּרְשׁוּ: דְּאֵין מְחַשְּׁבִין מִשָּׁנָה זוֹ עַל שָׁנָה הַבָּאָה; וְאַף עַל פִּי שֶׁאֵין מְגַלִּין — מִכָּל מָקוֹם אֵין מְחַשְּׁבִין, לִסְמוֹךְ עַל אוֹתוֹ חֶשְׁבּוֹן, שֶׁלֹּא לַעֲשׂוֹת חֶשְׁבּוֹן אַחֵר. וְ"שָׁלֹשׁ שָׁנִים" — פֵּרֵשׁ רַשִׁ״י: רְצוּפוֹת, שֶׁלֹּא יָבוֹא נִיסָן וּפֶסַח בַּקַּיִץ. (וְצָרִיךְ לוֹמַר: דְּהַיְנוּ לְצֹרֶךְ, דְּאִלּוּ מִפְּנֵי אָבִיב וּתְקוּפָה — מִמֵּילָא לֹא שַׁיָּךְ, וּמוּבָן.) וְהִקְשׁוּ הַתּוֹסָפוֹת: אִם כֵּן, שְׁתֵּי שָׁנִים נַמִּי; וּפֵרְשׁוּ כְּפֵרוּשׁ ב׳ דְּרַשִׁ״י — שֶׁאֵין קוֹבְעִים בְּפַעַם אֶחָד לִהְיוֹת ג׳ שָׁנִים מְעֻבָּרוֹת, כָּל אַחַת בִּזְמַנָּהּ; אֶלָּא שֶׁלְּפִי פֵּרוּשָׁם דִּלְעֵיל — אֵין זֶה נָכוֹן; עַל כָּל פָּנִים, עַיֵּן שָׁם. וְהִנֵּה הָרַמְבַּ״ם לֹא זָכַר כָּל זֶה. וְנִרְאֶה לִי: דִּמְפָרֵשׁ "מִשָּׁנָה לַחֲבֶרְתָּהּ" כְּפֵרוּשׁ הַתּוֹסָפוֹת, וְלֹא כְּפֵרוּשָׁם — דְּאֵינְהוּ קָאָמְרֵי "לַחְשׁוֹב", וְהוּא סָבוּר "לִקְבּוֹעַ"; וְהַיְנוּ בָּרַיְתָא דִּלְעֵיל: דְּאֵין מְעַבְּרִין הַשָּׁנָה לִפְנֵי רֹאשׁ הַשָּׁנָה. וּמַה שֶּׁאָמַר "שָׁלֹשׁ שָׁנִים" — נִרְאֶה לִי לְפָרֵשׁ, דְּרוֹצֶה לוֹמַר: לֹא מִבַּעְיָא "מִשָּׁנָה לַחֲבֶרְתָּהּ" — דְּלֹא שֶׁאֵין מִתְעַסֵּק בְּעִנְיְנֵי עִבּוּר; אֶלָּא אֲפִלּוּ בְּשָׁנָה שֶׁמִּתְעַסֵּק בְּעִנְיַן עִבּוּר — סָלְקָא דַּעְתָּךְ אָמֵינָא: כֵּיוָן דְּמִתְעַסֵּק בְּעִנְיַן עִבּוּר, יְעַבֵּר גַּם עַל שָׁנִים אֲחֵרִים; קָא מַשְׁמַע לַן דְּלֹא. וְהוּא הַדִּין דַּאֲפִלּוּ ב׳ שָׁנִים — דְּלֹא; אֶלָּא נָקַט ג׳ לְרְבוּתָא דְּר״ש, דִּפְלִיג גַּם בְּג׳, דְּרָאִיתוֹ מִדְּרַבִּי עֲקִיבָא — הָיוּ מִג׳ שָׁנִים. (וְכֵן מֻכְרָח לוֹמַר לְיַשֵּׁב פֵּרוּשׁ קַמָּא דְּרַשִׁ״י מִקֻּשְׁיַת הַתּוֹסָפוֹת.) מִיהוּ, לִי קָשֶׁה אַפֵּרוּשׁ זֶה דְּרַשִׁ״י: מַאי מִשּׁוּם רְאָיָה דְּקָאָמְרֵי חֲכָמִים? וְאוּלַי לֹא גָּרִיס כֵּן לְפִי פֵּרוּשׁוֹ זֶה. וְכָל זֶה לְעִנְיַן לִקְבּוֹעַ; אֲבָל לַחְשׁוֹב — שַׁפִּיר דָּמֵי. וְאִם כֵּן, כְּבָר כָּתַב הָרַמְבַּ״ם כָּל עִנְיַן בָּרַיְתָא זוֹ, לְפִי פֵּרוּשׁ זֶה — כֵּן נִרְאֶה לְפִי עֲנִיּוּת דַּעְתִּי. אֲבָל מִכָּל מָקוֹם, הִנֵּה לְפִי פֵּרוּשׁ רַשִׁ״י וְהַתּוֹסָפוֹת יָצְאוּ דִּינִים מְחֻדָּשִׁים מִבָּרַיְתָא זוֹ, רָאוּי לְדַעְתָּם):
