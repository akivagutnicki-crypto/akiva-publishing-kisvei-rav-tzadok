~ וְאִם כֵּן עַכְשָׁיו כֻּלָּם פְּטוּרִים, וְלָזֶה הִסְכִּים בְּשַׁאֲגַת אַרְיֵה (סִימָן מ״ט). וְכָתַב עוֹד, דַּאֲפִלּוּ הָיוּ חַיָּבִים — הָא בָּא סַנְחֵרִיב וּבִלְבֵּל כָּל הָעַמִּים, וְכֻלָּם פְּטוּרִים, כִּדְאִיתָא כַּהַאי גַּוְנָא בִּבְרָכוֹת בְּעַמּוֹנִים:

**הָאָב** חַיָּב לָמוּל אֶת בְּנוֹ (וְעַיֵּן בַּתַּשְׁבֵּ״ץ חֵלֶק ג׳ סִימָן ח׳) — דְּכוֹפִין לְהָאָב אִם אֵינוֹ רוֹצֶה, עַיֵּן שָׁם, שֶׁכָּתַב לְפָשׁוּט. וְעַיֵּן בְּתוֹלְדוֹת אָדָם לְהָרַשְׁבָּ״א (סִימָן שכ״א) כָּתַב דְּאֵין כּוֹפִין לָאָב — עַיֵּן שָׁם בְּהֵיטֵב, וְצָרִיךְ לְעַיֵּן לְדִינָא. וְאֵין הָאִשָּׁה חַיֶּבֶת לָמוּל אֶת בְּנָהּ. אֲפִלּוּ מֵת הָאָב, אוֹ דְּלֵיתֵיהּ, קָמָן אֶלָּא הַמִּצְוָה — אָז הִיא עַל הַיּוֹרְשִׁים הַיּוֹתֵר גְּדוֹלִים שֶׁיָּרְשׁוּ זְכוּתוֹ שֶׁל אָב גַּם בָּזֶה; וְאִי לֵיתְנַיְהוּ — גַּם כֵּן הַמִּצְוָה עַל אַפּוֹטְרוֹפּוֹס (אֲבָל יוֹרְשִׁים עֲדִיפֵי מֵאַפּוֹטְרוֹפּוֹס). עָבַר הָאָב וְלֹא מָל, אוֹ שֶׁאֵין כָּל אֶחָד מֵהַנִּזְכָּרִים — הֲרֵי הַבֵּית דִּין חַיָּבִים לְמוּלוֹ; וּמָלִין בְּעַל כָּרְחוֹ שֶׁל אָב, אִי אִיתֵיהּ קָמָן וְאֵינוֹ רוֹצֶה (אֲבָל בְּלָאו הָכֵי — אֵין מָלִין בְּנוֹ שֶׁל אָדָם שֶׁלֹּא מִדַּעְתּוֹ; וְאִם מָל שֶׁלֹּא מִדַּעְתּוֹ — חַיָּב לִתֵּן לוֹ, אִם הָאָב מוֹהֵל, עֲשָׂרָה זְהוּבִים. וְשִׁעוּרָם חֵלֶק שִׁשָּׁה עָשָׂר מֵה׳ סְלָעִים דְּפִדְיוֹן הַבֵּן, שֶׁיִּתְבָּאֵר שִׁעוּרוֹ בִּמְקוֹמוֹ. מִיהוּ בִּזְמַן הַזֶּה אֵין גּוֹבִין אוֹתוֹ בְּבֵית דִּין, וְאִם תָּפַס — אֵין מוֹצִיאִין מִיָּדוֹ. וַאֲפִלּוּ לָצֵאת יְדֵי שָׁמַיִם — יֵשׁ אוֹמְרִים דְּחַיָּב לְשַׁלֵּם, וְיֵשׁ חוֹלְקִים, וְכֵן עִקָּר). וְאִם לֹא מָלוּהוּ בֵּית דִּין — חַיָּב הוּא כְּשֶׁיַּגְדִּיל לָמוּל אֶת עַצְמוֹ, כְּמוֹ שֶׁנִּתְבָּאֵר. וְאֵין הָאָב אוֹ הַבֵּית דִּין חַיָּבִין כָּרֵת כְּשֶׁלֹּא מָלוּהוּ. וְאִם הָאָב יוֹדֵעַ בְּעַצְמוֹ לָמוּל — יֵשׁ אוֹמְרִים שֶׁמִּצְוָה עָלָיו מֻטֶּלֶת, וְאֵין רַשַּׁאי לְכַבֵּד לְאַחֵר לָמוּל, וְאִם כִּבֵּד — בִּטֵּל מִצְוָה. וְיֵשׁ חוֹלְקִים עַל זֶה, דִּבְכָל מָקוֹם שְׁלוּחוֹ שֶׁל אָדָם כְּמוֹתוֹ, וּלְפִיכָךְ רַשַּׁאי לְכַבֵּד אַחֵר, וְהוּא שְׁלוּחוֹ; וְיִהְיֶה הָאָב עוֹמֵד עַל גַּבָּיו לְהוֹדִיעַ שֶׁהוּא שְׁלוּחוֹ — וְכֵן עִקָּר. וְצָרִיךְ לְדַקְדֵּק שֶׁלֹּא יְכַבֵּד לְקָטָן, וַאֲפִלּוּ בֶּן י״ג שָׁנִים, אֶלָּא שֶׁאֵין יָדוּעַ שֶׁהֵבִיא שְׂעָרוֹת, לְפִי שֶׁאֵין שְׁלִיחוּת לְקָטָן וְאֵינוֹ כְּמוֹתוֹ. וּמִיהוּ מִכָּל מָקוֹם מִצְוָה לְכַתְּחִלָּה שֶׁיָּמוּל הָאָב בְּעַצְמוֹ, לְפִי שֶׁמִּצְוָה בּוֹ יוֹתֵר מִבִּשְׁלוּחוֹ. וְאִם הָאָב אֵינוֹ יוֹדֵעַ לָמוּל, וְיֵשׁ מוֹהֵל שֶׁאֵינוֹ רוֹצֶה לָמוּל בְּחִנָּם — יֵשׁ לְבֵית דִּין לִגְעוֹר בּוֹ, שֶׁאֵין זֶה דַּרְכָּם שֶׁל זֶרַע אַבְרָהָם אָבִינוּ עָלָיו הַשָּׁלוֹם; אַדְּרַבָּא, הַמּוֹהֲלִים מְהַדְּרִים אַחַר הַמִּצְוָה. וְאִם עֲדַיִן עוֹמֵד בְּמִרְדּוֹ, וְאֵין יַד הָאָב מַשֶּׂגֶת לָתֵת לוֹ שְׂכָרוֹ — הֲוֵי כְּמִי שֶׁאֵין לוֹ אָב, שֶׁבֵּית דִּין חַיָּבִים לְמוּלוֹ; וְלָכֵן בֵּית דִּין כּוֹפִין אוֹתוֹ, כְּשֶׁאֵין אַחֵר לָמוּל. וְאִם אֵין בִּידֵיהֶם לָכוֹפוֹ, אוֹ שֶׁאֵין שָׁהוּת — יְכוֹלִים לְהַטְעוֹתוֹ, וְאוֹמְרִים לוֹ: "בּוֹא וּמוֹל, וְנִתֵּן לְךָ כִּרְצוֹנְךָ", וְאַחַר כָּךְ יְכוֹלִים לִשְׂחוֹק בּוֹ וְלֵאמֹר: "לְהַשְׁטוֹת נִתְכַּוַּנּוּ" (כֵּן כָּתַב יַעְבֵּ״ץ). וְנִרְאֶה לִי דִּמַיְירֵי כְּשֶׁמְּבַקֵּשׁ הַרְבֵּה מָעוֹת, שֶׁאֵין בְּיַד הַבֵּית דִּין לִתֵּן לוֹ; אֲבָל בְּלָאו הָכֵי — עַד חֹמֶשׁ מְעוֹתֵיהֶם מִדּוּעַ
