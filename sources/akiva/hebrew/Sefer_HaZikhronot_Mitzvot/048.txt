~ הַדָּבָר. כָּל זֶה לְפֵרוּשׁ רַשִׁ״י בַּגְּמָרָא, פֶּרֶק בֵּן סוֹרֵר וּמוֹרֶה (ע׳ עַמּוּד ב׳). אֲבָל הָרַמְבַּ״ם כָּתַב: "וְאֵין עוֹלִין בַּלַּיְלָה, אֶלָּא בַּנֶּשֶׁף, קֹדֶם עֲלוֹת הַשַּׁחַר" (נִרְאֶה דְּצָרִיךְ לוֹמַר "הַשֶּׁמֶשׁ"). נִרְאֶה דִּמְפָרֵשׁ "אַחְרִיפוּ" הַיְנוּ קֹדֶם עֲלוֹת הַשֶּׁמֶשׁ, בְּסוֹף הַלַּיְלָה — לֹא בִּתְחִלַּת הַלַּיְלָה. וְצָרִיךְ לְהָבִין אֵיךְ מְיַשֵּׁב הַקֻּשְׁיָא בָּזֶה. וְאוּלַי דְּרוֹצֶה לוֹמַר: בְּשָׁעָה שֶׁאֵינוֹ לֹא יוֹם וְלֹא לַיְלָה, דְּגַם בְּשַׁחֲרִית יֵשׁ בֵּין הַשְּׁמָשׁוֹת, כַּיָּדוּעַ; וְזֶהוּ לְשׁוֹן "נֶשֶׁף", כִּדְאִיתָא (פֶּרֶק קַמָּא דִּבְרָכוֹת): "נְשַׁף לֵילְיָא וְאָתֵי יְמָמָא". וְלָכֵן אָמַר פַּעַם אַחַת "בַּיּוֹם", וְלֹא "בַּלַּיְלָה", וּפַעַם אַחַת "בַּלַּיְלָה" וְלֹא "בַּיּוֹם"; וְכַהַאי גַּוְנָא (פֶּרֶק קַמָּא דְּשַׁבָּת) בְּכַרְמְלִית — מַאי טַעְמָא קָרֵי לֵיהּ "רְשׁוּת הָרַבִּים"? לְפִי שֶׁאֵינוֹ רְשׁוּת הַיָּחִיד; מַאי טַעְמָא קָרֵי לֵיהּ "רְשׁוּת הַיָּחִיד"? לְפִי שֶׁאֵינוֹ רְשׁוּת הָרַבִּים. וְנִרְאֶה דְּהִכְרִיחוֹ לְפָרֵשׁ כֵּן לְיַשֵּׁב קֻשְׁיַת תּוֹסָפוֹת, דִּבּוּר הַמַּתְחִיל "וְאֵין" וְכוּלֵי — "הָא" וְכוּלֵי; דִּלְפִי זֶה נִיחָא, דִּלְאוֹר עִבּוּרוֹ לְמַעֵט יוֹם ל׳, וְאַלָּא בַּלַּיְלָה לְמַעֵט יוֹם ל״א. מִיהוּ מִכָּל מָקוֹם נִרְאֶה דַּעַת רַשִׁ״י עִקָּר בִּלְשׁוֹן הַגְּמָרָא; וְגַם לְדַעַת רַשִׁ״י יֵשׁ לְפָרֵשׁ כֵּן — אֵין עוֹלִין בְּיוֹם ל״א אֶלָּא בַּלַּיְלָה. וְעוֹד דְּקָשֶׁה לְהָרַמְבַּ״ם: אֵיךְ יֹאכַל אָז קֹדֶם הַתְּפִלָּה? וְעַיֵּן שָׁם בִּמְקוֹמוֹ. גַּם לֹא יָדַעְתִּי לָמָּה הִשְׁמִיט הָרַמְבַּ״ם "אַחְרִיפוּ" וּ"פוּקוּ" — לְהַשְׁכִּים לָצֵאת; גַּם לֹא יָדַעְתִּי אֵיךְ מְפָרֵשׁ לָהּ, וְצָרִיךְ עִיּוּן:

**אַחַר** שֶׁקִּדְּשׁוּ בֵּית דִּין הַחֹדֶשׁ (וְכֵן עִבּוּר הַשָּׁנָה) — מִצְוָה לָהֶם לְהוֹדִיעַ לָעָם, כְּדֵי שֶׁיֵּדְעוּ אֲמִתַּת רֹאשׁ חֹדֶשׁ, וְיֵדְעוּ הַמּוֹעֲדוֹת אֵיךְ לַעֲשׂוֹת (וְגַם בְּרֹאשׁ חֹדֶשׁ עַצְמוֹ נַפְקָא מִינָּהּ לֶאֱסוֹר בְּהֶסְפֵּד וְתַעֲנִית, וַעֲשִׂיַּת מְלָאכָה לְנָשִׁים). וּבָרִאשׁוֹנָה הָיוּ מַשִּׂיאִין מַשּׂוּאוֹת בְּכָל חֹדֶשׁ (כֵּן כָּתוּב בְּסֵפֶר טוּרֵי אֶבֶן, דִּבּוּר הַמַּתְחִיל "הָיוּ"): כְּשֶׁהָיָה נִרְאֶה בִּזְמַנּוֹ, לְקַדְּשׁוֹ בְּיוֹם ל׳ — הָיוּ מַשִּׂיאִין בַּלַּיְלָה שֶׁלְּאַחֲרָיו, לֵיל ל״א; אֲבָל כְּשֶׁהָיָה מְעֻבָּר — לֹא הָיוּ מַשִּׂיאִין כְּלָל, וְהָיוּ הַכֹּל יוֹדְעִים: כְּשֶׁלֹּא הִשִּׂיאוּ — שֶׁהוּא מְעֻבָּר, וּכְשֶׁהִשִּׂיאוּ — שֶׁנִּרְאָה בִּזְמַנּוֹ. וְאִם חָל יוֹם ל׳ בְּעֶרֶב שַׁבָּת, וְאִי אֶפְשָׁר לְהַשִּׂיא בְּלֵיל ל״א, שֶׁהוּא בְּשַׁבָּת — מַשִּׂיאִין בְּלֵיל ל״ב, בְּמוֹצָאֵי שַׁבָּת; וְהַכֹּל יוֹדְעִים שֶׁהָיָה חָסֵר, שֶׁהֲרֵי אִם לָאו — אֵין מַשִּׂיאִים (וּבְעִבּוּר שָׁנָה, עַיֵּן בְּטוּרֵי אֶבֶן, שֶׁכָּתַב: אֶפְשָׁר גַּם אָז הָיוּ שְׁלוּחִין יוֹצְאִים עַל זֶה, אוֹ הָיוּ מַשִּׂיאִים אֵיזֶה יָמִים בַּחֹדֶשׁ לְהוֹדִיעַ שֶׁהוּא אֲדָר שֵׁנִי). וְאֵין מַשִּׂיאִים אֶלָּא בַּלַּיְלָה, דִּבַיּוֹם — "שְׁרַגָּא בְּטִיהֲרָא מַאי מְהַנֵּי":

**וְכֵיצַד** הָיוּ מַשִּׂיאִין? מְבִיאִין כְּלֻנְסָאוֹת שֶׁל אֶרֶז אֲרֻכִּים, וַעֲצֵי קָנִים, וַעֲצֵי שֶׁמֶן, וּנְעֹרֶת שֶׁל פִּשְׁתָּן, וְכוֹרֵךְ בִּמְשִׁיחָה, וְעוֹלֶה לְרֹאשׁ הָהָר וּמַצִּית בָּהֶן הָאוּר, וּמוֹלִיךְ וּמֵבִיא, וּמַעֲלֶה וּמוֹרִיד, עַד שֶׁרוֹאֶה חֲבֵרוֹ עוֹשֶׂה כֵּן בְּרֹאשׁ הָהָר הַב׳; וְכֵן בְּרֹאשׁ הָהָר הַג׳, וְהָאֲחֵרִים. וּמֵהָאַחֲרוֹן, הַסָּמוּךְ לְבָבֶל — לֹא זָזוּ
