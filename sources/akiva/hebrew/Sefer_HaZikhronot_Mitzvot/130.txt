~ לְשֶׁיֵּחָשֵׁב וְכוּלֵי, עַיֵּן שָׁם. וּלְפִי עֲנִיּוּת דַּעְתִּי, לֹא יָדַעְתִּי מַה צֹּרֶךְ לְבַקָּשַׁת רַחֲמִים; עַד שֶׁכָּתַב בִּתְבוּאוֹת שׁוֹר שָׁם, דִּבְלֹא רַחֲמִים אֵין מְכַפֶּרֶת, וְלֹא דָּמֵי לְקָרְבָּן דְּוַדַּאי מְכַפֵּר, עַיֵּן שָׁם. וּלְדַעְתִּי — הֲלֹא כְּבָר הִבְטִיחָנוּ יִתְבָּרַךְ לִהְיוֹת "מְשַׁלְּמִים פָּרִים שְׂפָתֵינוּ", וְדָמֵי מַמָּשׁ לְקָרְבָּן; וִ"יהִי רָצוֹן" — לִרְוָחָא דְּמִלְּתָא הוּא דְּהֻצְרַךְ הַטּוּר. וְאוּלַי, גַּם בְּקָרְבָּן הָיָה מַצְרִיךְ אֲמִירַת "יְהִי רָצוֹן" טֶרֶם הַקְרָבָתוֹ, שֶׁיְּכֻפַּר עַל יְדֵי הֶעָוֹן אֲשֶׁר בָּא בַּעֲבוּרוֹ; וּפָשׁוּט:

**הַכֹּל** חַיָּבִים בְּמִצְוָה זוֹ — אֶחָד אֲנָשִׁים וְאֶחָד נָשִׁים, וַעֲבָדִים, טֻמְטוּם וְאַנְדְּרוֹגִינוֹס; שֶׁנֶּאֱמַר "בְּמִכְסַת נְפָשֹׁת" (מְכִילְתָּא; גְּמָרָא צ״א עַמּוּד ב׳). יָכוֹל מֵבִיא אֲנִי אַף אֶת הַחוֹלֶה וְהַזָּקֵן, שֶׁאֵינָם יְכוֹלִים לֶאֱכוֹל כְּזַיִת? תַּלְמוּד לוֹמַר: "אִישׁ לְפִי אָכְלוֹ" (שָׁם; גְּמָרָא ס״א עַמּוּד א׳) — מִי שֶׁיָּכוֹל לֶאֱכוֹל; יָצָא מִי שֶׁאֵין יָכוֹל לֶאֱכוֹל כְּזַיִת (דִּסְתָם אֲכִילָה בִּכְזַיִת, כִּדְאִיתָא בְּפֶרֶק "כֵּיצַד מְבָרְכִין" וּבִשְׁאָר דּוּכְתֵּי). וַאֲפִלּוּ חֲבוּרָה שֶׁל מֵאָה, שֶׁכָּל אֶחָד אֵין יָכוֹל לֶאֱכוֹל כְּזַיִת — אַף עַל פִּי שֶׁבַּחֲבוּרָתָן יְכוֹלִים לֶאֱכוֹל כְּזַיִת — מִכָּל מָקוֹם, כָּל אֶחָד אֵין רָאוּי שֶׁיִּשָּׁחֵט עָלָיו הַקָּרְבָּן פֶּסַח, דְּאֵינוֹ בַּר חִיּוּב, כֵּיוָן שֶׁאֵין יָכוֹל לֶאֱכוֹל כְּזַיִת — אֹנֶס רַחֲמָנָא פַּטְרֵיהּ; וַהֲרֵי כֻּלָּם פְּטוּרִים (מִשְׁנָה צ״א עַמּוּד א׳, לְפֵירוּשׁ הָרַמְבַּ״ם פֶּרֶק ב׳ הֲלָכָה ב׳). וְעַיֵּן שָׁם בַּכֶּסֶף מִשְׁנֶה, מַה שֶּׁכָּתַב: "וּמְפָרֵשׁ רַבֵּינוּ, דְּבָעֵינַן שֶׁלֹּא יְהֵא בָּהֶם אֲפִלּוּ אֶחָד שֶׁאֵין יָכוֹל לֶאֱכוֹל כְּזַיִת". וְהוּא תָּמוּהַּ: דְּמָה אִיכְפַּת לַן אִם אֶחָד אֵין יָכוֹל לֶאֱכוֹל כְּזַיִת, כֵּיוָן דַּהֲאֲחֵרִים יְכוֹלִים — הָוֵה "לְאוֹכְלָיו וְשֶׁלֹּא לְאוֹכְלָיו". וְאוּלַי טָעוּת סוֹפֵר בִּדְבָרָיו, וְצָרִיךְ לוֹמַר "אֲפִלּוּ אֶחָד שֶׁיָּכוֹל" וְכוּלֵי, וְקָאֵי אַמַּתְנִיתִין דְּאַסְרָהּ; וְאוּלַי גַּם תֵּבַת "דְּבָעֵינַן" טָעוּת סוֹפֵר, וְכַוָּנָתוֹ כְּמוֹ שֶׁנִּתְבָּאֵר לְעֵיל:

**וְהִנֵּה** נִרְאֶה לִי, דְּהוֹכָחַת הָרַמְבַּ״ם לְפֵירוּשׁוֹ — דְּעַיֵּן בְּתוֹסָפוֹת, דִּבּוּר הַמַּתְחִיל "עֲשָׂרָה" וְכוּלֵי, עַיֵּן שָׁם. וְהִנֵּה הַמַּשְׁמָעוּת מִלְּשׁוֹן הַמִּשְׁנָה (עַיֵּן בְּמַהַרְשָׁ״א; וְאֵין דְּבָרָיו נְכוֹנִים — דְּלָא דָּמֵי לִישָּׁנָא "וַאֲפִלּוּ" וְכוּלֵי, דְּמוֹרֶה דִּסְמִיךְ לְדִבְרֵי רַבִּי יוֹסֵי, לִלְשׁוֹן "וְאֵין" וְכוּלֵי, דְּמַשְׁמַע דִּמְסוּבָּב אַ"אֵין שׁוֹחֲטִין" דְּרֵישָׁא, דְּהוּא דִּבְרֵי רַבִּי יְהוּדָה — עַל זֶה אָמְרוּ "אֵין" וְכוּלֵי, וְאֵינוֹ סָמִיךְ לְדִבְרֵי רַבִּי יוֹסֵי כְּלָל), וְהַבָּרַיְתָא — דְּזֶה מִדִּבְרֵי רַבִּי יוֹסֵי, וּמַשְׁמַע דְּרַבִּי יְהוּדָה פָּלֵיג עַל זֶה. וּבֶאֱמֶת, קֻשְׁיַית הַתּוֹסָפוֹת לֹא הֵבַנְתִּי: דְּהָא מַתְנִינַן "דְּשֶׁלֹּא יְבִיאוּהוּ" וְכוּלֵי — הַיְנוּ אַף בִּיכוֹלִים לֶאֱכוֹל כְּזַיִת, וְתַקַּנְתָּא דְּרַבָּנָן הוּא, וְלֹא מִדְּבַר תּוֹרָה (וְהָכָא בְּפָסוּל מִדְּבַר תּוֹרָה עָסְקִינַן); וְעוֹד, אֵינוֹ אֶלָּא לְכַתְּחִלָּה, וּבְדִיעֲבַד כָּשֵׁר; וְעוֹד, אַף לְכַתְּחִלָּה — הֵיכִי דְּאִי אֶפְשָׁר בְּעִנְיָן אַחֵר, וַדַּאי רַשָּׁאִים וְחַיָּבִים לַעֲשׂוֹת בִּפְנֵי עַצְמָם; וְאִלּוּ הָכָא — מִפְּסוּלָא דְּאִי אֶפְשָׁר אַיְירִינַן. אֲבָל בֶּאֱמֶת, טְפֵי הֲוָה לְהוּ לְהַקְשׁוֹת מֵהָא דִּמְפֹרָשׁ סְתָמָא בְּכַמָּה מְקוֹמוֹת — בְּפֶרֶק "תָּמִיד
