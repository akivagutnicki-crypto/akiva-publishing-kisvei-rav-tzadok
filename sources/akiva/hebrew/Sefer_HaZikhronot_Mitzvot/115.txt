~ כָּלְבּוֹ) — זֵכֶר לִסְעוּדָה שֶׁהָיוּ עוֹשִׂין לְעֵדֵי הָרְאִיָּה. וְאִם חָל בְּשַׁבָּת — מִן הַדִּין יַעֲשֶׂה סְעוּדַת רֹאשׁ חֹדֶשׁ בְּאֶחָד בְּשַׁבָּת; אֲבָל לֹא נָהֲגוּ כֵן. וְלָכֵן יַמְשִׁיךְ סְעוּדָה ג׳ עַד בַּלַּיְלָה, וְיָצָא יְדֵי חוֹבַת סְעוּדַת רֹאשׁ חֹדֶשׁ. (וּבִשְׁיָרֵי כְּנֶסֶת הַגְּדוֹלָה כָּתַב לְהוֹסִיף מַאֲכָל אֶחָד מִבְּכָל הַשַּׁבָּתוֹת. וְלִי נִרְאֶה תִּקּוּן אַחֵר — לְהַקְדִּים אוֹ לְאַחֵר, כְּמוֹ שֶׁכָּתַב לְעִנְיַן סְעוּדַת שַׁבָּת בְּמִי שֶׁרָגִיל תָּמִיד בְּמַעֲדַנִּים. מִיהוּ, עַכְשָׁיו לֹא נָהֲגוּ בְּכָל זֶה; וְנִרְאֶה לִתֵּן טַעַם — מִשּׁוּם דְּבַבֵּית יוֹסֵף כָּתַב בְּשֵׁם הָרַ״ן, דְּאֵין מִצְוָה כְּלָל עַכְשָׁיו בִּסְעוּדַת רֹאשׁ חֹדֶשׁ; וְלָכֵן, עַל כָּל פָּנִים, אִם אָכַל סְעוּדָה — אַף עַל גַּב דִּבְלָאו רֹאשׁ חֹדֶשׁ נַמִּי אֲכָלָהּ — שַׁפִּיר דָּמֵי. וְכֵן הַדִּין בְּמִי שֶׁרָגִיל בְּמַעֲדַנִּים בְּיוֹם חֹל — דְּאֵין צָרִיךְ לְהוֹסִיף מַאֲכָל, אוֹ לְהַקְדִּים וּלְאַחֵר, בִּשְׁבִיל רֹאשׁ חֹדֶשׁ. וְדַע, דְּמַשְׁמַע בְּסִימָן קפ״ח דְּאֵין צָרִיךְ דַּוְקָא פַּת, אֶלָּא דַּי בִּשְׁאָר מִינֵי מְגָדִים):

**וְאָסוּר** בְּתַעֲנִית, מִטַּעַם הַנִּזְכָּר לְעֵיל. (מִיהוּ, בְּרֹאשׁ חֹדֶשׁ נִיסָן וְרֹאשׁ חֹדֶשׁ אָב — יֵשׁ אוֹמְרִים שֶׁמִּצְוָה לְהִתְעַנּוֹת בּוֹ, שֶׁאָז מֵתוּ צַדִּיקִים — אַהֲרֹן וּבָנָיו; וְאֵין הֲלָכָה כֵּן. וּבְנֵי וַוירְמַיְישָׁא מִתְעַנִּין בְּרֹאשׁ חֹדֶשׁ סִיוָן; אֲבָל אֵין לְמֵדִים מֵהֶם לְעָלְמָא, שֶׁהָיָה אֶצְלָם גְּזֵרָה רָעָה. וּבְעִנְיַן רֹאשׁ חֹדֶשׁ תִּשְׁרֵי, שֶׁהוּא רֹאשׁ הַשָּׁנָה — עַיֵּן בִּמְקוֹמוֹ). וְאֵין גּוֹזְרִים תַּעֲנִית עַל הַצִּבּוּר (וְאִם גָּזְרוּ — מַחֲלֹקֶת הַפּוֹסְקִים אִם חָל) בּוֹ, אֶלָּא אִם כֵּן גָּזְרוּ תַּעֲנִית כָּךְ וְכָךְ יָמִים, וְהִתְחִילוּ — אֲפִלּוּ יוֹם אֶחָד — קֹדֶם רֹאשׁ חֹדֶשׁ; דְּאָז מִתְעַנִּין בְּרֹאשׁ חֹדֶשׁ וּמַשְׁלִימִין. וְיָחִיד שֶׁקִּבֵּל עָלָיו תַּעֲנִית — בֵּין שֶׁהִתְחִיל בֵּין שֶׁלֹּא הִתְחִיל — אִם קִבְּלוֹ בִּלְשׁוֹן קַבָּלַת תַּעֲנִית בְּעָלְמָא — אֵין צָרִיךְ הַתָּרָה; וְאִם קִבְּלוֹ בִּלְשׁוֹן "הֲרֵי עָלַי", שֶׁהוּא לְשׁוֹן נֶדֶר — צָרִיךְ הַתָּרָה. וְאִם נִשְׁבַּע — חָלָה הַשְּׁבוּעָה, דְּאֵינוֹ אֶלָּא אִסּוּר דְּרַבָּנָן; וְיֵשׁ אוֹמְרִים דְּאֵין חָלָה, דְּהוּא מִדְּאוֹרַיְתָא. וְהַלְּבוּשׁ כָּתַב: אַף דְּמִדְּרַבָּנָן — מִכָּל מָקוֹם אֵין חָלָה, דַּחֲכָמִים עָשׂוּ חִזּוּק. וְלָכֵן נִרְאֶה דְּיַתִּירוּ לוֹ, דְּמַתִּירִין שְׁבוּעוֹת גַּם כֵּן מִדּוֹחַק, אוֹ לִדְבַר מִצְוָה כִּי הָכָא. וְהַמִּתְעַנֶּה עַל יְדֵי נִדְרוֹ, אוֹ שֶׁמִּתְעַנֶּה תַּעֲנִית חֲלוֹם — צָרִיךְ לְמֵיתַב תַּעֲנִית לְתַעֲנִיתוֹ; אֲבָל בְּרֹאשׁ חֹדֶשׁ נִיסָן וְאָב — דִּבְלָאו הָכֵי יֵשׁ אוֹמְרִים דְּמִצְוָה לְהִתְעַנּוֹת, כַּנִּזְכָּר לְעֵיל — אֵין צָרִיךְ לְמֵיתַב תַּעֲנִיתָא לְתַעֲנִיתוֹ:

**וּכְשֵׁם** שֶׁאֵין מִתְעַנִּין — כָּךְ אֵין מַסְפִּידִין, מִטַּעַם הַנִּזְכָּר לְעֵיל. וּמִיהוּ, הַנָּשִׁים מְעַנּוֹת — שֶׁכֻּלָּן אוֹמְרוֹת כְּאַחַת, וּמְטַפְּחוֹת — שֶׁמַּכּוֹת כַּף אֶל כַּף; אֲבָל לֹא מְקוֹנְנוֹת — שֶׁתְּהֵא הָאַחַת אוֹמֶרֶת וַאֲחֵרוֹת עוֹנוֹת אַחֲרֶיהָ — שֶׁזֶּהוּ עִקַּר הַהֶסְפֵּד. וְאַחַר שֶׁנִּקְבַּר הַמֵּת — לֹא מְעַנּוֹת וְלֹא מְטַפְּחוֹת גַּם כֵּן. וְכֵן אֵין אוֹמְרִים צִדּוּק הַדִּין וְקַדִּישׁ, וְיִתְבָּאֵר בִּמְקוֹמָם בְּהִלְכוֹת אֲבֵלוּת. וּמֻתָּר לְהִתְעַנּוֹת תַּעֲנִית שָׁעוֹת; אֲבָל לְדַעַת הָאוֹמְרִים שֶׁהוּא דְּאוֹרַיְתָא — יֵשׁ לֶאֱסוֹר:
