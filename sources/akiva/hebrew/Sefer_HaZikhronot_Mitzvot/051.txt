~ רע״א) בָּזֶה עַל דִּבְרֵי הָרַמְבַּ״ם בְּפֵרוּשׁ הַמִּשְׁנָיוֹת, עַיֵּן שָׁם; וְעַיֵּן בַּמְּפָרֵשׁ (פֶּרֶק ד׳ הֲלָכָה ה׳), עַיֵּן מַה שֶּׁכָּתַב "גָּלֻיּוֹת יִשְׂרָאֵל", עַיֵּן שָׁם). וּמַתְנִיתִין, עַל כָּרְחֲךָ בְּבַיִת שֵׁנִי מִתּוֹקְמָא — דִּמְקַמֵּי הָכֵי, בְּבַיִת רִאשׁוֹן, לֹא הָיוּ צְדוֹקִים, וְלֹא הָיוּ שְׁלוּחִים, אֶלָּא מַשִּׂיאִים מַשּׂוּאוֹת; וְאִם כֵּן שַׁפִּיר לֹא חָשֵׁיב תַּמּוּז וְטֵבֵת, וְאֵין רְאָיָה. מִיהוּ, לְפִי מַה שֶּׁכָּתַבְתִּי לְעֵיל בְּשֵׁם הַיְרוּשַׁלְמִי, דִּבְרַבִּי תִּיקְּנָה לִשְׁלוּחִים — צָרִיךְ לוֹמַר הָא דְּקָאָמַר "בִּזְמַן שֶׁבֵּית הַמִּקְדָּשׁ קַיָּם" וְכוּלֵי — הַיְנוּ לְפִי תַּקָּנַת רַבִּי, הָיָה כָּךְ רָאוּי לִהְיוֹת; וְאִם כֵּן נִיחָא רְאָיָתוֹ קְצָת, דַּהֲוֵי לֵיהּ לוֹמַר נַמִּי כֵּן — אַבַּיִת רִאשׁוֹן — אַטֵּבֵת וְתַמּוּז; וְדָחוּק, דְּיֵשׁ לוֹמַר בְּבַיִת שֵׁנִי הֲוֵי צְדוֹקִים, גַּם כֵּן לְכָךְ נָקְטֵיהּ. וְעַל כָּל פָּנִים, מַה שֶּׁכָּתַב דְּלָא גָּרְסִינַן "אַף", וּדְתָמִיד שִׁשָּׁה חֳדָשִׁים רַק לֵיתָא דְּבַיִת שֵׁנִי — אֵינוֹ כֵּן, כַּנִּזְכָּר לְעֵיל. מִיהוּ מִדִּבְרֵי רַשִׁ״י נִרְאֶה דִּמְפָרֵשׁ "אֵין שָׁלוֹם" — וְאֵין שְׁמָד אַחַר הַחֻרְבָּן, עַיֵּן שָׁם; וְכֵן הוּא לְהֶדְיָא בַּטּוּר אוֹרַח חַיִּים (סִימָן תקנ״ג). וּבֶאֱמֶת צָרִיךְ עִיּוּן גָּדוֹל עַל הָרַמְבַּ״ם, דְּאֵיךְ כָּתַב: וּבְתִשְׁעָה בְּאָב נָהֲגוּ לְהִתְעַנּוֹת גַּם בִּזְמַן הַבַּיִת, עַיֵּן שָׁם; וְהָא טַעְמָא מִשּׁוּם דְּנִכְפְּלוּ צָרוֹת, וְהֵם חֻרְבַּן בַּיִת ב׳ וּבֵיתָר — זֶה לֹא הָיָה אָז בְּעֵת בַּיִת ב׳. וּבְיוֹתֵר קָשֶׁה: לָמָּה שֶׁכָּתְבוּ הַתּוֹסָפוֹת, דִּבּוּר הַמַּתְחִיל "הוֹאִיל" וְכוּלֵי, וְעוֹד וְכוּלֵי, וְתֵרוּצָם — זֶה עִקָּר, כְּמוֹ שֶׁכָּתַב בְּסֵפֶר טוּרֵי אֶבֶן; אִם כֵּן זֶה וַדַּאי לֹא הָיָה בְּעֵת בַּיִת ב׳, וְאַמַּאי הִתְעַנּוּ? וְצָרִיךְ עִיּוּן גָּדוֹל. וּלְפִי זֶה, דִּבְרֵי סֵפֶר טוּרֵי אֶבֶן נְכוֹנוּ, וְדִבְרֵי הָרַמְבַּ״ם צָרִיךְ עִיּוּן. וְדַע: דְּנִרְאֶה לִי, אַף בְּעֵת שְׁמָד, דְּמִתְעַנִּין כָּל צוֹמוֹת — יֵשׁ לוֹמַר דְּלֹא נָפְקֵי אַתַּמּוּז וְטֵבֵת; וְזֶה פָּשׁוּט לְפִי טַעַם שֶׁכָּתַבְתִּי, דְּדָמֵי לְתִשְׁעָה בְּאָב בְּעֵת שָׁלוֹם. מִיהוּ לְפִי טַעְמוֹ שֶׁל סֵפֶר טוּרֵי אֶבֶן — אֵינוֹ כֵּן, עַיֵּן שָׁם; וּלְפִי עֲנִיּוּת דַּעְתִּי — כְּמוֹ שֶׁכָּתַבְתִּי. וְכֵן מַשְׁמַע מִדִּבְרֵי הָרַמְבַּ״ם הַנִּזְכָּר לְעֵיל, דִּלְדִידֵיהּ אַחַר הַחֻרְבָּן הֲוֵי עֵת שְׁמָד, וַאֲפִלּוּ הָכֵי לֹא חָשֵׁיב תַּמּוּז וְטֵבֵת; כֵּן נִרְאֶה לְפִי עֲנִיּוּת דַּעְתִּי:

**וְעַל** אֱלוּל — מִפְּנֵי רֹאשׁ הַשָּׁנָה, שֶׁיֵּדְעוּ מָתַי הִתְחִיל אֱלוּל, וְיַעֲשׂוּ רֹאשׁ הַשָּׁנָה בְּיוֹם ל׳ לֶאֱלוּל, דְּרֹב שָׁנִים אֵין אֱלוּל מְעֻבָּר. וְאַף עַל פִּי שֶׁסָּפֵק הוּא בְּיָדָם, שֶׁמָּא יְעַבְּרוּהוּ — הוֹלְכִים אַחַר רֹב שָׁנִים, הוֹאִיל וְאִי אֶפְשָׁר לָהֶם לָדַעַת; זֶהוּ דַּעַת רַשִׁ״י. אֲבָל תּוֹסָפוֹת, דִּבּוּר הַמַּתְחִיל "וְעַל", וְהָרַמְבַּ״ם סְבִירָא לְהוּ: דְּאִם לֹא נוֹדַע לָהֶם קְבִיעוּת תִּשְׁרֵי — עוֹשִׂים מִסָּפֵק יוֹם ל׳ לֶאֱלוּל וְיוֹם ל״א שְׁנֵיהֶם קֹדֶשׁ; שֶׁתּוֹעֶלֶת יְדִיעַת חֹדֶשׁ אֱלוּל — דְּלֹא יִצְטָרְכוּ ג׳ יָמִים, שֶׁיֵּדְעוּ אֵיזֶהוּ יוֹם ל׳ וְל״א לֶאֱלוּל. וְנִרְאֶה דַּעַת רַשִׁ״י מִמַּה שֶּׁכָּתוּב (כ״ט עַמּוּד ב׳): "הָא מִיקַּלְקַל רֹאשׁ הַשָּׁנָה" וְכוּלֵי; וּלְפֵרוּשׁ רַשִׁ״י נִיחָא, אַךְ לְפֵרוּשׁ הַתּוֹסָפוֹת דָּחוּק לְשׁוֹן "מִיקַּלְקַל". וְעַיֵּן שָׁם בַּתּוֹסָפוֹת, דִּבּוּר הַמַּתְחִיל "וְאַל". וְשׁוּב מָצָאתִי בְּסֵפֶר טוּרֵי אֶבֶן שָׁם מִזֶּה, עַיֵּן שָׁם. וְקֻשְׁיַת תּוֹסָפוֹת, דִּבּוּר הַמַּתְחִיל "וְעַל", אַרַשִׁ״י — יֵשׁ לוֹמַר דְּהָא קָשֶׁה: אֵיךְ סָלְקָא דַּעְתָּךְ דְּרַשִׁ״י יִסְבּוֹר גַּם עַכְשָׁיו רֹאשׁ הַשָּׁנָה יוֹם אֶחָד? וְאַדְּרַבָּא, מַשְׁמַע בְּכַמָּה מְקוֹמוֹת דְּרֹאשׁ הַשָּׁנָה ב׳ יָמִים חָמִיר מִשְּׁאָר יוֹם טוֹב, דִּקְדֻשָּׁה
