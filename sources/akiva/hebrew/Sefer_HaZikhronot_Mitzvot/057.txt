**אַף עַל פִּי** שֶׁאָמַרְנוּ לְעֵיל שֶׁאֵין לָהֶם לְעֵדִים לָבוֹא אֶלָּא מִמַּהֲלַךְ יוֹם וָלַיְלָה, שֶׁלֹּא יָבוֹאוּ לְאַחַר שֶׁעָשׂוּ בֵּית דִּין חֹדֶשׁ מְעֻבָּר, כְּמוֹ שֶׁנִּתְבָּאֵר — מִכָּל מָקוֹם, אִם בָּאוּ עֵדִים רְחוֹקִים ד׳ אוֹ ה׳ יָמִים אַחַר שֶׁעִבְּרוּ — שׁוֹמְעִין דִּבְרֵיהֶם. וְאַף עַל פִּי שֶׁלֹּא יָפֶה עָשׂוּ בְּמַה שֶּׁבָּאוּ לְהָעִיד אַחַר שֶׁעִבְּרוּ בֵּית דִּין, שֶׁאֲפִלּוּ עִבְּרוּ בֵּית דִּין בְּמֵזִיד, שֶׁלֹּא כַּדִּין — הֲוֵי עִבּוּר, כְּמוֹ שֶׁיִּתְבָּאֵר לְהַלָּן; מִכָּל מָקוֹם, הוֹאִיל וְדָחֲקוּ וּבָאוּ, וַאֲפִלּוּ בְּסוֹף הַחֹדֶשׁ, וְהֵעִידוּ שֶׁרָאוּ הַחֹדֶשׁ בִּזְמַנּוֹ — מְקַבְּלִין דִּבְרֵיהֶם (עַיֵּן בְּפֶרֶק ג׳ בְּלֶחֶם מִשְׁנֶה, הֲלָכָה א׳). אֲבָל מְאַיְּמִין עֲלֵיהֶם אִיּוּם גָּדוֹל, וּמַטְרִיחִין אוֹתָן בִּשְׁאֵלוֹת, וּמַטְרִיחִין עֲלֵיהֶם בִּבְדִיקוֹת, וּמְדַקְדְּקִין בָּעֵדוּת, וּמִשְׁתַּדְּלִין בֵּית דִּין שֶׁלֹּא יְקַדְּשׁוּ חֹדֶשׁ זֶה, הוֹאִיל וְיָצָא שְׁמוֹ מְעֻבָּר. וְאִם עָמְדוּ בְּעֵדוּתָן, וְנִמְצֵאת מְכֻוֶּנֶת, וְהָעֵדִים אֲנָשִׁים יְדוּעִים וּנְבוֹנִים, וְנֶחְקְרָה עֵדוּתָן כָּרָאוּי — מְקַדְּשִׁין אוֹתוֹ, וּמוֹנִין אוֹתוֹ חֹדֶשׁ מִיּוֹם ל׳, הוֹאִיל וְנִרְאָה הַיָּרֵחַ בְּלֵילוֹ. וְאִם הֻצְרְכוּ בֵּית דִּין לְהַנִּיחַ חֹדֶשׁ זֶה מְעֻבָּר כְּמוֹ שֶׁהָיָה — מַנִּיחִין אוֹתוֹ. (וְהוּא הַדִּין לְעַבֵּר בִּתְחִלָּה לְצֹרֶךְ, וְשֶׁלֹּא לִשְׁמוֹעַ לָעֵדִים — וּבֶאֱמֶת דָּמֵי מַמָּשׁ; וְלָאו דַּוְקָא נָקַט "לְהַנִּיחַ" וְכוּלֵי, אֶלָּא לְפִי שֶׁבָּא לִכְתּוֹב הַדְּבָרִים עַל הַסֵּדֶר — לָכֵן כָּתַב כֵּן; וְעַיֵּן מַה שֶּׁכָּתַבְתִּי לְעֵיל, וּפָשׁוּט.) וְזֶה הוּא שֶׁאָמְרוּ חֲכָמִים: מְעַבְּרִין אֶת הַחֹדֶשׁ לְצֹרֶךְ. וְיֵשׁ מִן הַחֲכָמִים הַגְּדוֹלִים מִי שֶׁחוֹלֵק בָּזֶה, וְאוֹמֵר: לְעוֹלָם אֵין מְעַבְּרִין אֶת הַחֹדֶשׁ לְצֹרֶךְ; הוֹאִיל וּבָאוּ עֵדִים — מְקַדְּשִׁין, וְאֵין מְאַיְּמִין עֲלֵיהֶם. וְאֵין מַחְלֹקֶת הַחֲכָמִים אֶלָּא בִּשְׁאָר חֳדָשִׁים, חוּץ מִנִּיסָן וְתִשְׁרֵי; אוֹ בְּעֵדֵי נִיסָן וְתִשְׁרֵי שֶׁבָּאוּ אַחַר שֶׁעָבְרוּ הָרְגָלִים, שֶׁכְּבָר נַעֲשָׂה מַה שֶּׁנַּעֲשָׂה, וְעָבַר זְמַן הַקָּרְבָּנוֹת וְהַמּוֹעֲדוֹת. אֲבָל אִם בָּאוּ הָעֵדִים בְּנִיסָן וְתִשְׁרֵי קֹדֶם חֲצִי הַחֹדֶשׁ — מְקַבְּלִין עֵדוּתָן, וְאֵין מְאַיְּמִין עֲלֵיהֶם כְּלָל; שֶׁאֵין מְאַיְּמִין עַל עֵדִים שֶׁהֵעִידוּ עַל הַחֹדֶשׁ שֶׁרָאוּהוּ בִּזְמַנּוֹ, כְּדֵי לְעַבְּרוֹ. אֲבָל מְאַיְּמִין עַל עֵדִים שֶׁנִּתְקַלְקְלָה עֵדוּתָן — וַהֲרֵי הַדָּבָר גְּנַאי שֶׁלֹּא תִּתְקַיֵּם הָעֵדוּת, וְיִתְעַבֵּר הַחֹדֶשׁ — מְאַיְּמִין עֲלֵיהֶן, כְּדֵי שֶׁתִּתְקַיֵּם הָעֵדוּת, וְיִתְקַיֵּם הַחֹדֶשׁ בִּזְמַנּוֹ. וְכֵן אִם בָּאוּ הָעֵדִים לְהָזִים אֶת הָעֵדִים שֶׁרָאוּהוּ בִּזְמַנּוֹ, קֹדֶם שֶׁיְּקַדְּשׁוּהוּ בֵּית דִּין — הֲרֵי אָנוּ מְאַיְּמִין עַל הַמְּזִימִין, עַד שֶׁלֹּא תִּתְקַיֵּם הַהֲזָמָה, וְיִתְקַדֵּשׁ הַחֹדֶשׁ בִּזְמַנּוֹ (מַשְׁמַע דִּבְזֶה אֵין מַחְלֹקֶת, וְהַכֹּל מוֹדִים). וְהוּא הַדִּין: שֶׁאִם רָאוּ צֹרֶךְ בַּדָּבָר, וְהֻצְרְכוּ בֵּית דִּין לְהַנִּיחַ אוֹתוֹ חֹדֶשׁ מְקֻדָּשׁ — מַנִּיחִין, כַּנִּזְכָּר לְעֵיל בִּמְעֻבָּר:

**וְהִנֵּה** כָּל זֶה — דִּבְרֵי הָרַמְבַּ״ם. וְנִרְאֶה לִי דְּנַפְקָא לֵיהּ מֵהַגְּמָרָא (כ׳ עַמּוּד א׳): "אִינִי? וְהָתָנֵי רב״ש" וְכוּלֵי — "לָא קַשְׁיָא: כָּאן לְעַבְּרוֹ, כָּאן לְקַדְּשׁוֹ" וְכוּלֵי; "וְכִי הָא דְּאָמַר רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי: מְאַיְּמִין" וְכוּלֵי. וְהִנֵּה, מֵהָא דְּקָאָמַר "וְכִי הָא" וְכוּלֵי — מַשְׁמַע דְּרַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי מִלְּתָא אַחֲרִיתָא קָאָמַר. וְעוֹד, דְּאִי רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי נַמִּי מִלְּצֹרֶךְ — אִם כֵּן, אַדְּרַבָּא, רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי פָּלִיג עַל זֶה; דְּמִתְּחִלָּה קָאָמַר מְעַבְּרִין לְצֹרֶךְ, וְאֵין צָרִיךְ לְאַיֵּם כְּלָל; וְרַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי
