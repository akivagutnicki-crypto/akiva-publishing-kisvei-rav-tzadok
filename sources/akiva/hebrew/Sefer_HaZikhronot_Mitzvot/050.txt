~ יָמִים — מִכָּל מָקוֹם נַפְקָא מִינָּהּ לְהוּ לָדַעַת יוֹם טוֹב אַחֲרוֹן לְעִנְיַן מֵת, וְכַהַאי גַּוְנָא. דְּמֵהַאי טַעְמָא הוֹלְכִין שְׁלוּחֵי נִיסָן אַף שֶׁאֵין מַגִּיעִין שְׁלוּחֵי תִּשְׁרֵי, כְּמוֹ שֶׁיִּתְבָּאֵר. אַךְ בְּסֻכּוֹת — לְפִי מַה שֶּׁכָּתוּב בַּגְּמָרָא, דְּאָזְלִינַן אַחַר רֹב, אֱלוּל אֵין מְעֻבָּר לְעִנְיַן מֵת וְכַהַאי גַּוְנָא לְהָקֵל; אִם כֵּן אֵין צָרִיךְ לֵילֵךְ עַד יוֹם טוֹב אַחֲרוֹן אֶלָּא בִּמְעֻבָּר, עַד כָּאן. וְלִי נִרְאֶה: גַּם בְּפֶסַח — וַדַּאי רֹב אֲדָר אֵין מְעֻבָּר, אַף לְמַאן דְּאָמַר פְּעָמִים מָלֵא פְּעָמִים חָסֵר, מִכָּל מָקוֹם רֹב פְּעָמִים חָסֵר; מִכָּל שֶׁכֵּן דֶּאֱלוּל, דְּיֵשׁ לָחוּשׁ גַּם כֵּן לְקִלְקוּל מוֹעֲדוֹת בְּבָבֶל, עַיֵּן הֵיטֵב בַּגְּמָרָא (י״ט עַמּוּד ב׳), וְתָבִין; וְעַיֵּן מַה שֶּׁכָּתַבְתִּי לְעֵיל בְּעִנְיַן ב׳ הַיָּמִים. וְאִם כֵּן קָשֶׁה: מַאי טַעְמָא אָזְלֵי שְׁלוּחֵי נִיסָן בְּמָקוֹם שֶׁאֵין מַגִּיעִים שְׁלוּחֵי תִּשְׁרֵי? וְעַל כָּרְחֲךָ, דְּאַף דְּרֹב אֵינוֹ מְעֻבָּר — מִכָּל מָקוֹם מֻכְרָחִי לְהַחְמִיר. אִי נַמִּי, אַף דְּלֹא יַחְמִירוּ — מִכָּל מָקוֹם אָזְלֵי שְׁלוּחֵי, אַף אַחַר, לְהָקֵל בְּאֵיזֶה דְּבָרִים. וְהוּא הַדִּין לְיוֹם טוֹב אַחֲרוֹן דְּפֶסַח וּדְסֻכּוֹת — דִּצְרִיךְ. וּלְפִי עֲנִיּוּת דַּעְתִּי, לִכְאוֹרָה בְּפֶסַח אַף אַחַר פֶּסַח צָרִיךְ לְמֵיזַל, מִשּׁוּם שָׁבוּעוֹת, דְּהוּא נ׳ יוֹם מִפֶּסַח. וּבְהָא דְּקָאָמְרִינַן (כ״א רֵישׁ עַמּוּד א׳): "תְּרֵי יַרְחֵי חַסִּירֵי" וְכוּלֵי — מוּכָח הָא לָאו הָכֵי לֵית לֵיהּ קָלָא, אַף בַּחֹדֶשׁ וּמֶחֱצָה, עַיֵּן שָׁם. מִיהוּ יֵשׁ לוֹמַר: עַד שָׁבוּעוֹת, דַּהֲוֵי יוֹתֵר מִב׳ חֳדָשִׁים — אִית לֵיהּ קָלָא, אַף אֶחָד יַרְחָא; וְצָרִיךְ לְעַיֵּן:

**וְעַל** אָב — מִפְּנֵי הַתַּעֲנִית שֶׁל תִּשְׁעָה בְּאָב. וּבִזְמַן שֶׁיֵּשׁ שָׁלוֹם, שֶׁאֵין מִתְעַנִּים, כְּמוֹ שֶׁיִּתְבָּאֵר בִּמְקוֹמוֹ שֶׁמִּצְוָה לִשְׂמוֹחַ בּוֹ — אֵין שְׁלוּחִים יוֹצְאִים עַל אָב. אַף עַל פִּי שֶׁמִּצְוָה לִשְׂמוֹחַ בּוֹ — מִכָּל מָקוֹם בִּשְׁבִיל זֶה אֵין שְׁלוּחִים יוֹצְאִים; כְּמוֹ שֶׁלֹּא יָצְאוּ שְׁלוּחִים בִּזְמַן שֶׁהָיְתָה מְגִלַּת תַּעֲנִית עַל כָּל יָמִים הַכְּתוּבִים בִּמְגִלַּת תַּעֲנִית, דְּלָא לְהִתְעַנָּאָה וּדְלָא לְמִסְפַּד; כֵּיוָן דְּמִדְּחוּ בְּלָאו הָכֵי כְּשֶׁאֵין שָׁלוֹם, וְכֵן עַכְשָׁיו. מַה שֶּׁאֵין כֵּן תִּשְׁעָה בְּאָב, דְּהוּכְפְּלוּ בּוֹ צָרוֹת, כִּדְקָאָמְרִינַן בַּגְּמָרָא; וְכֵן חֲנֻכָּה וּפוּרִים, דְּאַף עַכְשָׁיו לֹא בָּטְלוּ, כְּמוֹ כָּל מְגִלַּת תַּעֲנִית. כֵּן נִרְאֶה לְפִי עֲנִיּוּת דַּעְתִּי חִלּוּק זֶה; וּרְאָיָה מִלְּשׁוֹן הַמִּשְׁנָה: "בְּאָב מִפְּנֵי הַתַּעֲנִית", וְלֹא קָאָמַר "עַל תִּשְׁעָה בְּאָב" — אֶלָּא דַּוְקָא מִפְּנֵי הַתַּעֲנִית קָאָמַר, לֹא בִּזְמַן שָׁלוֹם. וְרָאִיתִי בְּסֵפֶר טוּרֵי אֶבֶן (י״ח עַמּוּד ב׳), דִּבּוּר הַמַּתְחִיל "בִּזְמַן שֶׁבֵּית הַמִּקְדָּשׁ קַיָּם", כָּתַב גַּם כֵּן כֵּן; רַק דְּכָתַב דְּכֵן בַּחֲנֻכָּה וּפוּרִים, עַיֵּן שָׁם; וְעַיֵּן מַה שֶּׁכָּתַבְתִּי לְהַלָּן. וּרְאָיָתוֹ — דְּאִם כֵּן, בִּזְמַן שֶׁבֵּית הַמִּקְדָּשׁ קַיָּם, לִיפּוֹק עַל תַּמּוּז וְטֵבֵת, מִפְּנֵי הָאִסּוּר תַּעֲנִית. וְעַיֵּן עוֹד שָׁם, דִּבּוּר הַמַּתְחִיל "עַל ו׳", וְדִבּוּר הַמַּתְחִיל "עַל אָב", וְדִבּוּר הַמַּתְחִיל "וּכְשֶׁהָיָה", כָּתַב כֵּן, וּדְלָא גָּרְסִינַן בְּמַתְנִיתִין "אַף עַל אִיָּר" — דִּבְכָל חַד אִיכָּא ו׳ חֳדָשִׁים בִּזְמַן שֶׁבֵּית הַמִּקְדָּשׁ קַיָּם, אַפֵּיק אָב וְעָיֵיל אִיָּר, עַיֵּן שָׁם. וְהִנֵּה לֹא עִיֵּן בְּפֵרוּשׁ הַמִּשְׁנָיוֹת לְהָרַמְבַּ״ם, דְּשָׁם כָּתַב דִּבְזְמַן בַּיִת שֵׁנִי לֹא הָיָה שָׁלוֹם וְלֹא שְׁמָד, עַיֵּן שָׁם (וְהַיְנוּ דְּהוּקְשָׁה לוֹ: דְּאִם כֵּן מַשְׁמַע בְּדוֹכְתֵי טוּבָא דְּבִימֵי תַּנָּאִים הָיוּ שְׁמָדוֹת וּגְזֵרוֹת. וְעַיֵּן בַּתַּשְׁבֵּ״ץ (חֵלֶק ב׳ סוֹף סִימָן
