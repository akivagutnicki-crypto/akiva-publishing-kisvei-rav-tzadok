~ וְתִתְחַדֵּשׁ לֹא פָּחוֹת וְלֹא יוֹתֵר מִכ״ט יוֹם, י״ב שָׁעוֹת, תשצ״ג; וְלָכֵן, בְּהִתְקַבֵּץ ב׳ פְּעָמִים הַי״ב שָׁעוֹת, יִהְיֶה יוֹם אֶחָד שָׁלֵם, וְיִהְיֶה הַחֹדֶשׁ מִל׳ יוֹם; וְיִתְבָּאֵר זֶה לְהַלָּן:

**וְאֵין** נִזְקָקִים לַיָּרֵחַ בְּלֵיל ל״א — בֵּין שֶׁנִּרְאָה בֵּין שֶׁלֹּא נִרְאָה — שֶׁאֵין לְךָ חֹדֶשׁ לְבָנָה יוֹתֵר מִל׳ יוֹם, וְלֹא פָּחוֹת מִכ״ט יוֹם. וְעַיֵּן בְּרֹאשׁ הַשָּׁנָה (דַּף כ׳ עַמּוּד א׳), מוּכָח דְּגַם בְּל״א מִצְוָה לְקַדֵּשׁ עַל רְאִיָּה. וְחֹדֶשׁ שֶׁיִּהְיֶה כ״ט יוֹם, וְיוֹם ל׳ רֹאשׁ חֹדֶשׁ — נִקְרָא "חֹדֶשׁ חָסֵר"; וְאִם לֹא יֵרָאֶה הַיָּרֵחַ בְּלֵיל ל״א, וְיִהְיֶה הַחֹדֶשׁ שֶׁעָבַר ל׳ יוֹם — נִקְרָא "חֹדֶשׁ מְעֻבָּר" וְ"חֹדֶשׁ מָלֵא". וְיָרֵחַ שֶׁנִּרְאָה בְּלֵיל ל׳ — הוּא הַנִּקְרָא "נִרְאָה בִּזְמַנּוֹ"; וְאִם נִרְאָה בְּל״א — נִקְרָא "בְּלֵיל עִבּוּרוֹ". וּמַהוּ הַנִּקְרָא "מְעֻבָּר" — עַיֵּן בְּרַשִׁ״י רֹאשׁ הַשָּׁנָה (כ״ב עַמּוּד ב׳) דִּבּוּר הַמַּתְחִיל "לְאוֹר", וּבִפְסָחִים (ב׳ עַמּוּד ב׳) רַשִׁ״י דִּבּוּר הַמַּתְחִיל "לְאוֹר", יְעֻיַּן שָׁם הֵיטֵב; וְעַיֵּן עוֹד סַנְהֶדְרִין (י׳ עַמּוּד ב׳) רַשִׁ״י דִּבּוּר הַמַּתְחִיל "יוֹם" וְדִבּוּר הַמַּתְחִיל "אוֹר". וְזֶהוּ עִבּוּר שָׁנָה, שֶׁהַבֵּית דִּין מְחַשְּׁבִין אִם צָרִיךְ לְעַבֵּר הַשָּׁנָה לַעֲשׂוֹתָהּ י״ג חֳדָשִׁים. וְקִדּוּשׁ הַחֹדֶשׁ — שֶׁהַבֵּית דִּין מְחַשְּׁבִין כְּדֶרֶךְ שֶׁחוֹשְׁבִין הָאִצְטַגְנִינִים, לֵידַע אִם אֶפְשָׁר לַיָּרֵחַ שֶׁיֵּרָאֶה בִּזְמַנּוֹ, בְּלֵיל ל׳, אוֹ אִי אֶפְשָׁר. אִם אֶפְשָׁר שֶׁיֵּרָאֶה — יוֹשְׁבִין וּמְצַפִּין כָּל יוֹם ל׳ לְעֵדִים: אִם בָּאוּ עֵדִים, וּדְרָשׁוּם וַחֲקָרוּם, וְנֶאֶמְנוּ דִּבְרֵיהֶם — מְקַדְּשִׁין אוֹתוֹ יוֹם שֶׁיִּהְיֶה רֹאשׁ חֹדֶשׁ; וְאִם לֹא נִרְאָה, וְלֹא בָּאוּ עֵדִים — מַשְׁלִימִין ל׳, וְיִהְיֶה חֹדֶשׁ מְעֻבָּר. וְאִם יָדְעוּ בְּחֶשְׁבּוֹן שֶׁאִי אֶפְשָׁר שֶׁיֵּרָאֶה — אֵין יוֹשְׁבִים יוֹם ל׳, וְאֵין מְצַפִּין לְעֵדִים; כְּגוֹן שֶׁהִגִּיעַ לָהֶם בְּחֶשְׁבּוֹנָם שֶׁמּוֹלַד הַלְּבָנָה וּתְחִלַּת חִדּוּשָׁהּ כַּחֲצוֹת הַיּוֹם, וּכְבָר אָמַרְנוּ שֶׁלִּבְנֵי מַעֲרָב מִתְכַּסֵּית שִׁשָּׁה שָׁעוֹת וְלִבְנֵי מִזְרָח י״ח שָׁעוֹת; וְהֵם בַּמַּעֲרָב, וּבָאוּ עֵדִים וְאָמְרוּ שֶׁרָאוּהָ קֹדֶם שִׁשָּׁה שָׁעוֹת, אוֹ בַּמִּזְרָח קֹדֶם י״ח שָׁעוֹת — יוֹדְעִים שֶׁהֵם עֵדֵי שֶׁקֶר, אוֹ שֶׁנִּרְאָה לָהֶם דְּמוּת לְבָנָה מִן הֶעָבִים, וְאֵינָהּ לְבָנָה בֶּאֱמֶת לְקַדְּשָׁהּ:

**כֵּיצַד** מְקַדְּשִׁין הַחֹדֶשׁ? כָּל מִי שֶׁרָאוּי לְהָעִיד, שֶׁרָאָה הַיָּרֵחַ — בָּא לְבֵית דִּין. שֶׁאֵין הַדָּבָר מָסוּר לְכָל אֶחָד, כְּמוֹ הַשַּׁבָּת, שֶׁכָּל אֶחָד מוֹנֶה שִׁשָּׁה וּמְקַדֵּשׁ שְׁבִיעִי; אֶלָּא לְבֵית דִּין מָסוּר, שֶׁנֶּאֱמַר: "הַחֹדֶשׁ הַזֶּה לָכֶם" — עֵדוּת זוֹ תְּהֵא מְסוּרָה לָכֶם. וְחָצֵר גְּדוֹלָה הָיְתָה בִּירוּשָׁלַיִם, וּמְרֻוַּחַת, וְהָיְתָה נִקְרֵאת "בֵּית יַעְזֵק" (שֶׁהָיְתָה מְגֻדֶּרֶת, כְּמוֹ שֶׁנֶּאֱמַר "וַיְעַזְּקֵהוּ" וְגוֹמֵר — כֵּן כָּתְבוּ רַשִׁ״י וְתוֹסְפוֹת יוֹם טוֹב. אֲבָל בִּירוּשַׁלְמִי אִיתָא שֶׁשָּׁם הָיוּ עוֹזְקִין אֶת הַהֲלָכָה, הַמד״א "וַיְעַזְּקֵהוּ" וְגוֹמֵר); וּלְשָׁם כָּל הָעֵדִים מִתְכַּנְּסִים. וְיֵשׁ אוֹמְרִים: דַּוְקָא כְּשֶׁבָּאוּ בְּשַׁבָּת וְחִלְּלוּ שַׁבָּת — כּוֹנְסוּם לְשָׁם (רַשִׁ״י). וְאֶפְשָׁר עַכְשָׁיו, אַחַר תַּקָּנַת רַבָּן גַּמְלִיאֵל, אַף בְּשַׁבָּת אֵין צָרִיךְ לְכַנְּסָם. וְהָכֵי מִסְתַּבְּרָא, דְּחָצֵר זֶה לָמָּה? יָבוֹאוּ לִמְקוֹם מוֹשַׁב בֵּית דִּין. וְהָיוּ עוֹשִׂין
