~ לְהוֹסִיף פָּחוֹת מִג׳ חוּמָשִׁים (נִרְאֶה דְּצָרִיךְ לוֹמַר שְׁנִיִּים — הַמַּגִּיהַּ); וְכָל שֶׁכֵּן בְּק׳ יָמִים — שֶׁאֵין צָרִיךְ לְהוֹסִיף כְּלָל. וְאִם כֵּן, צָרִיךְ עִיּוּן: מַדּוּעַ הוֹסִיף הָרַמְבַּ״ם ג׳ חוּמָשִׁין (נִרְאֶה דְּצָרִיךְ לוֹמַר שְׁנִיִּים — הַמַּגִּיהַּ)? וְאוּלַי רוֹצֶה לוֹמַר שֶׁקָּרוֹב לְאִצְטַגְנִינוּת יוֹתֵר מִן הָרִאשׁוֹן; וְכֵן מְדֻיָּק לִכְאוֹרָה בִּלְשׁוֹנוֹ, עַיֵּן שָׁם:

**אֲבָל** מִכָּל מָקוֹם דָּחוּק, דְּמֶרְחָק רַב מִן תתי״ו — שֶׁהוּא דַּעַת אַלְבַּתָּאנִי — לְתתקצ״ג. וְתוּ צָרִיךְ עִיּוּן: כֵּיוָן דְּבֵית דִּין עַל פִּי רְאִיָּה חָשְׁבוּ חֶשְׁבּוֹנוֹ שֶׁל רַב אַדָּא — מַדּוּעַ הוּא לֹא כָּתַב כֵּן, כֵּיוָן שֶׁהוּא בָּא לִכְתּוֹב הַדִּינִים לְבֵית דִּין? וְצָרִיךְ עִיּוּן: מַדּוּעַ לֹא חָשַׁב בְּחֶשְׁבּוֹנוֹ שֶׁל רַב אַדָּא, וְיַגִּיעַ מַהֲלַךְ הַשֶּׁמֶשׁ מַעֲלָה אַחַת בְּיוֹם אֶחָד וְשע״ז חֲלָקִים וְנ״ח רְגָעִים וְי״א חֶלְקֵי י״ח בְּרֶגַע; כִּי זֶהוּ הַנִּתּוֹסָף עַל ש״ס יָמִים — ה׳ יָמִים, ה׳ שָׁעוֹת, תתקצ״ז חֲלָקִים, מ״ח רֶגַע — וּכְשֶׁיֵּחָלֵק זֶה לְכָל הַש״ס יָמִים, יַגִּיעַ לְכָל יוֹם חֵלֶק כַּנִּזְכָּר לְעֵיל; שֶׁבְּיוֹם שָׁלֵם, עִם חֵלֶק הַנִּזְכָּר — מַהֲלַךְ הַשֶּׁמֶשׁ מַעֲלָה אַחַת. וְאִם בָּאנוּ לָדַעַת מַהֲלַךְ הַשֶּׁמֶשׁ בְּיוֹם אֶחָד, בִּלְעֲדֵי חֵלֶק הַנִּזְכָּר — כַּמָּה? הִנֵּה נַעֲשֶׂה מִן שע״ז חֲלָקִים, נ״ח רֶגַע, וְי״א חֶלְקֵי י״ח בְּרֶגַע — יָמִים שְׁלֵמִים. כֵּיצַד? שֶׁנִּכְפְּלֵם בְּי״ח, לַעֲשׂוֹת מִן חֶלְקֵי הָרֶגַע רְגָעִים שְׁלֵמִים; וְאַחַר כָּךְ בְּע״ו, כְּדֵי לַעֲשׂוֹת מִן הָרְגָעִים חֲלָקִים; וְאַחַר כָּךְ בְּאֶלֶף וּפ׳, כְּדֵי שֶׁיֵּעָשׂוּ הַחֲלָקִים שָׁעוֹת; וְאַחַר כָּךְ בְּכ״ד, לַעֲשׂוֹת הַשָּׁעוֹת יָמִים; וְיַגִּיעוּ מִסְפַּר הַיָּמִים תקי״ו אֶלֶף וְתשצ״א, וּמִסְפַּר כְּפוּלַת חֵלֶק הַנִּזְכָּר — ל״ה מִילְיָאן תנ״ח אֶלֶף וְתק״ס. וְנִמְצָא חֵלֶק הַנִּזְכָּר, הַנִּתּוֹסָף עַל הַיּוֹם לְתַשְׁלוּם מַעֲלָה אַחַת — הוּא א׳ מִל״ה מִילְיָאן וְתנ״ח אֶלֶף וְתק״ס, מִן תקי״ו אֶלֶף וְתשצ״א יָמִים; אוֹ תקי״ו אֶלֶף וְתשצ״א חֶלְקֵי ל״ה מִילְיָאן תנ״ח אֶלֶף וְתק״ס בַּיּוֹם (*). וְנִמְצָא הַיּוֹם — ל״ה מִילְיָאן תנ״ח אֶלֶף וְתק״ס חֶלְקֵי ל״ה מִילְיָאן תנ״ח אֶלֶף וְתק״ס, וְהַנּוֹסָף — תקי״ו אֶלֶף וְתשצ״א חֲלָקִים הַנִּזְכָּרִים; וּקְבוּצָתָם יַחַד הוּא הִלּוּךְ מַעֲלָה אַחַת. וְלָכֵן נַחֲלֹק מַעֲלָה אַחַת עַל קְבוּצַת ב׳ מִסְפָּרִים אֵלּוּ — וְהוּא ל״ה מִילְיָאן תתקע״ה אֶלֶף וְשנ״א:

(*) אָמַר הַמַּגִּיהַּ: מִן דִּבּוּר הַמַּתְחִיל "הִנֵּה נַעֲשֶׂה מִן שע״ז חֲלָקִים" וְכוּלֵי, עַד "וְנִמְצָא" וְכוּלֵי — לְשׁוֹנוֹ אֵינוֹ מוּבָן לִי. אַךְ הַמְכֻוָּן: שֶׁנִּכְפֹּל הַשע״ז חֲלָקִים בְּע״ו, לַעֲשׂוֹתָם רְגָעִים; וְאַחַר כָּךְ נִכְפֹּל כָּל הָרְגָעִים בְּי״ח, לַעֲשׂוֹת מִן כֻּלָּם חֶלְקֵי י״ח; וְיַגִּיעַ הַמִּסְפָּר תקי״ו אֶלֶף וְתשצ״א. וְכֵן נַעֲשֶׂה מֵהַיּוֹם שָׁלֵם: נִכְפֹּל הַכ״ד שָׁעוֹת בְּאֶלֶף וּפ׳, לַעֲשׂוֹת מֵהַשָּׁעוֹת חֲלָקִים; וְאַחַר כָּךְ בְּע״ו, לַעֲשׂוֹתָם רְגָעִים; וְאַחַר כָּךְ בְּי״ח, לַעֲשׂוֹתָם חֶלְקֵי י״ח, כַּנִּזְכָּר לְעֵיל; וְיַגִּיעַ הַמִּסְפָּר ל״ה מִילְיָאן תנ״ח אֶלֶף תק״ס:
