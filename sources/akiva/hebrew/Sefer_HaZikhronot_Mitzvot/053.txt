~ צָרִיךְ ב׳ יָמִים — דַּחֲזָקָה שֶׁאֵין בֵּית דִּין מִתְעַצְּלִים. וּפֵרוּשׁוֹ כַּנִּזְכָּר לְעֵיל: דְּרֹב אֱלוּל אֵין מְעֻבָּר, וְאֵין בֵּית דִּין מִתְעַצְּלִים לִגְבּוֹת עֵדוּת וּלְקַדְּשׁוֹ בִּזְמַנּוֹ; וְכַהַאי גַּוְנָא בִּמְנָחוֹת (דַּף ס״ח עַמּוּד א׳) לְעִנְיַן עֹמֶר: חֲזָקָה אֵין בֵּית דִּין מִתְעַצֵּל, עַיֵּן שָׁם. (וּבְקָרְבַּן הָעֵדָה, בְּפֵרוּשׁוֹ לִירוּשַׁלְמִי, עַיֵּן שָׁם: פֵּרֵשׁ שֶׁאֵין בֵּית דִּין מִתְעַצְּלִים לִשְׁלוֹחַ שְׁלוּחִים אִם לֹא נִרְאָה בִּזְמַנּוֹ. וְלֹא דַּק, דְּהָא מַשְׁמַע דְּבָבֶל הוּא, דְּאֵין שְׁלוּחִים מַגִּיעִים; דְּאִי שְׁלוּחִים מַגִּיעִים — מַאי סָלְקָא דַּעְתָּךְ לָחוּשׁ כְּלָל? אִם כֵּן, מָה הוֹעִילוּ שְׁלוּחִים, וּבְכָל אֶרֶץ יִשְׂרָאֵל יַעֲשׂוּ ב׳ יָמִים טוֹבִים? וְעוֹד, מַאי שְׁנָא יוֹם הַכִּפּוּרִים דְּאִי לָאו אִיהוּ שֶׁלֹּא יָחוּשׁוּ, וְאֵיךְ עָבְדוּ שְׁאָר יָמִים טוֹבִים? וְעוֹד, מַאי קָאָמַר "אִם לֹא נִרְאָה בִּזְמַנּוֹ"? הָא אַף אִי נִרְאָה בִּזְמַנּוֹ שׁוֹלְחִים שְׁלוּחִים, וְהֵם מַקְדִּימִים לָבוֹא; וְאִם כֵּן, אַדְּרַבָּא, רְאָיָה מִדְּלָא בָּאוּ — שֶׁמִּכָּל מָקוֹם דְּלֹא נִקְבַּע בִּזְמַנּוֹ. אֲבָל כָּל דְּבָרָיו תְּמוּהִים, וְהַבָּרוּר כַּנִּזְכָּר לְעֵיל.) וְאַף עַל גַּב דְּלֹא סָמְכִינַן אַשְּׁאָר יָמִים טוֹבִים בְּבָבֶל אַהָךְ רֻבָּא, כַּנִּזְכָּר לְעֵיל — יוֹם הַכִּפּוּרִים, דְּסַכַּנְתָּא לְהִתְעַנּוֹת ב׳ יָמִים, סָמְכִינַן אַחֲזָקָה זוֹ. וְאִם כֵּן, דְּעִקַּר טַעְמָא הוּא מִכֹּחַ רֻבָּא הַנִּזְכָּר לְעֵיל — אִם כֵּן זֶהוּ נָכוֹן תָּמִיד; אֲבָל בְּאד״ו, דִּסְפֵק גָּמוּר הוּא, כַּנִּזְכָּר לְעֵיל, וְלֵיכָּא רֻבָּא — שַׁפִּיר א״ל יוֹם הַכִּפּוּרִים צָרִיךְ שְׁנֵי יָמִים, וְאָתֵי שַׁפִּיר דִּבְרֵי רַשִׁ״י:

**וְעַל** תִּשְׁרֵי — מִפְּנֵי תַּקָּנַת הֶחָג. וְנִזְכָּר לְעֵיל דִּצְרִיךְ עַד יוֹם טוֹב אַחֲרוֹן תָּמִיד, דְּלֹא כְּטוּרֵי אֶבֶן. וְעַיֵּן בְּסֵפֶר טוּרֵי אֶבֶן (י״ט עַמּוּד ב׳), דִּבּוּר הַמַּתְחִיל "כַּמָּה עִבּוּר", הִקְשָׁה: יַתְקִינוּ לִהְיוֹת תָּמִיד אֱלוּל חָסֵר, וְלֹא יִצְטָרֵךְ שְׁלוּחִים; וְכָתַב: אַתִּשְׁרֵי יֵשׁ לוֹמַר — פְּעָמִים מִשּׁוּם יַרְקָא אוֹ מִתַּיָּא; אֲבָל מִנִּיסָן קָשֶׁה, וְהִנִּיחַ בְּקֻשְׁיָא. וּבֶאֱמֶת גַּם אַתִּשְׁרֵי לֹא נִיחָא, דְּיִתְקְנוּ אד״ו תָּמִיד מָלֵא, עַיֵּן שָׁם בַּתּוֹסָפוֹת דִּבּוּר הַמַּתְחִיל "וְעַל". אֲבָל הַפָּשׁוּט, כִּדְאִיתָא בַּגְּמָרָא וְתוֹסָפוֹת שָׁם: דְּמִצְוָה לְקַדֵּשׁ עַל פִּי הָרְאִיָּה, וְיִתְקַלְקְלוּ הַמּוֹעֲדוֹת כְּשֶׁלֹּא נְקַדֵּשׁ עַל פִּי רְאִיָּה; וְהָכֵי נַמִּי כֵּן, וְעָבְדִינַן שְׁלוּחִים, מַה שֶּׁאֵין כֵּן חֹדֶשׁ עִבּוּר, דְּשָׁם כַּמָּה דְּבָעִינַן מוֹסְפִינַן, וּפָשׁוּט:

**וְעַל** כִּסְלֵו — מִפְּנֵי חֲנֻכָּה. וּפָשׁוּט דְּיֵלְכוּ עַד סוֹפָהּ שֶׁל חֲנֻכָּה. (וְעַיֵּן בְּסֵפֶר טוּרֵי אֶבֶן (י״ח עַמּוּד ב׳), דִּבּוּר הַמַּתְחִיל "בִּזְמַן שֶׁבֵּית הַמִּקְדָּשׁ קַיָּם", לִדְבָרָיו שָׁם דִּבַּחֲנֻכָּה וּפוּרִים יַחְמִירוּ מִסָּפֵק, עַיֵּן שָׁם הֵיטֵב; אִם כֵּן, אִם קִדְּשׁוּ בִּזְמַנּוֹ — צָרִיךְ לֵילֵךְ עַד לֵיל ט׳ דַּחֲנֻכָּה, וְיוֹמוֹ, לִמְנוֹעַ מִבִּרְכַּת נֵר חֲנֻכָּה, וּקְרִיאַת הַתּוֹרָה וְהַלֵּל בַּט׳, לִסְפֵיקָא דִּידְהוּ דִּמְעֻבָּר; עַיֵּן שָׁם, וְתָבִין. מִיהוּ לְקַמָּן יִתְבָּאֵר, דִּלְפִי עֲנִיּוּת דַּעְתִּי אֵינוֹ כֵּן):

**וְעַל** אֲדָר — מִפְּנֵי פּוּרִים. (וְעַיֵּן בְּלֶחֶם מִשְׁנֶה, הִקְשָׁה: מִפְּנֵי מָה לֹא אָמַר "עַל אֲדָר ב׳ מִפְּנֵי הַפּוּרִים"? וְהִנִּיחַ בְּצָרִיךְ עִיּוּן. וְלֹא קָשֶׁה מִידֵּי, דְּתוֹסָפוֹת (י״ט עַמּוּד ב׳) דִּבּוּר הַמַּתְחִיל
