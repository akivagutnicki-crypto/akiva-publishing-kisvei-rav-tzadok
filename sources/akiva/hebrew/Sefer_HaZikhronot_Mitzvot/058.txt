~ קָאָמַר: "מְאַיְּמִין" — הָא אִם אִיְּמוּ וְלֹא הוֹעִיל, מַשְׁמַע דְּאֵין מְעַבְּרִים; וּכְמוֹ שֶׁכָּתַב הָרַמְבַּ״ם. לְכָךְ מְפָרֵשׁ הָרַמְבַּ״ם דְּרַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי מִלְּתָא אַחֲרִיתָא קָאָמַר, וְלָאו אַלְּצֹרֶךְ קָאֵי, אֶלָּא כְּשֶׁכְּבָר עִבְּרוּ אוֹ כְּבָר קִדְּשׁוּ. וּמַשְׁמַע לֵיהּ לְהָרַמְבַּ״ם עוֹד הָכֵי, מִדְּקָאָמַר "עֵדִים" בְּלְקַדְּשׁוֹ; וְאִי לְכַתְּחִלָּה לְקַדְּשׁוֹ לְצֹרֶךְ קָאָמַר — "עֵדִים" מַאן דְּכַר שְׁמַיְיהוּ? וְאֵיזֶה עֵדִים נִתְפּוֹס לְאַיֵּם עֲלֵיהֶם? הֲלֹא כָּל הָעָם לֹא רָאוּ, וְעַל לֹא רָאוּ — אֵיךְ שַׁיָּךְ עֵדִים, וְעַל מִי נְאַיֵּם? לְכָךְ מְפָרֵשׁ דְּמַיְירֵי כְּשֶׁכְּבָר הֵעִידוּ עֵדִים, וּנְאַיֵּם עֲלֵיהֶם לְהַחֲזִיק עֵדוּתָן, וְשֶׁלֹּא יְזוֹמוּם; וּבָזֶה רַק מְאַיְּמִין. אֲבָל אִם אִיְּמוּ וְלֹא הוֹעִיל — חוֹזְרִין וּמוֹנִין כָּרָאוּי, כַּנִּזְכָּר לְעֵיל. וּמְדַמֵּי לָהּ גְּמָרָא לְעִבּוּר לְצֹרֶךְ, שֶׁלֹּא בְּאִיּוּם; וּכְמוֹ דְּמַפְלִיג רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי כָּאן בֵּין קִדּוּשׁ לְעִבּוּר — הָכֵי נַמִּי שָׁם:

**וְהִנֵּה** בַּסּוֹף אַסֵּיק: רָבָא אָמַר — הָא דְּתָנֵי רַבָּה בַּר שְׁמוּאֵל, אֲחֵרִים הוּא, וְכוּלֵי, עַיֵּן שָׁם. וְרָבָא בָּא לְיַשֵּׁב דִּבְרֵי רַבִּי יְהוּדָה נְשִׂיאָה אַחַר רַבָּה בַּר שְׁמוּאֵל; אָתֵי דְּעוּלָּא — כְּבָר יִשֵּׁב רָבָא בְּעַצְמוֹ לְעֵיל: כָּאן לְקַדְּשׁוֹ, כָּאן לְעַבְּרוֹ. וּלְרָבָא, אִם כֵּן לְדִידָן — דְּלָא קַיְימָא לַן כַּאֲחֵרִים — בֵּין מְעַבְּרִין בֵּין מְקַדְּשִׁין לְצֹרֶךְ; וְהוּא הַדִּין לְאַיֵּם — דְּדָמֵי לִלְצֹרֶךְ, כַּנִּזְכָּר לְעֵיל; וּכְדִבְרֵי רַבִּי יְהוּדָה נְשִׂיאָה, דְּגַם לְקַדֵּשׁ מְאַיְּמִין, וְלֵית לֵיהּ תֵּרוּץ אַבָּיֵי. וְקַיְימָא לַן אַבָּיֵי וְרָבָא — הֲלָכָה כְּרָבָא; אִם כֵּן הֲלָכָה כְּרָבָא, דִּמְאַיְּמִין וְכֵן לְצֹרֶךְ עָבְדִינַן בְּכָל עִנְיָן, בֵּין לְקַדְּשׁוֹ בֵּין לְעַבְּרוֹ. אַךְ הִנֵּה בַּגְּמָרָא: אָמַר בָּתַר הָכֵי רַב דִּימִי מִנְּהַרְדְּעָא: מַתְנֵי אִיפְּכָא וְכוּלֵי, עַיֵּן שָׁם. וְהִנֵּה, לְדִבְרֵי רַב דִּימִי אַתְיָין: תָּנֵי רַבָּה בַּר שְׁמוּאֵל, וְדִבְרֵי רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי, וְדִבְרֵי רַבִּי יְהוּדָה נְשִׂיאָה — עַל נָכוֹן; וּפְלִיגֵי אַדְּעוּלָּא, וְאֵין מְאַיְּמִין, וּלְצֹרֶךְ אֶלָּא לְקַדְּשׁוֹ — לֹא לְעַבְּרוֹ; וּלְדִידֵיהּ וַדַּאי כֵּן עִבְּרוּהוּ. וְזֶהוּ "יֵשׁ מֵהַחֲכָמִים הַגְּדוֹלִים" וְכוּלֵי — וְהוּא רַב דִּימִי הַנִּזְכָּר. וּבִלְקַדְּשׁוֹ — כּוּלֵּי עָלְמָא מוֹדוּ דִּמְאַיְּמִין, וְכֵן לְצֹרֶךְ, דִּכְהָדָדֵי נִינְהוּ, כַּנִּזְכָּר לְעֵיל; כִּי פְּלִיגֵי — בִּלְעַבְּרוֹ לְאִיּוּם וּלְצֹרֶךְ. וּמַה שֶּׁכָּתוּב דִּבְנִיסָן וְתִשְׁרֵי כּוּלֵּי עָלְמָא מוֹדוּ — הִנֵּה זֶה כָּתַב בְּ"יֵרָאֶה לִי" וְכוּלֵי; וְהָרַמְבַּ״ם מִדְּנַפְשֵׁיהּ קָאָמַר לָהּ. וְהִנֵּה בַּגְּמָרָא אָמַר אַבָּיֵי כַּהַאי גַּוְנָא לְעִנְיַן לְקַדְּשׁוֹ, לְהַיֵּשׁ מְפָרְשִׁים שֶׁבַּתּוֹסָפוֹת, דִּבּוּר הַמַּתְחִיל "הָא"; אַךְ לְפֵרוּשׁ רַשִׁ״י הוּא אַדְּרַבָּא לְהֵפֶךְ. אֲבָל בְּלָאו הָכֵי — הָא רָבָא לֵית לֵיהּ דְּאַבָּיֵי. אַךְ הִנֵּה רָאִיתִי בִּירוּשַׁלְמִי (רֵישׁ פֶּרֶק ג׳ הֲלָכָה א׳): אִיתָא — כ״ז בְּשֵׁם רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי: מְעַמְעֲמִין (פֵּרוּשׁ: מְאַיְּמִין) עַל הַנִּרְאֶה לְעַבְּרוֹ, וְלֹא עַל שֶׁלֹּא נִרְאֶה לְקַדְּשׁוֹ; ר״ב בְּשֵׁם רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי: מְעַמְעֲמִין עַל שֶׁלֹּא נִרְאֶה לְקַדְּשׁוֹ, וְלֹא עַל הַנִּרְאֶה לְעַבְּרוֹ, וְכוּלֵי; וְלָא פְּלִיגִין — מַה דְּאָמַר ר״ז בִּשְׁאָר כָּל הַחֳדָשִׁים, מַה דְּאָמַר ר״ב בְּנִיסָן וְתִשְׁרֵי, עַד כָּאן:
