~ לְכָל הַתּוֹרָה כֻּלָּהּ, אוֹ לַעֲרָלוֹת, אִם לֹא שֶׁמָּלִין תּוֹךְ ח׳ מִשּׁוּם סַכָּנָה — אָז אֵין לְהַקְפִּיד מִי יָמוּל. וְאַף דְּהַכֹּל כְּשֵׁרִים, מִכָּל מָקוֹם אִם יֵשׁ יִשְׂרָאֵל גָּדוֹל שֶׁיּוֹדֵעַ לָמוּל — קוֹדֵם לְכֻלָּם. וִיהַדֵּר שֶׁיִּהְיֶה מוֹהֵל צַדִּיק וְתַלְמִיד חָכָם, שֶׁיָּבוֹא טָהוֹר וְיִטַּפֵּל עִם הַטָּהוֹר. וְרִי״עַ כָּתַב שֶׁכֹּהֵן קוֹדֵם לְיִשְׂרָאֵל, וְלָא יָדְעִנָא מְנָא לֵיהּ; וְעַל כָּל פָּנִים אִם הַיִּשְׂרָאֵל צַדִּיק טְפֵי — וַדַּאי הוּא קוֹדֵם. וְאוֹנֵן מָל אֲפִלּוּ בֶּן אַחֵר, הֵיכִי דְּלֵיכָּא אַחֵר לָמוּל (וְאֵין שָׁהוּת גַּם כֵּן לִקְבּוֹר הַמֵּת, כַּנִּזְכָּר לְעֵיל); אֲבָל הֵיכִי דְּאִיכָּא אַחֵר — לֹא יָמוּל, אֲפִלּוּ אַבָּל אַחַר ג׳ יָמִים בֶּן אַחֵר. אֲבָל בְּנוֹ — נִרְאֶה דְּיָמוּל הָאָבֵל, אַף דְּאִיכָּא אַחֵר, דְּמִצְוָה בּוֹ יוֹתֵר מִבִּשְׁלוּחוֹ; אֲבָל אוֹנֵן לֹא יָמוּל אַף בְּנוֹ, הֵיכִי דְּיֵשׁ אַחֵר, שֶׁהֲרֵי הוּא פָּטוּר מִכָּל הַמִּצְוֹת. וְאֵין טַעֲנַת קֻרְבָה טַעֲנָה לִתֵּן לוֹ, אֶלָּא יִתֵּן לְמִי שֶׁצַּדִּיק וְתַלְמִיד חָכָם יוֹתֵר (וְגַם "טוֹב שָׁכֵן קָרוֹב מֵאָח רָחוֹק"). וּמִי שֶׁהָיָה אֵצֶל אֵיזֶה בָּנָיו מוֹהֵל — יִתֵּן לוֹ גַּם אֶל הָאֲחֵרִים; וְכֵן אַחַר מוֹתוֹ, כְּשֶׁנּוֹלַד וְאֵין הַיּוֹרְשִׁים מוֹהֲלִים בְּעַצְמָם — יִתְּנוּ לְמִי שֶׁהָיָה אֲבִיהֶם רָגִיל לִתֵּן (אִם לֹא שֶׁנִּתְקַלְקֵל בֵּינְתַיִם). וְאִם אָמַר הָאָב לְאֶחָד — אֲפִלּוּ קֹדֶם שֶׁנּוֹלַד הַבֵּן — שֶׁאִם יִוָּלֵד לוֹ בֵּן יִהְיֶה הוּא מוֹהֵל, אָסוּר לוֹ לַחֲזוֹר, מִשּׁוּם "שְׁאֵרִית יִשְׂרָאֵל לֹא יַעֲשׂוּ עַוְלָה" וְגוֹמֵר, וּמֻתָּר לִקְרוֹתוֹ רָשָׁע; אִם לֹא שֶׁהָיָה בְּטָעוּת, כְּגוֹן שֶׁחָשַׁב שֶׁהוּא לֹא יִהְיֶה בְּעֵת הַמִּילָה בְּבֵיתוֹ, וְנִזְדַּמֵּן שֶׁהוּא בְּבֵיתוֹ (כֵּן כָּתַב רִי״עַ, אֶלָּא שֶׁהֶאֱרִיךְ בְּחִנָּם, לְפִי מַה שֶּׁכָּתַב הוּא כְּהַשַּׁ״ךְ דְּאָסוּר לָאָב לְכַבֵּד לְאַחֵר כְּשֶׁיָּכוֹל לָמוּל בְּעַצְמוֹ — אִם כֵּן הַדָּבָר פָּשׁוּט. מִיהוּ אַף לְדִידַן הַדִּין כֵּן), אוֹ שֶׁבָּא לוֹ אוֹהֲבוֹ אוֹ אָדָם צַדִּיק, דְּאָנַן סָהֲדֵי דְּאִלּוּ הֲוָה מִתְּחִלָּה לֹא הָיָה מְכַבֵּד — רַשַּׁאי לַחֲזוֹר. וּבְכָל עִנְיָן, אִם חָזַר בּוֹ — אַף עַל פִּי שֶׁעָשָׂה שֶׁלֹּא כַּדִּין, מִכָּל מָקוֹם חֲזָרָתוֹ מוֹעֶלֶת; וַאֲפִלּוּ קִנְיָן — אֵין מוֹעִיל, דַּהֲוֵי קִנְיַן דְּבָרִים. אֲבָל אִם נִשְׁבַּע לוֹ (רִי״עַ כָּתַב דְּאֵין חָלָה, דְּאֵינוֹ נִשְׁבָּע שֶׁיָּמוּל הוּא דַּוְקָא, שֶׁמָּא יָמוּת; וְאִי שֶׁלֹּא יָמוּל אַחֵר — נִשְׁבָּע לְבַטֵּל מִצְוָה. וּדְבָרָיו תְּמוּהִין, דְּיֵשׁ טוּבֵי כַּהַאי גַּוְנָא. וּפָשׁוּט דְּהַשְּׁבוּעָה שֶׁאִם יוּכַל הוּא — שֶׁלֹּא יָמוּל אַחֵר), אֲפִלּוּ קֹדֶם שֶׁנּוֹלַד — בְּכָל גַּוְנָא כּוֹפִין אוֹתוֹ לְקַיֵּם שְׁבוּעָתוֹ. מִיהוּ אִם יֵשׁ לוֹ הֶתֵּר עַל הַשְּׁבוּעָה, וְכֵן כְּשֶׁנִּזְדַּמֵּן אוֹהֲבוֹ אוֹ צַדִּיק, אוֹ הוּא עַצְמוֹ כַּנִּזְכָּר לְעֵיל — רַשַּׁאי לְהַתִּיר שְׁבוּעָתוֹ. וְאִם נָתַן לְאֶחָד וְלֹא הָיָה בָּעִיר, וְחָשַׁב שֶׁלֹּא יִהְיֶה גַּם בְּעֵת הַמִּילָה, וְכִבֵּד לְאַחֵר, וְאַחַר כָּךְ בָּא הָרִאשׁוֹן — יִמְהֲלֶנּוּ, דְּוַדַּאי לֹא הֲדַר מִינֵּיהּ. וְאִם כִּבֵּד לְאֶחָד, וּבָא אַחֵר וְחָטַף מִינֵּיהּ — אֵין צָרִיךְ לִתֵּן י׳ זְהוּבִים. וְכָתַב הַסְּמָ״ע: דַּוְקָא שֶׁשָּׁמַע הַבְּרָכָה וְעָנָה אָמֵן; וְעַיֵּן בְּשַׁ״ךְ. כָּתוּב בְּסֵפֶר הַזִּכְרוֹנוֹת (מִמַּהֲרַ״שׁ אָבוּהַב): רָאוּי לְתַקֵּן שֶׁלֹּא יִהְיֶה אָדָם רַשַּׁאי לָמוּל עַד שֶׁיִּטּוֹל רְשׁוּת מִבֵּית דִּין, וּלְשׁוּם קִצְבָּה לְשָׁנִים, עַד כ״ה אוֹ ל׳ שָׁנָה לֹא יוּתַּן רְשׁוּת לְשׁוּם אָדָם, מִלְּבַד
