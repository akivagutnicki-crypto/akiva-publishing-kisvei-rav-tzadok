**וְהִנֵּה** בְּמַעֲלָה אַחַת יֵשׁ תשע״ז מִילְיָאן וְת״ר אֶלֶף חֲמִישִׁיִּים. וּכְשֶׁנַּחֲלֹק מִסְפָּר זֶה עַל הַקּוֹדֵם — יַגִּיעֵנוּ מְנָתוֹ כ״א פַּעַם, וְעוֹד נִשְׁאָר כ״ב מִילְיָאן קי״ז אֶלֶף תרכ״ט. וְזֶהוּ הַמַּגִּיעַ לְל״ה מִילְיָאן תתקע״ה אֶלֶף וְשנ״א חֵלֶק בְּמַעֲלָה (שֶׁהוּא ל״ה תנ״ח תק״ס חֵלֶק בַּיּוֹם, שֶׁהוּא אֶחָד מִח״י בְּרֶגַע): כ״א חֲמִישִׁיּוֹת, וְכ״ב מִילְיָאן קי״ז אֶלֶף תרכ״ט חֶלְקֵי ל״ה מִילְיָאן תתקע״ה אֶלֶף שנ״א. וּכְשֶׁנִּכְפֹּל מִסְפָּר זֶה עִם ל״ה מִילְיָאן תנ״ח אֶלֶף וְתק״ס, וְנַעֲשֶׂה בִּקְבוּצָה הַכּוֹפֵל — מִכָּל ס׳ חֲמִישִׁיִּים רְבִיעִיּוֹת אֶחָד, וּמִכָּל ס׳ רְבִיעִיִּים שְׁלִישִׁי א׳, וְכֵן מִכָּל ס׳ שְׁלִישִׁיִּים, וּמִס׳ שְׁנִיִּים — יִהְיֶה קְבוּצוֹת הַכֹּל: נ״ט חֲלָקִים, ח׳ שְׁנִיִּים, י״ז שְׁלִישִׁיִּים, שִׁבְעָה רְבִיעִיִּים, מ״ו חֲמִישִׁיִּים, וְעוֹד ד׳ מִילְיָאן תתל״ז אֶלֶף וְרל״ד חֶלְקֵי ל״ה מִילְיָאן וְתתקע״ה אֶלֶף וְשנ״א מֵחֲמִישִׁיֵּי הַמ״ז:

**וְזֶהוּ** מַהֲלַךְ הַשֶּׁמֶשׁ בְּיוֹם תָּמִים (וּמַהֲלָכָהּ בַּחֵלֶק הַנּוֹסָף — הוּא כְּפוּלַת הַנִּזְכָּר לְעֵיל עִם תקי״ו אֶלֶף וְתשצ״א; וְאַחֲרֵי שֶׁנַּעֲשֶׂה, כַּנִּזְכָּר לְעֵיל, מֵחֲמִישִׁיִּים רְבִיעִיִּים וּשְׁלִישִׁיִּים וּשְׁנִיִּים — יִהְיֶה קְבוּצַת הַכֹּל: נ״א שְׁנִיּוֹת, מ״ב שְׁלִישִׁיּוֹת, נ״ב רְבִיעִיּוֹת, י״ג חֲמִישִׁיּוֹת, וְעוֹד ל״א מִילְיָאן קל״ח אֶלֶף וְקי״ז חֶלְקֵי ל״ה מִילְיָאן תתקע״ה אֶלֶף וְשנ״א מֵחֲמִישִׁיֵּי הַי״ד; וּקְבוּצַת ב׳ הַמַּהֲלָכִים הַנִּזְכָּרִים — מַעֲלָה שְׁלֵמָה), לְפִי חֶשְׁבּוֹן זֶה דְּרַב אַדָּא. וּלְפִי זֶה, לֹא יִתּוֹסְפוּ בְּי׳ יָמִים ב׳ שְׁנִיִּים שְׁלֵמִים; וְכָל שֶׁכֵּן שֶׁלֹּא יִתּוֹסֵף בְּק׳ יוֹם כְּלוּם:

אָמַר הַמַּעְתִּיק ש״ג בהרמ״ז זַצַלְלָה״ה: עַד כָּאן נִמְצָא בָּזֶה בִּכְתַב יַד קָדְשׁוֹ, וְנִשְׁאַר נְיָר חָלָק — דַּף שָׁלֵם, וְג׳ רְבִיעֵי עַמּוּד — וְאַחַר כָּךְ מַתְחִיל פֶּרֶק ד׳, בְּמִנְהֲגֵי וְדִינֵי רֹאשׁ חֹדֶשׁ:

# פֶּרֶק ד׳ — בְּמִנְהֲגֵי וְדִינֵי רֹאשׁ חֹדֶשׁ

**כְּבָר** זָכַרְנוּ שֶׁעַכְשָׁיו אֵין לָנוּ קִדּוּשׁ בֵּית דִּין הַחֹדֶשׁ. וּמִכָּל מָקוֹם נָהֲגוּ בַּשַּׁבָּת שֶׁלִּפְנֵי רֹאשׁ חֹדֶשׁ לְהַכְרִיז: "בְּיוֹם פְּלוֹנִי יִהְיֶה רֹאשׁ חֹדֶשׁ"; וְאֵין זֶה זֵכֶר לְקִדּוּשׁ בֵּית דִּין, אֶלָּא הוֹדָעָה לָעָם; וּמִכָּל מָקוֹם נָהֲגוּ לְאָמְרוֹ מְעֻמָּד, דֻּגְמַת קִדּוּשׁ בֵּית דִּין. וְרֹאשׁ חֹדֶשׁ תִּשְׁרֵי לֹא נָהֲגוּ לְהַכְרִיז — שֶׁאָמְרוּ רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה: "אֵיזֶהוּ חַג שֶׁהַחֹדֶשׁ מִתְכַּסֶּה בּוֹ" וְכוּלֵי, שֶׁצָּרִיךְ לְהַעֲלִימוֹ מִן הַשָּׂטָן. וּבִקְצָת מְקוֹמוֹת אַף רֹאשׁ חֹדֶשׁ אָב לֹא נָהֲגוּ לְהַכְרִיז (מִשּׁוּם
