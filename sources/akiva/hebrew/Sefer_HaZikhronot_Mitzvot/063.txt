~ וְהַצִּנָּה, שֶׁמִּפְּנֵי זֶה לֹא יִמָּנְעוּ מִלַּעֲלוֹת, לְפִי שֶׁאֶפְשָׁר לָהֶם לְהִתְחַמֵּם בִּבְגָדִים; וְלֹא מִפְּנֵי הַטֻּמְאָה, כְּגוֹן שֶׁהָיוּ רֹב הַקָּהָל אוֹ רֹב הַכֹּהֲנִים טְמֵאִים (וְכָל שֶׁכֵּן מִעוּטָן, וּפָשׁוּט), וְאֵין יְכוֹלִים לִמְצוֹא אֵפֶר פָּרָה לִטְהַר; אוֹ שֶׁהָיָה נָשִׂיא חוֹלֶה, וַאֲמָדוּהוּ לָמוּת אֵיזֶה יָמִים קֹדֶם הַפֶּסַח, וְיִהְיוּ טְמֵאִים, וְכַהַאי גַּוְנָא — אֵין אוֹמְרִים לְעַבֵּר, וְיִהְיֶה פְּנַאי לִטְהַר וְלַעֲשׂוֹת הַפֶּסַח בְּטָהֳרָה; אֶלָּא יַעֲשׂוּהוּ בְּטֻמְאָה (כְּשֶׁרֻבָּן טְמֵאִים). מִיהוּ אִם עִבְּרוּ, וְעִבְּרוּ מִפְּנֵי הַטֻּמְאָה — מְעֻבֶּרֶת. אֲבָל מִפְּנֵי שְׁאָר דְּבָרִים הַנִּזְכָּרִים — אַף דִּיעֲבַד אֵינָהּ מְעֻבֶּרֶת:

**וְיֵשׁ** דְּבָרִים שֶׁאֵין מְעַבְּרִין בִּשְׁבִילָן, אֲבָל עוֹשִׂין אוֹתָן סַעַד לְשָׁנָה שֶׁצְּרִיכָה עִבּוּר. וְאֵלּוּ הֵן: מִפְּנֵי הַגְּדָיִים וְהַטְּלָאִים שֶׁעֲדַיִן לֹא נוֹלְדוּ, אוֹ שֶׁהֵן מְעַט; וּמִפְּנֵי הַגּוֹזָלוֹת שֶׁלֹּא פָּרְחוּ — אֵין מְעַבְּרִין בִּשְׁבִיל זֶה, כְּדֵי שֶׁיִּהְיוּ הַגְּדָיִים וְהַטְּלָאִים מְצוּיִין לִפְסָחִים, וְהַגּוֹזָלוֹת לִרְאִיָּה, אוֹ לְמִי שֶׁנִּתְחַיֵּב בְּקָרְבַּן הָעוֹף; אֲבָל עוֹשִׂין אוֹתָן סַעַד לְשָׁנָה, לְיַשְּׁבוֹ וּלְיַפּוֹתוֹ לְעִנְיַן הָעִבּוּר (רַשִׁ״י וְתוֹסָפוֹת רֹאשׁ הַשָּׁנָה כ״א עַמּוּד ב׳, דִּבּוּר הַמַּתְחִיל "עַל"). כֵּיצַד? אוֹמְרִין: שָׁנָה זוֹ צְרִיכָה עִבּוּר, מִפְּנֵי הַתְּקוּפָה שֶׁמָּשְׁכָה, אוֹ מִפְּנֵי הָאָבִיב וּפֵרוֹת הָאִילָן שֶׁלֹּא הִגִּיעוּ; וְעוֹד, שֶׁהַגְּדָיִים קְטַנִּים, וְהַגּוֹזָלוֹת רַכִּים — זֶהוּ דַּעַת הָרַמְבַּ״ם. וּמַשְׁמַע דִּבְהָנָךְ דִּלְעֵיל — מִפְּנֵי שֶׁלֶג וְצִנָּה — אֵין עוֹשִׂין סַעַד; וְלֹא יָדַעְתִּי מַדּוּעַ. גַּם צֹרֶךְ סַעַד זֶה לָמָּה — לֹא נִתְבָּרֵר לִי יָפֶה. אֲבָל רַשִׁ״י זִכְרוֹנוֹ לִבְרָכָה פֵּרֵשׁ: שֶׁאֵין מְעַבְּרִין בִּשְׁבִילָן לְבַד, אֲבָל עוֹשִׂין אוֹתָן סַעַד לְאָבִיב אוֹ פֵּרוֹת הָאִילָן — שֶׁאֵין מְעַבְּרִין בִּשְׁבִיל אֶחָד לְבַד, וְאִם יֵשׁ עוֹד זֶה — עוֹשִׂין סַעַד וּמְעַבְּרִין; עַיֵּן שָׁם. וְכֵן דַּעַת הָרַב רַבִּי אַהֲרֹן בְּרַבִּי מְשֻׁלָּם וְהָרמ״ךְ, הַמּוּבָאִים בְּכֶסֶף מִשְׁנֶה (הֲלָכָה ט״ז); וְצָרִיךְ עִיּוּן אֵיךְ לֹא רָאוּ שֶׁדַּעַת הָרַמְבַּ״ם אֵינוֹ כֵּן, וְאֵינְהוּ לְתָרוּצֵי דִּבְרֵי הָרַמְבַּ״ם חָקְרוּ, עַיֵּן שָׁם. מִיהוּ צָרִיךְ עִיּוּן, דְּרַשִׁ״י סוֹתֵר דִּבְרֵי עַצְמוֹ, דְּמִדְּבָרָיו בְּרֹאשׁ הַשָּׁנָה (כ״א עַמּוּד ב׳), דִּבּוּר הַמַּתְחִיל "עַל נִיסָן" — לֹא מַשְׁמַע כֵּן, עַיֵּן שָׁם, וְצָרִיךְ עִיּוּן. וְעַיֵּן בְּסֵפֶר תּוֹרַת חַיִּים — מַיְיתֵי דִּבְרֵי הַתּוֹסָפוֹת בְּרֹאשׁ הַשָּׁנָה שָׁם, דְּהוּא דְּלָא כְּרַשִׁ״י; וְלֹא עִיֵּן, דְּכֵן הוּא גַּם כֵּן בְּרַשִׁ״י שָׁם, עַיֵּן שָׁם. וְעַיֵּן שָׁם בְּסֵפֶר תּוֹרַת חַיִּים, דְּאַסֵּיק הָעִקָּר דְּלָא כְּרַשִׁ״י, אֶלָּא כְּהָרַמְבַּ״ם; וְצָרִיךְ עִיּוּן לְדִינָא:

**אֵין** מְעַבְּרִים אֶלָּא בַּיּוֹם; וְאִם עִבְּרוּהָ בַּלַּיְלָה — אֵינָהּ מְעֻבֶּרֶת (וּבְבֵין הַשְּׁמָשׁוֹת — צָרִיךְ עִיּוּן; עַיֵּן לְעֵיל לְעִנְיַן קִדּוּשׁ. מִיהוּ, כֵּיוָן דְּבְקַל יוּכְלוּ לְתַקֵּן לָשׁוּב לְעַבֵּר — וַדַּאי יַחְזְרוּ וִיעַבְּרוּ). וּרְשׁוּת לְבֵית דִּין לַחְשׁוֹב וְלִקְבּוֹעַ, וְלֵידַע אֵיזוֹ שָׁנָה תִּהְיֶה מְעֻבֶּרֶת, אֲפִלּוּ לְכַמָּה שָׁנִים; אֲבָל אֵין אוֹמְרִים "שָׁנָה פְּלוֹנִית מְעֻבֶּרֶת" אֶלָּא אַחַר רֹאשׁ הַשָּׁנָה — אוֹמְרִים "שָׁנָה זוֹ מְעֻבֶּרֶת". וְדָבָר זֶה מִפְּנֵי הַדֹּחַק
