~ וְיִהְיֶה קָשֶׁה לוֹ לָשׁוּב. (נִרְאֶה לִכְאוֹרָה, דְּהָכֵי מְפָרֵשׁ הָרַב הַגְּמָרָא דְּסוֹף פֶּרֶק קַמָּא דְּקִדּוּשִׁין: "בְּעָבַר וְשָׁנָה — נַעֲשָׂה כְּהֶתֵּר" — דְּהַיְנוּ כְּשֶׁלֹּא שָׁב בֵּינְתַיִם. וּלְפִי זֶה, אִצְטַמִּיד דְּבָרֵינוּ דִּלְעֵיל — דִּבְשָׁב בֵּינְתַיִם אֵין נֶעֱנָשׁ עַל הַהִרְהוּר; וְנוֹסָף עוֹד — דַּאֲפִלּוּ שָׁב בֵּין פַּעַם רִאשׁוֹן לַב׳ — שׁוּב אֵין נֶעֱנָשׁ, לְפִי זֶה):

**וְעַל כָּל פָּנִים**, יַעֲשֶׂה תְּשׁוּבָה כְּשֶׁהוּא בְּכֹחוֹ וּבִגְבוּרָתוֹ (פֶּרֶק קַמָּא דַּעֲבוֹדָה זָרָה, דַּף כ׳); שֶׁנֶּאֱמַר: "אַשְׁרֵי אִישׁ" — וְלֹא אִשָּׁה? בִּתְמִיָּה! — אֶלָּא: אַשְׁרֵי מִי שֶׁעוֹשֶׂה תְּשׁוּבָה כְּשֶׁהוּא אִישׁ. (וַיִּקְרָא רַבָּה פָּרָשָׁה י״ח:) וְכֵן הוּא אוֹמֵר: "וּזְכֹר אֶת בּוֹרְאֶיךָ בִּימֵי בְּחוּרֹתֶיךָ" — עַד שֶׁכֹּחֲךָ עָלֶיךָ, בְּיַלְדוּתְךָ — שׁוּב בִּתְשׁוּבָה. וּמִכָּל מָקוֹם, אִם לֹא שָׁב אֶלָּא בְּזִקְנוּתוֹ — אֲפִלּוּ עָבַר כָּל יָמָיו, וְשָׁב בְּיוֹם מִיתָתוֹ — הֲרֵי זוֹ מִתְקַבֶּלֶת; שֶׁנֶּאֱמַר: "עַד אֲשֶׁר לֹא תֶחְשַׁךְ הַשֶּׁמֶשׁ וְהָאוֹר וְהַיָּרֵחַ וְהַכּוֹכָבִים, וְשָׁבוּ הֶעָבִים אַחַר הַגָּשֶׁם" — שֶׁהוּא יוֹם הַמִּיתָה; מִכְּלָל, שֶׁאִם זָכַר בּוֹרְאוֹ קֹדֶם שֶׁיָּמוּת וְשָׁב — נִסְלַח לוֹ. וְכֵן הוּא אוֹמֵר: "תָּשֵׁב אֱנוֹשׁ עַד דַּכָּא", וְדָרְשׁוּ חֲכָמֵינוּ זִכְרוֹנָם לִבְרָכָה: עַד דִּכְדּוּכָהּ שֶׁל נֶפֶשׁ:

**וְאַף עַל פִּי** שֶׁהַתְּשׁוּבָה יָפָה תָּמִיד — בַּעֲשָׂרָה יָמִים שֶׁמִּן רֹאשׁ הַשָּׁנָה עַד אַחַר יוֹם הַכִּפּוּרִים הוּא יָפֶה בְּיוֹתֵר, וּמִתְקַבֶּלֶת מִיָּד; שֶׁנֶּאֱמַר: "דִּרְשׁוּ ה׳ בְּהִמָּצְאוֹ". בַּמֶּה דְּבָרִים אֲמוּרִים — בְּיָחִיד; אֲבָל צִבּוּר — כָּל זְמַן שֶׁעוֹשִׂין תְּשׁוּבָה וְצוֹעֲקִין בְּלֵב שָׁלֵם — נַעֲנִין; שֶׁנֶּאֱמַר: "מִי כַּה׳ אֱלֹקֵינוּ בְּכָל קָרְאֵנוּ אֵלָיו". וְיוֹם הַכִּפּוּרִים עַצְמוֹ — הוּא זְמַן תְּשׁוּבָה לַכֹּל, לְיָחִיד וְלָרַבִּים; שֶׁנֶּאֱמַר בּוֹ: "כִּי בַיּוֹם הַזֶּה יְכַפֵּר עֲלֵיכֶם" וְגוֹ׳. וְאֵין יוֹם הַכִּפּוּרִים מְכַפֵּר אֶלָּא בִּתְשׁוּבָה; וּלְפִיכָךְ חַיָּבִים הַכֹּל לַעֲשׂוֹת תְּשׁוּבָה וּלְהִתְוַדּוֹת בְּיוֹם הַכִּפּוּרִים, כְּדֶרֶךְ שֶׁיִּתְבָּאֵר לְקַמָּן. וְכֵן חַיָּב כָּל אֶחָד לְהִתְוַדּוֹת וְלַעֲשׂוֹת תְּשׁוּבָה כְּשֶׁהוּא חוֹלֶה וְנוֹטֶה לָמוּת, קֹדֶם מִיתָתוֹ, כְּדֵי שֶׁיָּמוּת מִתּוֹךְ הַתְּשׁוּבָה (וּבְסִפְרֵי לְמָדוּהוּ מִקְּרָא, דְּהִסְמִיךְ "הַנֶּפֶשׁ הַהִוא" לְ"וְהִתְוַדּוּ" — לוֹמַר, שֶׁבִּשְׁעַת נְטִילַת נֶפֶשׁ טָעוּן וִדּוּי), וּכְדֶרֶךְ שֶׁיִּתְבָּאֵר לְקַמָּן:

**וְאַל** תֹּאמַר (רַמְבַּ״ם פֶּרֶק ז׳) שֶׁאֵין תְּשׁוּבָה אֶלָּא מֵעֲבֵרוֹת שֶׁיֵּשׁ בָּהֶם מַעֲשֶׂה — כְּזְנוּת וְגָזֵל וְכַיּוֹצֵא; אֶלָּא גַּם מִדֵּעוֹת וּמִדּוֹת הָרָעוֹת שֶׁבּוֹ — כְּכַעַס וְקִנְאָה, וְתַאֲוָה וְכָבוֹד, וְכַמּוֹהֶם — צָרִיךְ לָשׁוּב עֲלֵיהֶם וּלְהִתְוַדּוֹת. וְאֵלּוּ הָעֲוֹנוֹת קָשִׁים מֵאוֹתָם שֶׁיֵּשׁ בָּהֶם מַעֲשֶׂה; שֶׁבִּזְמַן שֶׁאָדָם נִשְׁקָע בְּאֵלּוּ — קָשֶׁה הוּא לִפְרוֹשׁ מֵהֶם; וְלָכֵן צָרִיךְ לְדַקְדֵּק מְאֹד אַחֲרֵיהֶם. וְכֵן הוּא אוֹמֵר: "יַעֲזֹב רָשָׁע דַּרְכּוֹ" וְגוֹ׳:

**כֵּיצַד** הִיא הַתְּשׁוּבָה וְהַוִּדּוּי? "וְהִתְוַדּוּ" הָאָמוּר בַּכָּתוּב — קִבְּלוּ רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה, שֶׁהוּא וִדּוּי דְּבָרִים: שֶׁיִּהְיֶה מוֹדֶה לְפָנָיו יִתְבָּרַךְ עַל כָּל הַחֲטָאִים שֶׁעָשָׂה,
