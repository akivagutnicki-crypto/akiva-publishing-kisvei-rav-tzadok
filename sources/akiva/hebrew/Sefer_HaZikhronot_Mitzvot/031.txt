**נוֹהֵג** בֵּין בִּבְהֵמָה בֵּין בְּחַיָּה, בֵּין בְּכוֹי, בֵּין בִּטְהוֹרִים (תל״מ לְהַטּוּר, וְעַיֵּן פְּרִי מְגָדִים רֵישׁ סִימָן ס״ד) בֵּין בִּטְמֵאִים; וְיֵשׁ אוֹמְרִים דְּאֵין נוֹהֵג בִּטְמֵאִים (וְנַפְקָא מִינָּהּ דְּלֶהֱוֵי בְּרִיָּה, וְעוֹד טוּבָא). וְנוֹהֵג בָּאָדָם (מִלְּבַד הָאִסּוּר דִּבְשַׂר אָדָם, שֶׁיִּתְבָּאֵר בִּמְקוֹמוֹ). מִיהוּ שַׁמְנִינוּתוֹ נִרְאֶה פָּשׁוּט דְּלָא אֲסִירָא בָּאָדָם, וְכֵן כָּתַב רִי״עַ. אֲבָל אֵינוֹ נוֹהֵג בָּעוֹף, מִפְּנֵי שֶׁהַבָּשָׂר שֶׁעַל הַקּוּלִית שֶׁלָּהֶם אֵינוֹ עָגוֹל סְבִיבוֹת הָעֶצֶם, וְלָא דָּמֵי לְכַף. וְאִם נִמְצָא עוֹף שֶׁיֵּשׁ לוֹ כַּף עָגוֹל, אוֹ בְּהֵמָה שֶׁאֵין כַּפָּהּ עָגוֹל — הַדָּבָר סָפֵק אִם אָסוּר בּוֹ גִּיד, וְאָסוּר, וְאֵין לוֹקִין עָלָיו. וְכָל שֶׁאֵינוֹ טָעוּן שְׁחִיטָה, כְּדָגִים (בְּחַיּוֹת שֶׁבַּיָּם שֶׁיֵּשׁ לָהֶם רַגְלַיִם כְּרַגְלֵי בְּהֵמָה, עַיֵּן פְּרִי חָדָשׁ סִימָן פ״ג) — אֵין גִּידוֹ אָסוּר. וְכֵן הַשָּׁלִיל — יֵשׁ אוֹמְרִים שֶׁאֵין אִסּוּר גִּיד נוֹהֵג בּוֹ, אֶלָּא כְּשֶׁהִפְרִיס עַל גַּבֵּי קַרְקַע וְהוּא בֶּן ט׳ חַי — אָסוּר גִּידוֹ מִדְּרַבָּנָן, מִשּׁוּם מַרְאִית הָעַיִן, כְּמוֹ דְּצָרִיךְ שְׁחִיטָה מִשּׁוּם מַרְאִית הָעַיִן. וְיֵשׁ חוֹלְקִים, וְאוֹמְרִים דְּלָא דָּמֵי גִּיד לִשְׁחִיטָה, וּבְגִיד נוֹהֵג בְּבֶן ט׳ חַי אֲפִלּוּ לֹא הִפְרִיס עַל גַּבֵּי קַרְקַע, כָּל שֶׁיָּדוּעַ שֶׁכָּלוּ חֳדָשָׁיו. וְיֵשׁ אוֹמְרִים עוֹד (עַיֵּן בְּמנ״י כְּלָל צ׳) שֶׁאֲפִלּוּ בֶּן ח׳, וְלֹא הִפְרִיס עַל גַּבֵּי קַרְקַע — אִסּוּר גִּיד נוֹהֵג בּוֹ (וְאֶפְשָׁר אֲפִלּוּ מֵת), דְּמִשְּׁעַת יְצִירָה נֶאֱסַר גִּיד; וְיֵשׁ לְהַחְמִיר. וְנוֹהֵג בֵּין בְּחֻלִּין בֵּין בְּמֻקְדָּשִׁים, בֵּין בִּבְהֵמוֹת זְכָרִים בֵּין בִּבְהֵמוֹת נְקֵבוֹת:

**אֵיזֶהוּ** גִּיד הַנָּשֶׁה? בַּיָּרֵךְ — בֵּין בִּימִינִי בֵּין בִּשְׂמָאלִי — יֵשׁ ב׳ גִּידִים: אֶחָד פְּנִימִי, סָמוּךְ לָעֶצֶם הַקּוּלִית, וְאֶחָד חִיצוֹן, סָמוּךְ לַבָּשָׂר; וּשְׁנֵיהֶם אֲסוּרִים, וְצָרִיךְ לְחַטֵּט אַחֲרֵיהֶם בְּשָׁרְשָׁם בָּעֶצֶם. אֶלָּא שֶׁהַפְּנִימִי אָסוּר מִן הַתּוֹרָה, שֶׁהוּא הַפָּשׁוּט בְּכָל הַיָּרֵךְ, וְהוּא עַל הַכַּף מַמָּשׁ. וְיֵשׁ אוֹמְרִים דְּגַם פְּנִימִי אֵינוֹ אָסוּר מִן הַתּוֹרָה אֶלָּא מַה שֶּׁעַל הַכַּף, אֲבָל לְמַעְלָה וּלְמַטָּה עַד סוֹפוֹ הוּא מִדְּרַבָּנָן, וְכֵן עִקָּר; וְחִיצוֹן כֻּלּוֹ רַק מִדְּרַבָּנָן. וְקְנוּקְנוֹת — וְהֵם הַגִּידִים הַדַּקִּים שֶׁבִּשְׁנֵיהֶם — אֲסוּרִים גַּם כֵּן מִדְּרַבָּנָן. אֲבָל שׁוּמָנָם — אַף אִסּוּר דְּרַבָּנָן אֵין בּוֹ, אֲבָל יִשְׂרָאֵל קְדוֹשִׁים הֵם, וְנָהֲגוּ בּוֹ אִסּוּר. וְרָאשֵׁי הַקְּנוּקְנוֹת נִכְנָסִים בְּרָאשֵׁי הָעֲצָמוֹת; עַל כֵּן צָרִיךְ הַמְנַקֵּר לִשְׁבּוֹר רָאשֵׁי הָעֲצָמוֹת, כְּדֵי לְהוֹצִיא רָאשֵׁי הַקְּנוּקְנוֹת מֵעִקָּרָן. וְזֶה לְשׁוֹן להפ״ס בְּסֵדֶר הַנִּקּוּר: "הַפלאנקי״ן — בִּלְשׁוֹן רוּסְיָא קאליא״ן — וְכוּלֵי, וּבְתוֹכוֹ יֵשׁ גִּיד אֶחָד לָאֹרֶךְ, וְצוּרָתוֹ כְּצוּרַת עַקְרָב, וְאָסוּר מִשּׁוּם גִּיד הַנָּשֶׁה" וְכוּלֵי; "הצע״ך בראטי״ן" וְכוּלֵי, "מַרְאֵה הַכֶּסֶף שֶׁבָּהּ מְגָרְרִין וּמְקַלְּפִין הֵיטֵב, וּמוֹצִיאִים מִתּוֹכָן גִּידִין מֻנָּחִים בְּשִׁפּוּעַ, לִפְעָמִים ב׳, לִפְעָמִים ג׳, וַאֲסוּרִים מִשּׁוּם גִּיד הַנָּשֶׁה" וְכוּלֵי; "וְעוֹד יֵשׁ גִּיד אֶחָד עוֹמֵד בְּתוֹךְ חוּט לָבָן, וְהוּא סוֹף גִּיד הַנָּשֶׁה עַצְמָהּ; וְעוֹד יֵשׁ בְּשִׁפּוּלֵי גִּיד אֶחָד בְּצוּרַת ג׳, בְּתַחְתִּית הַקיילכדי״ג ברעטי״ל" וְכוּלֵי, "בְּתוֹכָהּ יֵשׁ גִּידִין
