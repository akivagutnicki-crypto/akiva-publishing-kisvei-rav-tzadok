~ מוֹצִיא דִּבָּה). וְנָהֲגוּ לוֹמַר קֹדֶם הַהַכְרָזָה "מִי שֶׁעָשָׂה נִסִּים" וְכוּלֵי (עַיֵּן בַּלְּבוּשׁ (סִימָן תכ״א) ב׳ טְעָמִים: אֶחָד — מִשּׁוּם דְּזוֹ מִצְוָה רִאשׁוֹנָה אַחַר יְצִיאַת מִצְרַיִם, לְכָךְ מַזְכִּירִין אוֹתָהּ וּמִתְפַּלְּלִין עַל הַגְּאֻלָּה; ב׳ — שֶׁמִּתְפַּלְּלִים שֶׁיְּקַבֵּץ נִדָּחֵינוּ, וְנַחְזוֹר לְקַדֵּשׁ עַל פִּי הָרְאִיָּה; וְעַיֵּן שָׁם מַה שֶּׁכָּתַב עַל טַעַם זֶה. וְלִי נִרְאֶה, דְּוַדַּאי עַכְשָׁיו לֹא יַחְזוֹר קִדּוּשׁ הָרְאִיָּה לִמְקוֹמוֹ עַד כִּי יָשׁוּב ה׳ אֶת שְׁבוּת עַמּוֹ. וְהַנָּכוֹן מַה שֶּׁכָּתַב ר״י עֶמְדִּין: כִּי חִדּוּשׁ יִשְׂרָאֵל בְּחִדּוּשׁ לְבָנָה מִתְלֵי תַּלְיָין אַהֲדָדֵי, לָכֵן מַזְכִּירִין גַּם כֵּן חִדּוּשָׁן שֶׁל יִשְׂרָאֵל בִּגְאֻלָּתָן. וְהַפָּשׁוּט — כִּי רָאוּי לִהְיוֹת זֵכֶר הַגְּאֻלָּה וְהַבַּקָּשָׁה עָלֶיהָ שָׁגוּר בְּפִינוּ, וּנְבַקֵּשׁ בְּכָל יוֹם שֶׁיָּבוֹא, וְטֶרֶם הַחֹדֶשׁ בְּכָל חֹדֶשׁ, וְטֶרֶם הַשָּׁנָה בְּכָל שָׁנָה — שֶׁיָּבוֹא). וּבְנֻסַּח "יְהִי רָצוֹן" שֶׁנָּהֲגוּ לִפְנֵי "מִי שֶׁעָשָׂה נִסִּים" — עַיֵּן בְּהר״ד אַבּוּדִרְהַם, דְּפוּס פְּרָאג (דַּף ע״ח עַמּוּד ד׳), נֻסָּח אַחֵר מֵאֲשֶׁר לְפָנֵינוּ:

**נָהֲגוּ** קְצָת לְהִתְעַנּוֹת בְּעֶרֶב רֹאשׁ חֹדֶשׁ; וּכְשֶׁחָל רֹאשׁ חֹדֶשׁ בְּיוֹם א׳ — מִתְעַנִּין בְּה׳; וְאִם חָל בְּשַׁבָּת — מִתְעַנִּין בְּו׳ (וּבְאֵלִיָּה רַבָּה כָּתַב: וְזֶהוּ כְּשֶׁאֵין אוֹמְרִים סְלִיחוֹת; אֲבָל הָאוֹמְרִים סְלִיחוֹת — לֹא יִתְעַנּוּ אֶלָּא בְּה׳). וְצָרִיךְ גַּם כֵּן לְהַשְׁלִים, אִם הִשְׁלִים בְּפַעַם רִאשׁוֹן; וְאִם בְּפַעַם רִאשׁוֹן לֹא הִשְׁלִים — אֵין צָרִיךְ לְהַשְׁלִים, אֲפִלּוּ בְּחֹל. מִיהוּ, בְּמָקוֹם שֶׁעוֹשִׂין מִמֶּנּוּ תַּעֲנִית צִבּוּר, וְקוֹרִין "וַיְחַל", וְהוּא נִמְנֶה עִמָּהֶם — צָרִיךְ לְהַשְׁלִים:

**וּבְעִנְיַן** אֵימַת עֶרֶב רֹאשׁ חֹדֶשׁ — יֵשׁ אוֹמְרִים שֶׁהַיְנוּ יוֹם שֶׁלִּפְנֵי הַמּוֹלָד. אֲבָל יֵשׁ נוֹהֲגִים לְהִתְעַנּוֹת בְּיוֹם הַמּוֹלָד עַד שֶׁיַּגִּיעַ הַמּוֹלָד, וְאָז אוֹכְלִים, אֲפִלּוּ חָל הַמּוֹלָד בַּחֲצִי הַיּוֹם. וַאֲחֵרִים נָהֲגוּ לְהִתְעַנּוֹת בְּיוֹם שֶׁלִּפְנֵי רֹאשׁ חֹדֶשׁ, וַאֲפִלּוּ אַחַר הַמּוֹלָד. וְכָל מִי שֶׁנָּהַג בְּאֶחָד מֵאֵלּוּ, וּבָא לְשַׁנּוֹת דְּרָכָיו — צָרִיךְ הַתָּרָה. וְנָהֲגוּ בְּרֹב מְקוֹמוֹת לֵאמֹר הַמִּתְעַנִּים סְלִיחוֹת וּתְחִנּוֹת; וּמִנְהָג טוֹב לְאָמְרָם קֹדֶם תְּפִלַּת הַמִּנְחָה (נִרְאֶה לִי: קֹדֶם זְמַן תְּפִלַּת הַמִּנְחָה, וְהַיְנוּ מִנְחָה גְּדוֹלָה, מִו׳ וּמֶחֱצָה), דְּמִן הַמִּנְחָה וּלְמַעְלָה חָל רֹאשׁ חֹדֶשׁ; וּלְפִיכָךְ אֵין אוֹמְרִים תַּחֲנוּן בְּמִנְחָה (סִימָן קל״א, סְעִיף ו׳; וְעַיֵּן בְּהר״ד אַבּוּדִרְהַם, דְּפוּס פְּרָאג, דַּף מ״ג עַמּוּד ד׳, עַיֵּן שָׁם), וְלֹא "צִדְקָתְךָ צֶדֶק" אִם חָל בְּשַׁבָּת. וּמִכָּל מָקוֹם, מִנְהַג הָעוֹלָם לוֹמַר הַסְּלִיחוֹת בְּמִנְחָה; אֲבָל אִם חָל בְּעֶרֶב שַׁבָּת — נָהֲגוּ בִּקְצָת מְקוֹמוֹת שֶׁלֹּא לְאָמְרָם. וּבְכָל כְּגוֹן דָּא — פּוּק חֲזִי מַאי עַמָּא דְּבַר. וְרָאוּי לְמַהֵר בִּסְעוּדַת הַלַּיְלָה, שֶׁלֹּא לִכְנוֹס לְרֹאשׁ חֹדֶשׁ כְּשֶׁמְּעֻנֶּה:

**מִצְוַת** חֲכָמִים לְהַרְבּוֹת בִּסְעוּדַת רֹאשׁ חֹדֶשׁ; וְאַסְמְכִינְהוּ אַקְּרָא, דְּהֻקַּשׁ לְמוֹעֵד, דִּכְתִיב: "וּבְיוֹם שִׂמְחַתְכֶם וּבְמוֹעֲדֵיכֶם וּבְרָאשֵׁי חָדְשֵׁיכֶם". וְעוֹד (אֵלִיָּה רַבָּה בְּשֵׁם
