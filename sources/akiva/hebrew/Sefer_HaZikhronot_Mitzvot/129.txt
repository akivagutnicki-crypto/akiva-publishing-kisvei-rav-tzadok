~ מֵעֵת לְעֵת) מְקַדְּשִׁים; וְכֵן מְקַדְּשִׁים קֹדֶם תִּשְׁעָה בְּאָב וְקֹדֶם יוֹם הַכִּפּוּרִים. נֻסַּח קִדּוּשׁ הַלְּבָנָה — נֶאֱמַר מִשְּׁמוֹ, אֶפֶס לֹא אִשְׁתְּמִיט מִי מֵהָעוֹמְדִים לְפָנָיו לוֹמַר שֶׁנָּהַג כֵּן, וְהוּא: 'בָּרוּךְ אַתָּה ה׳' וְכוּלֵי, 'וְלַלְּבָנָה אָמַר שֶׁתִּתְחַדֵּשׁ — בְּאוֹר יְקָר — לְעֲטֶרֶת תִּפְאֶרֶת' וְכוּלֵי, 'בָּרוּךְ אַתָּה ה׳ מְקַדֵּשׁ חֳדָשִׁים'. 'כְּשֵׁם שֶׁאֲנִי רוֹקֵד כְּנֶגְדֵּךְ וְאֵינִי נוֹגֵעַ בָּךְ — כָּךְ אִם יִרְקְדוּ אֲחֵרִים כְּנֶגְדִּי, לֹא יִגְּעוּ בִּי', עַד כָּאן; וְכֵן הוּא בְּמַסֶּכֶת סוֹפְרִים שֶׁעִם הַגָּהוֹתָיו זִכְרוֹנוֹ לִבְרָכָה. אֵין אוֹמְרִים שׁוּם פְּסוּקִים — לֹא לִפְנֵי הַקִּדּוּשׁ וְלֹא לְאַחֲרָיו; רַק מַתְחִיל 'בָּרוּךְ אַתָּה ה׳' וְכוּלֵי, וּמְסַיֵּם 'סִימָן טוֹב' וְכוּלֵי, וְהוֹלֵךְ לוֹ", עַד כָּאן לְשׁוֹנוֹ. וְזֶהוּ לְפִי נֻסָּח שֶׁלָּנוּ, דְּאוֹמְרִים "סִימָן טוֹב" וְכוּלֵי אַחַר "שָׁלוֹם עֲלֵיכֶם", קֹדֶם פָּסוּק "קוֹל דּוֹדִי" וְכוּלֵי. וּמַה שֶּׁכָּתַב "לֹא לִפְנֵי הַקִּדּוּשׁ" — הוּא לְפִי שֶׁנָּהֲגוּ לוֹמַר לְפָנָיו "הַלְלוּ אֶת ה׳ מִן הַשָּׁמַיִם" וְכוּלֵי. וְהֵיכִי דִּנְהוֹג — נְהוֹג; נַהֲרָא נַהֲרָא וּפַשְׁטֵיהּ:

# מִצְוַת עֲשֵׂה ד׳

**(ה) מִצְוַת** עֲשֵׂה לִשְׁחוֹט בְּיוֹם י״ד בְּנִיסָן, בֵּין הָעַרְבַּיִם, אֶת קָרְבַּן הַפֶּסַח, בְּבֵית הַבְּחִירָה, בָּעֲזָרָה, כִּשְׁאָר הַקֳּדָשִׁים; שֶׁנֶּאֱמַר: "וְהָיָה לָכֶם לְמִשְׁמֶרֶת עַד אַרְבָּעָה עָשָׂר יוֹם לַחֹדֶשׁ הַזֶּה, וְשָׁחֲטוּ אֹתוֹ כֹּל קְהַל עֲדַת יִשְׂרָאֵל בֵּין הָעַרְבָּיִם". וְטַעַם מִצְוָה זוֹ — כְּדֵי לִזְכּוֹר הַנִּסִּים הַגְּדוֹלִים שֶׁנַּעֲשׂוּ בִּיצִיאַת מִצְרַיִם, וְשֶׁפָּסַח ה׳ עַל בָּתֵּי אֲבוֹתֵינוּ בְּמִצְרַיִם, וְהִצִּילָם מֵהַמַּשְׁחִית. וּמִי שֶׁבִּטֵּל מִצְוָה זוֹ בְּזָדוֹן, וְעָבַר יוֹם י״ד וְלֹא הִקְרִיב, וְהוּא לֹא טָמֵא, וּבְדֶרֶךְ רְחוֹקָה לֹא הָיָה — הֲרֵי זֶה חַיָּב כָּרֵת. בַּמֶּה דְּבָרִים אֲמוּרִים — כְּשֶׁלֹּא עָשָׂה גַּם הַשֵּׁנִי, בֵּין בְּזָדוֹן וּבֵין בִּשְׁגָגָה; עַיֵּן לְקַמָּן בְּפָרָשַׁת בְּהַעֲלֹתְךָ — יִתְבָּאֵר כָּל פְּרָטֵי מִשְׁפָּטִים אֵלּוּ. וְאִם בִּטְּלָהּ בִּשְׁגָגָה — אֲפִלּוּ שָׁגַג גַּם בַּשֵּׁנִי וְלֹא עֲשָׂאוֹ — פָּטוּר מֵחַטָּאת; שֶׁזֶּה אַחַת מִג׳ חֲטָאִים, שֶׁאַף שֶׁיֵּשׁ בִּזְדוֹנָן כָּרֵת — אֵין בְּשִׁגְגָתָן חַטָּאת, מִפְּנֵי שֶׁאֵין עַל זֶה לָאו, אֶלָּא מִצְוַת עֲשֵׂה בִּלְבַד:

**וְאֵין** מִצְוָה זוֹ נוֹהֶגֶת אֶלָּא בִּזְמַן שֶׁבֵּית הַמִּקְדָּשׁ קַיָּם. וּלְפִי שֶׁאָמְרוּ זִכְרוֹנָם לִבְרָכָה, שֶׁהָעוֹסֵק בְּתוֹרַת קָרְבָּן — כְּאִלּוּ הִקְרִיב — רָאוּי לַעֲסוֹק בְּמִשְׁפָּטָיו גַּם בַּזְּמַן הַזֶּה, בְּעֶרֶב פֶּסַח אַחַר חֲצוֹת. וְיֹאמַר מִקֹּדֶם: "יְהִי רָצוֹן מִלְּפָנֶיךָ, אֱלֹקֵינוּ וֵאלֹקֵי אֲבוֹתֵינוּ, שֶׁתְּהֵא (אֲמִירָה אוֹ) לִמּוּד זֶה חָשׁוּב וּמְקֻבָּל וּמְרֻצֶּה לְפָנֶיךָ כְּאִלּוּ עֲשִׂיתִיו"; כִּמְבֹאָר בְּאֹרַח חַיִּים (סִימָן א׳), דְּצָרִיךְ נַמִּי אֲמִירַת "יְהִי רָצוֹן", לְשֶׁיַּעֲלֶה וְיֵחָשֵׁב כְּאִלּוּ וְכוּלֵי. וְעַיֵּן שָׁם בְּבֵית יוֹסֵף, דְּגַם בְּלֹא "יְהִי רָצוֹן" — קְרִיאַת הַפָּרָשָׁה מְכַפֶּרֶת קְצָת מִיהַת. וְעַיֵּן בִּתְבוּאוֹת שׁוֹר (סְעִיף קָטָן ו׳), כָּתַב דְּאַכַּתִּי "יְהִי רָצוֹן" קָאָמַר, וְצָרִיךְ רַחֲמִים
