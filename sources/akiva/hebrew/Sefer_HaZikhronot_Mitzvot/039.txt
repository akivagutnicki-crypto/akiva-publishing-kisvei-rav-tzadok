~ לָהֶם סְעוּדוֹת גְּדוֹלוֹת, כְּדֵי שֶׁיִּהְיוּ רְגִילִין לָבוֹא. זוּג שֶׁבָּא רִאשׁוֹן — בּוֹדְקִין אוֹתוֹ רִאשׁוֹן; וּמַכְנִיסִין אֶת הַגָּדוֹל שֶׁבָּהֶן, וְשׁוֹאֲלִים לוֹ: כֵּיצַד רְאִיתָ אוֹתָהּ? בִּצְפוֹן הַשֶּׁמֶשׁ אוֹ בִּדְרוֹמָהּ? וְכַמָּה רְחָבָה, וְכַמָּה גְּבוֹהָה מֵהָאָרֶץ לְפִי רְאוּת הָעַיִן? וּלְהֵיכָן הָיוּ רָאשֵׁי קַרְנַיִם נוֹטִין? וְהַבֵּית דִּין מְחַשְּׁבִין זֶה, כְּדַרְכֵי הָאִצְטַגְנִינִים, וְיוֹדְעִין הַלְּבָנָה כְּשֶׁתֵּרָאֶה בְּחֹדֶשׁ זֶה — אֵיךְ תֵּרָאֶה; אִם נִמְצְאוּ דְּבָרָיו מְכֻוָּנִים לַחֶשְׁבּוֹן — מַכְנִיסִים אֶת חֲבֵרוֹ; אִם נִמְצְאוּ דִּבְרֵיהֶם מְכֻוָּנִים — עֵדוּתָן קַיֶּמֶת. וּשְׁאָר כָּל הַזּוּגוֹת שׁוֹאֲלִין אוֹתָן רָאשֵׁי דְּבָרִים — לֹא שֶׁצְּרִיכִין לָהֶם, אֶלָּא כְּדֵי שֶׁלֹּא יֵצְאוּ בְּפַחֵי נֶפֶשׁ, כְּדֵי שֶׁיִּהְיוּ רְגִילִין לָבוֹא. וְאַחַר כָּךְ, אַחַר שֶׁהִתְקַיְּמוּ הָעֵדוּת, רֹאשׁ בֵּית דִּין — וְהוּא הַנָּשִׂיא — אוֹמֵר: "מְקֻדָּשׁ", וְכָל הָעָם עוֹנִין אַחֲרָיו ב׳ פְּעָמִים: "מְקֻדָּשׁ, מְקֻדָּשׁ". וְאֵין מְקַדְּשִׁין אֶת הַחֹדֶשׁ אֶלָּא בִּשְׁלֹשָׁה, וְאֵין מְחַשְּׁבִין אֶלָּא בְּג׳. וְאֵין מְקַדְּשִׁין אֶלָּא חֹדֶשׁ שֶׁנִּרְאָה בִּזְמַנּוֹ (בַּגְּמָרָא כ״א עַמּוּד א׳, מַשְׁמַע דְּלוֹי סָבִירָא לֵיהּ מְקַדְּשִׁין בְּיוֹם ל״א, דְּלָא כְּרַבִּי אֶלְעָזָר בְּרַבִּי צָדוֹק. גַּם שָׁם עַמּוּד ב׳ בְּרַשִׁ״י, דִּבּוּר הַמַּתְחִיל "שְׁלוּחִין" — מוֹקֵי כֵּן סְתָמָא דְּדִבְרֵי אַבָּיֵי מִדְּמַתְנִיתִין וּדְבָרַיְיתָא, עַיֵּן שָׁם; אוּלַי סָבִירָא לֵיהּ כֵּן הֲלָכָה, וְצָרִיךְ עִיּוּן. וְעַיֵּן סַנְהֶדְרִין (י׳ עַמּוּד ב׳) בְּרַשִׁ״י, כָּתַב כְּרַבִּי אֶלְעָזָר בְּרַבִּי צָדוֹק, עַיֵּן שָׁם). אֲבָל נִרְאָה בְּיוֹם ל״א — אֵין מְקַדְּשִׁין אוֹתוֹ, שֶׁכְּבָר קִדְּשׁוּהוּ שָׁמַיִם (עַיֵּן בְּרַמְבַּ״ן סוֹף מִצְוַת עֲשֵׂה קנ״ג, וְאֵין רְאָיָתוֹ מֵרַבִּי אֶלְעָזָר בְּרַבִּי צָדוֹק מוּבֶנֶת, דְּהָא טַעְמָא קָאָמַר "שֶׁכְּבָר" וְכוּלֵי). וְאֵין מְקַדְּשִׁין, וְלֹא מְקַבְּלִין הָעֵדוּת, אֶלָּא בַּיּוֹם; וְאִם קִדְּשׁוּהוּ בַּלַּיְלָה — אֵינוֹ מְקֻדָּשׁ. וַאֲפִלּוּ נֶחְקְרוּ עֵדִים כְּבָר בַּיּוֹם, וַהֲוֵי תְּחִלַּת דִּין בַּיּוֹם; אוֹ שֶׁרָאוּ בֵּית דִּין וְכָל יִשְׂרָאֵל בַּיּוֹם, וְנִתְפַּרְסֵם הַדָּבָר לַכֹּל, וְלֹא הִסְפִּיקוּ לוֹמַר "מְקֻדָּשׁ הַחֹדֶשׁ" עַד בַּלַּיְלָה — אֵין מְקַדְּשִׁים בַּלַּיְלָה; בֵּין שֶׁרָאוּ בְּכ״ט — אֵין מְקַדְּשִׁים בְּלֵיל ל׳; בֵּין שֶׁרָאוּ בְּל׳ וְלֹא הִסְפִּיקוּ — אֵין מְקַדְּשִׁים בְּלֵיל ל״א, אֲפִלּוּ לְדִבְרֵי הָאוֹמֵר שֶׁיִּתְבָּאֵר לְהַלָּן, שֶׁאִם בָּאוּ אַחַר כָּךְ עֵדִים — מְקַדְּשִׁים לְמַפְרֵעַ. וְיֵשׁ אוֹמְרִים (חִדּוּשֵׁי הָרַשְׁבָּ״א, דְּלָא כְּהַטּוּרֵי אֶבֶן, עַיֵּן שָׁם) — אֲפִלּוּ לְדִבְרֵי הַסּוֹבְרִים שֶׁאֵין מְקַדְּשִׁים לְמַפְרֵעַ — מִכָּל מָקוֹם, כָּל זְמַן שֶׁלֹּא הִפְסִיק זְמַן שֶׁאֵין מְקַדְּשִׁים בּוֹ בֵּינְתַיִם, וְנִגְבְּתָה הָעֵדוּת בִּזְמַנּוֹ — מְקַדְּשִׁים אַף אַחַר כָּךְ. אֲבָל מִכָּל מָקוֹם אֵין מְקַדְּשִׁים בַּלַּיְלָה; וְאַף עַל פִּי שֶׁעַל יְדֵי זֶה יִדָּחֶה, וִיהֵא לְמָחָר רֹאשׁ חֹדֶשׁ, וִיהֵא מְעֻבָּר:

**וּלְפִיכָךְ** בֵּית דִּין עַצְמָם, שֶׁרָאוּ בְּלֵיל ל׳ — אַף עַל פִּי שֶׁרְאִיַּת הַבֵּית דִּין עַצְמָם אֵינָהּ פְּחוּתָה מִשְּׁמִיעַת דִּבְרֵי הָעֵדִים — מִכָּל מָקוֹם, הוֹאִיל וְרָאוּ בַּלַּיְלָה, בְּשָׁעָה שֶׁאֵין רְאוּיִים לְדִין, אִם כֵּן לֹא הָיוּ יְכוֹלִים אָז לַעֲשׂוֹת דִּין, וְיִצְטָרְכוּ לְקַדֵּשׁ אַחַר כָּךְ בַּיּוֹם עַל פִּי רְאִיָּתָם בַּלַּיְלָה; וְאֵין רְאִיָּתָם אֶלָּא כְּעֵדוּת, כֵּיוָן שֶׁלֹּא יָכְלוּ לְקַדְּשׁוֹ תֵּכֶף לִרְאִיָּה, וְקַיְימָא לַן אֵין עֵד נַעֲשֶׂה דַּיָּן, כְּמוֹ שֶׁיִּתְבָּאֵר בִּמְקוֹמוֹ. לְפִיכָךְ צְרִיכִים לְהוֹשִׁיב עוֹד שְׁנַיִם אֲחֵרִים אֵצֶל אֶחָד מֵהֶם, וְיָעִידוּ
