~ בִּבְרִיתוֹ שֶׁל אַבְרָהָם אָבִינוּ", וַ"אֲשֶׁר קִדֵּשׁ יְדִיד מִבֶּטֶן". וְיֵשׁ חוֹלְקִים גַּם בָּזֶה, וּמַשְׁמַע דְּהַחוֹלְקִים גַּם עַל "אֲשֶׁר קִדֵּשׁ יְדִיד מִבֶּטֶן" — שֶׁלֹּא לְבָרֵךְ, וּדְלָא כְּר״י עֶמְדִּין. מִיהוּ אוֹמֵר "אֱלֹקֵינוּ וֵאלֹקֵי אֲבוֹתֵינוּ, קַיֵּם" וְכוּלֵי. וְנִרְאֶה דַּוְקָא גֵּר שֶׁנִּתְגַּיֵּר, דְּקָטָן שֶׁנּוֹלַד דְּמֵי; אֲבָל מְהָלוֹ לְקָטָן עוֹבֵד כּוֹכָבִים אוֹ מוּמָר, דְּצָרִיךְ לְהַטִּיף מִמֶּנּוּ דַּם בְּרִית — מְבָרֵךְ כָּל הַבְּרָכוֹת, כְּמִילָה גְּמוּרָה, דְּמִילַת עוֹבֵד כּוֹכָבִים לָאו כְּלוּם הוּא מִדְּאוֹרַיְתָא. וְכֵן מְהָלוֹ תּוֹךְ ח׳, כְּשֶׁצָּרִיךְ אַחַר כָּךְ לְהַטִּיף דַּם בְּרִית — שֶׁיְּבָרֵךְ. וְאִם מְהָלוֹ תּוֹךְ ח׳ מִפְּנֵי הַסַּכָּנָה, וְכָל שֶׁכֵּן תִּינוֹק שֶׁמֵּת קֹדֶם שֶׁנִּמּוֹל, שֶׁנָּהֲגוּ לְמוּלוֹ — לֹא יְבָרֵךְ. וּכְשֶׁחוֹזְרִים עַל צִיצִין הַמְעַכְּבִים הַמִּילָה — צָרִיךְ לַחֲזוֹר וּלְבָרֵךְ כָּל הַבְּרָכוֹת עַד "כּוֹרֵת הַבְּרִית"; אֲבָל אַל יֹאמַר "אֱלֹקֵינוּ וֵאלֹקֵי אֲבוֹתֵינוּ, קַיֵּם" וְכוּלֵי, שֶׁכְּבָר אָמַר. מִכָּל שֶׁכֵּן מָל וְלֹא פָּרַע, אַחַר כָּךְ שֶׁפּוֹרֵעַ — צָרִיךְ לְבָרֵךְ. אֲבָל אִם מָל כָּהֹגֶן, וְנִמְשְׁכָה עָרְלָתוֹ — נִרְאֶה לִי שֶׁלֹּא לְבָרֵךְ כְּלָל. וְאַנְדְּרוֹגִינוֹס, וְכֵן כָּל שֶׁמִּילָתוֹ סָפֵק — אֵין מְבָרְכִים עַל מִילָתוֹ; אֲבָל אוֹמְרִים "אֱלֹקֵינוּ וֵאלֹקֵי אֲבוֹתֵינוּ, קַיֵּם" וְכוּלֵי. וּמַמְזֵר — מְבָרְכִים עַל מִילָתוֹ, כְּיִשְׂרָאֵל גָּמוּר, עַד "כּוֹרֵת הַבְּרִית"; אֲבָל לֹא יֹאמַר "אֱלֹקֵינוּ וֵאלֹקֵי אֲבוֹתֵינוּ, קַיֵּם" וְכוּלֵי, שֶׁאֵין מְבַקְשִׁים עָלָיו רַחֲמִים. וְגַם יְפַרְסֵם בִּשְׁעַת מִילָתוֹ שֶׁהוּא מַמְזֵר, כְּגוֹן לְמוּלוֹ בַּחֲצַר בֵּית הַכְּנֶסֶת. בַּמֶּה דְּבָרִים אֲמוּרִים — בְּאֵשֶׁת אִישׁ; אֲבָל פְּנוּיָה שֶׁיָּלְדָה, הֵיכִי דְּלֵיכָּא חֲשַׁשׁ סְפֵק מַמְזֵרוּת אוֹ חָלָל לִפְסוֹל לִכְהֻנָּה — אֵין לְפַרְסְמוֹ. אֲבָל מִכָּל מָקוֹם נָכוֹן לְבַיֵּשׁ יוֹלְדָיו, וְשֶׁלֹּא לָמוּל בְּבֵית הַכְּנֶסֶת כְּכָל יְלָדִים. וְנִרְאֶה דִּמְבַקְשִׁים גַּם כֵּן עָלָיו רַחֲמִים, כֵּיוָן דְּוָלָד כָּשֵׁר הוּא:

**אִם** יֵשׁ ב׳ תִּינוֹקוֹת לָמוּל — יְבָרֵךְ בְּרָכָה אַחַת לִשְׁנֵיהֶם, וַאֲפִלּוּ ב׳ מָלִים (וְנִרְאֶה בִּלְבַד בְּמָקוֹם אֶחָד; הָא לָאו הָכֵי — מְקוֹם הֶפְסֵק הֲוֵי, אַף דְּהָלַךְ לִשְׁמוֹעַ הַבְּרָכָה). וְאַל יַעֲשֶׂה כֵּן — הָאֶחָד יְבָרֵךְ "עַל הַמִּילָה", וְעוֹלֶה גַּם לָרִאשׁוֹן, וְהַשֵּׁנִי יְבָרֵךְ "אֲשֶׁר קִדֵּשׁ יְדִיד מִבֶּטֶן", וְעוֹלֶה גַּם לָרִאשׁוֹן — אִם מִתְכַּוְּנִים זֶה לְהוֹצִיא וְזֶה לָצֵאת. וַאֲפִלּוּ אֵין הַנַּעַר לְפָנָיו בִּשְׁעַת הַבְּרָכָה — כֵּיוָן שֶׁדַּעְתּוֹ עָלָיו (וְאֶפְשָׁר דִּכְהַאי גַּוְנָא אוֹמֵר "אֱלֹקֵינוּ וֵאלֹקֵי אֲבוֹתֵינוּ, קַיֵּם אֶת הַיְלָדִים הָהֵם לַאֲבוֹתָם וּלְאִמּוֹתָם, וְיִקָּרְאוּ שְׁמָם" וְכוּלֵי, וְכוֹלְלָם בְּיַחַד; אֲבָל לֹא מַשְׁמַע כֵּן לִכְאוֹרָה מֵרַמָּ״א), וּבִלְבַד שֶׁלֹּא יַסִּיחַ דַּעְתּוֹ בֵּינְתַיִם. אֲבָל אִם שָׂח בֵּינְתַיִם בִּדְבָרִים שֶׁאֵינָם מִצָּרְכֵי הַמִּילָה, אוֹ שֶׁלֹּא הָיָה דַּעְתּוֹ מִתְּחִלָּה עַל הַשֵּׁנִי — צָרִיךְ לְבָרֵךְ "אֲשֶׁר קִדֵּשׁ יְדִיד מִבֶּטֶן" עַל הָא׳, וְלַחֲזוֹר וּלְבָרֵךְ עַל מִילַת הַשֵּׁנִי. אֲבָל שָׂח בְּצָרְכֵי הַמִּילָה — לָא הֲוֵי הֶפְסֵק. וַאֲפִלּוּ שָׁכַח וּבֵרֵךְ הָרִאשׁוֹן "אֲשֶׁר קִדֵּשׁ יְדִיד מִבֶּטֶן" — אֵין צָרִיךְ הַשֵּׁנִי לְבָרֵךְ "עַל הַמִּילָה" הַב׳, אֲפִלּוּ הוּא מוֹהֵל אַחֵר, דִּבְרָכָה דְּמֵעִנְיָנֵיהּ אֵינָהּ הֶפְסֵק. וְאִם ב׳ הַיְלָדִים שֶׁל אָדָם אֶחָד — הָאָב אוֹמֵר "לְהַכְנִיסָם בִּבְרִיתוֹ שֶׁל אַבְרָהָם אָבִינוּ", וְכֵן אוֹמֵר "קַיֵּם אֶת הַיְלָדִים" וְגוֹמֵר. וִיתַקֵּן כָּל צָרְכֵי הַמִּילָה לְכָל אֶחָד בִּפְנֵי עַצְמוֹ — כְּגוֹן הַנֵּר בִּמְקוֹמוֹת
