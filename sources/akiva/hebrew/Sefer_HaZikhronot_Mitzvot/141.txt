~ בְּחָכְמָה וּבִתְבוּנָה, קָאֵי. וּמַה שֶּׁכָּתַב מִדִּבְרֵי רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה: "וְדַע מַה שֶּׁתָּשִׁיב לְאֶפִּיקוֹרוֹס" — אֲנִי תָּמֵהַּ: הֲלֹא הַתְחָלָתוֹ "הֱוֵי שָׁקוּד לִלְמוֹד תּוֹרָה", וְאַחַר כָּךְ אָמַר "וְדַע" וְכוּלֵי; מַשְׁמַע, דְּעַל יְדֵי הַשְּׁקִידָה בַּתּוֹרָה הִיא הַיְּדִיעָה לְהָשִׁיב לְאֶפִּיקוֹרוֹס (כַּאֲשֶׁר כֵּן הוּא הָאֱמֶת); וְאַיֵּה מָקוֹם לְלִמּוּד מוֹפֵת הַפִילוֹסוֹפְיָא? וּכְבָר הָרַשְׁבָּ״א זִכְרוֹנוֹ לִבְרָכָה וְדַעֲמֵיהּ, כַּת גְּדוֹלָה מֵרַבָּנֵי זְמַנּוֹ, הִסְכִּימוּ בְּחֵרֶם שֶׁלֹּא יְלַמֵּד אָדָם אֶת בְּנוֹ חָכְמוֹת שֶׁלְּאַחַר הַטֶּבַע — וְהִיא חָכְמַת הַפִילוֹסוֹפְיָא — בַּבּוֹרֵא וּבְמִדּוֹתָיו:

**וּמִכָּל** מָקוֹם, הוֹאִיל וּלְהָרַמְבַּ״ם וּבַעַל "חוֹבַת הַלְּבָבוֹת" — וְשֶׁעִמָּהֶם כַּת גְּדוֹלָה מִגְּדוֹלֵי הָרִאשׁוֹנִים — עִקַּר מִצְוַת "אָנֹכִי" הוּא הַיְּדִיעָה בְּמוֹפֵת, אָמַרְתִּי לְהַזְכִּיר מוֹפֵת אֶחָד, הַיּוֹתֵר בָּרוּר וְהַיּוֹתֵר חָזָק שֶׁבָּהֶם, עַל מְצִיאוּת הַבּוֹרֵא יִתְבָּרַךְ וַאֲמִתּוּתוֹ. וְאַקְדִּים הַקְדָּמוֹת: א) הַמִּקְרֶה לֹא יַתְמִיד, וְהַמַּתְמִיד אֵינֶנּוּ מִקְרִיִּי, אֲבָל נִפְעָל עַל יְדֵי פּוֹעֵל מְכַוֵּן. ב) לֹא יִפְעַל פּוֹעֵל אֶת עַצְמוֹ, אֲבָל יִתְפַּעֵל עַל יְדֵי זוּלָתוֹ (כִּי אִם נֹאמַר שֶׁיִּפְעַל — קֹדֶם הֱוָיָתוֹ הָיָה אֶפֶס, וְאַחַר הֱוָיָתוֹ כְּבָר נִפְעַל). ג) כָּל שֶׁיֵּשׁ לוֹ תְּחִלָּה — יֵשׁ לוֹ תִּכְלָה, וְכָל שֶׁאֵין לוֹ תְּחִלָּה — אֵין לוֹ תִּכְלָה; וְכֵן בְּהֵפֶךְ. ד) סִבּוֹת וּמְסֻבָּבִים, אוֹ פּוֹעֲלִים וְנִפְעָלִים, בְּלֹא תַּכְלִית — שֶׁקֶר (וּמִלְּבַד הֱיוֹת דָּבָר זֶה פָּשׁוּט בַּשֵּׂכֶל — עוֹד מֻכְרָח לְפִי הַקְדָּמָה ג׳ הַנִּזְכֶּרֶת לְעֵיל: כִּי אַחַר שֶׁבְּיַד הָאָדָם לְכַלּוֹת מְסֻבָּב אוֹ פְּעֻלָּה אַחַת, וְלֹא יָשׁוּב לִפְעוֹל עוֹד — וְנִמְצָא יֵשׁ לוֹ תִּכְלָה, וְעַל כָּרְחֲךָ הָיָה לוֹ תְּחִלָּה, וְאֵינוֹ בִּלְתִּי בַּעַל תַּכְלִית). וְהַתּוֹלָדָה מֵהַקְדָּמוֹת אֵלּוּ: אָנוּ רוֹאִים, בְּדִמְיוֹן, מִין מִמִּינֵי הַצְּמָחִים אוֹ מִין חַי (אוֹ שְׁאָר דָּבָר שֶׁיִּהְיֶה) — נִשְׁאַל: מֵאַיִן הוּא? הֲלֹא אוֹכֵל מִזֶּרַע; וְהַזֶּרַע — מִצּוֹמֵחַ אוֹ חַי כַּיּוֹצֵא בּוֹ; וְהַצּוֹמֵחַ אוֹ חַי הַהוּא — גַּם כֵּן מִזֶּרַע; וְהַזֶּרַע הַהוּא — מִצּוֹמֵחַ אוֹ חַי כַּיּוֹצֵא בּוֹ. וְכֵן נָשׁוּב לִשְׁאוֹל עַל הַצּוֹמֵחַ אוֹ הַחַי, וְנָשׁוּב לִשְׁאוֹל עַל הַזֶּרַע — וְסִבּוֹת וּמְסֻבָּבִים עַד בִּלְתִּי בַּעַל תַּכְלִית — שֶׁקֶר; וּבְמִקְרֶה אֵינָם, כִּי רוֹאִים אָנוּ אוֹתָם מַתְמִידִים; וְעַל כָּרְחֲךָ נִפְעֲלוּ, וְעַל יְדֵי פּוֹעֵל זוּלָתָם. וְנִשְׁאַל עַל הַפּוֹעֵל הַהוּא — מִי פְּעָלוֹ? עַד שֶׁבְּהֶכְרֵחַ לָבוֹא בְּסוֹף כָּל דָּבָר שֶׁנִּפְעַל עַל יְדֵי הָאֱלוֹקַּ — וְהוּא רִאשׁוֹן שֶׁאֵין רֵאשִׁית לְרֵאשִׁיתוֹ; וְכֵיוָן שֶׁאֵין לוֹ תְּחִלָּה — אֵין לוֹ תִּכְלָה; וְהוּא רִאשׁוֹן וְהוּא אַחֲרוֹן, וּמִבַּלְעָדָיו אֵין אֱלֹקִים:
