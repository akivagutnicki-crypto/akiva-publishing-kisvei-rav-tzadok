~ כְּסִדְרָן שֶׁהָיוּ בְּמַעֲשֵׂה בְרֵאשִׁית — עַל אַחַת מִכָּל אֵלּוּ מְבָרֵךְ: "בָּרוּךְ אַתָּה ה׳ אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם עוֹשֶׂה מַעֲשֵׂה בְרֵאשִׁית", מִפְּנֵי שֶׁכֵּן הָיוּ בִּשְׁעַת קְבִיעָתָן וּתְחִלַּת תְּלִיָּתָן בָּרָקִיעַ — כָּל אֶחָד בִּמְקוֹמוֹ הַמְיֻחָד לוֹ. וְיֵשׁ אוֹמְרִים, דִּפְלִיגֵי לְעֵיל אַבִּרְכַּת חַמָּה בִּתְקוּפָתָהּ — פְּלִיגֵי נַמִּי אַהֲנָךְ בְּרָכוֹת; וְהָעִקָּר דִּמְבָרְכִין. מִיהוּ, נִרְאֶה דְּאֵין חִיּוּב לְדַקְדֵּק אַחַר זֶה, לִרְאוֹת אִם נִזְדַּמְּנוּ כֵּן — לְבָנָה בְּטָהֳרָתָהּ, וְכוֹכָבִים וּמַזָּלוֹת כְּסִדְרָן; וַאֲפִלּוּ יוֹדֵעַ עַל פִּי חֶשְׁבּוֹן שֶׁצְּרִיכִים לִהְיוֹת בִּזְמַן הַהוּא כֵּן — אֵין צָרִיךְ לְדַקְדֵּק אַחֲרֵיהֶם לִרְאוֹתָם; אֶלָּא שֶׁאִם נִזְדַּמֵּן שֶׁרָאָם כַּנִּזְכָּר לְעֵיל — מְבָרֵךְ עֲלֵיהֶם:

**הָרוֹאֶה** לְבָנָה בְּחִדּוּשָׁהּ, בִּתְחִלַּת חֹדֶשׁ — כְּשֶׁאוֹרָהּ מִתְחַדֵּשׁ וְזוֹרֵחַ, וּזְרִיחָתָהּ נִכֶּרֶת עַל גַּבֵּי קַרְקַע, שֶׁנֶּהֱנִין מֵאוֹרָהּ (וְלָכֵן נָהֲגוּ לְדַקְדֵּק אַחַר צֵל) — מִצְוַת חֲכָמִים לְבָרֵךְ (וְגַם הַסּוּמָא חַיָּב לְבָרֵךְ; אֲבָל נָשִׁים פְּטוּרוֹת, וְאֵין לָהֶם לְהַחְמִיר כְּלָל בָּזֶה עַל עַצְמָם לְבָרֵךְ — אַף בְּלִי שֵׁם וּמַלְכוּת — מִפְּנֵי שֶׁהֵם גָּרְמוּ פְּגַם הַלְּבָנָה בְּחֵטְא חַוָּה. וְאָבֵל — אִם יַשְׁלִים אֲבֵלוּתוֹ עַד י׳ בַּחֹדֶשׁ — יַמְתִּין מִלְּקַדֵּשׁ בַּאֲבֵלוּתוֹ; וְאִם לָאו — לֹא יַמְתִּין עַד לֵיל י״א. וְר״י עֶמְדִּין כָּתַב: עַד ז׳, אִם לֹא יַשְׁלִימוֹ — לֹא יַמְתִּין, וִיקַדֵּשׁ בַּאֲבֵלוּתוֹ. קָטָן, מִשֶּׁהִגִּיעַ לְחִנּוּךְ — רָאוּי לְחַנְּכוֹ בָּהּ) מְעֻמָּד, וְרַגְלָיו מְיֻשָּׁרוֹת, אַחֲרֵי תְלוֹתוֹ עֵינָיו לְהִסְתַּכֵּל בָּהּ (בִּתְחִלַּת הַבְּרָכָה; וּלְדַעַת יֵשׁ אוֹמְרִים — כָּל מֶשֶׁךְ הַבְּרָכָה; אֲבָל אַחַר כָּךְ אֵין לְהִסְתַּכֵּל בָּהּ): "בָּרוּךְ אַתָּה ה׳ אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם אֲשֶׁר בְּמַאֲמָרוֹ בָּרָא שְׁחָקִים" וְגוֹ׳, וּמְסַיֵּם בְּשֵׁם: "בָּרוּךְ אַתָּה ה׳ מְחַדֵּשׁ חֳדָשִׁים". וְאוֹמֵר: "סִימָן טוֹב תְּהֵא (לְשׁוֹן נְקֵבָה, עַל הַלְּבָנָה) לָנוּ וּלְכָל יִשְׂרָאֵל" ג׳ פְּעָמִים; וּבְאֵלִיָּה רַבָּה, בְּשֵׁם כמ״ת — לוֹמַר רַק תֵּבוֹת "סִימָן טוֹב" בִּלְבַד ג׳ פְּעָמִים. וְאוֹמֵר "בָּרוּךְ יוֹצְרֵךְ" וְכוּלֵי; וְאַחַר כָּךְ רוֹקֵד (וְיִזָּהֵר שֶׁלֹּא יִכְרַע בִּרְכָּיו בִּרְקִידָתוֹ, דְּמֶחֱזֵי כְּכוֹרֵעַ לַלְּבָנָה; אֶלָּא זוֹקֵף עַל אֶצְבְּעוֹת רַגְלָיו), וְאוֹמֵר "כְּשֵׁם שֶׁאֲנִי רוֹקֵד" וְכוּלֵי; וְאוֹמֵר "תִּפֹּל עֲלֵיהֶם" וְגוֹ׳, וּלְמַפְרֵעַ "כָּאָבֶן יִדְּמוּ" וְכוּלֵי (וְיֵשׁ אוֹמְרִים: עַד "קָנִיתָ", וּלְמַפְרֵעַ "קָנִיתָ זוּ עַם" וְכוּלֵי) — ג׳ פְּעָמִים. וְכָתַב בַּטּוּר לוֹמַר אַחַר כָּךְ "אָמֵן וְאָמֵן הַלְלוּיָהּ"; וְלֹא נָהֲגוּ כֵן. וּבַלְּבוּשׁ כָּתַב כָּאן לוֹמַר "סִימָן טוֹב" וְכוּלֵי. וְאַחַר כָּךְ אוֹמֵר לַחֲבֵרוֹ ג׳ פְּעָמִים "שָׁלוֹם עָלֶיךָ" (נֻסַּח הָאֲרִיזַ״ל: "עֲלֵיכֶם"), וּמֵשִׁיב — הֲוֵי כְּשׁוֹאֵל. וְאַחַר כָּךְ אוֹמֵר "דָּוִד מֶלֶךְ יִשְׂרָאֵל חַי וְקַיָּם". וְאַחַר כָּךְ — קַבָּלָה מֵרַבֵּינוּ יְהוּדָה הֶחָסִיד — לוֹמַר פָּסוּק "קוֹל דּוֹדִי הִנֵּה" וְכוּלֵי, "דּוֹמֶה דוֹדִי" וְכוּלֵי, עַד "מִן הַחֲרַכִּים"; וְיֵשׁ אוֹמְרִים עוֹד "שִׁיר לַמַּעֲלוֹת אֶשָּׂא" וְכוּלֵי, וְ"הַלְלוּ אֵל בְּקָדְשׁוֹ" וְכוּלֵי, וְ"תָנָא דְּבֵי רַבִּי יִשְׁמָעֵאל: אִלְמָלֵא" וְכוּלֵי, "אָמַר אַבָּיֵי" וְכוּלֵי:

**אַחַר כָּךְ** יְנַעֵר שׁוּלֵי בְּגָדָיו, וְיֵלֵךְ לְבֵיתוֹ בְּלֵב טוֹב, שֶׁזֶּה כְּאִלּוּ קִבֵּל פְּנֵי הַשְּׁכִינָה; וּמֻבְטָח שֶׁלֹּא יָמוּת בְּאוֹתוֹ חֹדֶשׁ בְּמִיתָה מְשֻׁנָּה (וְעַיֵּן
