~ גַּם שְׁנַת תֹּהוּ בְּמִסְפַּר הַשָּׁנִים, וּבְיִחוּד לְעִנְיַן תְּקוּפוֹת הַחַמָּה; שֶׁתְּקוּפָה רִאשׁוֹנָה הָיְתָה בִּשְׁנַת תֹּהוּ, שֶׁהֲרֵי הַחַמָּה נוֹצְרָה קֹדֶם יְצִירַת אָדָם הָרִאשׁוֹן. וְהִנֵּה יְצִירַת הַחַמָּה בְּיוֹם ד׳, כַּאֲשֶׁר בָּא בַּכָּתוּב; וּלְפִי שֶׁקִּבְּלוּ חֲכָמִים שֶׁכָּל מַעֲשֵׂה בְרֵאשִׁית בְּקוֹמָתָן נִבְרְאוּ, וְעִקַּר אוֹר חַמָּה וּבְהִירוּתָהּ נִרְאֶה בְּסוֹף שָׁעָה ג׳ — אִם כֵּן הָיְתָה יְצִירַת הַחַמָּה תְּחִלַּת שָׁעָה ד׳ שֶׁל יוֹם ד׳, וְאָז תְּקוּפַת תִּשְׁרֵי. וְכֵיוָן שֶׁמֶּרְחַק תְּקוּפָה מֵחֲבֶרְתָּהּ ז׳ שָׁעוֹת וּמֶחֱצָה — אִם כֵּן הָיְתָה תְּקוּפַת נִיסָן הַקּוֹדֶמֶת בִּתְחִלַּת לֵיל ד׳, שֶׁהוּא מֶרְחַק ט״ו שָׁעוֹת מִתְּקוּפַת תִּשְׁרֵי — י״ב שְׁעוֹת הַלַּיְלָה וְג׳ שְׁעוֹת הַיּוֹם — ב׳ פְּעָמִים ז׳ שָׁעוֹת וּמֶחֱצָה שֶׁל תְּקוּפַת נִיסָן וְתַמּוּז:

**וּמֵעַתָּה** נוּכַל לֵידַע כָּל תְּקוּפוֹת הַשָּׁנָה, בִּהְיוֹת הַמֶּרְחָק בֵּין תְּקוּפָה לִתְקוּפָה רַק ז׳ שָׁעוֹת וּמֶחֱצָה, וְהַמֶּרְחָק בֵּין תְּקוּפוֹת שָׁנָה לְשֶׁל אַחֲרֶיהָ — ד׳ פְּעָמִים ז׳ שָׁעוֹת וּמֶחֱצָה, שֶׁהֵם יוֹם וּרְבִיעַ הַיּוֹם; וּבְב׳ שָׁנִים — הַמֶּרְחָק ב׳ יָמִים וּמֶחֱצָה. נִמְצָא: כָּל ב׳ שָׁנִים מִתְגַּלְגְּלִים הַתְּקוּפוֹת עַל אוֹתָם שָׁעוֹת, רַק בְּב׳ שָׁנִים — הָא׳ הֵם בַּלַּיְלָה, וּבְב׳ אֲחֵרִים הֵם בַּיּוֹם; כִּי בַּעֲבוֹר ב׳ שָׁנִים תָּשׁוּב תְּקוּפַת נִיסָן לִיפּוֹל בִּתְחִלַּת הַיּוֹם, וְכֵן תְּשׁוּבֶינָה כָּל הַתְּקוּפוֹת. וּבְד׳ שָׁנִים — הַמֶּרְחָק ה׳ יָמִים; וְאָז שָׁבָה תְּקוּפַת נִיסָן לִמְקוֹמָהּ הָרִאשׁוֹן, לִתְחִלַּת הַלַּיְלָה; וְכֵן שָׁבִים כָּל הַתְּקוּפוֹת בְּכָל ד׳ שָׁנִים לִמְקוֹמָם הָרִאשׁוֹן בְּשָׁעוֹת, אֲבָל רְחוֹקִים זוֹ מִזּוֹ ה׳ יָמִים. וּבְז׳ פְּעָמִים ד׳ שָׁנִים — שֶׁהוּא כ״ח שָׁנִים — חוֹזְרִים הַתְּקוּפוֹת כֻּלָּם לִמְקוֹמָם הָרִאשׁוֹן, בֵּין בְּיָמִים בֵּין בְּשָׁעוֹת; וְחוֹזֶרֶת תְּקוּפַת נִיסָן לִתְחִלַּת לֵיל ד׳, שֶׁכְּבָר עָבְרוּ ז׳ פְּעָמִים ה׳ יָמִים, שֶׁהֵם ה׳ שָׁבוּעוֹת שְׁלֵמוֹת. וְלָכֵן כָּל כ״ח שָׁנִים נִקְרָא "מַחֲזוֹר", שֶׁהַחַמָּה חוֹזֶרֶת לִמְקוֹמָהּ, לִהְיוֹת הַתְּקוּפוֹת בִּמְקוֹמָם הָרִאשׁוֹן. וּלְפִי שֶׁיֵּשׁ מַחֲזוֹר שֶׁל י״ט שָׁנָה, כַּנִּזְכָּר לְעֵיל — נִקְרָא לָזֶה שֶׁל כ״ח "מַחֲזוֹר גָּדוֹל":

**וְכָל** ד׳ שָׁנִים שֶׁבְּמַחֲזוֹר גָּדוֹל — חוֹזְרִים הַתְּקוּפוֹת לִתְחִלָּתָן בְּשָׁעוֹת, וְלֹא בְּיָמִים; רַק מֶרְחַק ה׳ יָמִים בֵּין אֶחָד לַחֲבֵירוֹ. כֵּיצַד? בִּתְחִלַּת ד׳ שָׁנִים הָרִאשׁוֹנִים — תְּקוּפַת נִיסָן בְּרֹאשׁ לֵיל ד׳, כְּמוֹ שֶׁנִּתְבָּאֵר; בַּשְּׁנִיִּים — בְּרֹאשׁ לֵיל ב׳; בַּשְּׁלִישִׁיִּים — בְּרֹאשׁ לֵיל ז׳; וּבָרְבִיעִיִּים — בְּרֹאשׁ לֵיל ה׳; וּבַחֲמִישִׁיִּים — בְּרֹאשׁ לֵיל ג׳; בַּשִּׁשִּׁיִּים — בְּרֹאשׁ לֵיל א׳; בַּשְּׁבִיעִיִּים — בְּרֹאשׁ לֵיל ו׳. וְסִימָנָם: דבי״ז הגא״ו. וְכָל ב׳ שָׁנִים הַשְּׁנִיִּים בַּהז׳ הַנִּזְכָּרִים — הַתְּקוּפוֹת חוֹזְרִים בְּשָׁעוֹת, אֲבָל לֹא בָּעֵת; כִּי מַה שֶּׁהָיָה בַּלַּיְלָה — שָׁב לִהְיוֹת בַּיּוֹם, אוֹ לְהֵפֶךְ, כְּמוֹ שֶׁנִּתְבָּאֵר. וּתְקוּפַת נִיסָן — תְּחִלַּת ב׳ שָׁנִים הָאַחֲרוֹנִים שֶׁל הַז׳ — תָּמִיד בִּתְחִלַּת הַיּוֹם;
