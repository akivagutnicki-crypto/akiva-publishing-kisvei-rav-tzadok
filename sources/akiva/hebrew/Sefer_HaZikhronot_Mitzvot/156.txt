**וְנִמְצְאוּ** אֹפְנֵי הַחִיּוּבִים לְנִטְמָא בַּמִּקְדָּשׁ ג׳: שָׁהָה כְּשִׁעוּר, אוֹ הִשְׁתַּחֲוָה, אוֹ יָצָא בַּאֲרוּכָּה. וְכָל דְּבָרִים אֵלּוּ — הֲלָכָה לְמֹשֶׁה מִסִּינַי הֵן:

**וּלְפִיכָךְ** נִרְאֶה לִי, דִּמְצֹרָע שֶׁנִּצְטָרַע בִּירוּשָׁלַיִם, אוֹ מִי שֶׁנִּקְרָה אוֹ רָאָה זִיבָה בְּהַר הַבַּיִת — דְּכִי גְּמִירֵי, בָּעֲזָרָה הוּא דִּגְמִירֵי; אֲבָל בְּמַחֲנֵה לְוִיָּה וְיִשְׂרָאֵל — אֵינוֹ לוֹקֶה בְּכָל הָנֵי. מִיהוּ, צָרִיךְ לְעַיֵּן לְעִנְיַן שְׁהִיָּה, דְּיֵשׁ לוֹמַר דְּהַדָּבָר תָּלוּי בְּמַחֲלֹקֶת רַשִׁ״י וְתוֹסָפוֹת: דִּלְפֵירוּשׁ רַשִׁ״י (ט״ז עַמּוּד ב׳), דִּבּוּר הַמַּתְחִיל "טֻמְאָה" — דִּ"וְלֹא יְטַמְּאוּ" לֹא קָאֵי אַנִּטְמָא בַּמִּקְדָּשׁ — אִם כֵּן, בְּב׳ מַחֲנוֹת לֹא יִהְיֶה כְּלָל אַזְהָרָה אַנִּטְמָא שָׁם. אֲבָל לְמַה שֶּׁכָּתְבוּ שָׁם הַתּוֹסָפוֹת, דִּבּוּר הַמַּתְחִיל "תְּנָהוּ" — דִּ"וְלֹא יְטַמְּאוּ" קָאֵי גַּם אַנִּטְמָא בַּמִּקְדָּשׁ — אִם כֵּן, גַּם בְּב׳ מַחֲנוֹת יֵשׁ אַזְהָרָה אַנִּטְמָא שָׁם; וְאִם כֵּן יֵשׁ לוֹמַר, אַדְּרַבָּא: דְּכֵיוָן דְּלֹא גְּמִירֵי שִׁעוּר שְׁהִיָּה — אִם כֵּן, בִּשְׁהִיָּה כְּשִׁעוּר יְצִיאָה וּכְנִיסָה — לוֹקֶה; וּכְמוֹ שֶׁכָּתְבוּ הַתּוֹסָפוֹת (י״ז עַמּוּד א׳), דִּבּוּר הַמַּתְחִיל "אוֹ אֵין", עַיֵּן שָׁם. כֵּן נִרְאֶה לִי. וְצָרִיךְ לְעַיֵּן, לְפִי זֶה, בְּדִין יָצָא בַּדֶּרֶךְ אֲרוּכָּה — אִם צָרִיךְ גַּם כֵּן שִׁעוּר שְׁהִיַּית הֲלִיכַת דֶּרֶךְ אֲרוּכָּה טְפֵי מֵהַקְּצָרָה כְּשִׁעוּר יְצִיאָה וּכְנִיסָה, אוֹ אֲפִלּוּ בְּפָחוֹת דַּי, אוֹ אֲפִלּוּ בְּיוֹתֵר לֹא סַגִּי לְחַיֵּב; וְצָרִיךְ עִיּוּן:

**כְּבָר** זָכַרְנוּ אֳפָנִים הַמֻּתָּרִים לִיכָּנֵס טָמֵא בַּמִּקְדָּשׁ, בְּסוֹף סִימָן שֶׁלִּפְנֵי זֶה. זוּלַת בָּזֶה נִשְׁתַּנָּה הַלָּאו מֵהָעֲשֵׂה: בְּקָרְבְּנוֹת הַצִּבּוּר, וְשֶׁקָּבוּעַ לָהֶם זְמַן, הַקְּרֵבִים בְּטֻמְאָה, שֶׁנִּתְבָּאֵר לְעֵיל שָׁם דְּלֹא הֻתְּרוּ לִיכָּנֵס אֶלָּא לָעֲזָרָה, אֲבָל אִם נִכְנְסוּ לַהֵיכָל — טְעוּנִים שִׁלּוּחַ, עַיֵּן שָׁם — לְעִנְיַן הַלָּאו פְּטוּרִים, אִם נִכְנְסוּ לַהֵיכָל, מִכָּרֵת וּמַלְקוֹת בְּמֵזִיד, אוֹ קָרְבָּן בְּשׁוֹגֵג; שֶׁנֶּאֱמַר: "וִישַׁלְּחוּ מִן הַמַּחֲנֶה... אֶל מִחוּץ לַמַּחֲנֶה תְּשַׁלְּחוּם" — כָּל הֵיכִי דְּקָרֵינַן "אֶל מִחוּץ לַמַּחֲנֶה תְּשַׁלְּחוּם" — קָרֵינַן "וִישַׁלְּחוּ מִן הַמַּחֲנֶה"; וְכָל הֵיכִי דְּלָא קָרֵינַן "אֶל מִחוּץ לַמַּחֲנֶה" — כְּגוֹן הָכָא, דִּצְרִיכִים שֶׁיִּהְיוּ בָּעֲזָרָה — לָא קָרֵינַן בְּהוּ "וִישַׁלְּחוּ מִן הַמַּחֲנֶה". וְיֶתֶר פְּרָטֵי דִּינִים אֵלּוּ — נִתְבָּאֲרוּ בַּסִּימָן שֶׁלִּפְנֵי זֶה, וְיִתְבָּאֲרוּ בְּפָרָשַׁת תֵּצֵא (סִימָן ......), עַיֵּן שָׁם:

# מִצְוַת עֲשֵׂה שֶׁל תְּשׁוּבָה — סִימָן ...

**מִצְוַת** עֲשֵׂה שֶׁצִּוָּה הַבּוֹרֵא יִתְבָּרַךְ לְכָל אִישׁ אֲשֶׁר יֶחֱטָא וְאָשֵׁם — לְהִתְוַדּוֹת לְפָנָיו יִתְבָּרַךְ; שֶׁנֶּאֱמַר: "דַּבֵּר אֶל בְּנֵי יִשְׂרָאֵל: אִישׁ אוֹ אִשָּׁה כִּי יַעֲשׂוּ מִכָּל חַטֹּאת הָאָדָם לִמְעֹל מַעַל בַּה׳, וְאָשְׁמָה הַנֶּפֶשׁ הַהִוא — וְהִתְוַדּוּ אֶת
