**וּמָקוֹם** זֶה — הוּא נוֹטֶה מִתַּחַת קַו הַשָּׁוֶה, הַמְסַבֵּב בְּאֶמְצַע הָעוֹלָם מִן הַמִּזְרָח לְמַעֲרָב, כְּנֶגֶד רוּחַ צְפוֹנִית, מִן כ״ט מַעֲלוֹת עַד ל״ה מַעֲלוֹת, שֶׁהוּא כְּמוֹ צ׳ פַּרְסָאוֹת, וִירוּשָׁלַיִם בְּאֶמְצָעָם; וְזֶה הוּא הַנִּקְרָא "רֹחַב הָעוֹלָם", וְהוּא מִצָּפוֹן לְדָרוֹם. וְ"אֹרֶךְ הָעוֹלָם" נִקְרָא מִן הַמַּעֲרָב לְמִזְרָח; וּבַחֲצִי כַּדּוּר הָעֶלְיוֹן, שֶׁאָנוּ בּוֹ, יֵשׁ ק״פ מַעֲלוֹת מִן קְצֵה הַמַּעֲרָב עַד קְצֵה הַמִּזְרָח (וְכֵן בַּחֲצִי כַּדּוּר הַתַּחְתּוֹן, וְשֶׁשָּׁם מְדִינַת אָמֶעֶרִיקָא, וְאֵין לָנוּ עֵסֶק בָּהּ), וְאֶמְצָעוֹ הוּא צ׳ מַעֲלוֹת מִן הַמַּעֲרָב וְצ׳ מִן הַמִּזְרָח. וּמָקוֹם זֶה רָחוֹק מִקְּצֵה הַמַּעֲרָב בִּכְמוֹ ס״ג מַעֲלוֹת עַד ס״ט מַעֲלוֹת; וַהֲרֵי הוּא נוֹטֶה מֵאֶמְצַע הַיִּשּׁוּב כְּנֶגֶד רוּחַ מַעֲרָב בִּכְמוֹ כ״א מַעֲלוֹת עַד כ״ז מַעֲלוֹת, וִירוּשָׁלַיִם בְּאֶמְצָעָן. וְעַל הַמָּקוֹם הַזֶּה נִתְיַסְּדוּ כָּל הַחֶשְׁבּוֹנוֹת הַבָּאִים, וּשְׁעוֹת הַתְּקוּפָה וְהַמּוֹלָד הֵם שָׁמָּה בִּירוּשָׁלַיִם. וְכֵיוָן שֶׁבֵּאַרְנוּ שֶׁשְּׁעוֹת הַיּוֹם וְהַלַּיְלָה מִשְׁתַּנִּים, כְּמוֹ שֶׁאָמַרְנוּ — לְפִיכָךְ רָאִינוּ לְהַעְתִּיק לוּחַ שֶׁיִּסֵּד הַיָּשָׁ״ר מִקַּנְדִּיָּא, עַל "אֵיזֶהוּ מְקוֹמָן" שֶׁל בְּנֵי גּוֹלָה בְּאֵיזֶה מְדִינוֹת — מֶרְחָקָן בְּאֹרֶךְ וְרֹחַב; וּמִמֶּנּוּ יֵדַע הָרוֹצֶה זְמַן וָעֵת וָרֶגַע תְּקוּפוֹת וּמוֹלָדוֹת הָאֶמְצָעִי וְהָאֲמִתִּי בְּנָקֵל, אַחֲרֵי שֶׁיֵּדַע שְׁעָתָן וּזְמַנָּם בִּירוּשָׁלַיִם, כְּפִי חֶשְׁבּוֹן שֶׁיִּתְבָּאֵר, אִם יִרְצֶה הַשֵּׁם:

**(וּמִתּוֹךְ** זֶה נוּכַל גַּם כֵּן לַחְשׁוֹב וְלֵידַע רֶגַע לִקּוּי הַחַמָּה — כְּשֶׁיִּהְיֶה רֶגַע מוֹלַד הָאֲמִתִּי בְּאֹפֶן חֲצִי הַיּוֹם שֶׁלָּנוּ מַמָּשׁ, אוֹ קָרוֹב מִמֶּנּוּ בִּכְדֵי שָׁעָה אַחַת. וְכֵן נוּכַל לֵידַע רֶגַע לִקּוּי הַלְּבָנָה — כְּשֶׁיָּחוּל לִהְיוֹת עַל פִּי חֶשְׁבּוֹנֵנוּ בַּלַּיְלָה שֶׁלּוֹ, בַּחֲצִי חֹדֶשׁ הָאֲמִתִּי; שֶׁיִּהְיֶה הַלְּבָנָה בְּרֹאשׁ הַתְּלִי אוֹ זְנָבוֹ (עַיֵּן לְקַמָּן בֵּאוּר עִנְיָנָם) בִּשְׁעַת הַמּוֹלָד הָאֲמִתִּי, אוֹ בַּחֲצִי חֹדֶשׁ הָאֲמִתִּי — אָז יִהְיֶה בְּוַדַּאי לִקּוּי הַחַמָּה אוֹ הַלְּבָנָה. וְאֵין כָּאן מָקוֹם לְהַאֲרִיךְ מִזֶּה):

**מַהֲלַךְ** הַשֶּׁמֶשׁ בְּיוֹם אֶחָד תָּמִים — שֶׁהוּא כ״ד שָׁעוֹת — נ״ט חֲלָקִים, ח׳ שְׁנִיּוֹת; וּבְי׳ יָמִים — ט׳ מַעֲלוֹת, נ״א חֲלָקִים, כ״ג שְׁנִיּוֹת; וּבְק׳ יוֹם — צ״ח מַעֲלוֹת, ל״ג חֲלָקִים, נ״ג שְׁנִיּוֹת; וּשְׁאֵרִית מַהֲלָכָהּ בְּאֶלֶף יוֹם, אַחַר שֶׁנַּשְׁלִיךְ כָּל ש״ס מַעֲלוֹת — נִשְׁאָר רס״ה מַעֲלוֹת, ל״ח חֲלָקִים, נ׳ שְׁנִיּוֹת; וּמַהֲלָכָהּ בְּכ״ט יוֹם לְפִי זֶה — כ״ח מַעֲלוֹת, ל״ה חֲלָקִים, וּשְׁנִיָּה אַחַת; וּבְשָׁנָה סְדוּרָה — שֶׁהִיא שנ״ד יוֹם — מַהֲלָכָהּ שמ״ח מַעֲלוֹת, נ״ה חֲלָקִים, ט״ו שְׁנִיּוֹת. אֵלֶּה הֵם דִּבְרֵי הָרַמְבַּ״ם. וְהִנֵּה הוֹסִיף בְּי׳ יוֹם ג׳ שְׁנִיּוֹת יוֹתֵר עַל י׳ פְּעָמִים נ״ט ח׳; וּבְק׳ יוֹם הוֹסִיף ג׳
