~ בְּנֵי אָדָם הֵם מַקְפִּידִים דַּוְקָא עַל מִין א׳. וְנִרְאֶה, דְּמִי שֶׁמַּקְפִּיד שֶׁזֶּה מֻבְחָר אֵלָיו — יַעֲשֶׂה מֵאוֹתוֹ הַמִּין, דְּצָרִיךְ לַעֲשׂוֹת הַמִּצְוָה מִמַּה שֶּׁהוּא מֻבְחָר אֶצְלוֹ; אִם לֹא שֶׁלֹּא יוּכַל לְהַשִּׂיג אֶלָּא הָאַחֵר. וּבִזְמַנָּם הָיוּ רֹב (תּוֹסָפוֹת ג׳ עַמּוּד ב׳, דִּבּוּר הַמַּתְחִיל "מֵאֵלֶיהָ") פִּסְחֵיהֶם טָלֶה (וְכֵן מַשְׁמַע סַנְהֶדְרִין י״א עַמּוּד א׳: "וְאִמְּרַיָּא דְּעַרְקִין" וְכוּלֵי, וְלֹא זָכַר גְּדָיִים, כְּמוֹ שֶׁנִּזְכַּר בַּבָּרַיְתָא שָׁם; שְׁמַע מִינַּהּ דְּרֹב פִּסְחֵיהֶם הָיוּ טְלָאִים), מִפְּנֵי שֶׁבְּשָׂרוֹ הָיָה נִבְחָר לְרֹב בְּנֵי אָדָם; אוֹ אֶפְשָׁר דְּהָיוּ שְׁכִיחִים טְפֵי בְּאֶרֶץ יִשְׂרָאֵל מִגְּדָיִים:

**וְכֵיצַד** מַעֲשָׂיו? מְבִיאוֹ לָעֲזָרָה (וְאֵין צָרִיךְ סְמִיכָה), וְשׁוֹחֲטוֹ (אֲפִלּוּ זָר) בְּכָל מָקוֹם בָּעֲזָרָה, בְּיוֹם י״ד בְּנִיסָן, אַחַר חֲצוֹת הַיּוֹם (וְאִם שְׁחָטוֹ קֹדֶם חֲצוֹת — פָּסוּל), וְאַחַר גְּמַר עֲבוֹדַת תָּמִיד שֶׁל בֵּין הָעַרְבַּיִם, וְהַקְטָרַת קְטֹרֶת שֶׁל בֵּין הָעַרְבַּיִם, וַהֲטָבַת הַנֵּרוֹת. וְאִם עָבַר וּשְׁחָטוֹ אַחַר חֲצוֹת, קֹדֶם לַתָּמִיד שֶׁל בֵּין הָעַרְבַּיִם — כָּשֵׁר; "וְלֹא יִזְרוֹק דַּם הַפֶּסַח עַד שֶׁיִּשְׁחוֹט הַתָּמִיד וְיִזְרְקוּ אֶת דָּמוֹ" וְכוּלֵי — כֵּן כָּתַב בְּסֵדֶר הַיּוֹם (וְעַיֵּן מַה שֶּׁכָּתַב בשוד״י, סִימָן נ״ז, עַל דִּבְרֵי הָרַב יַעֲבֵ״ץ עָלָיו). וּבִמְחִילַת כְּבוֹדוֹ, אִשְׁתְּמִיטְתֵּיהּ גְּמָרָא זְבָחִים (צ״א עַמּוּד א׳), דְּדַוְקָא עַד שֶׁיִּזְרוֹק דַּם הַתָּמִיד, אֲבָל לֹא עַד שֶׁיִּשְׁחוֹט; דְּאִי אַכַּתִּי לֹא שַׁחְטֵיהּ לַתָּמִיד — יִזְרוֹק דַּם הַפֶּסַח מִקֹּדֶם, וְאַחַר כָּךְ יִשְׁחָטֶנּוּ, עַיֵּן שָׁם. וְהָרַמְבַּ״ם סָתַם כִּלְשׁוֹן הַמִּשְׁנָה, "עַד שֶׁיִּזְרוֹק" וְכוּלֵי; וּבְפֶרֶק ט׳ (מֵהִלְכוֹת תְּמִידִין וּמוּסָפִין, הֲלָכָה ד׳) כָּתַב: "וְאִם נִשְׁחֲטוּ שְׁנֵיהֶם כְּאֶחָד" וְכוּלֵי, וְלֹא בֵּאֵר רְבוּתָא יְתֵירָה — אִם נִשְׁחֲטוּ אֲפִלּוּ זֶה אַחַר זֶה, עַיֵּן שָׁם; וְצָרִיךְ עִיּוּן לִכְאוֹרָה. וְאוּלַי סָמַךְ עַצְמוֹ אַכָּאן, בְּהִלְכוֹת פֶּסַח, דְּרַק "עַד שֶׁיִּזְרוֹק":

**וְרָאִיתִי** בְּסֵפֶר אַמְתַּחַת בִּנְיָמִין (פֶּרֶק ה׳ דִּפְסָחִים, מִשְׁנָה ג׳), עַיֵּן שָׁם — דְּבָרִים תְּמוּהִים. מַה שֶּׁכָּתַב: "הַיְנוּ מִשּׁוּם דְּתָדִיר וְשֶׁאֵינוֹ תָּדִיר — תָּדִיר קוֹדֵם" וְכוּלֵי — נֶעְלַם מִמֶּנּוּ תּוֹסָפוֹת פְּסָחִים (נ״ט עַמּוּד א׳), דִּבּוּר הַמַּתְחִיל "יְאֻחַר", דְּנֶגֶד סְבָרַת "תָּדִיר" — יֵשׁ סְבָרָא דְּ"עָלֶיהָ הַשְׁלֵם", עַיֵּן שָׁם; וְהוּא מֻכְרָח, דְּאִי לָאו הָכֵי — לָמָּה לִי כְּלָל לְ"בָעֶרֶב" וּ"בֵין הָעַרְבַּיִם", עַיֵּן שָׁם. וּמַה שֶּׁכָּתַב: "מִשּׁוּם דִּכְתִיב 'עֶרֶב' וּ'בֵין הָעַרְבַּיִם' — לֹא הָיוּ מַקְדִּימִין" וְכוּלֵי "בִּזְרִיקָה, דְּלֹא נ׳ אֶלָּא בִּשְׁחִיטָה" — נֶעְלַם מִמֶּנּוּ דִּבְרֵי רַשִׁ״י זְבָחִים שָׁם, דְּלֹא נִזְכַּר אֶלָּא קְרָא דְּ"עֶרֶב" וּ"בֵין הָעַרְבַּיִם", לֹא זוּלַת, עַיֵּן שָׁם. אֲבָל בֶּאֱמֶת, וַדַּאי גַּם בִּזְרִיקָה יֶשְׁנוֹ לְחִיּוּב זֶה, דִּלְאַחֲרוֹ מִתָּמִיד; וּמַאי דְּלֹא כְּתִיב אֶלָּא בִּשְׁחִיטָה — הַיְנוּ דְּמִמֵּילָא נִשְׁמַע לִזְרִיקָה, דְּעַל כָּרְחֲךָ שְׁחִיטָה קוֹדֶמֶת. וּמַה שֶּׁכָּתַב: "וּמִשּׁוּם הָכֵי, הֵיכִי דְּנִשְׁחַט" וְכוּלֵי — לֹא יָדַעְתִּי מַאי "מִשּׁוּם הָכֵי" דְּקָאָמַר; הָא אִי טַעְמָא מִשּׁוּם תָּדִיר — נַמִּי מַאי נַפְקָא מִינַּהּ? תִּקְדַּם שְׁחִיטַת תָּמִיד, דִּתְדִירָה! וּמַה שֶּׁכָּתַב: "וְכֵן פָּסַק הָרַמְבַּ״ם" וְכוּלֵי — כְּבָר זָכַרְנוּ, דְּהָרַמְבַּ״ם לֹא כָּתַב אֶלָּא בְּשָׁחֲטוּ שְׁנֵיהֶם
