~ יוֹדְעִים שֶׁקָּבְעוּ בּוֹ בֵּית דִּין הַחֹדֶשׁ, שֶׁאֵין הַשְּׁלוּחִים יוֹצְאִים בְּיוֹם טוֹב. וַאֲפִלּוּ בִּירוּשָׁלַיִם עַצְמָהּ, שֶׁהוּא מְקוֹם בֵּית דִּין, פְּעָמִים רַבּוֹת הָיוּ עוֹשִׂין יוֹם טוֹב שֶׁל רֹאשׁ הַשָּׁנָה ב׳ יָמִים — שֶׁאִם לֹא בָּאוּ עֵדִים כָּל יוֹם ל׳ (וְזֶהוּ לְדִידָן, אַחַר תַּקָּנַת רַבָּן יוֹחָנָן בֶּן זַכַּאי. וְנִרְאֶה דַּעַת הָרַמְבַּ״ם כְּדַעַת רַבֵּינוּ חֲנַנְאֵל, בַּתּוֹסָפוֹת בֵּיצָה ה׳ עַמּוּד ב׳, דִּבּוּר הַמַּתְחִיל "מִי", עַיֵּן שָׁם הֵיטֵב) — נוֹהֲגִין הָיוּ בְּאוֹתוֹ הַיּוֹם, שֶׁמְּצַפִּין לְעֵדִים, קֹדֶשׁ, וּלְמָחָר קֹדֶשׁ. וְהוֹאִיל וְהָיוּ עוֹשִׂין אוֹתוֹ ב׳ יָמִים אַף בִּזְמַן הָרְאִיָּה — הִתְקִינוּ שֶׁיִּהְיוּ עוֹשִׂין אַף בְּנֵי אֶרֶץ יִשְׂרָאֵל אוֹתוֹ תָּמִיד ב׳ יָמִים, אַף בִּזְמַן הַזֶּה, שֶׁקּוֹבְעִים עַל פִּי הַחֶשְׁבּוֹן. וְלָמַדְנוּ, שֶׁאֲפִלּוּ יוֹם טוֹב שֵׁנִי שֶׁל רֹאשׁ הַשָּׁנָה מִדִּבְרֵי סוֹפְרִים, אֶלָּא דַּעֲדִיף מִשְּׁאָר יוֹם טוֹב שֵׁנִי:

**וְעַיֵּן** מַה שֶּׁכָּתַבְתִּי לְעֵיל (בְּפֶרֶק א׳), בְּשִׁיטַת רַשִׁ״י, דִּבְכָל אֶרֶץ יִשְׂרָאֵל עָשׂוּ בִּזְמַן שֶׁקִּדְּשׁוּ עַל פִּי רְאִיָּה רֹאשׁ הַשָּׁנָה רַק יוֹם אֶחָד, עַיֵּן שָׁם. וְנִרְאֶה לִי: דְּמִכָּל מָקוֹם עַכְשָׁיו, אַף לְרַשִׁ״י, עָבְדִינַן בְּאֶרֶץ יִשְׂרָאֵל כֻּלָּהּ רֹאשׁ הַשָּׁנָה ב׳ יָמִים; כֵּיוָן דִּבִירוּשָׁלַיִם עָבְדִינַן ב׳ יָמִים, אֲפִלּוּ לְפֵרוּשׁ רַשִׁ״י, כְּמוֹ שֶׁכָּתַבְתִּי שָׁם — דְּכֵן בִּזְמַן שֶׁהָיָה רְאִיָּה, מִשּׁוּם דְּאֶפְשָׁר לְהִתְבָּרֵר, עַיֵּן שָׁם; וְכֵיוָן דְּכָל אֶרֶץ יִשְׂרָאֵל רְאוּיָה לִמְקוֹם מוֹשַׁב בֵּית דִּין, וּפְעָמִים יָשְׁבוּ בְּאוּשָׁא, וּפְעָמִים בְּעֵין טַב — לְכָךְ עַכְשָׁיו תָּמִיד יַעֲשׂוּ ב׳ יָמִים בְּרֹאשׁ הַשָּׁנָה בְּכָל אֶרֶץ יִשְׂרָאֵל, כַּנִּזְכָּר לְעֵיל. וּבָזֶה הָיָה אֶפְשָׁר לְיַשֵּׁב דַּעַת הָרֹא״שׁ, סוֹף פֶּרֶק "אֵין עוֹמְדִין", הַנִּזְכָּר לְעֵיל, מִקֻּשְׁיוֹת הַזִּכְרוֹן יוֹסֵף וְהָעֲטֶרֶת רֹאשׁ; דִּרְאָיָתוֹ מֵרֹאשׁ הַשָּׁנָה שַׁפִּיר הֲוֵי רְאָיָה גְּדוֹלָה, דְּעַכְשָׁיו עָבְדִינַן ב׳ יָמִים, כַּנִּזְכָּר לְעֵיל; וְהָבֵן, דְּזֶה נָכוֹן. וּמִכָּל מָקוֹם, הַמְחֻוָּר וַאֲמִתִּי — מַה שֶּׁכָּתַבְתִּי לְעֵיל:

**וְאֵין** עֲשִׂיַּת יוֹם טוֹב יוֹם אֶחָד תַּלְיָא בִּקְרִיבַת הַמָּקוֹם. כֵּיצַד? אִם יִהְיֶה הַמָּקוֹם בֵּינוֹ וּבֵין יְרוּשָׁלַיִם מַהֲלַךְ ה׳ יָמִים אוֹ פָּחוֹת, שֶׁבְּוַדַּאי אֶפְשָׁר שֶׁיַּגִּיעוּ לָהֶן שְׁלוּחִין — אֵין אוֹמְרִים שֶׁאַנְשֵׁי מָקוֹם זֶה עוֹשִׂין יוֹם אֶחָד; שֶׁמִּי יֹאמַר לָנוּ שֶׁהָיוּ הַשְּׁלוּחִים יוֹצְאִים לְמָקוֹם זֶה? שֶׁמָּא לֹא הָיוּ שְׁלוּחִים יוֹצְאִים לְאוֹתוֹ מָקוֹם, מִפְּנֵי שֶׁלֹּא הָיָה שָׁם מוֹשַׁב בְּנֵי יִשְׂרָאֵל בְּשָׁעָה שֶׁקָּבְעוּ עַל פִּי הָרְאִיָּה, אוֹ שֶׁהָיָה חֵרוּם בַּדֶּרֶךְ — כְּדֶרֶךְ שֶׁהָיָה בֵּין יְהוּדָה וְגָלִיל בִּימֵי חַכְמֵי הַמִּשְׁנָה — אוֹ שֶׁהָיוּ הַכּוּתִים בְּאוֹתוֹ דֶּרֶךְ מוֹנְעִים הַשְּׁלוּחִין. וְאִלּוּ הָיָה הַדָּבָר תָּלוּי בִּקְרִיבַת הַמָּקוֹם — הָיוּ כָּל בְּנֵי מִצְרַיִם עוֹשִׂין יוֹם אֶחָד, שֶׁהֲרֵי אֶפְשָׁר שֶׁיַּגִּיעוּ לָהֶם שְׁלוּחֵי תִּשְׁרֵי, שֶׁאֵין בֵּין יְרוּשָׁלַיִם לְמִצְרַיִם עַל דֶּרֶךְ אַשְׁקְלוֹן אֶלָּא מַהֲלַךְ ח׳ יָמִים אוֹ פָּחוֹת; וְכֵן רֹב סוּרְיָא. הָא לָמַדְתָּ, שֶׁאֵין הַדָּבָר תָּלוּי בִּהְיוֹת הַמָּקוֹם קָרוֹב, אֶלָּא עִקַּר הַדָּבָר כָּךְ הוּא: כָּל מָקוֹם שֶׁיֵּשׁ
