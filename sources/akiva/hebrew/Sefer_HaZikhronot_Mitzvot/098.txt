~ קֶשֶׁת, גְּדִי, דְּלִי, דָּגִים. וְאֵלּוּ הַי״ב צוּרוֹת לֹא הָיוּ מְכֻוָּנוֹת אֶלָּא בִּזְמַן הַמַּבּוּל, שֶׁאָז הֶעֱלוּ לָהֶם שֵׁמוֹת אֵלּוּ; אֲבָל בִּזְמַנִּים אֵלּוּ כְּבָר סָבְבוּ מְעַט, שֶׁגַּם לַכּוֹכָבִים כֻּלָּם יֵשׁ סְבִיבָה קְצָת, כְּמוֹ שֶׁיִּתְבָּאֵר. וְכָל אֵלּוּ הַצּוּרוֹת הֵם בָּעִגּוּל הַיּוֹתֵר גָּדוֹל, בְּאֶמְצַע הֶקֵּף הַכַּדּוּר, וְיֵשׁ לָהֶם רֹחַב קְצָת, הַחֲגוֹרָה שֶׁבְּאֶמְצַע הַגּוּף — לָכֵן קְרָאוּם "חֲגוֹרַת הַמַּזָּלוֹת". וְיֵשׁ עוֹד ל״ו צוּרוֹת זוּלַת אֵלּוּ, יְדוּעִים לְחַכְמֵי הַתְּכוּנָה, וְהֵם לִצְפוֹן חֲגוֹרָה זוֹ וְלִדְרוֹמָהּ; וּלְפִי שֶׁאֵין בָּהֶם צֹרֶךְ לְחֶשְׁבּוֹן הָרְאִיָּה — לֹא זְכַרְנוּם:

**וְהוֹסִיפוּ** עוֹד לַחֲלוֹק הַגַּלְגַּל בִּכְלָלוֹ לְש״ס מַעֲלוֹת — כָּל מַזָּל שְׁלֹשִׁים מַעֲלוֹת; וְכָל מַעֲלָה לְס׳ חֲלָקִים, וְכָל חֵלֶק לְס׳ שְׁנִיִּים, וְכָל שְׁנִיָּה לְס׳ שְׁלִישִׁים; וְכֵן לִרְבִיעִיִּים, וְלַחֲמִישִׁיִּים, וּלְיֶתֶר הַמִּסְפָּרִים — כְּפִי מַה שֶּׁיְּבִיאֵנוּ הָעִנְיָן — כָּל אֶחָד ס׳ פְּעָמִים כַּחֲבֵרוֹ (וּמַה שֶּׁבָּחֲרוּ בְּמִסְפָּרִים אֵלּוּ — לִהְיוֹתָם מִתְחַלְּקִים לְכַמָּה חֲלָקִים). וְכָל זֶה לְמַעַן דַּעַת כַּמָּה מַעֲלוֹת מִן הַגַּלְגַּל. וְנִמְשָׁךְ מִזֶּה, שֶׁבְּכָל שָׁעָה הַגַּלְגַּל מְסַבֵּב בְּגִלְגּוּלוֹ ט״ו מַעֲלוֹת — חֵלֶק כ״ד מִן ש״ס — כִּי בְּכָל כ״ד שָׁעָה סוֹבֵב כֻּלּוֹ עַל צִירָיו. וּמִזֶּה נוּכַל לִשְׁפּוֹט עַל מֶרְחֲקֵי שְׁעוֹת הַזְּרִיחָה וְהַשְּׁקִיעָה בָּאֲרָצוֹת הָרְחוֹקוֹת זוֹ מִזּוֹ. כֵּיצַד? כְּשֶׁנַּחֲלֹק כָּל כַּדּוּר הָאָרֶץ גַּם כֵּן לְש״ס חֲלָקִים:

**וְיָדַעְנוּ** כִּי כַּדּוּר הָאָרֶץ — חוּט כ״ד אֶלֶף מִיל יָסֹב אוֹתוֹ (וִידִיעַת זֶה גַּם כֵּן נוּכַל לָדַעְתּוֹ מֵהֶקֵּף הַגַּלְגַּל בְּמַעֲלָה אַחַת, בִּמְדִידַת מַעֲלָה א׳ בָּאָרֶץ; וְאֵין זֶה מְקוֹמוֹ לְבָאֲרוֹ. וְעַיֵּן בַּמְּפָרֵשׁ, פֶּרֶק ג׳ מִיְּסוֹדֵי הַתּוֹרָה, הֲלָכָה ח׳), וְהֵם ה׳ אֲלָפִים וְד׳ מֵאוֹת פַּרְסָאוֹת אַשְׁכְּנַזִּית; וּבְחָלְקֵנוּ אוֹתוֹ לְש״ס מַעֲלוֹת — יַגִּיעַ מְנַת כָּל מַעֲלָה ט״ו פַּרְסָאוֹת. וּבְכָל כ״ד שָׁעָה הַגַּלְגַּל סוֹבֵב סְבִיב כָּל כַּדּוּר הָאָרֶץ, וּבְשָׁעָה אַחַת יָסֹב ט״ו מַעֲלוֹת, שֶׁהֵם ט״ו פְּעָמִים ט״ו, שֶׁהֵם רכ״ה פַּרְסָאוֹת. וְיָדַעְתָּ בְּדִמְיוֹן: אִם זָרַח הַשֶּׁמֶשׁ בִּמְקוֹמְךָ בִּתְחִלַּת שָׁעָה ח׳ אַחַר חֲצוֹת הַלַּיְלָה — תֵּדַע שֶׁבָּעִיר הָרְחוֹקָה מִמְּךָ לְצַד מִזְרָח רכ״ה פַּרְסָאוֹת זָרְחָה בְּשָׁעָה שְׁלֵמָה מִקֹּדֶם; וְהִפּוּךְ הַדָּבָר בָּעִיר הָרְחוֹקָה לְצַד מַעֲרָב מֶשֶׁךְ הַנִּזְכָּר — שֶׁתִּזְרַח בְּשָׁעָה שֶׁלְּאַחַר כָּךְ, תְּחִלַּת שָׁעָה ט׳. וְכֵן אַתָּה דָּן לְמֶחֱצָה, וְלִשְׁלִישׁ, וְלִרְבִיעַ, וּלְיֶתֶר חֲלָקִים, וּלְכִפְלוֹ, וּלְמְרֻבָּעוֹ, וּלְכָל מֶרְחָק שֶׁתִּרְצֶה. "וְדַעַת לְנָבוֹן נָקָל":

**כַּאֲשֶׁר** הִסְתַּכְּלוּ חַכְמֵי הַתּוֹכְנִים פַּעַם בְּפַעַם בְּמַעֲמָד וּמַצָּב הַכּוֹכָבִים — רָאוּ כִּי כָּל הַכּוֹכָבִים יֵשׁ לָהֶם קְצָת תְּנוּעָה בִּלְעֲדֵי תְּנוּעָה הַיּוֹמִית,
