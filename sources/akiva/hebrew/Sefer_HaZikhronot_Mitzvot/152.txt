~ לְעֵיל בַּסִּימָן שֶׁלִּפְנֵי זֶה. וְדִין טֻמְטוּם וְאַנְדְּרוֹגִינוֹס — נִתְבָּאֵר בַּסִּימָן שֶׁלִּפְנֵי זֶה; וְכֵן דִּין הַכֵּלִים נִתְבָּאֵר שָׁם, שֶׁאִסּוּר הַלָּאו גַּם אַכֵּלִים קָאֵי:

**כֵּיצַד** הֵם חִלּוּקֵי הַג׳ מַחֲנוֹת כָּאן, לְעִנְיַן הַלָּאו דִּכְנִיסָה? מַחֲנֶה א׳ — מַחֲנֵה יִשְׂרָאֵל — אֵינָהּ אֶלָּא יְרוּשָׁלַיִם בִּלְבַד; וּמְצֹרָע הַנִּכְנָס בָּהּ — לוֹקֶה. אֲבָל שְׁאָר עֲיָרוֹת הַמֻּקָּפִים חוֹמָה — אַף עַל פִּי שֶׁאֵינוֹ רַשַּׁאי לִיכָּנֵס בָּהּ, שֶׁנֶּאֱמַר "בָּדָד יֵשֵׁב", וְטָעוּן שִׁלּוּחַ מַחֲנֶה, כְּמוֹ שֶׁנִּתְבָּאֵר לְעֵיל — מִכָּל מָקוֹם, אֵינוֹ לוֹקֶה אִם נִכְנַס (וְאֶפְשָׁר דְּלוֹקֶה מַכַּת מַרְדּוּת). מַחֲנֶה ב׳ — הַר הַבַּיִת (אֲבָל חֵיל וְעֶזְרַת נָשִׁים, שֶׁנִּתְבָּאֵר לְעֵיל דְּהֶחְמִירוּ חֲכָמִים — אֵין בָּם מַלְקוֹת), וְהִיא מַחֲנֵה לְוִיָּה; וְעַל מַחֲנֶה זוֹ נֶאֱמַר לָאו בִּפְנֵי עַצְמוֹ בְּפָרָשַׁת תֵּצֵא, עַיֵּן שָׁם. וּמְצֹרָע שֶׁדָּחַק וְנִכְנַס לָהּ — לוֹקֶה שְׁמוֹנִים. זֶהוּ דַּעַת הָרַמְבַּ״ם. (וְכָתַב הַכֶּסֶף מִשְׁנֶה: "וְקָשֶׁה, דַּהֲוָה לֵיהּ לוֹמַר דְּאִם נִכְנַס לָעֲזָרָה — לוֹקֶה ק״כ, שֶׁטִּמֵּא עוֹד מַחֲנֵה שְׁכִינָה", עַיֵּן שָׁם — שֶׁדָּחַק. וְלִי נִרְאֶה: מִלְּבַד דְּיֵשׁ לוֹמַר פָּשׁוּט, מִשּׁוּם דְּזָכַר לְהַלָּן עַל מַחֲנֵה שְׁכִינָה כָּרֵת — לֹא מַלְקוֹת (אַף עַל פִּי שֶׁכָּל חַיָּבֵי כְּרֵתוֹת לוֹקִין) — לֹא זְכָרוֹ כָּאן; עוֹד נִרְאֶה, דְּהָא וַדַּאי — אַף לֹא בְּהַר הַבַּיִת, אֶלָּא בִּירוּשָׁלַיִם גּוּפָהּ — מִמָּקוֹם לְמָקוֹם, אִם הִתְרוּ בּוֹ, וְחָזְרוּ וְהִתְרוּ בּוֹ: "אַל תִּכָּנֵס" — וְהוּא נִכְנָס, "אַל תִּכָּנֵס" — וְהוּא נִכְנָס — לוֹקֶה עַל כָּל כְּנִיסָה וּכְנִיסָה; וְלֹא הֻצְרַךְ הָרַב לְבָאֵר זֶה, שֶׁאִם עָבַר עַל לָאו אֶחָד כַּמָּה פְּעָמִים — לוֹקֶה עַל כָּל פַּעַם. וְלֹא בֵּאֵר הָרַמְבַּ״ם אֶלָּא: בְּנִכְנָס לְהַר הַבַּיִת — לוֹקֶה שְׁמוֹנִים, מִפְּנֵי ב׳ לָאוִין: א׳ — "וְלֹא יְטַמְּאוּ אֶת מַחֲנֵיהֶם", עַל הַכְּנִיסָה לִירוּשָׁלַיִם; וְהַב׳ — אַלָּאו דְּפָרָשַׁת תֵּצֵא, אַכְּנִיסָה לְהַר הַבַּיִת. אֲבָל נִכְנַס לָעֲזָרָה — לֹא הֻצְרַךְ לְבָאֵר, דְּזֶהוּ לָאו דְּהָכָא — דְּ"וְלֹא יְטַמְּאוּ" — וְלָאו אֶחָד הוּא; וּפָשׁוּט.) מַחֲנֶה ג׳ — הָעֲזָרָה, וְהִיא מַחֲנֵה שְׁכִינָה; הַנִּכְנָס בָּהּ — עָנוּשׁ כָּרֵת, כְּמוֹ שֶׁנִּתְבָּאֵר:

**אֵיזֶהוּ** הַטָּמֵא שֶׁחַיָּב כָּרֵת עַל הַמִּקְדָּשׁ? כָּל שֶׁנִּטְמָא בְּטֻמְאָה מִן הַמֵּת שֶׁהַנָּזִיר מְגַלֵּחַ עָלֶיהָ — שֶׁכְּבָר נִתְפָּרֵשׁ בִּמְקוֹמוֹ, בְּמִצְוֹת וְחִיּוּבֵי הַנָּזִיר; אוֹ שֶׁנָּגַע בְּאָדָם אוֹ בְּכֵלִים שֶׁנִּטְמְאוּ בְּאוֹתָן הַטֻּמְאוֹת שֶׁהַנָּזִיר מְגַלֵּחַ עֲלֵיהֶן — שֶׁהֲרֵי הוּא שֵׁנִי לָרִאשׁוֹן שֶׁנָּגַע בְּמֵת (פְּרָט זֶה הוּא מִדִּבְרֵי הָרַמְבַּ״ם; אֲבָל הָרַאֲבַ״ד חוֹלֵק עָלָיו, וְכָתַב: דִּכְמוֹ שֶׁאֵין הַנָּזִיר מְגַלֵּחַ עַל מַגָּעוֹ בְּכֵלִים שֶׁנָּגְעוּ בְּמֵת — וַאֲפִלּוּ עַל חֶרֶב (בְּזוֹ מַחֲלֹקֶת הַפּוֹסְקִים, יִתְבָּאֵר בִּמְקוֹמוֹ) — כֵּן אֵין חַיָּב עַל מַגָּעָן עַל בִּיאַת מִקְדָּשׁ); אוֹ שֶׁנִּטְמָא בִּשְׁאָר אֲבוֹת הַטֻּמְאָה שֶׁל תּוֹרָה, שֶׁנִּתְפָּרְשׁוּ בִּמְקוֹמָן. כְּלָלוֹ שֶׁל דָּבָר: כָּל הַטָּעוּן בִּיאַת מַיִם מִן הַתּוֹרָה (לְפִי דַּעַת הָרַאֲבַ״ד לֵיתֵיהּ לְהַאי כְּלָלָא) — חַיָּב כָּרֵת עַל בִּיאַת מִקְדָּשׁ; וַאֲפִלּוּ אַחַר שֶׁטָּבַל, עַד שֶׁיַּעֲרִיב שִׁמְשׁוֹ (אָזִיל לְטַעְמֵיהּ, שֶׁנִּתְבָּאֵר לְעֵיל דַּעְתּוֹ, דִּמְחֻסַּר כִּפּוּרִים אֵינוֹ בְּכָרֵת;
