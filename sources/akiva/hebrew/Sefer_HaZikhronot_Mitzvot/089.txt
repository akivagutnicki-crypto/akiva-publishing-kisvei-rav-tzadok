~ וְסִימָן לְרָאשֵׁי הַיָּמִים שֶׁל הַז׳ פְּעָמִים ב׳ שָׁנִים אֵלּוּ שֶׁבְּמַחֲזוֹר גָּדוֹל — הוּא: וד״ב זהג״א. שֶׁבַּד׳ שָׁנִים רִאשׁוֹנִים — תְּקוּפַת נִיסָן שֶׁל תְּחִלַּת שְׁנַת ג׳, שֶׁהוּא תְּחִלַּת ב׳ שָׁנִים אַחֲרוֹנִים שֶׁלּוֹ — בִּתְחִלַּת יוֹם ו׳; וּבַשְּׁנִיִּים — בִּתְחִלַּת יוֹם ד׳; וְכֵן כֻּלָּם:

**וְהִנֵּה** בִּהְיוֹת תְּקוּפַת נִיסָן בִּתְחִלַּת הַלַּיְלָה — הִנֵּה תָּבוֹא תְּקוּפַת תַּמּוּז שֶׁאַחֲרֶיהָ בְּז׳ שָׁעוֹת וּמֶחֱצָה בַּלַּיְלָה, וּתְקוּפַת תִּשְׁרֵי בְּסוֹף ג׳ שָׁעוֹת עַל הַיּוֹם (הַיּוֹם וְהַלַּיְלָה נַחְשׁוֹב פֹּה תָּמִיד לְי״ב שָׁעוֹת — בֵּין בְּיָמִים הָאֲרֻכִּים, בֵּין בְּיָמִים קְצָרִים. וּלְפִי שֶׁאֵינוֹ שָׁוֶה, לָכֵן הִתְחַכְּמוּ חַכְמֵי הָעִבּוּר הָאַחֲרוֹנִים לַחְשׁוֹב הַמֵּעֵת לְעֵת מֵחֲצִי הַיּוֹם, אוֹ מֵחֲצִי הַלַּיְלָה הָא׳ עַד חֲצִי הַב׳, שֶׁזֶּהוּ תָּמִיד שָׁוֶה. וְעַיֵּן בְּתוֹסָפוֹת עֵירוּבִין (נ״ו עַמּוּד א׳), דִּבּוּר הַמַּתְחִיל "וְאֵין" — מַה שֶּׁהִנִּיחוּ בְּקֻשְׁיָא, עַיֵּן שָׁם. וְהַנִּרְאֶה לַעֲנִיּוּת דַּעְתִּי: דְּתָמִיד נַחְשׁוֹב הַיּוֹם י״ב שָׁעוֹת, וְכֵן הַלַּיְלָה; וּמִן שָׁעָה ז׳ אַחַר חֲצוֹת הַלַּיְלָה — מַתְחִיל יוֹם, אַף בִּימוֹת תְּקוּפַת תַּמּוּז וְטֵבֵת; וְכֵן מִן שָׁעָה ז׳ אַחַר חֲצוֹת הַיּוֹם — מַתְחִיל לַיְלָה, תָּמִיד. וְהַמּוֹתָר — מַה שֶּׁהַיּוֹם לוֹוֶה מִן הַלַּיְלָה בִּתְקוּפַת תַּמּוּז, אוֹ הַלַּיְלָה מִן הַיּוֹם בִּתְקוּפַת טֵבֵת — אֵינוֹ נֶחְשָׁב לַיּוֹם אוֹ לַלַּיְלָה כְּלָל. וְעַיֵּן בִּנְקוּדוֹת הַכֶּסֶף, בְּיוֹרֶה דֵּעָה (סִימָן קפ״ד, סְעִיף קָטָן ב׳) — כָּתַב כֵּן מַמָּשׁ לְעִנְיַן עוֹנָה דְּנִדָּה, לְדַעַת אֲבִי אָסָף, עַיֵּן שָׁם. כֵּן נִרְאֶה לְפִי עֲנִיּוּת דַּעְתִּי.) וּתְקוּפַת טֵבֵת — בְּי׳ שָׁעוֹת וּמֶחֱצָה עַל הַיּוֹם; וּתְקוּפַת נִיסָן שֶׁל שָׁנָה שֶׁאַחַר כָּךְ — בְּו׳ שָׁעוֹת עַל הַלַּיְלָה שֶׁלְּאַחֲרָיו; וְתַמּוּז — בְּאֶחָד וּמֶחֱצָה עַל הַיּוֹם; וְתִשְׁרֵי — בְּט׳ בַּיּוֹם; וְטֵבֵת — בְּד׳ שָׁעוֹת וּמֶחֱצָה בַּלַּיְלָה שֶׁלְּאַחֲרָיו:

**זֶהוּ** סִדְרָן שֶׁל ב׳ שָׁנִים רִאשׁוֹנִים; וְהַשְּׁנִיִּים גַּם כֵּן כַּנִּזְכָּר לְעֵיל, שֶׁהַתְחָלַת תְּקוּפַת נִיסָן בִּתְחִלַּת הַיּוֹם, וְשָׁב מַה שֶּׁהָיָה לַיְלָה לְיוֹם, וְיוֹם לְלַיְלָה. וְהַסִּימָן לַתְּקוּפוֹת: אזג״י ואט״ד (אוֹתִיּוֹת מִתְחַלְּפוֹת. הָרִאשׁוֹן — שָׁלֵם, וְהַב׳ — יִלְוֶה אֵלָיו עוֹד מֶחֱצָה; הַג׳ — שָׁלֵם, וְהַד׳ — יִלְוֶה וְכוּלֵי; וְכֵן כֻּלָּם): אזג״י — לְשָׁנָה רִאשׁוֹנָה, וְאט״ד — לַשְּׁנִיָּה:

**וְאַחֲרֵי** יָדַעְנוּ אֶת כָּל אֵלֶּה — הִנֵּה נִגְלְתָה יְדִיעַת תְּקוּפַת כָּל שָׁנָה. כֵּיצַד? הֲרֵי שֶׁנִּרְצֶה לֵידַע תְּקוּפַת תִּשְׁרֵי שֶׁל שְׁנַת הַתר״ט לִבְרִיאַת עוֹלָם: נַשְׁלִיךְ כָּל כ״ח שָׁנִים שֶׁעָבְרוּ מִבְּרִיאַת עוֹלָם עַד שָׁנָה זוֹ (וְהַתְחָלַת מִנְיַן הַשָּׁנִים מִנִּיסָן), כְּמוֹ שֶׁבֵּאַרְנוּ, שֶׁבְּכָל כ״ח חוֹזֶרֶת חַמָּה לִמְקוֹמָהּ; וְהַמּוֹתָר מִכ״ח הוּא ז׳ שָׁנִים שְׁלֵמִים, עַד תְּקוּפַת נִיסָן תר״ח, וּתְקוּפַת תִּשְׁרֵי רְצוּיֵינוּ הִיא מִשְּׁנַת הַח׳. וְהִנֵּה כְּבָר עָבְרוּ ד׳ שָׁנִים, שֶׁהוּא הֶקֵּף קָטָן הָרִאשׁוֹן, וְגַם ב׳ שָׁנִים מִן ד׳ הַשְּׁנִיִּים. וּכְפִי סִימָן וד״ב זהג״א — הִנֵּה הָיָה תְּקוּפַת תְּחִלַּת
