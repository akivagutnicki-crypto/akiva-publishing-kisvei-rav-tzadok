~ רְבִיעַ שָׁנָה, עַד שֶׁחוֹזֶרֶת לִנְקֻדַּת סַרְטָן. וְהִנֵּה, מִן סַרְטָן עַד גְּדִי — הוּא חֲצִי הַכַּדּוּר וַחֲצִי אוֹפַן הַמִּישׁוֹר; וּמִן גְּדִי עַד סַרְטָן — חֲצִי הַשֵּׁנִי שֶׁל הַכַּדּוּר הַשְּׁמֵימִי:

**וְהִנֵּה** הָעִגּוּלִים הַמַּחְשְׁבִיִּים שֶׁל הַחַמָּה, עַל יְדֵי גַּלְגַּל הַיּוֹמִי — הֵם בְּכָל הֶקֵּף הַכַּדּוּר. אִם כֵּן, אוֹתָם עִגּוּלִים מְדֻמִּים שֶׁעָשְׂתָה בִּהְיוֹתָהּ בִּנְקֻדַּת סַרְטָן — הִיא הָעִגּוּלָה שֶׁעָשְׂתָה בְּעָמְדָהּ בִּנְקֻדַּת גְּדִי; וְכֵן עִגּוּלָה שֶׁל נְקֻדַּת מֹאזְנַיִם — הִיא אוֹתָהּ הָעִגּוּלָה שֶׁל נְקֻדַּת טָלֶה; וְכֵן כָּל הָעִגּוּלוֹת כְּפוּלוֹת. וְנִמְצָא כְּלָלוּת הָעִגּוּלִים הֵם רַק קפ״ב (בִּלְעֲדֵי הַמּוּתָרוֹת): שֶׁמֵּרֹאשׁ סַרְטָן עַד רֹאשׁ גְּדִי — שֶׁהוּא מִצָּפוֹן לְדָרוֹם; וְאַחַר כָּךְ, מֵרֹאשׁ גְּדִי עַד רֹאשׁ סַרְטָן — הֵם אוֹתָם הָעִגּוּלִים עַצְמָם, אֲבָל לְצָפוֹן מִדָּרוֹם. וְכָל הָעִגּוּלִים הַמַּחְשְׁבִיִּים שֶׁאָמַרְנוּ — נִקְרָאִים "מְסִבָּה הַיּוֹמִית". וְאַרְבַּע תְּקוּפוֹת הַשָּׁנָה הֵם אֵלּוּ: כְּשֶׁהַחַמָּה בְּרֹאשׁ סַרְטָן — וְזֶה בְּרֹב הַשָּׁנִים בְּיֶרַח תַּמּוּז, וְלָכֵן נִקְרֵאת "תְּקוּפַת תַּמּוּז"; וּכְשֶׁהוּא בְּרֹאשׁ מֹאזְנַיִם — וְהוּא תְּקוּפַת תִּשְׁרֵי; וּכְשֶׁהוּא בְּרֹאשׁ גְּדִי — וְהוּא תְּקוּפַת טֵבֵת; וּכְשֶׁהוּא בְּרֹאשׁ טָלֶה — וְהוּא תְּקוּפַת נִיסָן:

**וּלְהָבִין** הַדָּבָר נְצַיֵּר, וּלְמַרְאִית עֵינֵינוּ נִשְׁפּוֹט בַּצִּיּוּר הַזֶּה: הָעִגּוּל א׳ ב׳ ג׳ ד׳ ה׳ ו׳ — הוּא עִגּוּלַת הָרָקִיעַ; מִתְּחִלַּת צִיר הַדְּרוֹמִי

~ עוֹלָה עַד לְאוֹפַן הַמִּישׁוֹר, וּפוֹגֶשֶׁת בּוֹ בִּנְקֻדָּה אַחַת (אֵיזוֹ שֶׁהִיא — נְקֻדַּת תְּחִלַּת טָלֶה אוֹ תְּחִלַּת גְּדִי), וְחוֹזֶרֶת וְיוֹרֶדֶת לַצִּיר הַצְּפוֹנִי, וּמִמֶּנָּה וּלְמַטָּה לַחֲצִי הַשֵּׁנִי שֶׁל הַכַּדּוּר, וְחוֹזֶרֶת לִפְגּוֹשׁ בִּנְקֻדָּה בְּקַו הַשָּׁוֶה, וּמִשָּׁם חוֹזֶרֶת
