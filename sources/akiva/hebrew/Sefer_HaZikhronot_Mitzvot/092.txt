~ הַלָּלוּ, שֶׁעָבְרוּ רצ״ד מַחֲזוֹרִים קְטַנִּים, וְיִתְרוֹנָם עוֹלֶה י״ז יָמִים, י״ח שָׁעוֹת, ל׳ חֲלָקִים; וְאַחַר נְכַיֵּן מִזֶּה ז׳ ט׳ תרמ״ב — הוּא יוֹתֵר עַל י׳ יָמִים; יוּכַל לַחְשׁוֹב י׳ יָמִים, וְלֹא יִצְטָרֵךְ לְדַקְדֵּק עוֹד בְּנִכּוּי ז׳ ט׳ תרמ״ב; וְכֵן כַּהַאי גַּוְנָא בְּיִתְרוֹנֵי הַשָּׁנִים. וְאִם יִשָּׁאֵר לוֹ יוֹתֵר מִכ״ט י״ב תשצ״ג (כְּגוֹן בִּתְקוּפַת טֵבֵת שֶׁקֹּדֶם שְׁנַת הָעִבּוּר) — יֵדַע שֶׁהַתְּקוּפָה הִיא בַּחֹדֶשׁ שֶׁאַחֲרָיו (בִּשְׁבָט). וְכֵן לְהֵפֶךְ: אִם יִגְרַע ז׳ ט׳ תרמ״ב — עוֹדֵף עַל הַמְּגוֹרָע; הִנֵּה, בְּכָךְ יָמִים וְשָׁעוֹת תִּקְדַּם הַתְּקוּפָה לְהַמּוֹלָד. אֲבָל זֶה אִי אֶפְשָׁר כְּלָל בְּעִתִּים אֵלּוּ, שֶׁכְּבָר עָבְרוּ כָּל כָּךְ מַחֲזוֹרִים קְטַנִּים; אֲבָל הָיָה קֹדֶם זְמַן רַב, שֶׁעֲדַיִן לֹא הִשְׁלִימוּ הַשָּׁעָה וְתפ״ה חֲלָקִים — יִתְרוֹן הַמַּחֲזוֹרִים קְטַנִּים — לְז׳ ט׳ תרמ״ב. אוּלָם בְּעִתִּים הַלָּלוּ — תָּמִיד הַתְּקוּפָה כְּמוֹ י׳ יָמִים בַּחֹדֶשׁ, כַּנִּזְכָּר, לִפְחוֹת; אִם לֹא כְּשֶׁהַחֹדֶשׁ נִדְחֶה מִיּוֹם הַמּוֹלָד. וְלֹא חָשׁוּ מְיַסְּדֵי חֶשְׁבּוֹן הָעִבּוּר לִקְבּוֹעַ עוֹד חֹדֶשׁ עִבּוּר בִּשְׁבִיל זֶה הַשָּׁעָה וְתפ״ה חֲלָקִים, הוֹאִיל וּמִתְקַבֵּץ בְּרֹב הַשָּׁנִים לְיָמִים הַרְבֵּה; לְפִי שֶׁרָאוּ שֶׁעַד סוֹף הָאֶלֶף הַשִּׁשִּׁי — שֶׁהוּא יְמֵי הָעוֹלָם — לֹא יַגִּיעַ קִבּוּץ זֶה, אַחַר נִכּוּי ז׳ ט׳ תרמ״ב, לַחֲצִי חֹדֶשׁ, שֶׁיִּהְיֶה פֶּסַח אוֹ סֻכּוֹת שֶׁלֹּא בִּזְמַנּוֹ בְּשָׁנָה רִאשׁוֹנָה שֶׁל מַחֲזוֹר קָטָן. וְגַם מְהֵרָה יִבָּנֶה בֵּית הַמִּקְדָּשׁ, וְיָשׁוּב חֶשְׁבּוֹן הָרְאִיָּה לִמְקוֹמוֹ; לְפִיכָךְ לֹא חָשׁוּ לַדָּבָר. כֵּן נִרְאֶה לִי:

**חֶשְׁבּוֹנוֹת** הַלָּלוּ מֵהַתְּקוּפָה שֶׁבֵּאַרְנוּ — הוּא הַכֹּל לְפִי דַּעַת שְׁמוּאֵל, הַסּוֹבֵר שֶׁשְּׁנַת הַחַמָּה הוּא שס״ה וּרְבִיעַ, וְשֶׁיֵּשׁ יִתְרוֹן בְּכָל מַחֲזוֹר קָטָן שָׁעָה וְתפ״ה חֲלָקִים. אֲבָל דַּעַת רַב אַדָּא עָלָיו הַשָּׁלוֹם — וְכֵן דַּעַת הַרְבֵּה מֵחַכְמֵי אֻמּוֹת הָעוֹלָם — שֶׁמֶּשֶׁךְ הַשָּׁנָה פָּחוֹת מִשס״ה וּרְבִיעַ. וְהוּא חִלֵּק עוֹד אֶת כָּל חֵלֶק לְע״ו רְגָעִים, וְאָמַר שֶׁהַשָּׁנָה הוּא שס״ה יוֹם, וְה׳ שָׁעוֹת, וְתתקצ״ז חֲלָקִים, וּמ״ח רְגָעִים. וְתוֹסֶפֶת שְׁנַת הַחַמָּה עַל שְׁנַת הַלְּבָנָה — שֶׁהִיא שנ״ד ח׳ תתע״ו, כְּמוֹ שֶׁנִּתְבָּאֵר לְעֵיל — י׳ יָמִים, כ״א שָׁעוֹת, קכ״א חֲלָקִים, מ״ח רְגָעִים. וּבְי״ט שָׁנָה — שֶׁהוּא מַחֲזוֹר קָטָן, שֶׁיֵּשׁ בָּהֶם ז׳ עִבּוּרִים — לֹא יִשָּׁאֵר שׁוּם תּוֹסָפוֹת וְיִתְרוֹן כְּלָל; וְנִמְצָא בְּכָל י״ט שָׁנָה שָׁוָה שְׁנוֹת הַחַמָּה לְהַלְּבָנָה. וְזוֹ הָיָה מְקוֹר מַחְשֶׁבֶת רַב אַדָּא עָלָיו הַשָּׁלוֹם, לְבַל יִהְיֶה הַיִּתְרוֹן שָׁעָה וְתפ״ה חֲלָקִים מִכָּל מַחֲזוֹר:

**וּלְפִי** דַּעְתּוֹ — אֵין לָנוּ מַחֲזוֹר גָּדוֹל שֶׁל חַמָּה כְּלָל; אֲבָל מַחֲזוֹר קָטָן שֶׁל לְבָנָה, שֶׁהוּא שֶׁל י״ט שָׁנָה — גַּם הוּא מַחֲזוֹר לַחַמָּה, שֶׁחוֹזֶרֶת לִהְיוֹת רְחוֹקָה מִן הַמּוֹלָד כַּאֲשֶׁר הָיְתָה בִּתְחִלַּת הַבְּרִיאָה; וְנִסְתַּלְּקוּ כָּל חֶשְׁבּוֹנוֹת שֶׁאָמַרְנוּ
