~ נִמְצֵאת בְּמָקוֹם שֶׁאֵין שִׁלּוּחָהּ אֶלָּא מִדְּרַבָּנָן — כְּגוֹן בְּחֵיל וְעֶזְרַת נָשִׁים (אוֹ אֲפִלּוּ מִן הַתּוֹרָה, אֶלָּא שֶׁאֵין עוֹנְשִׁין עַל זְדוֹנוֹ כָּרֵת — כְּמוֹ שֶׁיִּתְבָּאֵר בַּסִּימָן שֶׁאַחַר זֶה, דְּאֵינָהּ אֶלָּא בָּעֲזָרָה לְבַד. וְנִרְאֶה לִי, דְּהוּא הַדִּין אִם נִמְצֵאת בָּעֲזָרָה, אֶלָּא שֶׁאֵין טֻמְאַת הַכְּלִי אֶלָּא טֻמְאָה מִדִּבְרֵי סוֹפְרִים; וְצָרִיךְ לְעַיֵּן בָּזֶה) — אֵינוֹ רַשַּׁאי לְהוֹצִיאָהּ בְּשַׁבָּת, אֶלָּא כּוֹפֶה עָלֶיהָ פְּסַכְתֵּר לְכַסּוֹתוֹ, עַד שֶׁתֶּחְשַׁךְ, וְיוֹצִיאוֹ:

**וְכָל** זֶה שֶׁאָמַרְנוּ — כְּשֶׁאֵינוֹ לְצֹרֶךְ. אֲבָל לְצֹרֶךְ הַמִּקְדָּשׁ — כָּל הַטְּמֵאִים מֻתָּרִים לִיכָּנֵס: כְּגוֹן שֶׁצָּרִיךְ לְהוֹצִיא הַטֻּמְאָה שֶׁבַּמִּקְדָּשׁ, אוֹ לִבְנוֹת וּלְתַקֵּן (וְכֵן לְתַקֵּן הַסֵּפֶר תּוֹרָה שֶׁבָּאָרוֹן — תּוֹסָפוֹת בָּבָא בַּתְרָא י״ד עַמּוּד א׳), וְאִי אֶפְשָׁר לַעֲשׂוֹת בִּטְהוֹרִים — רַשָּׁאִים הַטְּמֵאִים לַעֲשׂוֹת וּלְהִכָּנֵס לָעֲזָרָה וְלַהֵיכָל, וַאֲפִלּוּ לְקָדְשֵׁי קָדָשִׁים, אִם אִי אֶפְשָׁר זוּלָתָם; לְפִי שֶׁטֻּמְאַת הַמִּקְדָּשׁ עַכְשָׁיו עַל יָדָם — זֶהוּ תִּקּוּנוֹ, וְעָדִיף לֵיהּ. וַאֲפִלּוּ מְצֹרָעִים — נִרְאֶה דְּרַשָּׁאִים לִיכָּנֵס, אִם אִי אֶפְשָׁר בִּלְעָדָם. מִיהוּ, אֶפְשָׁר דַּאֲסוּרִים — דְּהָא מִיָּד בִּכְנִיסָתָם לִירוּשָׁלַיִם אִסּוּרָא עָבְדֵי, וּבִירוּשָׁלַיִם לָא מְתַקְּנֵי מִידֵּי. וּלְתַקֵּן בִּירוּשָׁלַיִם — וַדַּאי נִרְאֶה דִּמְצֹרָעִים אֲסוּרִים לִיכָּנֵס, אַף דְּאִי אֶפְשָׁר בִּלְעָדָם, דְּלָאו הֶכְרֵחַ הוּא:

**וּטְמֵאֵי** מֵתִים רַשָּׁאִים עוֹד לְהִכָּנֵס לַמִּקְדָּשׁ אֲפִלּוּ שֶׁלֹּא לְצֹרֶךְ הַמִּקְדָּשׁ, אֶלָּא לְצֹרֶךְ עַצְמָם; וְהוּא בְּקָרְבָּנוֹת שֶׁקָּבוּעַ לָהֶם זְמַן — כִּתְמִידִין וּמוּסָפִין וְכַיּוֹצֵא — שֶׁקְּרֵבִים בְּטֻמְאָה, אִם הָיוּ רֹב צִבּוּר אוֹ כֹּהֲנִים טְמֵאִים; דְּמֻתָּרִים כָּל הַטְּמֵאִים לִיכָּנֵס לָעֲזָרָה לְצֹרֶךְ הַקָּרְבָּן. וְטֻמְאָה זוֹ שֶׁהֻתְּרָה — אֵינָהּ אֶלָּא טֻמְאַת מֵת בִּלְבַד, מִפְּנֵי שֶׁהַקָּרְבָּן קָרֵב בִּטְמֵא מֵת, כְּמוֹ שֶׁיִּתְבָּאֵר בִּמְקוֹמוֹ. אֲבָל טְמֵא שֶׁרֶץ, וְזָבִים וְזָבוֹת, וּמְצֹרָע וְכַיּוֹצֵא — הוֹאִיל וְאֵין הַקָּרְבָּן קָרֵב בְּטֻמְאָתָם, אֲפִלּוּ הָיָה קָרֵב בִּטְמֵא מֵת — הֵם אֲסוּרִים לָבוֹא לַמִּקְדָּשׁ, וְצָרִיךְ לְשַׁלְּחָם. וַאֲפִלּוּ בַּמִּקְדָּשׁ גּוּפֵיהּ — לֹא הֻתַּר אֶלָּא בָּעֲזָרָה, שֶׁהֵם צְרִיכִים לָבוֹא שָׁם (תּוֹסָפוֹת פְּסָחִים צ״ה סוֹף עַמּוּד ב׳); אֲבָל מִן הַהֵיכָל — אִם דָּחֲקוּ וְנִכְנְסוּ, צָרִיךְ לְשַׁלְּחָם; שֶׁנֶּאֱמַר "וִישַׁלְּחוּ מִן הַמַּחֲנֶה" — אֲפִלּוּ מִקְצָת מַחֲנֶה, כְּגוֹן הָכָא, שֶׁאֵין מִשְׁתַּלְּחִין מִכָּל מַחֲנֵה שְׁכִינָה, שֶׁהֲרֵי מֻתָּרִים בָּעֲזָרָה — אֲפִלּוּ הָכֵי צָרִיךְ שִׁלּוּחַ. וּמִזֶּה נִלְמַד גַּם כֵּן: כְּשֶׁצָּרִיךְ לִבְנוֹת וּלְתַקֵּן בָּעֲזָרָה — שֶׁאֵין לוֹ לִידָּחֵק וְלֵילֵךְ לַהֵיכָל וְכַיּוֹצֵא בּוֹ, וְצָרִיךְ שִׁלּוּחַ מִשָּׁם; וּפָשׁוּט:

**וְנִתְבָּרֵר**, כִּי הֲרֵי שֶׁתַּרְשֵׁנוּ הַמַּלְכוּת לִבְנוֹת בֵּית הַמִּקְדָּשׁ בְּעִיר הַקֹּדֶשׁ בִּירוּשָׁלַיִם (כִּדְאִיתָא בַּיְרוּשַׁלְמִי דְּמַעֲשֵׂר שֵׁנִי וּשְׁאָר דּוּכְתֵּי, דְּמִתְּחִלָּה "בּוֹנֵה יְרוּשָׁלַיִם ה׳", וְאַחַר כָּךְ "נִדְחֵי יִשְׂרָאֵל יְכַנֵּס") — אַף עַל פִּי שֶׁכֻּלָּנוּ טְמֵאֵי מֵתִים,
