~ ב׳. וְאִם חָל — אֲפִלּוּ חֵלֶק א׳ קֹדֶם י״ח שָׁעוֹת, שֶׁחָל בְּי״ז שָׁעוֹת אֶלֶף ע״ט חֲלָקִים — אֵינוֹ נִדְחֶה, וְנִקְבָּע בְּיוֹמוֹ:

(ב) כְּשֶׁמּוֹלַד רֹאשׁ הַשָּׁנָה בְּג׳, בְּט׳ שָׁעוֹת וְר״ד חֲלָקִים עַל שָׁעָה י׳, וְהַשָּׁנָה הִיא שָׁנָה פְּשׁוּטָה — הַקְּבִיעוּת נִדְחֶה מִיּוֹמוֹ, וְקוֹבְעִים בְּיוֹם ה׳; שֶׁאִי אֶפְשָׁר בְּמָחֳרָתוֹ, בְּיוֹם ד׳, שֶׁלֹּא אד״ו רֹאשׁ הַשָּׁנָה. וְהַטַּעַם: לְפִי שֶׁאִם תִּתֵּן ד׳ ח׳ תתע״ו עַל ג׳ ט׳ ר״ד — שֶׁהוּא יִתְרוֹן הַשָּׁנָה הַפְּשׁוּטָה — יֵצֵא מוֹלַד תִּשְׁרֵי בַּשָּׁנָה הַבָּאָה בְּסוֹף ז׳, סוֹף י״ח שָׁעוֹת, וְהוּא מוֹלָד זָקֵן, וְיִדָּחֶה לְיוֹם ב׳, כַּנִּזְכָּר לְעֵיל. וְאִם יִקְבְּעוּ רֹאשׁ הַשָּׁנָה זֶה בְּיוֹמוֹ — יִהְיֶה בֵּין רֹאשׁ הַשָּׁנָה זֶה לְרֹאשׁ הַשָּׁנָה הַבָּא ו׳ יָמִים, וְיִצְטָרְכוּ לְפִי זֶה לַעֲשׂוֹת ח׳ חֳדָשִׁים מְלֵאִים בְּאוֹתָהּ שָׁנָה, כְּמוֹ שֶׁיִּתְבָּאֵר לְהַלָּן; וְאֵין עוֹשִׂין כָּל כָּךְ חֳדָשִׁים מְלֵאִים, שֶׁהַדָּבָר תָּמוּהַּ בְּעֵינֵי הַבְּרִיּוֹת. וְכָל שֶׁכֵּן כְּשֶׁהַמּוֹלָד מִגטר״ד וּלְמַעְלָה — שֶׁנִּדְחֶה. אֲבָל לְמַטָּה, אֲפִלּוּ רַק חֵלֶק אֶחָד — כְּגוֹן שֶׁנּוֹלַד בגטר״ג — אֵין זֶה נִדְחֶה, מִפְּנֵי שֶׁיָּחוּל מוֹלַד תִּשְׁרֵי הַבָּא בְּחֵלֶק אֶחָד קֹדֶם י״ח, וְלֹא יִדָּחֶה גַּם כֵּן:

(ג) כְּשֶׁמּוֹלַד רֹאשׁ הַשָּׁנָה בְּיוֹם ב׳, בְּט״ו שָׁעוֹת וְתקפ״ט חֲלָקִים מִשָּׁעָה ט״ז, וְכָל שֶׁכֵּן יֶתֶר עַל כֵּן — אִם הָיְתָה אוֹתָהּ הַשָּׁנָה מוֹצָאֵי מְעֻבֶּרֶת, שֶׁהָיְתָה הַשָּׁנָה הַסְּמוּכָה לָהּ מִלְּפָנֶיהָ מְעֻבֶּרֶת — אֵין קוֹבְעִין רֹאשׁ חֹדֶשׁ בְּיוֹמוֹ, אֶלָּא לְמָחֳרָתוֹ, בְּיוֹם ג׳. לְפִי שֶׁלְּפִי זֶה, עַל כָּרְחֲךָ, הָיָה מוֹלַד תִּשְׁרֵי מִשָּׁנָה שֶׁעָבְרָה: ג׳ י״ח, כַּאֲשֶׁר יֵצֵא לָנוּ אַחַר גִּירְעוֹן הַכ״א תקפ״ט מִן א׳ ט״ו תקפ״ט — יִהְיֶה הַנִּשְׁאָר ג׳ י״ח. וְאִם נִקְבַּע רֹאשׁ הַשָּׁנָה זֶה בְּיוֹמוֹ — אִם כֵּן אֵין בֵּין רֹאשׁ הַשָּׁנָה דְּאִשְׁתְּקַד לְשֶׁל עַכְשָׁיו, בִּשְׁנַת הָעִבּוּר, אֶלָּא ד׳ יָמִים; שֶׁהֲרֵי רֹאשׁ הַשָּׁנָה דְּאִשְׁתְּקַד, הָיָה מוֹלָד זָקֵן בְּיוֹם ג׳, וְנִדְחָה עַד יוֹם ה׳, לְפִי הַנִּזְכָּר לְעֵיל; וְאִם כֵּן, אִי אֶפְשָׁר אִם לֹא שֶׁנַּעֲשָׂה ח׳ חֳדָשִׁים חֲסֵרִים בַּשָּׁנָה שֶׁעָבְרָה — וְאֵין עוֹשִׂין כֵּן, שֶׁהַדָּבָר תָּמוּהַּ בְּעֵינֵי הַבְּרִיּוֹת. לְפִיכָךְ תִּדָּחֶה רֹאשׁ הַשָּׁנָה לְמָחֳרָתוֹ, וְיָבוֹאוּ שִׁשָּׁה חֳדָשִׁים מְלֵאִים וְשִׁבְעָה חֲסֵרִים. אֲבָל אִם יִהְיֶה הַמּוֹלָד אֲפִלּוּ רֶגַע אַחַת קֹדֶם מִט״ו תקפ״ט עַל יוֹם ב׳ — כְּגוֹן בְּט״ו תקפ״ח — אוֹ שֶׁלֹּא הָיְתָה הַשָּׁנָה מוֹצָאֵי עִבּוּר — הֲרֵי זֶה נִקְבָּע בְּיוֹמוֹ. כָּל אֵלּוּ הֵם הַדְּחִיּוֹת שֶׁל יוֹם הַקְּבִיעוּת, כמ״ש:

**כְּבָר** בֵּאַרְנוּ שֶׁחֲדָשָׁהּ שֶׁל לְבָנָה אֵינֶנּוּ בְּיָמִים שְׁלֵמִים, אֲבָל מִצְטָרֵף לְשָׁעוֹת וְלַחֲלָקִים. וְאִי אֶפְשָׁר לוֹמַר שֶׁרֹאשׁ חֹדֶשׁ יִהְיֶה בְּמִקְצַת מִשְּׁעוֹת הַיּוֹם, וְתִהְיֶה מִקְצַת שָׁעוֹת שֶׁעַד הַמּוֹלָד מֵחֹדֶשׁ שֶׁעָבַר — שֶׁנֶּאֱמַר: "עַד חֹדֶשׁ יָמִים"; מִפִּי הַשְּׁמוּעָה לָמְדוּ: שֶׁיָּמִים אַתָּה מְחַשֵּׁב לַחֹדֶשׁ, וְאִי אַתָּה מְחַשֵּׁב שָׁעוֹת
