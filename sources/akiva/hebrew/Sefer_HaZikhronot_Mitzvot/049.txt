~ מִשָּׁם אֶלָּא מוֹלִיךְ וּמֵבִיא, וּמַעֲלֶה וּמוֹרִיד, עַד שֶׁיִּהְיֶה רוֹאֶה כָּל בְּנֵי בָבֶל; אַנְשֵׁי עִיר פּוּמְבְּדִיתָא וְכַיּוֹצֵא — עִיר גְּדוֹלָה, הַסְּמוּכָה — כִּמְדוּרַת אֵשׁ, שֶׁכָּל אֶחָד נוֹטֵל אֲבוּקָה בְּיָדוֹ וְעוֹלֶה לְרֹאשׁ גַּגּוֹ, וְהֵמָּה הָיוּ מוֹדִיעִין לְכָל בְּנֵי בָבֶל. אַחַר כָּךְ, שֶׁקִּלְקְלוּ הַכּוּתִים — וּכְשֶׁהָיָה חֹדֶשׁ מְעֻבָּר הִשִּׂיאוּ הֵם מַשּׂוּאוֹת, לְהַכְשִׁיל בְּנֵי הַגּוֹלָה — הִתְקִינוּ רַבִּי וּבֵית דִּינוֹ (כֵּן הוּא בִּירוּשַׁלְמִי וּמַתְנִיתִין (י״ח עַמּוּד א׳): בִּזְמַן שֶׁבֵּית הַמִּקְדָּשׁ קַיָּם, עַיֵּן מַה שֶּׁכָּתוּב לְעֵיל) שֶׁיִּהְיוּ שְׁלוּחִים יוֹצְאִים, וּמוֹדִיעִים לָרַבִּים. וּשְׁלוּחִים אֵלּוּ אֵין מְחַלְּלִים לֹא אֶת יוֹם טוֹב וְלֹא אֶת יוֹם הַכִּפּוּרִים, וְאֵין צָרִיךְ לוֹמַר שַׁבָּת; שֶׁאֵין מְחַלְּלִים אֶת הַשַּׁבָּת לְקַיְּמוֹ, אֶלָּא לְקַדְּשׁוֹ בִּלְבַד. (עַיֵּן בְּחִדּוּשֵׁי הָרַשְׁבָּ״א, הֶאֱרִיךְ לְהָבִין הַטַּעַם. וּלְפִי עֲנִיּוּת דַּעְתִּי פָּשׁוּט: קִדּוּשׁ הַחֹדֶשׁ הוּא תָּמִיד בִּירוּשָׁלַיִם; וְאִם כֵּן, מִשּׁוּם קָרְבָּן אֵין צָרִיךְ שָׁלִיחַ, וְהָעִקָּר שָׁלִיחַ מִשּׁוּם קָרְבָּן. וְכֵן מִשּׁוּם תַּקָּנַת הַמּוֹעֲדוֹת — אֵין צָרִיךְ, לְפִי שֶׁכָּל יִשְׂרָאֵל עוֹלִים לָרֶגֶל לִירוּשָׁלַיִם, וְשָׁם יָדְעוּ; וּלְכָךְ לֹא צִוְּתָה תּוֹרָה לְחַלֵּל הַשַּׁבָּת בִּשְׁבִיל מוּעָטִים הַנִּשְׁאָרִים בְּבֵיתָם. וְאַף עַל פִּי שֶׁאַחַר כָּךְ, כַּמָּה פְּעָמִים הָיָה הַקִּדּוּשׁ שֶׁלֹּא בִּירוּשָׁלַיִם, וְנַפְקָא מִינָּהּ לְקָרְבָּן; וְגַם אַחַר הַחֻרְבָּן יִהְיֶה צָרִיךְ לִשְׁלִיחַ מִתַּקָּנַת הַמּוֹעֲדוֹת — מִכָּל מָקוֹם לֹא יָכְלוּ חֲכָמִים לְתַקֵּן, וּפָשׁוּט):

**וּלְפִי** שֶׁשְּׁלִיחוּת שְׁלוּחִים הִיא דָּבָר קָשֶׁה מֵהַשָּׂאַת מַשּׂוּאוֹת — לְפִיכָךְ אֵין הַשְּׁלוּחִים יוֹצְאִים עַל כָּל חֹדֶשׁ, אֶלָּא עַל חֹדֶשׁ שֶׁיֵּשׁ בּוֹ צֹרֶךְ לְהוֹדִיעַ. וְשִׁשָּׁה חֳדָשִׁים יֵשׁ בָּהֶן צֹרֶךְ, וַעֲלֵיהֶם הָיוּ הַשְּׁלוּחִים יוֹצְאִים אַחַר שֶׁנִּתְקַדֵּשׁ הַחֹדֶשׁ, בֵּין בִּזְמַנּוֹ בֵּין מְעֻבָּר. וְאַף עַל פִּי שֶׁמַּשּׂוּאוֹת לֹא הָיוּ מַשִּׂיאִין אֶלָּא כְּשֶׁנִּרְאָה בִּזְמַנּוֹ, כְּמוֹ שֶׁנִּתְבָּאֵר — יֵשׁ לוֹמַר: לְפִי שֶׁהַשָּׂאַת מַשּׂוּאוֹת רָחוֹק שֶׁיֶּאֱנַס (עַיֵּן בְּדַף כ״ג עַמּוּד א׳), שַׁפִּיר הָיוּ יוֹדְעִין כְּשֶׁלֹּא הִשִּׂיאוּ שֶׁמְּעֻבָּר; אֲבָל בְּשָׁלִיחַ — שְׁכִיחַ שֶׁיֶּאֱנַס בְּדַרְכּוֹ, וְלֹא יָבוֹא לְאֵיזֶה מָקוֹם; וְאִי לָא נָפְקֵי אַמְּעֻבָּר — כְּשֶׁיֶּאֱנַס וְלֹא יָבוֹא, יַחְשְׁבוּ דִּמְעֻבָּר הוּא, וְיַעֲשׂוּ בְּשִׁיתְסַר, וּבֶאֱמֶת חָסֵר. אֲבָל עַכְשָׁיו, דְּגַם אַמְּעֻבָּר נָפְקֵי — כִּי לֹא אָתֵי, כְּלָל יָדְעֵי דְּנֶאֱנַס, וְיַעֲשׂוּ שְׁנֵי יָמִים מִסָּפֵק (עַיֵּן בְּתוֹסָפוֹת (כ״ג עַמּוּד א׳) דִּבּוּר הַמַּתְחִיל "וְהָא", וְתָבִין). וְאִם נֹאמַר דְּכָאן בְּשָׁלִיחַ דִּשְׁכִיחַ אֳנָסֵי — גַּם הֵם יַחְשְׁבוּ דְּנֶאֱנַס, עַיֵּן שָׁם בַּתּוֹסָפוֹת, שֶׁכָּתְבוּ "תָּלוּ" וְכוּלֵי, דִּשְׁכִיחָא; אִם כֵּן, כָּל שֶׁכֵּן שֶׁיֵּשׁ כָּאן חֲשָׁשׁ אַמָּלֵא אֲמִתִּי, דְּלֹא יֵדְעוּ, דְּתָמִיד יַחְשְׁבוּ דְּנֶאֱנַס; לְכָךְ נָפֵיק אַתַּרְוַיְהוּ. כֵּן נִרְאֶה פָּשׁוּט (וּבְחִנָּם דָּחַק בְּסֵפֶר טוּרֵי אֶבֶן (י״ח עַמּוּד א׳) דִּבּוּר הַמַּתְחִיל "עַל ו׳ ח׳", עַיֵּן שָׁם):

**וְאֵלּוּ** הֵן: עַל נִיסָן — מִפְּנֵי חַג הַפֶּסַח, שֶׁיַּעֲשׂוּהוּ בִּזְמַנּוֹ (כָּתַב בְּסֵפֶר טוּרֵי אֶבֶן: דְּיֵלְכוּ עַד עֶרֶב יוֹם טוֹב הָאַחֲרוֹן, מִשּׁוּם יוֹם טוֹב אַחֲרוֹן; אַף עַל גַּב דְּעַל כָּל פָּנִים יַעֲשׂוּ ב׳ יָמִים, כֵּיוָן דְּעָבְדֵי בְּיוֹם טוֹב רִאשׁוֹן ב׳
