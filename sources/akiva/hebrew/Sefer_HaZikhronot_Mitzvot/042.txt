~ חֲכָמִים, כְּמוֹ כָּאן דִּמְשֶׁקִּלְקְלוּ תִּקְּנוּ חֲכָמִים וְכוּלֵי, וְהָתָם נַמִּי לְאַחַר קִלְקָלָה. וּמַה שֶּׁכָּתַב הוּא דְּלָאו כ״א ע״ה — לֹא מַשְׁמַע כֵּן, מִדְּמַאי וּשְׁאָרֵי. וְעַיֵּן שָׁם עוֹד מַה שֶּׁכָּתַב עַל "לְפִיכָךְ" וְכוּלֵי, עַיֵּן שָׁם. וְתָמוּהַּ, דִּלְפִי עֲנִיּוּת דַּעְתִּי יֵשׁ לוֹמַר בְּפָשׁוּט: חֲדָא בַּחֲבֶרְתָּהּ — דְּבִשְׁאָר עֵדוּת פָּסוּל רַק כָּאן, הוֹאִיל וְאֵין מְדַקְדְּקִין; לְפִיכָךְ מִדְּבַר תּוֹרָה כָּשֵׁר, וְאַחַר שֶׁקִּלְקְלוּ — הִשְׁווּ לִשְׁאָר עֵדֻיּוֹת, וּמוּבָן):

**מִשֶּׁקִּלְקְלוּ** הָאֶפִּיקוֹרְסִים, וְהָיוּ שׂוֹכְרִין אֲנָשִׁים לְהָעִיד שֶׁרָאוּ — וְהֵם לֹא רָאוּ — הִתְקִינוּ שֶׁלֹּא יְקַבְּלוּ בֵּית דִּין עֵדוּת הַחֹדֶשׁ אֶלָּא מֵעֵדִים שֶׁמַּכִּירִין אוֹתָן בֵּית דִּין שֶׁהֵן כְּשֵׁרִים, וְחוֹקְרִין וְדוֹרְשִׁין אוֹתָם אִם אֵינָם עֵדִים שְׂכוּרִים מֵהָאֶפִּיקוֹרְסִים — מִלְּבַד הַחֲקִירָה וְהַדְּרִישָׁה עַל גּוּף הָעֵדוּת, שֶׁכְּבָר נִתְבָּאֵר. (וְנִסְתַּלְּקָה קֻשְׁיַת בַּעַל לֶחֶם מִשְׁנֶה, מַה שֶּׁהִנִּיחַ בְּקֻשְׁיָא.) לְפִיכָךְ, אִם רָאוּ בְּנֵי הָעִיר שֶׁהָעֵדִים מִשָּׁם — שֶׁאֵין אוֹתָם הָעֵדִים אֲנָשִׁים יְדוּעִים לְבֵית דִּין — מְשַׁלְּחִים עִמָּהֶם עוֹד עֵדִים אֲחֵרִים, הַנִּכָּרִים לְבֵית דִּין, שֶׁיָּעִידוּ עַל אֵלּוּ שֶׁכְּשֵׁרִים הֵם; וְאַחַר כָּךְ מְקַבְּלִין מֵהֶם:

**וְצָרִיךְ** שֶׁיִּרְאוּ הָעֵדִים הַלְּבָנָה עַל פְּנֵי הָרָקִיעַ; וְאָז, אַף עַל פִּי שֶׁלֹּא נִתְכַּוְּנוּ לִרְאוֹת לְשֵׁם עֵדוּת, רַק לִרְאִיָּה בְּעָלְמָא — יְכוֹלִים לְהָעִיד. (עַיֵּן בְּתוֹסָפוֹת כְּרֵיתוֹת דַּף י״ב, וּבְסֵפֶר טוּרֵי אֶבֶן כ״ד עַמּוּד א׳ דִּבּוּר הַמַּתְחִיל "מִכָּל מָקוֹם", עַיֵּן שָׁם; וּלְפִי עֲנִיּוּת דַּעְתִּי — אַף הַתּוֹסָפוֹת מוֹדֵי בְּקִדּוּשׁ הַחֹדֶשׁ, עַיֵּן שָׁם.) אֲבָל אָמְרוּ שֶׁרָאוּהָ בֶּעָבִים, אוֹ רָאוּ צִלָּהּ בַּמַּיִם אוֹ בְּעֶשֶׁשִׁית — וְנִרְאֶה דְּאֵין שִׁעוּר לֶעָבִים, אֲפִלּוּ דַּקִּים וְקַלִּים — כֵּיוָן שֶׁלֹּא רָאוּהָ בָּרָקִיעַ, אֵינוֹ כְּלוּם, דְּאֵימוֹר עִיגּוּל שֶׁל עָב לָבָן הוּא. וַאֲפִלּוּ רָאוּ מִקְצָתָהּ (פֵּרוּשׁ: מִקְצַת הָרְאִיָּה, וּמוּסָב "מִקְצָתָהּ" אַרְאִיָּה, לֹא אַלְּבָנָה; וְעַיֵּן בַּמְּפָרֵשׁ, וְאֵין דְּבָרָיו מוּבָנִים לִי) בָּרָקִיעַ וּמִקְצָתָהּ בֶּעָבִים, אוֹ בַּמַּיִם וְעֶשֶׁשִׁית — אֵין זוֹ רְאִיָּה, כַּנִּזְכָּר לְעֵיל:

**הָיוּ** הָעֵדִים מַכְחִישִׁים זֶה אֶת זֶה בִּבְדִיקוֹת וַחֲקִירוֹת — כְּגוֹן שֶׁזֶּה אוֹמֵר "גָּבְהָהּ שְׁלֹשָׁה קוֹמוֹת", וְזֶה אוֹמֵר "גָּבְהָהּ חֲמִשָּׁה קוֹמוֹת" — אֵין מִצְטָרְפִים; וּמִצְטָרֵף אֶחָד מֵהֶם עִם אַחֵר שֶׁיָּעִיד כְּמוֹתוֹ. וְדַוְקָא כְּשֶׁאִי אֶפְשָׁר לְתָרֵץ דִּבְרֵיהֶם, כְּגוֹן כָּאן; דְּאַף עַל גַּב דְּאָדָם טוֹעֶה בְּגֹבַהּ קוֹמָה אַחַת, מִכָּל מָקוֹם אִי אֶפְשָׁר לָנוּ לוֹמַר שֶׁשְּׁנֵיהֶם טוֹעִין כָּל אֶחָד בְּקוֹמָה אַחַת, דְּלָא שַׁיָּךְ דְּיִטְעוּ שְׁנֵיהֶם (פְּסָחִים י״ב עַמּוּד א׳, תּוֹסָפוֹת דִּבּוּר הַמַּתְחִיל "אֶלָּא"). וְכֵן כְּשֶׁזֶּה אוֹמֵר ג׳ וְזֶה אוֹמֵר ד׳ וָחֵצִי, אוֹ כָּל שֶׁהוּא יוֹתֵר מִד׳ (סֵפֶר טוּרֵי אֶבֶן כ״ד עַמּוּד א׳ דִּבּוּר הַמַּתְחִיל "וְאֶחָד אוֹמֵר ה׳"). אֲבָל אֶחָד אוֹמֵר ב׳ קוֹמוֹת וְאֶחָד אוֹמֵר ג׳ קוֹמוֹת — מְתָרְצִים דִּבְרֵיהֶם לִהְיוֹת שָׁוִים, וְהָאֶחָד טוֹעֶה
