~ מוּסָף בִּמְקוֹם "יוֹצֵר" — לֹא יָצָא יְדֵי תְּפִלַּת "יוֹצֵר", אֲבָל יְדֵי מוּסָף יָצָא, אַף עַל פִּי שֶׁהִקְדִּימָהּ לְשַׁחֲרִית:

**אַחַר כָּךְ** מַחֲזִיר שְׁלִיחַ צִבּוּר הַתְּפִלָּה. וּמִנְהַג הַסְּפָרַדִּים לוֹמַר קְדֻשַּׁת "כֶּתֶר"; וּכְשֶׁמַּגִּיעַ לִ"מְלֹא כָל הָאָרֶץ כְּבוֹדוֹ" — אוֹמֵר "לְעֻמָּתָם" וְכוּלֵי, כְּבְשַׁחֲרִית; וּמִנְהָגֵנוּ לוֹמַר "נְקַדֵּשׁ", כְּבְשַׁחֲרִית. וְאַחַר שֶׁסִּיֵּם הַחֲזָרָה — אוֹמֵר קַדִּישׁ, וְשִׁיר שֶׁל "בָּרְכִי נַפְשִׁי"; וּבִקְצָת מְקוֹמוֹת אֵין נוֹהֲגִים כֵּן. (וְזֶהוּ לְשׁוֹן "הַנְהָגַת הַגְּרָ״א": "בִּתְפִלַּת עַרְבִית דְּרֹאשׁ חֹדֶשׁ — מַרְבִּים קְצָת בְּנֵרוֹת; וְהוּא זִכְרוֹנוֹ לִבְרָכָה לָבַשׁ כּוֹבַע שַׁבָּת לִכְבוֹד רֹאשׁ חֹדֶשׁ, שֶׁהוּא דְּאוֹרַיְתָא. וּבְהַלֵּל, פָּסוּק 'יֹאמַר נָא' וְ'יֹאמְרוּ נָא' — הַשְּׁלִיחַ צִבּוּר לְבַד אוֹמֵר (בְּיוֹם שֶׁמְּדַלְּגִין), וְהַצִּבּוּר שׁוֹמְעִין וְעוֹנִין 'הוֹדוּ'. בִּ'יהַלְלוּךָ' הַנֻּסָּח: 'יְהַלְלוּךָ ה׳ אֱלֹקֵינוּ כָּל מַעֲשֶׂיךָ'. וְהַקְּרִיאָה בְּפָרָשַׁת הַתָּמִיד: הָרִאשׁוֹן קוֹרֵא 'וַיְדַבֵּר', 'צַו', 'וְאָמַרְתָּ'; הַב׳ — 'אֶת הַכֶּבֶשׂ אֶחָד' וְכוּלֵי, עַד 'וּבְיוֹם הַשַּׁבָּת'; הַג׳ חוֹזֵר לְמַעְלָה, וּמַתְחִיל מִן 'עוֹלַת תָּמִיד' וְכוּלֵי, עַד 'וּבְרָאשֵׁי חָדְשֵׁיכֶם'; וְהַד׳ — 'וּבְרָאשֵׁי חָדְשֵׁיכֶם' עַד סוֹף. בְּמוּסָף הַנֻּסָּח: 'זִכָּרוֹן לְכֻלָּם הָיוּ' וְכוּלֵי — כִּי עַד 'מִזְבֵּחַ חָדָשׁ' הַכֹּל לְשׁוֹן עָבָר, וּמִ'מִּזְבֵּחַ' מַתְחִילִין בַּקָּשָׁה עַל הֶעָתִיד. אֵין אוֹמְרִים שִׁיר שֶׁל יוֹמוֹ, וְאוֹמְרִים בִּמְקוֹמוֹ, בְּב׳ יָמִים דְּרֹאשׁ חֹדֶשׁ, שִׁיר שֶׁל יוֹם — מִזְמוֹר ק״ד 'בָּרְכִי נַפְשִׁי'; וְדוֹחֶה אֲפִלּוּ שֶׁל שַׁבָּת, וְכָל שֶׁכֵּן שֶׁל חֲנֻכָּה", עַד כָּאן לְשׁוֹנוֹ):

# פֶּרֶק ה׳ — בְּבִרְכַּת קִדּוּשׁ לְבָנָה וְחַמָּה

**הָרוֹאֶה** הַחַמָּה בִּתְקוּפָתָהּ — וְהוּא בִּתְקוּפַת נִיסָן שֶׁל רֹאשׁ מַחֲזוֹר גָּדוֹל שֶׁל כ״ח שָׁנָה, שֶׁאָז הַתְּקוּפָה בִּתְחִלַּת לֵיל ד׳, וּכְמוֹ שֶׁנִּתְבָּאֵר בַּחֶשְׁבּוֹן (לְעֵיל פֶּרֶק ב׳) שֶׁכֵּן הָיָה בִּתְחִלַּת הַיְצִירָה — כְּשֶׁרוֹאֶה אוֹתָהּ בְּיוֹם ד׳ בַּבֹּקֶר, מִצְוַת חֲכָמִים לְבָרֵךְ: "בָּרוּךְ אַתָּה ה׳ אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם עוֹשֶׂה בְרֵאשִׁית" (בַּטּוּר וּרְאַ״שׁ אִיתָא: "עוֹשֶׂה מַעֲשֵׂה בְרֵאשִׁית"). וְרָאוּי לְבָרֵךְ בְּרֹב עָם; וְנָהֲגוּ בְּשַׁחֲרִית, כְּשֶׁיּוֹצְאִים מִבֵּית הַכְּנֶסֶת — מִתְאַסְּפִים הַכֹּל וּמְבָרְכִים. וְיַכְרִיז הַשַּׁמָּשׁ בָּעֶרֶב שֶׁלְּמָחָר יִזָּהֵר כָּל אֶחָד לְבָרֵךְ, שֶׁמָּא מִי לֹא יִהְיֶה מָחָר בְּבֵית הַכְּנֶסֶת בְּעֵת אֲסִיפַת הַקָּהָל. וְנָכוֹן לְהַקְדִּים הַבְּרָכָה כָּל מַה שֶּׁאֶפְשָׁר בַּבֹּקֶר, מִשּׁוּם זְרִיזִין מַקְדִּימִין; וְעוֹד, שֶׁמָּא תִּתְכַּסֶּה הַחַמָּה בֶּעָבִים; בְּדֶרֶךְ כְּלָל — אִם לֹא שֶׁמַּמְתִּין לְרֹב עָם. וְגַם אָז, אִם הוּא קְצָת יוֹם מְעֻנָּן, שֶׁיֵּשׁ לָחוּשׁ פֶּן תִּתְכַּסֶּה בֶּעָנָן — לֹא יַמְתִּין, וְיַקְדִּים לְבָרֵךְ
