~ נִשְׁחָט" וּשְׁאָר דּוּכְתֵּי — דְּשֶׁלֹּא לְאוֹכְלָיו פָּסוּל; וְאַטּוּ פָּלֵיג רַבִּי יְהוּדָה עַל זֶה? אֲבָל לְפִי דַּעַת הָרַמְבַּ״ם יֵשׁ לוֹמַר, דְּרַבִּי יְהוּדָה עַל כָּרְחֲךָ סְבִירָא לֵיהּ "אִישׁ לְפִי אָכְלוֹ" — הַיְנוּ בַּחֲבוּרָה, וְקָרֵי לְכָל הַחֲבוּרָה לְשׁוֹן יָחִיד. וְזֶה נִיחָא לְרַבִּי יוֹסֵי שַׁפִּיר: מֵ"אִישׁ לְפִי אָכְלוֹ" לָמַדְנוּ, לְכָל מִי שֶׁאֵין יָכוֹל לֶאֱכוֹל — פָּטוּר; וְאִם כֵּן, מִמֵּילָא חֲבוּרָה מֵאֲנָשִׁים פְּטוּרִים — אֵין מוֹעִיל דִּבְהִצְטָרְפָן יִהְיֶה כְּזַיִת. אֲבָל לְרַבִּי יְהוּדָה, עַל כָּרְחֲךָ רוֹצֶה לוֹמַר: קְרָא אַחֲבוּרָה — דְּבָעֵינַן "לְפִי אָכְלוֹ": שֶׁאִם אֵין יְכוֹלִים לֶאֱכוֹל כְּזַיִת — פְּטוּרִים, וְאִם יְכוֹלִים הַחֲבוּרָה לֶאֱכוֹל כְּזַיִת — חַיָּבִים. וְשַׁפִּיר יֵשׁ לוֹמַר, אַף דְּכָל אֶחָד אֵין יָכוֹל לֶאֱכוֹל — כֵּיוָן דְּכָל הַחֲבוּרָה יְכוֹלִים — רַשָּׁאִים; וְשַׁפִּיר פָּלֵיג רַבִּי יוֹסֵי אַהָא, וְאָתֵי שַׁפִּיר. וּמוּכָח כְּפֵירוּשׁ הָרַמְבַּ״ם:

**וְיֵשׁ** חוֹלְקִים, וְסוֹבְרִים דְּאַף דְּכָל אֶחָד אֵין יָכוֹל לֶאֱכוֹל — כֵּיוָן דְּכָל בְּנֵי הַחֲבוּרָה יְכוֹלִים לֶאֱכוֹל כְּזַיִת, שַׁפִּיר שׁוֹחֲטִים (פֵּירֵשׁ רַשִׁ״י שָׁם). וְצָרִיךְ לוֹמַר לְדִידֵיהּ, דְּחַיָּבִים בִּכְהַאי גַּוְנָא בְּקָרְבַּן פֶּסַח; דְּאֵין פְּטוּרִים אֶלָּא מִפְּנֵי הָאֹנֶס, דְּאִי אֶפְשָׁר לָהֶם לֶאֱכוֹל; וְכֵיוָן דְּיֵשׁ אוֹכְלִים — גַּם הֵם חַיָּבִים בְּאוֹתוֹ מִקְצָת, פָּחוֹת מִכְּזַיִת, לֶאֱכוֹל. וְצָרִיךְ עִיּוּן לְפִי זֶה: לָמָּה לָהּ לַגְּמָרָא (דַּף ס״א) לְמֵימַר "וְאִתְּקַשׁ אוֹכְלִין לִמְנוּיִין" וְכוּלֵי, עַיֵּן שָׁם, לְעִנְיַן "לְאוֹכְלָיו וְשֶׁלֹּא לְאוֹכְלָיו"? תִּיפּוֹק לֵיהּ, דִּ"לְאוֹכְלָיו וְשֶׁלֹּא לְאוֹכְלָיו" — גַּם הַ"שֶּׁלֹּא לְאוֹכְלָיו" חַיָּבִים, כֵּיוָן דְּיֵשׁ אוֹכְלָיו! (א״ה: שָׁם לֹא מָצָאתִי עִנְיַן "לְאוֹכְלָיו וְשֶׁלֹּא לְאוֹכְלָיו"; וְאוּלַי הַקֻּשְׁיָא סוֹבֶבֶת עַל שְׁאֵלַת הַגְּמָרָא, ס״ב עַמּוּד ב׳: "מַה בֵּין לִשְׁמוֹ וְשֶׁלֹּא לִשְׁמוֹ לְאוֹכְלָיו וְשֶׁלֹּא לְאוֹכְלָיו", עַד כָּאן.) וְאֶפְשָׁר לְיַשֵּׁב קְצָת, וְצָרִיךְ עִיּוּן. וְהָעִקָּר כְּדֵעָה רִאשׁוֹנָה:

**וְכָל** הַפְּטוּרִים מִן הָרְאִיָּה (שֶׁיִּתְבָּאֵר בְּפָרָשַׁת מִשְׁפָּטִים וּבְסוֹף פָּרָשַׁת רְאֵה) — חַיָּבִים בְּפֶסַח; חוּץ מִן חֵרֵשׁ שׁוֹטֶה וְקָטָן, שֶׁהֵם פְּטוּרִים מִכָּל הַמִּצְוֹת; וְחוּץ מִן הֶעָרֵל (וְעוֹבֵר בְּלָאו, כְּמוֹ שֶׁיִּתְבָּרֵר לְהַלָּן), וְהַטָּמֵא (יִתְבָּאֵר מִשְׁפָּטוֹ בְּפָרָשַׁת בְּהַעֲלֹתְךָ, בְּפֶסַח שֵׁנִי), וְהַזָּקֵן שֶׁאֵינוֹ יָכוֹל לַעֲלוֹת בְּרַגְלָיו מִירוּשָׁלַיִם לָעֲזָרָה — שֶׁהַתּוֹרָה פְּטָרָה בְּפֵירוּשׁ לְאֵלּוּ, וּפְטָרָהּ לַזָּקֵן בַּמֶּה שֶּׁנֶּאֱמַר "וּבְדֶרֶךְ לֹא הָיָה" — הָא אִם הָיָה בְּדֶרֶךְ רְחוֹקָה, פָּטוּר; וְזֶה דֶּרֶךְ רְחוֹקָה הוּא לַזָּקֵן (וְאַף עַל פִּי שֶׁגַּם בְּפֶסַח שֵׁנִי יִהְיֶה אוֹתוֹ דֶּרֶךְ רְחוֹקָה — מִכָּל מָקוֹם, מִפֶּסַח רִאשׁוֹן פְּטָרַתּוּ תּוֹרָה. כֵּן נִרְאֶה לַעֲנִיּוּת דַּעְתִּי לְיַשֵּׁב דִּבְרֵי הַתּוֹסָפוֹת (ג׳ עַמּוּד ב׳), דִּבּוּר הַמַּתְחִיל "מֵאֵלֶיהָ", בַּמֶּה שֶּׁכָּתְבוּ "אוֹ זָקֵן" וְכוּלֵי "דְּפָטוּר" וְכוּלֵי; עַיֵּן בְּמִשְׁנֶה לַמֶּלֶךְ מַה שֶּׁכָּתַב עֲלֵיהֶם. וְנִרְאֶה לִי דַּעַת הַתּוֹסָפוֹת: דְּעַיֵּן בָּרַמְבַּ״ם (פֶּרֶק ה׳, הֲלָכָה ט׳): "הָלַךְ וְלֹא" וְכוּלֵי, שֶׁהָיָה בִּירוּשָׁלַיִם וְהָיָה חוֹלֶה בְּרַגְלָיו, וְלֹא הִגִּיעַ לָעֲזָרָה עַד שֶׁעָבַר זְמַן הַקָּרְבָּן — הֲרֵי זֶה אָנוּס, וְאֵינוֹ בְּדֶרֶךְ רְחוֹקָה, עַיֵּן שָׁם. וְהִנֵּה רַשִׁ״י (צ״ד עַמּוּד א׳), דִּבּוּר הַמַּתְחִיל "חַיָּב", כָּתַב "שֶׁאֵין זֶה אֹנֶס" וְכוּלֵי, עַיֵּן שָׁם. וְצָרִיךְ לוֹמַר, דְּהוּא מְפָרֵשׁ גַּם דִּבְרֵי הַיְרוּשַׁלְמִי, "דְּחוֹלֶה בְּרַגְלָיו" (שֶׁהֵבִיא בַּכֶּסֶף מִשְׁנֶה), גַּם כֵּן בִּכְהַאי
