~ הָאָדָם יִסְתַּכֵּל שָׁוֶה לְכָל צַד. וְנִמְשָׁךְ שֶׁהָאֹפֶק מִשְׁתַּנֶּה כְּמוֹ כֵן מֵאָדָם אֶל אָדָם; וְהָאֹפֶק הַנִּרְאֶה לְאָדָם זֶה — אֵין נִרְאֶה לְאָדָם הָעוֹמֵד רָחוֹק מִמֶּנּוּ, אֲבָל יַתְחִיל בְּמָקוֹם אַחֵר כְּפִי הַמֶּרְחָק שֶׁבֵּינֵיהֶם, וְכֵן סִיּוּמוֹ:

**וְזֶה** מוּבָן מִמַּה שֶּׁנִּתְבָּאֵר הֱיוֹת נְקֻדַּת נֹכַח הָרֹאשׁ מֶרְכָּזִי; וְגַם נִרְאֶה לָעַיִן לַמִּסְתַּכֵּל בַּצּוּרָה שֶׁצִּיַּרְנוּ לְמַעְלָה:

**וְכַאֲשֶׁר** נַרְבֶּה לְהִסְתַּכֵּל בְּאָפְקוֹ — נִרְאֶה בְּחוּשׁ סִבּוּב קַעֲרוּרִית הַשָּׁמַיִם עִם כּוֹכָבֶיהָ, אוֹ עִם הַחַמָּה. אֲבָל נִרְאֶה, אַחַר הַהִסְתַּכְּלוּת פַּעַם בְּפַעַם, ב׳ נְקֻדּוֹת בְּקַעֲרוּרִיתוֹ — אַחַת בְּאֶמְצַע הַצָּפוֹן, וְאַחַת בְּאֶמְצַע הַדָּרוֹם (אָמְרֵינוּ "בְּאֶמְצַע" — אֵינוֹ בְּכָל מָקוֹם, אֲבָל רַק לַשּׁוֹכְנִים תַּחַת קַו מַשְׁוֵה הַיּוֹם, וְהוּא בְּקַו הָאָרֶץ הַמַּפְסִיק בֵּין צָפוֹן לְדָרוֹם, שֶׁמִּמֶּנּוּ לִקְצֵה הַדְּרוֹמִי כְּמִמֶּנּוּ לִקְצֵה הַצְּפוֹנִי, וְהַיּוֹם וְהַלַּיְלָה שָׁוִים שָׁם תָּמִיד י״ב שָׁעוֹת, כְּמוֹ שֶׁיִּתְבָּאֵר — שָׁם הַנְּקֻדּוֹת מַמָּשׁ עַל חֲצִי הָאֹפֶק הַצְּפוֹנִי וְהַדְּרוֹמִי; אֲבָל בְּאֹפֶק זוּלַת זֶה — הוּא מָשׁוּךְ מִן הָאֹפֶק לְפִי עֵרֶךְ מֶרְחָקוֹ מִקַּו הַמַּשְׁוֶה), הַשּׁוֹמְרִים מַצָּבָם לְעוֹלָם, בְּלִי סִבּוּב וְשִׁנּוּי; וְהֵם הַנִּקְרָאִים בְּשֵׁם "צִירֵי" אוֹ "קוֹטְבֵי הָעוֹלָם", כִּי עֲלֵיהֶם יְסוֹבֵב כָּל כַּדּוּר הַשָּׁמַיִם, וְהֵם שׁוֹקְטִים. וְהַצְּפוֹנִי נִקְרָא "צִיר" אוֹ "קֹטֶב הַצְּפוֹנִי", וַאֲשֶׁר בַּדָּרוֹם נִקְרָא "צִיר" אוֹ "קֹטֶב הַדְּרוֹמִי"; וְהֵם מַמָּשׁ בְּמֶרְכַּז הַשָּׁמַיִם:

**וְהַכּוֹכָבִים** הַסְּמוּכִים יַעֲשׂוּ סִבּוּבִים וְעִגּוּלִים סְבִיבָם, בְּאֹפֶן שֶׁיִּגְדְּלוּ הָעִגּוּלִים כָּל זְמַן שֶׁיִּרְחֲקוּ מֵהַצִּירִים; עַד כִּי הַכּוֹכָבִים הָעוֹמְדִים בָּאֶמְצַע מַמָּשׁ, בְּמֶרְחָק שָׁוֶה מִזֶּה וּמִזֶּה — הֵם הָעוֹשִׂים הָעִגּוּל הַיּוֹתֵר גָּדוֹל מִכֻּלָּם; וְהוּא הָעִגּוּל הַמַּתְחִיל בְּאֶמְצַע הָאֹפֶק הַמִּזְרָחִי, וְהוֹלֵךְ עַד בְּאֶמְצַע הָרָקִיעַ בְּגָבְהָהּ, וְעַד אֶמְצַע הָאֹפֶק הַמַּעֲרָבִי, וּמִשָּׁם אֶל תַּחַת הָאָרֶץ, עַד שֶׁיָּשׁוּב לְהַגִּיעַ לְאֶמְצַע אֹפֶק הַמִּזְרָחִי. וְעִגּוּל זֶה נִקְרָא "עִגּוּל מַשְׁוֵה הַיּוֹם" אוֹ "קַו הַשָּׁוֶה":

**וּבוֹ**, בָּעִגּוּל הַגָּדוֹל הַזֶּה — הוּא חֲגוֹרַת הַמַּזָּלוֹת, וּמְשׁוּכָה קְצָת מִמֶּנּוּ לְצַד צָפוֹן וּלְצַד דָּרוֹם. וְעִנְיַן חֲגוֹרָה זוֹ הוּא: כִּי הַמַּזָּלוֹת הָעוֹמְדִים בְּעִגּוּל זֶה — חִלְּקוּם הַתּוֹכְנִים הַקַּדְמוֹנִים לְי״ב צוּרוֹת, בַּעֲשׂוֹתָם מֵחִבּוּר אֵיזֶה כּוֹכָבִים תְּמוּנָה כָּךְ, וּקְרָאוּהָ בְּשֵׁם, לֵידַע מְקוֹם כָּל חֵלֶק וָחֵלֶק מִמֶּנּוּ; וְהֵם הַנִּקְרָאִים "י״ב מַזָּלוֹת". וְהֵם הֵם כּוֹכָבִים, הֵם הֵם מַזָּלוֹת — שֶׁהַמַּזָּלוֹת מְחֻבָּרִים מִכּוֹכָבִים. וְאֵלֶּה שְׁמוֹתָם: טָלֶה, שׁוֹר, תְּאוֹמִים, סַרְטָן, אַרְיֵה, בְּתוּלָה, מֹאזְנַיִם, עַקְרָב,
