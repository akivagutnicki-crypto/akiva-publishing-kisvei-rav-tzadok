~ לֹא יִתְּנוּ עֲבוּר מִצְוַת עֲשֵׂה). וְאִם בִּזְבְּזוּ בֵּית דִּין מָמוֹן, וְאַחַר כָּךְ נִתְעַשֵּׁר הָאָב — צָרִיךְ עִיּוּן אִם יְכוֹלִים לִגְבּוֹת מִמֶּנּוּ, אִי עֲבוּרוֹ נָתְנוּ, אוֹ כֵּיוָן שֶׁחָל עֲלֵיהֶם הַמִּצְוָה; וְצָרִיךְ עִיּוּן:

**אֵימָתַי** מָלִין? מִשֶּׁתָּנֵץ הַחַמָּה שֶׁל יוֹם ח׳ לְלֵדָתוֹ (וְאִם נוֹלַד בֵּין הַשְּׁמָשׁוֹת — שֶׁל יוֹם ט׳). וְאִם עָבַר וּמָל מִשֶּׁעָלָה עַמּוּד הַשַּׁחַר — יָצָא. וְהֵיכִי דְּאִי אֶפְשָׁר בְּעִנְיָן אַחֵר, אַף לְכַתְּחִלָּה כְּדִיעֲבַד דָּמֵי, וְרַשַּׁאי לְמוּלוֹ לְכַתְּחִלָּה בַּעֲלוֹת הַשַּׁחַר. אֲבָל אִם מָל בַּלַּיְלָה, אוֹ תּוֹךְ ח׳, אֲפִלּוּ בַּיּוֹם — לֹא יָצָא, וְצָרִיךְ לַחֲזוֹר לְהַטִּיף מִמֶּנּוּ דַּם בְּרִית. וְכָל הַיּוֹם עַד צֵאת הַכּוֹכָבִים כָּשֵׁר לְמִילָה, אֶלָּא שֶׁזְּרִיזִין מַקְדִּימִין לְמִצְוָה, וְלָכֵן יְזָרֵז עַצְמוֹ לְהַשְׁכִּים בַּבֹּקֶר (אִם יֵשׁ מִנְיַן עֲשָׂרָה) לָמוּל. וְיִתְפַּלֵּל מִקֹּדֶם, דִּתְפִלָּה וּקְרִיאַת שְׁמַע תְּדִירִים וְקוֹדְמוּ לְמִילָה; וְכֵן בְּיוֹם שֶׁיֵּשׁ בּוֹ קְרִיאַת הַתּוֹרָה אוֹ מוּסָף — גַּם כֵּן תְּדִירֵי וְקוֹדְמֵי. אֲבָל בְּרֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים נָהֲגוּ לָמוּל אַחַר קְרִיאַת הַתּוֹרָה, קֹדֶם מוּסָף. נִרְאֶה לְפִי שֶׁהַמּוּסָף נֶגֶד הַקָּרְבָּן דְּיוֹמוֹ, וְקָרְבַּן מוּסָף דְּרֹאשׁ הַשָּׁנָה אֵינוֹ תָּדִיר אֶלָּא פַּעַם אַחַת בַּשָּׁנָה, וּמִילָה לְגַבֵּיהּ כִּי תְּדִירָא דָּמְיָא. וּלְפִי זֶה הוּא הַדִּין בְּכָל יוֹם טוֹב הָיָה רָאוּי לְהַקְדִּים הַמִּילָה לְמוּסָף, אֲבָל לֹא נָהֲגוּ כֵּן אֶלָּא רֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים. אוּלַי לְפִי שֶׁשּׁוֹהִין הַרְבֵּה בִּתְפִלַּת מוּסָף, אָז תְּאֻחַר הַמִּילָה מְאֹד, לְכָךְ מַקְדִּימִין. וְלָכֵן אִם יֵשׁ חַזָּנִים וְכַהַאי גַּוְנָא שֶׁמַּאֲרִיכִים גַּם בְּמוּסָף דִּשְׁאָר יוֹם טוֹב — יְמוּלוּ קֹדֶם מוּסָף. וּמִילָה קוֹדֶמֶת לִתְקִיעַת שׁוֹפָר, דְּאֵינָהּ אֶלָּא פַּעַם אַחַת בַּשָּׁנָה, וּמִילָה כִּתְדִירָא לְגַבֵּיהּ (לְפִי זֶה לִכְאוֹרָה הוּא הַדִּין לִנְטִילַת לוּלָב, וְלֹא נָהֲגוּ הָכֵי; וְנִרְאֶה דִּנְטִילַת לוּלָב ז׳ יָמִים — אוּלַי חָשׁוּב קְצָת תָּדִיר נֶגֶד מִילָה). וְכֵן לְמִקְרָא מְגִלָּה — וְיֵשׁ אוֹמְרִים דִּקְרִיאַת מְגִלָּה קוֹדֶמֶת, מִשּׁוּם פִּרְסוּמֵי נִיסָּא; אֶלָּא דְּהֵיכִי דְּאֵין שָׁהוּת אֶלָּא לַעֲשׂוֹת חַד — יָמוּל, דְּהוּא דְּאוֹרַיְתָא, וּמִקְרָא מְגִלָּה דְּרַבָּנָן; וְכֵן נוֹהֲגִים. וְנִרְאֶה לִי מִילָה וַחֲתֻנָּה — אֲפִלּוּ מִילָה שֶׁלֹּא בִּזְמַנָּהּ קוֹדֶמֶת, אַף דַּחֲתֻנָּה שְׁכִיחָה טְפֵי, כִּדְמַשְׁמַע פֶּרֶק קַמָּא דְּחֻלִּין (דַּף ט׳ עַמּוּד א׳); מִכָּל מָקוֹם מִסְתַּבְּרָא דְּמִילָה קוֹדֶמֶת. וְכֵן מִילָה קוֹדֶמֶת לִקְבוּרַת הַמֵּת. בַּמֶּה דְּבָרִים אֲמוּרִים — מִילַת בְּנוֹ וּקְבוּרַת מֵתוֹ; אֲבָל קְבוּרַת מֵתוֹ וּמִילַת בֶּן אַחֵר, שֶׁאֵין מוֹהֵל אֶלָּא הוּא — קְבוּרַת מֵתוֹ קוֹדֵם, אִם לֹא שֶׁאֵין שָׁהוּת לַעֲשׂוֹת שְׁנֵיהֶם; וַאֲפִלּוּ מִילַת בְּנוֹ — אִם אֶפְשָׁר לִקְבּוֹר הַמֵּת עַד שֶׁלֹּא יַשְׁלִימוּ הַתְּפִלָּה, שֶׁלֹּא יִדָּחוּ אַף זְרִיזוּת הַמִּצְוָה — יַעֲשׂוּ כֵּן:

**הַכֹּל** כְּשֵׁרִים לָמוּל — אֲפִלּוּ נָשִׁים, עֲבָדִים וּקְטַנִּים (וְיֵשׁ חוֹלְקִים בְּנָשִׁים), וְעָרֵל שֶׁמֵּתוּ אֶחָיו מֵחֲמַת מִילָה. אֲבָל לֹא עוֹבֵד כּוֹכָבִים, וְיִשְׂרָאֵל מוּמָר
