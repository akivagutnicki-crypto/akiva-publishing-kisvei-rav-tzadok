~ קָאֵי אִתְּקוּפַת נִיסָן; וְהָרַמְבַּ״ם לְטַעְמֵיהּ, דְּהִשְׁמִיט תְּקוּפַת תִּשְׁרֵי, וְטַעְמוֹ כְּמוֹ שֶׁכָּתְבוּ הַתּוֹסָפוֹת (י״ג עַמּוּד ב׳), סוֹף דִּבּוּר הַמַּתְחִיל "שָׁמוֹר" — דְּקַיְימָא לַן כַּאֲחֵרִים, דְּלָא חָיְישִׁינַן לִתְקוּפַת תִּשְׁרֵי וְכוּלֵי, עַיֵּן שָׁם. אוֹ דְּשָׁם הָאִיבַּעְיָא בְּרשב״י — הֵיאַךְ סָבִירָא לֵיהּ; וּגְמָרָא דְּרֹאשׁ הַשָּׁנָה אָתְיָא כַּאֲחֵרִים, כְּמוֹ שֶׁכָּתְבוּ רַשִׁ״י וְתוֹסָפוֹת; וְאֵין לִפְשׁוֹט מִזֶּה אַדִּבְרֵי רשב״י. כֵּן נִרְאֶה לִי (וְעַיֵּן בְּהַשָּׂגַת הָרמ״ךְ, עַיֵּן שָׁם):

**וְעַל** שְׁלֹשָׁה אֲרָצוֹת הָיוּ סוֹמְכִין בְּאָבִיב וּבְפֵרוֹת הָאִילָן: עַל אֶרֶץ יְהוּדָה, וְעַל עֵבֶר הַיַּרְדֵּן, וְעַל הַגָּלִיל. וְאִם הִגִּיעַ אָבִיב אוֹ פֵּרוֹת הָאִילָן צָמְחוּ בִּשְׁתֵּי אֲרָצוֹת מֵאֵלּוּ, וּבְאַחַת לֹא — אֵין מְעַבְּרִין; וַאֲפִלּוּ אֶחָד מֵאֵלּוּ לֹא נָהֲגוּ גַּם בְּכֻלָּם — לְפִי שֶׁאֵין מְעַבְּרִין אֶלָּא עַל שְׁנֵיהֶם, כְּמוֹ שֶׁנִּתְבָּאֵר. וְאִם לֹא הִגִּיעַ הָאָבִיב וְגַם לֹא צָמְחוּ פֵּרוֹת הָאִילָן בִּשְׁתֵּי אֲרָצוֹת — אֲפִלּוּ הִגִּיעַ שְׁנֵיהֶם בַּשְּׁלִישִׁית — מְעַבְּרִין. וְאֵלּוּ הֵם דְּבָרִים שֶׁהֵן הָעִקָּר, שֶׁמְּעַבְּרִין בִּשְׁבִילָן, כְּדֵי שֶׁיִּהְיוּ הַשָּׁנִים שְׁנֵי הַחַמָּה. וְלִפְעָמִים בֵּית דִּין מְעַבְּרִין הַשָּׁנָה אַף זוּלַת ג׳ דְּבָרִים אֵלּוּ, לְפִי שֶׁמְּעַבְּרִים אֶת הַשָּׁנָה לְצֹרֶךְ — כְּשֶׁיִּרְאוּ בֵּית דִּין צֹרֶךְ בַּדָּבָר, הָיוּ מְעַבְּרִין אַף בִּלְעֲדֵי ג׳ דְּבָרִים שֶׁאָמַרְנוּ. וְעַל אֵיזֶה צֹרֶךְ מְעַבְּרִים? בִּזְמַן שֶׁבֵּית הַמִּקְדָּשׁ קַיָּם — מִפְּנֵי הַדְּרָכִים שֶׁאֵינָן מְתֻקָּנִים מִפְּנֵי יְמוֹת הַגְּשָׁמִים, וְאִי אֶפְשָׁר לַעֲלוֹת לָרֶגֶל וְלַעֲשׂוֹת קָרְבַּן פֶּסַח; וּמִפְּנֵי הַגְּשָׁרִים שֶׁנֶּהֶרְסוּ, וְנִמְצְאוּ הַנְּהָרוֹת מַפְסִיקִין, וּמוֹנְעִים הָעָם מִלַּעֲלוֹת לָרֶגֶל, וּמִסְתַּכְּנִים בְּעַצְמָם וּמֵתִים (כֵּן כָּתַב הָרַמְבַּ״ם. וְלֹא יָדַעְתִּי מַה צֹּרֶךְ לָזֶה — בְּלָאו הָכֵי נַמִּי, כְּמוֹ בִּדְרָכִים, שֶׁנִּמְנָעִים וְאִי אֶפְשָׁר לַעֲלוֹת); וּמִפְּנֵי תַּנּוּרֵי פְּסָחִים שֶׁנִּימּוֹקוּ בַּגְּשָׁמִים, וּצְרִיכִים לְתַקְּנָם וּלְיַבְּשָׁם — מְעַבְּרִים הַשָּׁנָה אַף עַל אַחַת מֵאֵלּוּ, עַד שֶׁיְּתַקְּנוּהוּ הַדְּרָכִים אוֹ הַגְּשָׁרִים אוֹ הַתַּנּוּרִים. וְכֵן מְעַבְּרִים מִפְּנֵי גָּלֻיּוֹת יִשְׂרָאֵל, שֶׁנִּשְׁמַע לְבֵית דִּין שֶׁנֶּעֶקְרוּ מִמְּקוֹמָם לַעֲלוֹת לָרֶגֶל וְלַעֲשׂוֹת קָרְבַּן פֶּסַח, וַעֲדַיִן לֹא הִגִּיעוּ, וְאֵין לָהֶם פְּנַאי לְהַגִּיעַ עַד הַפֶּסַח — מְעַבְּרִין הַשָּׁנָה, כְּדֵי שֶׁיַּגִּיעוּ; שֶׁאִם לֹא יַגִּיעוּ — לֹא יִהְיוּ רְגִילִין לָבוֹא. (וּפֵרוּשׁ "גָּלֻיּוֹת יִשְׂרָאֵל" — כָּתַב הַמְּפָרֵשׁ: דְּבַבַּיִת שֵׁנִי הָיוּ נִקְרָאִים כֵּן, עַיֵּן שָׁם; מַשְׁמַע דְּקָאֵי אַיּוֹשְׁבֵי אֶרֶץ יִשְׂרָאֵל. אֲבָל תּוֹסָפוֹת, דִּבּוּר הַמַּתְחִיל "וּמִפְּנֵי", כָּתְבוּ: דְּהַיְנוּ אַיּוֹשְׁבֵי חוּץ לָאָרֶץ; וְדִבְרֵיהֶם נְכוֹנִים. אֲבָל צָרִיךְ עִיּוּן מִמַּה שֶּׁכָּתְבוּ בִּפְסָחִים (ג׳ עַמּוּד ב׳), דִּבּוּר הַמַּתְחִיל "מֵאַלְיָה", דְּמֵחוּץ לָאָרֶץ פָּטוּר לַעֲלוֹת לָרֶגֶל וְלַעֲשׂוֹת קָרְבַּן פֶּסַח; וְעַיֵּן מַה שֶּׁכָּתַבְתִּי עַל דִּבְרֵיהֶם בְּשׁוּד״י (סִימָן נ״ז). וְאוּלַי, אַף עַל גַּב דִּפְטוּרִים — מִכָּל מָקוֹם, אִי הֶחְמִירוּ עַל עַצְמָם וְנֶעֶקְרוּ — מְעַבְּרִין בִּשְׁבִילָם; וְצָרִיךְ עִיּוּן):

**אֲבָל** אֵין מְעַבְּרִים מִפְּנֵי גָּלֻיּוֹת יִשְׂרָאֵל שֶׁעֲדַיִן לֹא נֶעֶקְרוּ מִמְּקוֹמָן (וּבַתּוֹסֶפְתָּא, פֶּרֶק ב׳, הַגִּרְסָא לְהֵפֶךְ — עַיֵּן בְּסֵפֶר תּוֹרַת חַיִּים); וְלֹא מִפְּנֵי הַשֶּׁלֶג
