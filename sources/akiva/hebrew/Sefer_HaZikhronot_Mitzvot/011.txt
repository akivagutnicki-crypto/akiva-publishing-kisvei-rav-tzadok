~ קִשּׁוּי, עַל יְדֵי שֶׁתִּתְחַמֵּם הַשִּׁדְרָה, וְעוֹד שֶׁבְּגָדָיו נוֹפְלִים עַל אַמָּתוֹ, וְכֵן יָדָיו מֻנָּחוֹת עַל הָאַמָּה, וּמִתְחַמֵּם וּבָא לִידֵי הוֹצָאַת זֶרַע לְבַטָּלָה. וַאֲפִלּוּ אִם אֵינוֹ יָשֵׁן — אָסוּר לִשְׁכַּב כָּךְ; וּלְפִיכָךְ אֲפִלּוּ לִישׁוֹן כָּךְ אָסוּר, כְּמוֹ שֶׁנִּתְבָּאֵר. וְיִזָּהֵר כְּשֶׁיָּשֵׁן שֶׁלֹּא יִהְיֶה אֵיבָרוֹ בְּגָלוּי, דִּשְׁמָּא יַקְרֶה וְיֵרָאֶה גְנוּתוֹ לָרַבִּים, אִם לֹא שֶׁיָּשֵׁן בְּבַיִת סָגוּר אוֹ אָפֵל (דָּבָר זֶה נִלְמָד מֵרַשִׁ״י נִדָּה י״ב עַמּוּד ב׳, עַיֵּן שָׁם תּוֹסָפוֹת דִּבּוּר הַמַּתְחִיל "לייט"):

**וּמִכָּל מָקוֹם** אַשְׁרֵי הַנִּזְהָר מִכָּל דָּבָר הַמֵּבִיא לִידֵי קֶרִי, וְאֵינוֹ רוֹאֶה קֶרִי אֲפִלּוּ לְאָנְסוֹ — וְ"קָדוֹשׁ" יֵאָמֵר לוֹ; שֶׁגַּם מֵרְאִיַּת קֶרִי לְאָנְסוֹ נִבְרָאִין שֵׁדִין וְרוּחִין וְלִילִין (עֵירוּבִין שָׁם). וֶאֱלִישָׁע נִשְׁתַּבַּח "אִישׁ קָדוֹשׁ" — בְּמָה שֶׁלֹּא רָאֲתָה עַל סְדִינוֹ טִפַּת קֶרִי, בְּפֶרֶק קַמָּא דִּבְרָכוֹת; וְכֵן בִּירוּשַׁלְמִי, אָמַר רַבִּי מָנָא, עַיֵּן שָׁם; וְעַיֵּן בְּוַיִּקְרָא רַבָּא (פָּרָשָׁה כ״ד). וְרַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה נָתְנוּ סְגֻלָּה לָזֶה: לְהַשְׂבִּיעַ עַצְמוֹ בְּדִבְרֵי תּוֹרָה קֹדֶם הַשֵּׁינָה, שֶׁנֶּאֱמַר: "וְשָׂבֵעַ יָלִין בַּל יִפָּקֶד רָע", שֶׁהוֹצָאַת זֶרַע לְבַטָּלָה נִקְרָא "רָע". בַּסְּפָרִים נִזְכָּרִים אֲמִירוֹת כַּמָּה פְּסוּקִים וּמִזְמוֹרִים וּתְפִלּוֹת לְהוֹעִיל לְהִנָּצֵל מִזֶּה; וַאֲנַחְנוּ אֵין לָנוּ אֶלָּא מַה שֶּׁאָמְרוּ רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה. וְגַם בְּלָאו הָכֵי כָּתְבוּ אֵיזֶה מְחַבְּרִים, וְכֵן כָּתוּב בְּסֵפֶר עֲבוֹדַת הַקֹּדֶשׁ, שֶׁיִּמְנַע מִזֶּה שֶׁלֹּא יִתְגָּרוּ בּוֹ בִּפְרָט כְּשֶׁיִּשְׁכַּח לְאָמְרָם. וְהָאֲרִיזַ״ל הָיָה נִזְהָר לִישׁוֹן עִם צִיצִית, וּמִצְוָה אַגּוֹנֵי וּמַצָּלֵי מֵחִטְאָה (סוֹטָה כ״א עַמּוּד א׳). גַּם הָיָה נִזְהָר לִהְיוֹת נֵעוֹר בְּלֵיל טְבִילָה לַעֲסוֹק בַּתּוֹרָה עַד חֲצוֹת (פֵּרוּשׁ: עַד שֶׁיִּרְצֶה לְשַׁמֵּשׁ), וְלֹא יִישַׁן מִקֹּדֶם, פֶּן יָבוֹא חַס וְשָׁלוֹם לִידֵי קֶרִי. "וְאַשְׁרֵי אִישׁ שֶׁלֹּא יִשְׁכָּחֶךָּ, וּבֶן אָדָם יִתְאַמֶּץ בָּךְ", וּמָגֵן הוּא לְכָל הַחוֹסִים בּוֹ:

**וְאָסוּר** לוֹ לְאָדָם לְהַרְהֵר בַּעֲבֵירָה, כְּגוֹן בְּאֵשֶׁת אִישׁ אוֹ שְׁאָר זְנוּת. וְאַף עַל פִּי שֶׁמַּחֲשָׁבָה רָעָה אֵין הַקָּדוֹשׁ בָּרוּךְ הוּא מְצָרְפָהּ לְמַעֲשֶׂה (קִדּוּשִׁין מ׳ עַמּוּד ב׳), מִכָּל מָקוֹם הִרְהוּר זֶה גּוֹרֵם לִידֵי הוֹצָאַת זֶרַע לְבַטָּלָה; וְהוֹצָאַת זֶרַע לְבַטָּלָה שֶׁבָּא עַל יְדֵי הַהִרְהוּר — כָּתְבוּ בַּסְּפָרִים דְּגָרוּעַ מֵהוֹצָאַת זֶרַע לְבַטָּלָה בְּיָדַיִם וְכַהַאי גַּוְנָא. וְעַל זֶה אָמְרוּ: הִרְהוּרֵי עֲבֵירָה קָשִׁין מֵעֲבֵירָה (יוֹמָא כ״ט עַמּוּד א׳; עַיֵּן שָׁם בְּרַשִׁ״י, פֵּרֵשׁ: "קָשִׁין לַגּוּף" וְכוּלֵי). וּבְסֵפֶר כַּד הַקֶּמַח, אוֹת ט׳ — טָהֳרָה, הֶאֱרִיךְ בָּזֶה וְהֵבִיא ד׳ פֵּרוּשִׁים דְּקָאֵי עַל הָעֲבֵירָה וְחֹמֶר עָנְשָׁהּ. (וְלִי נִרְאֶה פָּשׁוּט, דְּזֶה כֵּיוָן דְּמֵבִיא לִידֵי הוֹצָאַת זֶרַע לְבַטָּלָה, וַעֲבֵירָה עַצְמָהּ עַל פִּי רוֹב בִּפְנוּיָה — דִּזְנוּת דְּאֵשֶׁת אִישׁ לָא שְׁכִיחַ, כִּדְאִיתָא פֶּרֶק ב׳ דְּגִטִּין (דַּף י״ז עַמּוּד ב׳), וּפְנוּיָה לְדַעַת פּוֹסְקִים דְּרַבָּנָן — אִם כֵּן הַהִרְהוּר שֶׁמֵּבִיא לִידֵי הוֹצָאַת זֶרַע לְבַטָּלָה, הֶחָמוּר מִכָּל הָעֲבֵירוֹת, גָּרוּעַ הַרְבֵּה וְקָשֶׁה מִגּוּף הָעֲבֵירָה). וַאֲפִלּוּ בָּטוּחַ שֶׁלֹּא יָבוֹא לִידֵי הוֹצָאַת זֶרַע לְבַטָּלָה עַל יְדֵי הַהִרְהוּר, אוֹ שֶׁאִשְׁתּוֹ עִמּוֹ, וְאִם יִתְקַשֶּׁה יוּכַל לְהִנָּצֵל מֵהָאִסּוּר — מִכָּל מָקוֹם אָסוּר לְהַרְהֵר בְּדָבָר רָע בַּיּוֹם, שֶׁלֹּא
