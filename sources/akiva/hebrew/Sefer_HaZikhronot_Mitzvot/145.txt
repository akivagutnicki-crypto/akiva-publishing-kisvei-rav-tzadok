**וְהִנֵּה** הַכָּתוּב מָנָה ג׳ טְמֵאִים: "צָרוּעַ", "וְזָב", "וְטָמֵא לָנָפֶשׁ" — שֶׁהוּא טְמֵא מֵת. וְאֶפְשָׁר הָיָה לוֹמַר שֶׁשִּׁלּוּחַ שְׁלָשְׁתָּן שָׁוֶה; אֲבָל מִמַּה שֶּׁנֶּאֱמַר בְּפָרָשַׁת תַּזְרִיעַ בִּמְצֹרָע "בָּדָד יֵשֵׁב, מִחוּץ לַמַּחֲנֶה מוֹשָׁבוֹ" — לָמַדְנוּ שֶׁאֵין שִׁלּוּחוֹ לַמָּקוֹם שֶׁזָּב וּטְמֵא מֵת מְשֻׁלָּחִים. וְכֵיוָן שֶׁלָּמְדוּ שֶׁהַמְּצֹרָע, שֶׁטֻּמְאָתוֹ חֲמוּרָה, שֶׁמְּטַמֵּא בְּבִיאָה — שִׁלּוּחוֹ חָמוּר מִשִּׁלּוּחַ חֲבֵרָיו — הוּא הַדִּין לְכָל שֶׁטֻּמְאָתוֹ חֲמוּרָה, שֶׁשִּׁלּוּחוֹ חָמוּר; וּלְפִיכָךְ זָב, שֶׁמְּטַמֵּא מִשְׁכָּב וּמוֹשָׁב, וַאֲפִלּוּ מִתַּחַת הָאֶבֶן — מַה שֶּׁאֵין הַמֵּת מְטַמֵּא — גַּם שִׁלּוּחוֹ חָמוּר מִשִּׁלּוּחַ הַמֵּת. וַהֲרֵי לָמַדְנוּ ג׳ שִׁלּוּחִים, מְחֻלָּקִים לְג׳ טְמֵאִים שֶׁמָּנָה הַכָּתוּב. הָא כֵּיצַד? הַמְּצֹרָע (בַּמִּדְבָּר רַבָּה פָּרָשָׁה ו׳: "מִ'כָּל צָרוּעַ"; וְעַיֵּן בַּגְּמָרָא פְּסָחִים ס״ז עַמּוּד ב׳, פָּרֵיךְ אַהָךְ "כָּל", וּמְשַׁנֵּי "אִידֵי", וְצָרִיךְ לְעַיֵּן) — אֶחָד הַמֻּסְגָּר וְאֶחָד הַמֻּחְלָט — מְשַׁלְּחוֹ חוּץ לְג׳ מַחֲנוֹת, שֶׁהוּא חוּץ לִירוּשָׁלַיִם (וּלְעָרֵי מֻקָּפִים חוֹמָה). זָבִים וְזָבוֹת — וְהוּא הַדִּין לְמִשְׁכָּבָן וּמוֹשָׁבָן (תּוֹסֶפְתָּא סוֹף פֶּרֶק קַמָּא דְּכֵלִים) — נִדּוֹת וְיוֹלְדוֹת — מְשַׁלֵּחַ חוּץ לְב׳ מַחֲנוֹת, שֶׁהוּא חוּץ לְהַר הַבַּיִת, מַחֲנֵה לְוִיָּה. וְאֶחָד (בְּבַמִּדְבָּר רַבָּה שָׁם דָּרֵישׁ "מִכָּל"; אֲבָל גְּמָרָא דִּידַן דָּרֵישׁ "מִכָּל" — בַּעַל קֶרִי) זָב בַּעַל ג׳ רְאִיּוֹת, אוֹ בַּעַל ב׳ רְאִיּוֹת, וּבַעַל רְאִיָּה אַחַת — שָׁוִים. וּמִמַּה שֶּׁנֶּאֱמַר "וְכָל זָב" — לָמְדוּ חֲכָמִים (פְּסָחִים ס״ז עַמּוּד ב׳) דְּבַעַל קֶרִי יֵשׁ לוֹ גַּם כֵּן דִּין זָב. טְמֵא מֵת — וַאֲפִלּוּ הַמֵּת עַצְמוֹ — נִכְנָס לְמַחֲנֵה לְוִיָּה, שֶׁנֶּאֱמַר: "וַיִּקַּח מֹשֶׁה אֶת עַצְמוֹת יוֹסֵף עִמּוֹ" — "עִמּוֹ": בִּמְחִצָּתוֹ, שֶׁהוּא מְחִצַּת מַחֲנֵה לְוִיָּה; וּמִשְׁתַּלֵּחַ רַק מִמַּחֲנֶה אַחַת, שֶׁהוּא מַחֲנֵה שְׁכִינָה. וּמִמַּה שֶּׁנֶּאֱמַר "וְכֹל טָמֵא לָנָפֶשׁ" — לָמְדוּ חֲכָמִים (שָׁם): הוּא הַדִּין לִטְמֵא שֶׁרֶץ, וְנוֹגֵעַ בִּנְבֵלָה, וְכָל טְמֵאֵי מַגָּע — שֶׁדִּינָם כִּטְמֵא מֵת. וְכֵן בּוֹעֲלֵי נִדָּה (כְּשֶׁלֹּא רָאָה קֶרִי עבמ״ל, אִי נַמִּי כְּשֶׁטָּבַל לְקִרְיוֹ) — דִּינָם כִּטְמֵא מֵת. וְכֵן הָעוֹבְדֵי כּוֹכָבִים — אַף עַל פִּי שֶׁלְּעִנְיַן שְׁאָר דְּבָרִים גָּזְרוּ חֲכָמִים עֲלֵיהֶם שֶׁיִּהְיוּ כְּזָבִים לְכָל דִּבְרֵיהֶם — לְעִנְיַן שִׁלּוּחַ מַחֲנוֹת לֹא גְּזוּר עֲלַיְיהוּ, אֶלָּא כִּטְמֵא מֵת בִּלְבַד: לְשַׁלְּחָם חוּץ לְמַחֲנֶה אַחַת. וּמִמַּה שֶּׁנֶּאֱמַר בְּפָרָשַׁת חֻקַּת, בְּטֻמְאַת מֵת: "טָמֵא יִהְיֶה, עוֹד טֻמְאָתוֹ בוֹ" — שֶׁמְּיֻתָּרִים — דָּרְשׁוּ חֲכָמִים (נָזִיר מ״ה): "טָמֵא יִהְיֶה" — לְרַבּוֹת טְבוּל יוֹם; "עוֹד טֻמְאָתוֹ בוֹ" — לְרַבּוֹת מְחֻסַּר כִּפּוּרִים, אַף עַל פִּי שֶׁטָּבַל וְהֶעֱרִיב שִׁמְשׁוֹ — שֶׁגַּם כֵּן מִשְׁתַּלְּחִים מִמַּחֲנֶה אַחַת, כְּטֻמְאַת מֵת. וְעַיֵּן בָּרַמְבַּ״ם, כָּתַב לִמּוּד אַחֵר; וְעַיֵּן בְּתוֹסָפוֹת זְבָחִים (י״ז עַמּוּד א׳), שֶׁהֶאֱרִיכוּ בָּזֶה:

**חֲכָמִים** עֲלֵיהֶם הַשָּׁלוֹם הוֹסִיפוּ מַעֲלָה יְתֵרָה בְּבֵית עוֹלָמִים; וְהוּא, שֶׁחִלְּקוּ בְּמַחֲנֵה לְוִיָּה, וְאָמְרוּ שֶׁלֹּא יִהְיוּ טְמֵאֵי מֵתִים רַשָּׁאִים לְהִכָּנֵס בָּהּ אֶלָּא מִפֶּתַח הַר הַבַּיִת עַד הַחֵיל בִּלְבַד; אֲבָל מִן הַחֵיל וְאֵילָךְ — אָסְרוּ לִטְמֵא מֵת, וְעוֹבְדֵי כּוֹכָבִים, וּבוֹעֵל נִדָּה (וְנִרְאֶה, דְּהוּא הַדִּין טְמֵא שֶׁרֶץ וּשְׁאָרֵי) לִיכָּנֵס לְשָׁם; אֲבָל טְבוּל יוֹם וּמְחֻסַּר כִּפּוּרִים — מֻתָּרִים. וּמֵעֶזְרַת נָשִׁים וְאֵילָךְ — אָסְרוּ בִּימֵי יְהוֹשָׁפָט
