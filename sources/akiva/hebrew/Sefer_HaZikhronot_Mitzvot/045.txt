~ וְאַף עַל פִּי שֶׁמִּן הַתּוֹרָה אֵין צְרִיכִים לְעֵדִים הַמַּכִּירִין; מִכָּל מָקוֹם, כֵּיוָן דְּמִדְּרַבָּנָן לֹא מְקַבְּלִין מִינַּיְהוּ — אִם כֵּן אִי אֶפְשָׁר לְהָעִיד עַל הַחֹדֶשׁ אֶלָּא אִם כֵּן יְחַלְּלוּ (חִדּוּשֵׁי הָרַשְׁבָּ״א; וְעַיֵּן בְּטוּרֵי אֶבֶן, וְעַיֵּן מַה שֶּׁכָּתַבְתִּי בְּמָקוֹם אַחֵר). וַאֲפִלּוּ הָיָה רַק עֵד אֶחָד — עֵד רְאִיָּה אוֹ עֵד מַכִּיר — הֲרֵי זֶה הוֹלֵךְ וּמְחַלֵּל מִסָּפֵק, שֶׁמָּא יִמְצָא אַחֵר וְיִצְטָרֵף עִמּוֹ:

(**עַיֵּן** בְּלֶחֶם מִשְׁנֶה: עָשָׂה מַחְלֹקֶת בֵּין רַשִׁ״י לְתוֹסָפוֹת וְהָרַמְבַּ״ם — וְדַעַת רַשִׁ״י: דַּוְקָא הֵיכִי דְּיָדְעִינַן שֶׁיֵּשׁ שָׁם עֵד אַחֵר בְּאוֹתוֹ מָקוֹם שֶׁהוֹלֵךְ שָׁם לְהִצְטָרֵף עִמּוֹ, אַף שֶׁאֵין יוֹדְעִין שֶׁהוּא בְּבֵיתוֹ — מְחַלְּלִין; אֲבָל כְּשֶׁאֵין יוֹדְעִין שֶׁיֵּשׁ עֵד — לֹא, עַיֵּן שָׁם בַּאֲרִיכוּת. וְלֹא נִהְרָא לַעֲנִיּוּת דַּעְתִּי, דִּרְאִיַּת הַתּוֹסָפוֹת מֵאָב וּבְנוֹ — חֲזָקָה דְּמַשְׁמַע לְהֶדְיָא דִּבְשַׁבָּת קָאֵי, כְּמִשְׁנָה הַקּוֹדֶמֶת וְשֶׁאַחַר כֵּן. וְעוֹד, אִי בְּחוֹל — פְּשִׁיטָא, דִּלְמַאי מִלְּתָא? דִּלְחִיּוּבָא קָתָנֵי — בְּחוֹל, וּדְלָא כְּהַלֶּחֶם מִשְׁנֶה. (וּרְאָיַת הַלֶּחֶם מִשְׁנֶה מִמ׳ זוּג — יֵשׁ לִדְחוֹת, דְּשָׁם לֹא יָדְעוּ חַד מֵחַבְרֵיהּ; וּבְזוֹ וַדַּאי דִּמְחַלְּלִין, דְּאִם כֵּן לְעוֹלָם לֹא יְחַלֵּל, דְּתָמִיד יָחוּשׁ פֶּן יֵשׁ עֵדִים אֲחֵרִים. לְפִיכָךְ, אַף עַל פִּי שֶׁנִּרְאָה בַּעֲלִיל, אָמְרוּ שֶׁיְּחַלֵּל וְלֹא יָחוּשׁ שֶׁיֵּשׁ עֵדִים אֲחֵרִים; דְּאִם כֵּן תָּמִיד יָחוּשׁ, וּלְכָךְ אוֹתוֹ שֶׁעִכְּבָן — לָאו שַׁפִּיר עָבֵיד, דְּאַף עַל גַּב שֶׁיָּדַע שֶׁהָלְכוּ אֲחֵרִים, מִכָּל מָקוֹם הֵם לֹא יָדְעוּ, וּבְפַעַם אַחֶרֶת יִמָּנְעוּ.) וְעוֹד, דְּהַתּוֹסָפוֹת עַצְמָם, דִּבּוּר הַמַּתְחִיל "סַהֲדָא", הִסְכִּימוּ לְפֵרוּשׁ רַשִׁ״י — דְּכֵן הוּא בִּירוּשַׁלְמִי וְתוֹסֶפְתָּא — וּבְדִבּוּר הַמַּתְחִיל "וְחַד" כָּתְבוּ "מְחַלֵּל" וְכוּלֵי. אֲבָל קֻשְׁיָתוֹ בֶּאֱמֶת לֹא קָשֶׁה מִידֵּי, כְּמוֹ שֶׁכָּתוּב תּוֹסָפוֹת דִּבּוּר הַמַּתְחִיל "וְחַד" וְכוּלֵי: מַקְשֶׁה עַל לִישָׁנָא וְכוּלֵי, דְּמַשְׁמַע דְּחַד מְהֵימַן, עַיֵּן שָׁם. וְאִם כֵּן, כִּי פָּרִיךְ "אֲרַב נְהוֹרָאִי" — נַמִּי פָּרִיךְ מִלִּישָׁנָא דְּמַשְׁמַע דַּהֲוֵי מְהֵימַן, וְעַל זֶה מְשַׁנֵּי "אַחֲרִינָא" וְכוּלֵי: לְכָךְ מְהֵימַן; וְר״א אָמַר דִּידַע דְּשָׁם יֵשׁ סַהֲדָא, וּלְכָךְ מְהֵימַן; אֲבָל בְּלָאו הָכֵי קָשֶׁה קֻשְׁיַת הַמַּקְשֶׁה "אַמַּאי מְהֵימַן", כַּנִּזְכָּר לְעֵיל, וּפָשׁוּט. וְעַיֵּן עוֹד שָׁם בְּלֶחֶם מִשְׁנֶה, וּבְחִדּוּשֵׁי הָרַשְׁבָּ״א וּבְמַהַרְשָׁ״א. וְנִרְאֶה לִכְאוֹרָה דְּדַעַת רַשִׁ״י כְּהַלֶּחֶם מִשְׁנֶה, מִמַּה שֶּׁכָּתַב (כ״א עַמּוּד ב׳) דִּבּוּר הַמַּתְחִיל "אַרְבָּעִים" וְכוּלֵי: "זוּג אַחֵר זוּג"; וְכַוָּנָתוֹ דְּלֹא תִּקְשֶׁה: אַעֵדִים אִי הָלְכוּ בְּבַת אַחַת, אֵיךְ חִלְּלוּ שַׁבָּת אַרְבָּעִים זוּג? דַּי בְּחַד. וְקָשֶׁה: תִּיפּוֹק לֵיהּ, אֲפִלּוּ בְּבַת אַחַת, וּכְמוֹ בְּאָב וּבְנוֹ, דְּפֵרֵשׁ רַשִׁ״י שֶׁמָּא יִכָּשֵׁל בְּפִיו וְיִפָּסֵל וְכוּלֵי; אֶלָּא וַדַּאי דְּזֶהוּ רַק בְּחוֹל, אֲבָל שַׁבָּת אֵין מְחַלְּלִין מִסָּפֵק. מִיהוּ יֵשׁ לְיַשְּׁבָם קְצָת; גַּם יֵשׁ לְחַלֵּק בֵּין אַרְבָּעִים זוּג לְמִיעוּטֵי א׳ אוֹ ב׳. וְאֵין לַעֲשׂוֹת מַחְלֹקֶת בֵּין רַשִׁ״י לִשְׁאָר פּוֹסְקִים בְּחִנָּם):

**וְכֵן** אָב וּבְנוֹ יֵלְכוּ לְהָעִיד וִיחַלְּלוּ שַׁבָּת; וְאַף עַל פִּי שֶׁאֵין מִצְטָרְפִים זֶה עִם זֶה, שֶׁהַקְּרוֹבִים פְּסוּלִים — מִכָּל מָקוֹם, שֶׁמָּא יִפָּסֵל אֶחָד מֵהֶם, שֶׁיִּכָּשֵׁל בְּפִיו,
