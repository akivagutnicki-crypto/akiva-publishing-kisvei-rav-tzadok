~ ח׳ — אִם יוֹם הָאַחֲרוֹן שַׁבָּת אוֹ יוֹם טוֹב, אָסוּר לְמוּלוֹ, כֵּיוָן דְּהוּא סְפֵק יוֹם ט׳; אֶלָּא יִדָּחֶה לְיוֹם א׳. וַאֲפִלּוּ לֹא הוֹצִיא אֶלָּא רֹאשׁוֹ בֵּין הַשְּׁמָשׁוֹת חוּץ לַאֲוִיר הָעוֹלָם (וְהוּא הַדִּין שֶׁהָיָה בּוֹכֶה, דְּמִסְּתָמָא יָצָא רֹאשׁוֹ). אֲבָל אִם הָאֵם אוֹ הַמְיַלֶּדֶת אוֹמֶרֶת בָּרִי שֶׁהַתִּינוֹק הָיָה מֻנָּח בְּבִטְנָהּ כְּבִשְׁאָר הַיָּמִים, וְלֹא קָשְׁתָה לֵילֵד כְּלָל — אַף עַל פִּי שֶׁשְּׁמָעוּהוּ בּוֹכֶה, לֵית לַן בָּהּ; וְכֻלּוֹ יָצָא בְּשַׁבָּת וְיוֹם טוֹב. וּצְרִיכָה הַמְיַלֶּדֶת לְדַקְדֵּק, אִם הָאִשָּׁה מַקְשָׁה בְּבֵין הַשְּׁמָשׁוֹת, לִרְאוֹת פֶּן יָצָא רֹאשׁ הַוָּלָד; וְאִם לֹא רָאֲתָה — אוֹקְמֵי אַחֲזָקָה שֶׁלֹּא נוֹלַד (וּדְלָא כְּיַעְבֵּ״ץ, עַיֵּן שָׁם), עַד שֶׁרְאִינוּהוּ נוֹלָד. וְאֵין חִלּוּק בֵּין יוֹם טוֹב רִאשׁוֹן לְיוֹם טוֹב שֵׁנִי שֶׁל גָּלֻיּוֹת. וְיֵשׁ אוֹמְרִים דְּדוֹחֶה סְפֵק מִילָה לְיוֹם טוֹב שֵׁנִי שֶׁל גָּלֻיּוֹת, וְכֵן עִקָּר:

**וּכְשֶׁנִּרְאָה** ג׳ כּוֹכָבִים בֵּינוֹנִים — הוּא וַדַּאי לַיְלָה; וְאִם נִרְאוּ תֵּכֶף לְהוֹצָאַת רֹאשׁוֹ ג׳ כּוֹכָבִים בֵּינוֹנִים — הֲרֵי נוֹלַד בַּלַּיְלָה. וְאִם שָׁהוּ קְצָת, וְאַחַר כָּךְ רָאוּ ג׳ כּוֹכָבִים בֵּינוֹנִים — אִם לְפִי הַשִּׁהוּי נִרְאֶה לָהֶם שֶׁהָיָה יוֹם כְּשֶׁנּוֹלַד, אֵין לָהֶם אֶלָּא רְאוּת עֵינֵיהֶם, אֲפִלּוּ לִדְחוֹת שַׁבָּת וְיוֹם טוֹב. וְשִׁעוּר בֵּין הַשְּׁמָשׁוֹת — הַתְחָלָתוֹ הוּא כְּשִׁעוּר הִלּוּךְ ג׳ מִילִין וּרְבִיעַ לְאַחַר הַתְחָלַת שְׁקִיעַת הַחַמָּה, שֶׁהוּא לְעֵרֶךְ רְבִיעִית שָׁעָה קֹדֶם צֵאת הַכּוֹכָבִים. וְאִם נוֹלַד קֹדֶם — אֲפִלּוּ כְּבָר שָׁקְעָה הַחַמָּה — הֲרֵי הוּא כְּנוֹלָד בַּיּוֹם. וְכָל סְפֵק אִסּוּר דְּאוֹרַיְתָא אוֹ דְּרַבָּנָן בְּשַׁבָּת — כְּגוֹן אַנְדְּרוֹגִינוֹס וְטֻמְטוּם (וְאִם נִקְרַע קֹדֶם ח׳ וְנִמְצָא זָכָר — מַחְלֹקֶת), וְיוֹצֵא דֹּפֶן, וְנָכְרִית שֶׁיָּלְדָה וְאַחַר כָּךְ נִתְגַּיְּרָה, וּמִי שֶׁיֵּשׁ לוֹ ב׳ עָרְלוֹת — דְּהַיְנוּ ב׳ עוֹרוֹת זֶה עַל גַּב זֶה, אוֹ ב׳ גִּידִים — וְכֵן נוֹלַד מָהוּל, שֶׁצָּרִיךְ לְהַטִּיף דַּם בְּרִית, אוֹ נִמְשְׁכָה עָרְלָתוֹ אַחַר כָּךְ — כֻּלָּם, אֲפִלּוּ בְּיוֹם ח׳, אֵין מָלִין בְּשַׁבָּת. אֲבָל מָלוּהוּ תּוֹךְ ח׳, אוֹ מָלוּ עוֹבֵד כּוֹכָבִים אוֹ מוּמָר בַּח׳ — חוֹזְרִין וּמַטִּיפִין דַּם בְּרִית בַּח׳ בְּשַׁבָּת. וּמָלִין מַמְזֵר, אוֹ יִשְׂרְאֵלִית — אֵשֶׁת אִישׁ — אֲפִלּוּ שֶׁיָּלְדָה מֵעֶבֶד וְנָכְרִי, בַּח׳ בְּשַׁבָּת. וְכֵן יִשְׂרָאֵל שֶׁהֵמִיר וְהוֹלִיד בֵּן מִיִּשְׂרְאֵלִית — יֵשׁ אוֹמְרִים אֲפִלּוּ הִיא גַּם כֵּן מוּמֶרֶת. אֲבָל יִשְׂרָאֵל שֶׁנּוֹלַד לוֹ בֵּן מִנָּכְרִית — אֵין מִילָתוֹ דּוֹחָה שַׁבָּת. וְכֵן הַקָּרָאִים שֶׁבִּזְמַן הַזֶּה, וְאֵין צָרִיךְ לוֹמַר הַיִּשְׁמְעֵאלִים — אֲפִלּוּ יְדוּעִים מִבְּנֵי קְטוּרָה, וַאֲפִלּוּ לְדִבְרֵי הָאוֹמֵר דִּמְצֻוִּים עַל הַמִּילָה. מִיהוּ בַּקָּרָאִים — כְּשֶׁיָּדוּעַ שֶׁלֹּא יְדַבְּרוּ תּוֹעָה עַל הָרַבָּנִים, וְלֹא יִתְלוֹצְצוּ בְּדִבְרֵי חַכְמֵי הַמִּשְׁנָה וְהַשַּׁ״ס — מֻתָּר לָמוּל בְּנֵיהֶם בְּשַׁבָּת, דִּלְמָא נָפֵיק מִינַּיְהוּ זַרְעָא דִּמְעַלְיָא:

**בִּזְמַן** הַתַּלְמוּד — אִם לֹא הָיוּ רוֹחֲצִים הַיֶּלֶד קֹדֶם הַמִּילָה וְאַחֲרֶיהָ, וּבַיּוֹם ג׳ לַמִּילָה, בְּמַיִם חַמִּין — הָיָה מְסֻכָּן; לְפִיכָךְ נִזְקְקוּ לִכְתּוֹב דִּינֵיהֶם בְּשַׁבָּת. וּבִמְדִינוֹתֵינוּ נָהֲגוּ לְרָחֲצוֹ לִפְנֵי הַמִּילָה בְּחַמִּין שֶׁהוּחַמּוּ מֵאֶתְמוֹל, וּלְאַחַר
