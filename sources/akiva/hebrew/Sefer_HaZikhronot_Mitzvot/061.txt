**וְזֶהוּ** לְמַאן דְּאָמַר סַנְהֶדְרִין (י״ג עַמּוּד א׳): "יוֹם תְּקוּפָה מַתְחִיל"; דִּלְמַאן דְּאָמַר "גּוֹמֵר" — אֲפִלּוּ חָלָה תְּקוּפָה בְּט״ו בְּנִיסָן, אֵין יוֹם רִאשׁוֹן דְּפֶסַח בְּחֹדֶשׁ הָאָבִיב. אֲבָל קָשֶׁה: יְעַבְּרוּ לַאֲדָר, וְיַגִּיעַ תְּקוּפָה בְּט״ו; וּמַה צֹּרֶךְ לְחֹדֶשׁ מְעֻבָּר? וְצָרִיךְ לוֹמַר: דִּסְבִירָא לֵיהּ יוֹם תְּקוּפָה גּוֹמֵר, וּלְכָךְ בְּט״ו אֶפְשָׁר לְעַבֵּר לַאֲדָר; אֲבָל בְּט״ז — אִי אֶפְשָׁר. אֲבָל לְמַאן דְּאָמַר יוֹם תְּקוּפָה מַתְחִיל — בְּט״ז יְעַבְּרוּ לַאֲדָר, וּבְי״ז יְעַבְּרוּ לַשָּׁנָה:

**וְיֵשׁ** אוֹמְרִים (תּוֹסָפוֹת רֹאשׁ הַשָּׁנָה כ״א עַמּוּד א׳, דִּבּוּר הַמַּתְחִיל "כִּי"), שֶׁמַּה שֶּׁאָמַר הַכָּתוּב "שָׁמוֹר אֶת חֹדֶשׁ הָאָבִיב" — רוֹצֶה לוֹמַר: שֶׁנִּיסָן שֶׁל חַמָּה, שֶׁקָּרוּי "אָבִיב", יִהְיֶה בְּחֹדֶשׁ — רוֹצֶה לוֹמַר: כְּשֶׁהַלְּבָנָה בְּחִדּוּשָׁהּ; וְאֵין הַלְּבָנָה נִקְרֵאת "חֲדָשָׁה" אֶלָּא עַד ט״ו בְּנִיסָן, וּמִט״ו וְאֵילָךְ נִקְרֵאת "יְשָׁנָה". וְזֶהוּ גַּם כֵּן, עַל כָּרְחֲךָ, לְמַאן דְּאָמַר יוֹם תְּקוּפָה גּוֹמֵר; וַאֲפִלּוּ יְעַבְּרוּ לַאֲדָר — לְתַחוּל תְּקוּפָה בְּט״ו; יוֹם תְּקוּפָה גּוֹמֵר, וּמַתְחֶלֶת מִט״ו. וְאִם לֹא יְעַבְּרוּ לַאֲדָר — אֲפִלּוּ חָלָה בְּט״ו, שֶׁלֹּא כַּדִּין. וּלְמַאן דְּאָמַר יוֹם תְּקוּפָה מַתְחִיל — אִם חָלָה בְּט״ו, שַׁפִּיר דָּמֵי; וְאִם חָלָה בְּט״ז — יְעַבְּרוּ לַאֲדָר; וְאִם חָלָה בְּי״ז — יְעַבְּרוּ לַשָּׁנָה; וְעַיֵּן שָׁם בַּתּוֹסָפוֹת. וּבְסַנְהֶדְרִין (י״ג עַמּוּד ב׳), תּוֹסָפוֹת דִּבּוּר הַמַּתְחִיל "שָׁמוֹר" — וְצָרִיךְ עִיּוּן הֲלָכָה כְּמַאן. וּמִדִּבְרֵי הָרַמְבַּ״ם, שֶׁכָּתַב "אִם תִּהְיֶה בְּט״ז בְּנִיסָן", וְטַעְמוֹ כְּהַלְּבוּשׁ, כַּנִּזְכָּר לְעֵיל — מַשְׁמַע כְּמַאן דְּאָמַר דְּיוֹם תְּקוּפָה גּוֹמֵר, כַּנִּזְכָּר לְעֵיל. מִיהוּ לְמַה שֶּׁכָּתְבוּ הַגָּהוֹת, דְּהָרַמְבַּ״ם סָבִירָא לֵיהּ כְּטַעַם ב׳ הַנִּזְכָּר לְעֵיל — אִם כֵּן יֵשׁ לוֹמַר: דְּסָבִירָא לֵיהּ כְּדִבְרֵי רַשִׁ״י בְּרֹאשׁ הַשָּׁנָה שָׁם, דְּרַק י״ד יָמִים נִקְרֵאת "חֲדָשָׁה", וּבְט״ז — יְעַבְּרוּ שָׁנָה, וְיוֹם תְּקוּפָה מַתְחִיל; וְצָרִיךְ עִיּוּן:

**וְאֵין** מְעַבְּרִין עַל הָאָבִיב לְבַדּוֹ, אוֹ עַל פֵּרוֹת הָאִילָן לְבַדָּן — אֶלָּא בְּצֵרוּף שְׁנֵיהֶם מְעַבְּרִין. וְאַף עַל פִּי שֶׁהַתְּקוּפָה קֹדֶם לְט״ז בְּנִיסָן — וּמְעַבְּרִין עַל הַתְּקוּפָה לְבַדָּהּ; וְאַף עַל פִּי שֶׁאֵין חֲשָׁשׁ דְּאָבִיב וּפֵרוֹת הָאִילָן — לְפִי שֶׁהַתְּקוּפָה אִי אֶפְשָׁר לִטְעוֹת בָּהּ; אֲבָל אָבִיב וּפֵרוֹת הָאִילָן — אֶפְשָׁר שֶׁיִּטְעוּ בָּהֶן; וּלְפִיכָךְ אֵין דַּי בְּאַחַת אֶלָּא בִּשְׁתֵּיהֶן (תּוֹסָפוֹת סַנְהֶדְרִין י״א עַמּוּד ב׳, דִּבּוּר הַמַּתְחִיל "אוֹ"). וְעַיֵּן שָׁם בַּגְּמָרָא, אַסֵּיק זֶה בְּתֵיקוּ. אֲבָל הָרַמְבַּ״ם סָמַךְ אִדְּשָׁלַח לֵיהּ רַב הוּנָא בַּר אָבִין לְרָבָא (רֹאשׁ הַשָּׁנָה כ״א עַמּוּד א׳), דְּלֹא זָכַר אֶלָּא תְּקוּפָה בִּלְבַד (וְעוֹד רְאָיָה, מִמַּה שֶּׁכָּתוּב דְּאֵין מְעַבְּרִים קֹדֶם רֹאשׁ הַשָּׁנָה וְאַחַר רֹאשׁ הַשָּׁנָה, עַיֵּן שָׁם; וְאִי לֹא מִשּׁוּם תְּקוּפָה לְחוּד — פְּשִׁיטָא שֶׁקֹּדֶם רֹאשׁ הַשָּׁנָה אָבִיב וּפֵרוֹת הָאִילָן מְנָא יָדַע? עַיֵּן בְּסֵפֶר תּוֹרַת חַיִּים דַּף י״ב — כָּתַב זֶה). וְצָרִיךְ לוֹמַר: הָא דְּנִשְׁאַר הַגְּמָרָא בְּתֵיקוּ — אוֹ כְּמוֹ שֶׁכָּתְבוּ הַתּוֹסָפוֹת בְּכַמָּה דּוּכְתֵּי, דְּדֶרֶךְ הַגְּמָרָא דְּפָשִׁיט לֵיהּ בְּמָקוֹם אַחֵר, מַאי דְּהִנִּיחַ בְּדוּכְתָּא אַחֲרִינָא בְּתֵיקוּ; אוֹ דְּשָׁם אַיְירֵי לְעִנְיַן תְּקוּפַת תִּשְׁרֵי, עַיֵּן שָׁם בְּרַשִׁ״י וּבַתּוֹסָפוֹת; וּבְרֹאשׁ הַשָּׁנָה
