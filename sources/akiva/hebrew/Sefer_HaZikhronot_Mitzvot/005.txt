~ הַנְּקֵבָה, אִם גַּם אֲבוֹתָם יָצְאוּ יְדֵי פְּרִיָּה וּרְבִיָּה בְּאֵלֶּה הַנְּכָדִים שֶׁהִנִּיחוּ, אִם נֹאמַר זִכָּרוֹן אֶחָד עוֹלֶה לְכָאן וּלְכָאן. עוֹד אֲנִי מְסֻפָּק: אִם מֵתוּ גַּם הַנְּכָדִים וְנִשְׁאֲרוּ בְּנֵיהֶם, אִם גַּם בְּנֵי בְּנֵי בָנִים כְּבָנִים, וְצָרִיךְ עִיּוּן. (וְעַיֵּן בְּעֶרְכֵי הַכִּנּוּיִים בְּשֵׁם הַמַּבִּי״ט):

**אַף עַל פִּי** שֶׁקִּיֵּם פְּרִיָּה וּרְבִיָּה, אָסוּר לוֹ מִדְּרַבָּנָן לַעֲמוֹד בְּלֹא אִשָּׁה, שֶׁלֹּא יִתְגַּבֵּר יִצְרוֹ עָלָיו; וְאַסְמְכוּהָ רַבָּנָן אַקְּרָא, שֶׁנֶּאֱמַר: "לֹא טוֹב הֱיוֹת הָאָדָם לְבַדּוֹ" וְגוֹמֵר. וְהָרַמְבַּ״ן מְסֻפָּק, דְּאֶפְשָׁר אִסּוּר תּוֹרָה הוּא לֵישֵׁב בְּלֹא אִשָּׁה, מִקְּרָא דְּ"לֹא טוֹב". אֶלָּא הַחֲכָמִים אָמְרוּ שֶׁיִּשָּׂא אִשָּׁה בַּת בָּנִים, שֶׁנֶּאֱמַר: "וְלָעֶרֶב אַל תַּנַּח יָדֶךָ"; וְאַף עַל פִּי שֶׁהוֹלִיד בָּנִים בְּיַלְדוּתוֹ, יַחֲזוֹר וְיוֹלִיד בָּנִים גַּם כֵּן בְּזִקְנוּתוֹ. וּמִיהוּ אֵין כּוֹפִים עַל זֶה, דְּלָאו תַּקָּנָה הִיא אֶלָּא כְּעֵין יִשּׁוּב דֶּרֶךְ אֶרֶץ, וְאֵין מַחְמִירִים עָלָיו כָּל כָּךְ; אֲבָל לִשָּׂא אִשָּׁה כּוֹפִים (בְּמָקוֹם שֶׁאֵין גְּזֵרָה, כַּנִּזְכָּר לְעֵיל). וְכֵן אִם יֵשׁ לוֹ בָּנִים הַרְבֵּה וּמִתְיָרֵא שֶׁמָּא יָבוֹאוּ קְטָטוֹת וּמְרִיבוֹת בֵּין הַבָּנִים וּבֵין אִשְׁתּוֹ כְּשֶׁתִּהְיֶה בַּת בָּנִים — מֻתָּר לִשָּׂא אִשָּׁה שֶׁאֵינָהּ בַּת בָּנִים; אֲבָל אָסוּר לֵישֵׁב בְּלֹא אִשָּׁה מִשּׁוּם חֲשַׁשׁ קְטָטָה. וְנִרְאֶה דְּאָז יִשָּׂא אִשָּׁה בַּת בָּנִים, כֵּיוָן דְּבֵין כָּךְ וּבֵין כָּךְ יִהְיוּ קְטָטוֹת, וְלֵישֵׁב בְּלֹא אִשָּׁה כְּלָל אָסוּר; לְפִיכָךְ יִשָּׂא בַּת בָּנִים, אִם לֹא שֶׁכְּשֶׁיִּשָּׂא בַּת בָּנִים יִהְיוּ יוֹתֵר קְטָטוֹת — וְצָרִיךְ עִיּוּן, מַאן מְפַיֵּס. וְאִם מַכִּיר שֶׁאֵינוֹ בֶּן בָּנִים עוֹד וְאֵינוֹ רָאוּי לְהוֹלִיד עוֹד — רַשַּׁאי לִשָּׂא אִשָּׁה שֶׁאֵינָהּ בַּת בָּנִים. וְנִרְאֶה לִי דְּאִם יַמְתִּין אֵיזֶה שָׁנִים, יַשִּׂיא בָּנָיו וְלֹא יִהְיוּ קְטָטוֹת בֵּינֵיהֶם וּבֵין אִשְׁתּוֹ — יַמְתִּין, וְיִשָּׂא אַחַר כָּךְ אִשָּׁה בַּת בָּנִים; וְהוּא שֶׁאֵין יִצְרוֹ מִתְגַּבֵּר עָלָיו:

**הָאָב** חַיָּב לִבְנוֹ לְהַשִּׂיאוֹ אִשָּׁה; וְהוּא לִשָּׂא וּבְנוֹ לִשָּׂא — בְּנוֹ קוֹדֵם (פֶּרֶק קַמָּא דְּקִדּוּשִׁין. מִיהוּ אֵין כּוֹפִין אוֹתוֹ — תּוֹלְדוֹת אָדָם לְהָרַשְׁבָּ״א סִימָן שכ״א). פֵּרוּשׁ: שֶׁלְּפִי שֶׁדֶּרֶךְ הָאִישׁ לַחֲזוֹר אַחַר הָאִשָּׁה, כְּמוֹ שֶׁאִיתָא בְּרֵישׁ קִדּוּשִׁין, וְצָרִיךְ הָאָב לַחֲזוֹר בִּשְׁבִילוֹ. אֲבָל אֵין צָרִיךְ לִתֵּן לוֹ מָעוֹת וּלְהַלְבִּישׁוֹ; וְרַק בְּבַת לָמְדוּ חֲכָמִים (פֶּרֶק נַעֲרָה שֶׁנִּתְפַּתְּתָה), מִדִּכְתִיב "וְאֶת בְּנוֹתֵיכֶם תְּנוּ לַאֲנָשִׁים", שֶׁיַּלְבִּישֶׁהָ וְיַנְעִילֶהָ כְּדֵי שֶׁיִּקְפְּצוּ עָלֶיהָ בְּנֵי אָדָם — לֹא בְּבֵן. וּמִכָּל מָקוֹם אִם אִי אֶפְשָׁר בְּעִנְיָן אַחֵר לְהַשִּׂיאוֹ, נִרְאֶה לִי דִּמְחֻיָּב לְהַלְבִּישׁוֹ, וְגַם לְהַפְרִיז לוֹ נְדָן בְּעִנְיָן שֶׁיּוּכַל לְהַשִּׂיאוֹ אִשָּׁה כָּל דְּהוּא. וְאִם לֹא הִשִּׂיאוֹ אָבִיו — הוּא חַיָּב לְהַשִּׂיא עַצְמוֹ, כְּמוֹ שֶׁנִּתְבָּאֵר. וְאִם אֵין לוֹ כְּדֵי צָרְכּוֹ לְהַשִּׂיאוֹ — צְרִיכִים הַצִּבּוּר לְהַשִּׂיאוֹ. וּמוֹכְרִים סֵפֶר תּוֹרָה כְּדֵי לִשָּׂא אִשָּׁה לְקַיֵּם מִצְוַת פְּרִיָּה וּרְבִיָּה; וַאֲפִלּוּ קִיֵּם מִצְוַת פְּרִיָּה וּרְבִיָּה מוֹכְרִים סֵפֶר תּוֹרָה כְּדֵי
