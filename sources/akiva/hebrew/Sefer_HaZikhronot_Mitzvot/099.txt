~ עוֹד, בְּכָל ק׳ שָׁנָה כְּמוֹ מַעֲלָה וָחֵצִי, וּכְמוֹ כֵן מֵהַמִּזְרָח לְמַעֲרָב. וּלְשִׁבְעָה מֵהַכּוֹכָבִים — וְהֵם שְׁנֵי הַמְּאוֹרוֹת, חַמָּה וּלְבָנָה, וְה׳ כּוֹכָבִים קְטַנִּים, אֲשֶׁר נְקָבוּם הַתּוֹכְנִים בְּשֵׁמוֹת: שַׁבְּתַאי, צֶדֶק, מַאְדִּים, כּוֹכָב, נֹגַהּ, וְנִקְרָאִים "כּוֹכְבֵי לֶכֶת" — מָצְאוּ לָמוֹ בְּכָל לַיְלָה שִׁנּוּי מַצָּב בֵּין יֶתֶר הַכּוֹכָבִים; וּבִבְחִינוֹת שׁוֹנוֹת הִרְגִּישׁוּ כִּי יֵשׁ לָהֶם תְּנוּעָה בִּלְעֲדֵי תְּנוּעָה הַיּוֹמִית. מַעֲשֵׂי הָרְקִיעִים — תִּשְׁעָה, וְהֵם ט׳ גַּלְגַּלִּים הַסּוֹבְבִים סְבִיב הָאָרֶץ, וְהָאָרֶץ בְּמֶרְכָּזָם:

**א׳.** גַּלְגַּל הַיּוֹמִי, וְהוּא הַגַּלְגַּל הָעֶלְיוֹן מִכֻּלָּם, הַחוֹזֵר בְּכָל יוֹם מִן הַמִּזְרָח לְמַעֲרָב; וְהוּא הַמַּקִּיף וּמְסַבֵּב אֶת הַכֹּל, וּבְכֹחוֹ הַגָּדוֹל, בִּהְיוֹתוֹ מַקִּיף לְכָל הַגַּלְגַּלִּים כֻּלָּם — מַכְרִיחַ לְכָל הַגַּלְגַּלִּים שֶׁבְּתוֹכוֹ שֶׁיְּסַבְּבוּ גַּם כֵּן עִמּוֹ אֶת הָאָרֶץ; וְסִבּוּבוֹ בְּכָל כ״ד שָׁעָה (אֲבָל מִכָּל מָקוֹם כָּל אֶחָד מִתְגַּלְגֵּל תְּנוּעַת עַצְמוֹ גַּם כֵּן). וְהוּא הַנִּקְרָא "אוֹפַן הַמִּישׁוֹר", מִפְּנֵי שֶׁמַּהֲלָכוֹ בְּיֹשֶׁר דֶּרֶךְ קַו הַשָּׁוֶה, מִבְּלִי שֶׁיִּנְטֶה לְשׁוּם צַד — לֹא כְּיֶתֶר הַגַּלְגַּלִּים, שֶׁיִּתְבָּאֵר. וְאֵין בּוֹ שׁוּם כּוֹכָב וּמַזָּל; אֲבָל נַחְשׁוֹב בּוֹ גַּם כֵּן י״ב מַזָּלוֹת (וְכֵן נֶחְלְקוּ לְש״ס מַעֲלוֹת) מִפְּנֵי הַכּוֹכָבִים הַקְּבוּעִים בַּגַּלְגַּל שֶׁל אַחֲרָיו; וְכֵן כָּל גַּלְגַּל נֶחְלָק כֵּן, כְּדֵי לֵידַע כָּל מָקוֹם מִן הַגַּלְגַּל:

**ב׳.** גַּלְגַּל הַמַּזָּלוֹת, שֶׁבּוֹ קְבוּעִים הַכּוֹכָבִים וּמַזָּלוֹת כֻּלָּם. וְאַף עַל פִּי שֶׁיֵּשׁ עוֹד גַּלְגַּלִּים אַחֲרָיו — נִרְאִים הַכּוֹכָבִים לָנוּ בְּהִירִים, וּכְאִלּוּ כֻּלָּם, עִם הַלְּבָנָה וּשְׁאָר הַז׳ כּוֹכְבֵי לֶכֶת, בְּגַלְגַּל א׳; הוּא לִהְיוֹת עֶצֶם הַגַּלְגַּלִּים זַכִּים וְסַפִּירִיִּים וּבְהִירִים מְאֹד. וַחֲגוֹרַת הַמַּזָּלוֹת בּוֹ — בָּעִגּוּל הַיּוֹתֵר גָּדוֹל; אֲבָל הוּא נוֹטֶה מֵאוֹפַן הַמִּישׁוֹר שֶׁל גַּלְגַּל הַיּוֹמִי בְּב׳ צְדָדָיו כְּמוֹ כ״ד מַעֲלוֹת, וְאֵינוֹ דָּבוּק וּפוֹגֵעַ בְּאוֹפַן הַמִּישׁוֹר אֶלָּא בְּב׳ נְקֻדּוֹת — בִּנְקֻדַּת רֹאשׁ מַזַּל טָלֶה, וּבִנְקֻדַּת רֹאשׁ מֹאזְנַיִם (וּמִכָּל מָקוֹם נְשַׁעֵר כָּל אוֹפַן הַמִּישׁוֹר בַּמַּזָּלוֹת, לְפִי מַה שֶּׁהֵם הַמַּזָּלוֹת סָמוּךְ לוֹ). וְגַלְגַּל הַיּוֹמִי כּוֹפוֹ בְּמַהֲלָכוֹ בְּכָל יוֹם מִמִּזְרָח לְמַעֲרָב; וּמִכָּל מָקוֹם הוּא בְּעַצְמוֹ יֵשׁ לוֹ תְּנוּעָה בִּפְנֵי עַצְמוֹ גַּם כֵּן מִמִּזְרָח לְמַעֲרָב, בִּכְמוֹ מַעֲלָה א׳ בְּכָל שִׁבְעִים שָׁנָה. וּמַהֲלָכוֹ בְּמַהֲלַךְ אוֹפַן הַמִּישׁוֹר נִרְאֶה כִּמְעֻקָּל, וְהוֹלֵךְ בְּדֶרֶךְ עֲקַלָּתוֹן (וְנוּכַל לְנַסּוֹת הַדָּבָר בְּב׳ עִגּוּלִים: לָשׂוּם זֶה תּוֹךְ זֶה — לֹא בְּעִנְיָן שֶׁיַּקִּיף הָא׳ לַב׳ מִכָּל עֲבָרָיו, אֲבָל קְצָת נָטוּי מִמֶּנּוּ בְּב׳ רָאשָׁיו, וְדָבוּק רַק בִּנְקֻדָּה אַחַת בָּאֶמְצַע; וּנְגַלְגֵּל הָעֶלְיוֹן כְּפִי עִגּוּלוֹ, וְכֵן הַפְּנִימִי כְּפִי עִגּוּלוֹ — תִּרְאֶה גִּלְגּוּל הַפְּנִימִי הוֹלֵךְ עֲקַלְקַלּוֹת). וְנִמְצָאִים הַמַּזָּלוֹת מִתְמוֹגְגִים וְנֶעְתָּקִים מִמְּקוֹמָם, וּבְהֶמְשֵׁךְ הַזְּמַן לֹא יִהְיֶה טָלֶה וּמֹאזְנַיִם נְקֻדּוֹת הַפְּגִיעָה. אֲבָל אֵין דְּבָרֵינוּ בְּהִתְמוֹגְגוּת הַמַּזָּלוֹת — כִּי אֵין נַפְקָא מִינָּהּ אִם יִהְיֶה טָלֶה וּמֹאזְנַיִם נְקֻדּוֹת הַפְּגִיעָה אוֹ זוּלָתָם, וְכֵן בִּשְׁאָר הַמְּקוֹמוֹת; אֲבָל כַּוָּנָתֵנוּ לִקְרוֹא לָהֶם שֵׁם, לְמַעַן יִהְיֶה יָדוּעַ לָנוּ מְקוֹם רְצוֹנֵנוּ, וְנִקְרָאֵם בְּשֵׁמוֹת הַמַּזָּלוֹת, וְאֵין נַפְקָא מִינָּהּ אִם אֵין הַמַּזָּל הַהוּא בָּעֵת הַזֹּאת שָׁם:

**ג׳.** גַּלְגַּל כּוֹכַב
