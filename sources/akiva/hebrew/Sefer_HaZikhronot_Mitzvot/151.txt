~ נִהְיֶה רַשָּׁאִים לִבְנוֹתוֹ. מִיהוּ, נִרְאֶה: הַבַּעֲלֵי קֶרִי וְזָבִים — יִטְבְּלוּ לְקִרְיָן, וְיַעֲרִיב שִׁמְשָׁן; וְאַף עַל פִּי דְּאַכַּתִּי פָּשׁוּ גַּבַּיְיהוּ טֻמְאַת מֵת — מִכָּל מָקוֹם, הָא חֲמוּרָה טֻמְאַת בַּעַל קֶרִי לְעִנְיַן שִׁלּוּחַ, כְּמוֹ שֶׁנִּתְבָּאֵר (וּבְטֻמְאַת שֶׁרֶץ וּכְהַאי גַּוְנָא — יֵשׁ לְסַפֵּק אִם צְרִיכִים לִטְבּוֹל, כֵּיוָן דִּבְלָאו הָכֵי טְמֵאֵי מֵתִים, וְטֻמְאַת מֵת חֲמִירֵי; וְעַיֵּן חֻלִּין ק״ד עַמּוּד א׳ בְּתוֹסָפוֹת, וּבְמַהֲרַ״ם שִׁיף שָׁם, עַיֵּן שָׁם); אִם לֹא שֶׁאִי אֶפְשָׁר בְּעִנְיָן אַחֵר, שֶׁאֵין פְּנַאי לְהַמְתִּין — דְּאָז הֻתְּרוּ כִּטְמֵא מֵת, וּכְמוֹ שֶׁנִּתְבָּאֵר. וְהָרַחֲמָן יְזַכֵּנוּ לִרְאוֹת בִּמְהֵרָה בְיָמֵינוּ, אָמֵן:

# מִצְוַת לֹא תַעֲשֶׂה — סִימָן ש׳

**מִצְוַת** לֹא תַעֲשֶׂה, שֶׁהִזְהִיר הַבּוֹרֵא יִתְבָּרַךְ לְכָל טָמֵא מֵהִכָּנֵס לְמִקְדַּשׁ ה׳ וּלְיֶתֶר מַחֲנוֹת, כְּפִי מַה שֶּׁנִּתְבָּאֵר בַּסִּימָן שֶׁלִּפְנֵי זֶה; שֶׁנֶּאֱמַר: "וְלֹא יְטַמְּאוּ אֶת מַחֲנֵיהֶם". וּמַה הִיא טֻמְאָה זוֹ? הַכְּנִיסָה לַמִּקְדָּשׁ; אֲבָל אִם נָגַע בָּעֲזָרָה מֵאֲחוֹרֶיהָ — לֹא טִמְּאָהּ, וּפָטוּר; שֶׁנֶּאֱמַר בְּיוֹלֶדֶת (וַיִּקְרָא י״ב): "וְאֶל הַמִּקְדָּשׁ לֹא תָבֹא" — זֶה בִּנְיַן אָב לְכָל הַטְּמֵאִים, שֶׁאֲסוּרִים בְּבִיאָה בִּלְבַד בַּמַּחֲנוֹת. וְנִכְפְּלָה הָאַזְהָרָה בִּכְנִיסַת מַחֲנֵה לְוִיָּה לְקַמָּן בְּפָרָשַׁת תֵּצֵא, עַיֵּן שָׁם. וְהָעוֹבֵר, וְנִכְנַס לְמַחֲנֶה שֶׁהוּא אָסוּר לִיכָּנֵס בָּהּ — הֲרֵי זֶה לוֹקֶה אַרְבָּעִים. וְאִם נִכְנַס לָעֲזָרָה, שֶׁהוּא מֵעֶזְרַת יִשְׂרָאֵל וְלִפְנִים — אֶחָד מְצֹרָע וְאֶחָד טְמֵא מֵת וְכָל הַטְּמֵאִים — הֲרֵי זֶה בְּכָרֵת (מַכּוֹת י״ג); שֶׁנֶּאֱמַר (בַּמִּדְבָּר י״ט): "וְנִכְרְתָה הַנֶּפֶשׁ הַהִוא מִתּוֹךְ הַקָּהָל, כִּי אֶת מִקְדַּשׁ ה׳ טִמֵּא" (תּוֹרַת כֹּהֲנִים פָּרָשַׁת אַחֲרֵי); וְאוֹמֵר (וַיִּקְרָא י״ז): "וְאִם לֹא יְכַבֵּס, וּבְשָׂרוֹ לֹא יִרְחָץ — וְנָשָׂא עֲוֹנוֹ"; וְאָמְרוּ רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה, שֶׁבְּטֻמְאַת מִקְדָּשׁ הַכָּתוּב מְדַבֵּר, וְאֵין נְשִׂיאַת עָוֹן אֶלָּא כָּרֵת. וְכָפַל עוֹד כָּרֵת זֶה בִּלְשׁוֹן מִיתָה בְּזָבָה (וַיִּקְרָא ט״ו): "וְלֹא יָמֻתוּ בְּטֻמְאָתָם בְּטַמְּאָם אֶת מִשְׁכָּנִי אֲשֶׁר בְּתוֹכָם". וְזֶהוּ — אִם עָשָׂה בְּזָדוֹן; אֲבָל בִּשְׁגָגָה — הוּא מֵבִיא קָרְבַּן עוֹלֶה וְיוֹרֵד, כְּמוֹ שֶׁנֶּאֱמַר (וַיִּקְרָא ה׳): "אוֹ כִי יִגַּע בְּטֻמְאַת אָדָם" וְגוֹ׳. וַאֲפִלּוּ טְבוּל יוֹם וּמְחֻסַּר כִּפּוּרִים שֶׁנִּכְנְסוּ לְעֶזְרַת יִשְׂרָאֵל — חַיָּבִים כָּרֵת (וְהָרַמְבַּ״ם כָּתַב: בִּמְחֻסַּר כִּפּוּרִים — אֵינוֹ חַיָּב כָּרֵת וְלֹא מַלְקוֹת, אֶלָּא מַכַּת מַרְדּוּת; וְאֵין הֲלָכָה כֵּן):

**וְנוֹהֵג** בַּכֹּל — בִּזְכָרִים וּבִנְקֵבוֹת — וּבְכָל זְמַן; אֲפִלּוּ בִּזְמַן שֶׁאֵין בֵּית הַמִּקְדָּשׁ קַיָּם, מִכָּל מָקוֹם קְדֻשָּׁתוֹ עָלָיו, וַעֲנוּשׁ כָּרֵת כָּל הַנִּכְנָס לִמְקוֹם הָעֲזָרָה. זֶהוּ דַּעַת הָרַמְבַּ״ם וְהָרָאָ״ה; אֲבָל הָרַאֲבַ״ד כָּתַב (סוֹף פֶּרֶק ו׳ מֵהִלְכוֹת בֵּית הַבְּחִירָה), דְּהַנִּכְנָס עַתָּה שָׁם — אֵין בּוֹ כָּרֵת, עַיֵּן שָׁם. וְיֵשׁ לְעַיֵּן אִם לָאו יֵשׁ כָּאן; וְעַיֵּן מַה שֶּׁכָּתַבְתִּי
