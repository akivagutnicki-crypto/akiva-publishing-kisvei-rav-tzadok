~ הָאָב אֶת בְּנוֹ וְהָרַב אֶת עַבְדּוֹ. וְזֶה בִּתְנָאִים שְׁלֹשָׁה, הָא׳ — שֶׁיַּעֲשֶׂה בִּפְנֵי אֲנָשִׁים חֲכָמִים וּבְקִיאִים בַּדָּבָר:

**כֵּיצַד** מָלִין? לוֹקֵחַ כָּל דָּבָר הַכּוֹרֵת, וַאֲפִלּוּ צוֹר וּזְכוּכִית, חוּץ מִקְּרוּמִית שֶׁל קָנֶה, שֶׁקֵּסָמִים נִתָּזִין מִמֶּנָּה וְיָבוֹא לִידֵי כְּרוּת שָׁפְכָה. וּמִצְוָה מִן הַמֻּבְחָר בְּבַרְזֶל — בְּסַכִּין אוֹ בְּמִסְפָּרַיִם; וְנָהֲגוּ בְּסַכִּין. וְחוֹתֵךְ אֶת הָעָרְלָה — כָּל הָעוֹר הַחוֹפֶה אֶת הָעֲטָרָה, וְהוּא הַחוּט הַגָּבוֹהַּ הַסּוֹבֵב בֵּין הַבָּשָׂר שֶׁבְּרֹאשׁ הַגִּיד וּבֵין הַגִּיד עַצְמוֹ — עַד שֶׁתִּתְגַּלֶּה הָעֲטָרָה כֻּלָּהּ. וְאִם נִשְׁאֲרוּ אֵיזֶה צִיצִין: אִם הֵם חוֹפִים רֹב גָּבְהוֹ שֶׁל אוֹתוֹ חוּט (וּדְלָא כְּיֵשׁ אוֹמְרִים רֹב גָּבְהוֹ שֶׁל כָּל הַבָּשָׂר שֶׁבְּרֹאשׁ הַגִּיד), אוֹ רֹב הֶקֵּפוֹ, אַף עַל פִּי שֶׁאֵינוֹ רֹב גָּבְהוֹ — הֲרֵי זֶה מְעַכֵּב אֶת הַמִּילָה, וּכְאִלּוּ לֹא נִמּוֹל כְּלָל. וְאִם אֵין חוֹפִים אֶלָּא מְעַט, וְאֵין בּוֹ לֹא רֹב גָּבְהוֹ וְלֹא רֹב הֶקֵּפוֹ שֶׁל הַחוּט — לְכַתְּחִלָּה יִטּוֹל בְּחוֹל כָּל הַצִּיצִים הַגְּדוֹלִים, וְאִם לֹא נָטַל — אֵינוֹ מְעַכֵּב. וּמְשַׁעֲרִים זֶה אַחַר שֶׁנִּתְקַשָּׁה הַגִּיד, וּדְלָא כְּיֵשׁ אוֹמְרִים לְשַׁעֵר קֹדֶם שֶׁנִּתְקַשָּׁה. וְאַחַר כָּךְ פּוֹרְעִין הַקְּרוּם הָרַךְ שֶׁלְּמַטָּה מֵהָעוֹר בְּצִפֹּרֶן, וּמַחֲזִירוֹ לְכָאן וּלְכָאן, עַד שֶׁיֵּרָאֶה בְּשַׂר הָעֲטָרָה. וְצָרִיךְ שֶׁיְּדַקְדֵּק בָּזֶה, כִּי לִפְעָמִים נִמְצָא עוֹר דַּק, וְאֵינוֹ אֶלָּא לֵיחָה בְּעָלְמָא, וּפוֹרְעוֹ — וְעוֹר הַפְּרִיעָה הָאֲמִתִּי קַיָּם; לָכֵן יְדַקְדֵּק בָּזֶה. וְאִם מָל וְלֹא פָּרַע — כְּאִלּוּ לֹא מָל (כָּתַב רִי״עַ: דַּוְקָא קָטָן, אֲבָל הַמָּל אֶת הַגָּדוֹל אֵין צָרִיךְ לִפְרוֹעַ; וְלֹא נִרְאֶה לִי). וְאַחַר כָּךְ מוֹצְצִין הַמִּילָה, כְּדֵי שֶׁיֵּצֵא הַדָּם מִמְּקוֹמוֹת הָרְחוֹקִים, וְלֹא יָבוֹא לִידֵי סַכָּנָה; וְכָל מוֹהֵל שֶׁאֵינוֹ מוֹצֵץ — מַעֲבִירִין אוֹתוֹ. וְאַחַר שֶׁמּוֹצֵץ נוֹתְנִים עָלָיו אֲבַק סַמִּים הָעוֹצְרִים הַדָּם, כְּקֶמַח שֶׁל עֵץ רָקוּב מִתּוֹלַעַת וְהַדּוֹמֶה. וְנָהֲגוּ לִזְרוֹק הָעָרְלָה, וְכֵן דַּם הַמְּצִיצָה, בְּחוֹל וְעָפָר:

**מָל** יָפֶה, וְחָזַר וְנִמְשַׁךְ הָעָרְלָה — מִן הַתּוֹרָה אֵין צָרִיךְ לַחֲזוֹר לְמוּלוֹ, כֵּיוָן שֶׁכְּבָר נִמּוֹל פַּעַם אַחַת יָפֶה; אֲבָל מִדְּרַבָּנָן, אִם נִמְשַׁךְ עַד שֶׁאֵינוֹ נִרְאֶה נִמּוֹל בְּעֵת שֶׁמִּתְקַשֶּׁה — צָרִיךְ לַחְתּוֹךְ כָּל הַבָּשָׂר הַמָּשׁוּךְ, עַד שֶׁיִּהְיֶה נִרְאֶה כְּנִמּוֹל בִּשְׁעַת קִשּׁוּיוֹ. וְאִם נִרְאֶה נִמּוֹל — אֲפִלּוּ רַק מִעוּט הָעֲטָרָה, וְרֻבָּהּ מְכֻסָּה בָּעוֹר הַמָּשׁוּךְ — אֵין צָרִיךְ לְמוּלוֹ שֵׁנִית. וּמִכָּל מָקוֹם אִם אֶפְשָׁר יְתַקֵּן עַל יְדֵי שֶׁיִּמְשׁוֹךְ הָעוֹר וְיִדְחָקֶנָּה לַאֲחוֹרֶיהָ, וְיִקְשְׁרֶנָּה שָׁם עַד שֶׁתַּעֲמוֹד וְלֹא תַּחֲזוֹר לְמַטָּה. וְדַוְקָא כְּשֶׁיָּדוּעַ בְּוַדַּאי שֶׁבִּשְׁעַת הַמִּילָה נִמּוֹל יָפֶה וְנִפְרַע יָפֶה בָּעוֹר הָאֲמִתִּי; אֲבָל מִסְּתָמָא יֵשׁ לָחוּשׁ שֶׁמִּתְּחִלָּה לֹא נִמּוֹל יָפֶה, וְאָז תָּלוּי בַּחֲפִיַּת רֹב, כְּמוֹ שֶׁנִּתְבָּאֵר לְעֵיל. וְאִם נוֹלַד מָהוּל מְעַט —
