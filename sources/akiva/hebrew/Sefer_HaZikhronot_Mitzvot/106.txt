~ הִיא חֲצִי עִגּוּל הַיּוֹצֵא מֶרְכָּז, וּבִנְקֻדַּת ד׳ — חֲצִי עִגּוּל הַדּוֹמֶה; וּבֵינְתַיִם, מִנְּקֻדַּת ג׳ עַד ד׳, בְּכָל חֵלֶק וָחֵלֶק מִמֶּנּוּ יֶאֱרַךְ קֶשֶׁת הַהֶבְדֵּל שֶׁבֵּין ב׳ הַגַּלְגַּלִּים. וְהוּא מוּבָן לַמִּסְתַּכֵּל בַּצּוּרָה הַנִּזְכֶּרֶת לְעֵיל: כִּי, בְּדִמְיוֹן, בַּחֲצִי תְּקוּפָה — וְהוּא בַּחֲצִי רֶבַע הַגַּלְגַּל, וְהוּא בִּנְקֻדַּת ש׳ — וַחֲצִי רֶבַע גַּלְגַּל הַדּוֹמֶה הוּא בִּנְקֻדַּת ר׳, וְהֶבְדֵּלָם הוּא הַקֶּשֶׁת ר׳ ש׳. וּמִנְּקֻדַּת ד׳ עַד נְקֻדַּת ה׳ — יְקַצֵּר מַהֲלָכוֹ בְּפָחוֹת מֵחֲצִי (נִרְאֶה דְּצָרִיךְ לוֹמַר מֵרֶבַע — הַמַּגִּיהַּ) שָׁנָה, וּבְכָל נְקֻדָּה יִמְעַט בְּקֶשֶׁת הַהֶבְדֵּל, עַד בּוֹאוֹ לִנְקֻדַּת ה׳, שֶׁשָּׁם הֵם שָׁוִים; וּמִשָּׁם יָשׁוּב לִנְקֻדַּת ו׳, וְשָׁם יִהְיֶה הַהֶבְדֵּל קֶשֶׁת ו׳ י׳; וּמִשָּׁם יָשׁוּב לִנְקֻדַּת ג׳, שֶׁשָּׁם שָׁוִים. וְנִמְצָא: פְּעָמִים יַאֲרִיךְ חֶשְׁבּוֹן הָאֲמִתִּי, וּפְעָמִים יְקַצֵּר, וְהָאֶמְצָעִי — הַשָּׁוֶה תָּמִיד. וְעִקַּר חֶשְׁבּוֹן הָרְאִיָּה — הוּא לָדַעַת חֶשְׁבּוֹן הָאֲמִתִּי:

**וְהִנֵּה** אִלּוּ הָיָה כְּמוֹ שֶׁצִּיַּרְנוּ — שֶׁנְּקֻדַּת ג׳, שֶׁהִיא נְקֻדָּה הַיּוֹתֵר גְּבוֹהָה מִגַּלְגַּל הַשֶּׁמֶשׁ לְכַדּוּר אַרְצֵנוּ, מַזַּל סַרְטָן בְּדִמְיוֹן, וּנְקֻדַּת ה׳, שֶׁהִיא נְקֻדָּה הַיּוֹתֵר שְׁפֵלָה מִן הַשֶּׁמֶשׁ לְאַרְצֵנוּ, בְּרֹאשׁ מַזַּל גְּדִי, בְּדֶרֶךְ מָשָׁל, אוֹ כְּהַאי גַּוְנָא בִּשְׁאָר מְקוֹמוֹת — הָיָה כֵּן תָּמִיד בְּלִי שִׁנּוּי, הִנֵּה יָדַעְנוּ תָּמִיד שֶׁתְּקוּפוֹת שְׁתַּיִם הֵם אֲרֻכּוֹת וּשְׁתַּיִם קְצָרוֹת, וְכֵן הוּא בְּכָל שָׁנָה וְשָׁנָה. אֲבָל לְפִי שֶׁרָאוּ הַתּוֹכְנִים שֶׁאֵין תְּקוּפוֹת כָּל שָׁנָה שָׁוִים בִּזְמַנָּם — לְפִיכָךְ שָׂמוּ עוֹד לְגַלְגַּל הַדּוֹמֶה הַנִּזְכָּר מַהֲלָךְ, זוּלַת מַהֲלַךְ הַשֶּׁמֶשׁ בְּגַלְגַּלָּהּ הַיּוֹצֵא מֶרְכָּז: בְּכָל שִׁבְעִים שָׁנָה — מַעֲלָה אַחַת; וּמַהֲלָכָהּ מִמַּעֲרָב לְמִזְרָח, כְּמַהֲלַךְ הַשֶּׁמֶשׁ; וּמוֹשֶׁכֶת בְּמַהֲלָכָהּ גַּם כֵּן גַּלְגַּל הַשֶּׁמֶשׁ שֶׁבְּתוֹכָהּ. וְנִמְצָא, נְקֻדַּת הַג׳ — שֶׁהוּא גֹּבַהּ הַשֶּׁמֶשׁ — אִם הָיָה בִּתְחִלַּת מַזַּל סַרְטָן, בְּדִמְיוֹן, הִנֵּה בְּעוֹד שִׁבְעִים שָׁנָה יִהְיֶה בִּתְחִלַּת מַעֲלָה ב׳ מִמַּזַּל סַרְטָן; וּבְב׳ אֲלָפִים וּמֵאָה שָׁנָה יִהְיֶה בִּתְחִלַּת מַזַּל עַקְרָב (נִרְאֶה דְּצָרִיךְ לוֹמַר אַרְיֵה — הַמַּגִּיהַּ), כִּי הָלַךְ כְּבָר גַּלְגַּל הַדּוֹמֶה כָּל ל׳ מַעֲלוֹת סַרְטָן, בְּל׳ פְּעָמִים שִׁבְעִים, שֶׁהֵם ב׳ אֲלָפִים וְק׳ שָׁנָה:

**וּלְפִי** זֶה אֵין תְּקוּפוֹת כָּל שָׁנָה שָׁווֹת. כִּי, בְּדִמְיוֹן, בִּהְיוֹת מְקוֹם הַגֹּבַהּ — שֶׁהִיא נְקֻדַּת ג׳ — בִּתְחִלַּת סַרְטָן, וּתְהִי הַתְּקוּפָה עַד נְקֻדַּת ד׳ — הִנֵּה יִהְיֶה הַהֶבְדֵּל קֶשֶׁת ד״ח. אֲבָל כְּשֶׁיִּהְיֶה מְקוֹם הַגֹּבַהּ בַּחֲצִי עַקְרָב (נִרְאֶה דְּצָרִיךְ לוֹמַר אַרְיֵה — הַמַּגִּיהַּ), בְּדִמְיוֹן — הִנֵּה יִהְיֶה תְּקוּפַת מֹאזְנַיִם, בְּדִמְיוֹן, בִּנְקֻדַּת פ׳; וְיִהְיֶה מִמֶּנָּה לִנְקֻדַּת ר׳ שֶׁל תְּקוּפָה הַקּוֹדֶמֶת, שֶׁהִיא תְּקוּפַת סַרְטָן — הַהֶבְדֵּל בֵּינֵיהֶם קֶשֶׁת ר״ש וְקֶשֶׁת ע״פ; כִּי קַו ע׳ וְקַו ש׳ הוּא רְבִיעַ גַּלְגַּל יוֹצֵא הַמֶּרְכָּז, וְהוּא יוֹתֵר מִן קֶשֶׁת ד״ח אוֹ ו״י. וּמִמֶּנָּה לִנְקֻדַּת ס׳, תְּקוּפָה שֶׁלְּאַחֲרֶיהָ, שֶׁהִיא תְּקוּפַת גְּדִי — הַהֶבְדֵּל קֶשֶׁת נ״ס, וְהוּא פָּחוֹת מִקֶּשֶׁת ד״ח אוֹ ו״י. וְכֵן מִנְּקֻדַּת ס׳ לִנְקֻדַּת
