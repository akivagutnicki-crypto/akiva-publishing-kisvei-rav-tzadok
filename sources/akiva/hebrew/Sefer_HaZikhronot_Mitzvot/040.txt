~ הַשְּׁנַיִם בִּפְנֵי הַשְּׁלֹשָׁה בְּיוֹם ל׳, וִיקַדְּשׁוּהוּ הַשְּׁלֹשָׁה. אֲבָל אִם רָאוּ בְּיוֹם ל׳, הוֹאִיל וְהוּא זְמַן קִדּוּשׁ — מְקַדְּשִׁין בְּעַצְמָם (עַיֵּן בַּמְּפָרֵשׁ, שֶׁהִנִּיחַ בְּצָרִיךְ עִיּוּן שֶׁלֹּא יָדַע הַחִלּוּק בָּזֶה; וּבְמְחִילַת כְּבוֹדוֹ, הַחִלּוּק פָּשׁוּט בַּגְּמָרָא, כַּנִּזְכָּר). וְאִם רָאוּ סָמוּךְ לִשְׁקִיעַת הַחַמָּה שֶׁל יוֹם כ״ט, וְלֹא הִסְפִּיקוּ לוֹמַר "מְקֻדָּשׁ" עַד שֶׁחֲשֵׁכָה — מְקַדְּשִׁים בְּיוֹם ל׳ עַל פִּי עַצְמָם, בְּלִי שֶׁיַּעֲמִידוּ עֵדִים; כֵּיוָן שֶׁרָאוּ בְּשָׁעָה הָרְאוּיָה לְקַדֵּשׁ, אֵין צָרִיךְ לִהְיוֹת עֵדִים וִיהֵא אֵין עֵד נַעֲשֶׂה דַּיָּן (וְזֶהוּ דַּעַת הָרַלְבָּ״ח; וְלַעֲנִיּוּת דַּעְתִּי צָרִיךְ עִיּוּן בָּזֶה, וּרְאָיָתוֹ אֵין מֻכְרַחַת, עַיֵּן שָׁם). אִם שָׁמְעוּ הָעֵדוּת, אוֹ רָאוּ בְּעַצְמָם, בְּעוֹד יוֹם, וְלֹא הִסְפִּיקוּ לוֹמַר "מְקֻדָּשׁ" אֶלָּא בְּבֵין הַשְּׁמָשׁוֹת: אִם הוּא בְּבֵין הַשְּׁמָשׁוֹת דְּיוֹם ל׳, וְשָׁמְעוּ אוֹ רָאוּ בְּכ״ט — כֵּיוָן שֶׁעֲדַיִן יֵשׁ לָהֶם פְּנַאי לוֹמַר "מְקֻדָּשׁ" כָּל יוֹם ל׳ — אֲפִלּוּ עָבְרוּ וְאָמְרוּ "מְקֻדָּשׁ" בְּבֵין הַשְּׁמָשׁוֹת, צְרִיכִים לַחֲזוֹר וְלוֹמַר בְּל׳ "מְקֻדָּשׁ", דְּבֵין הַשְּׁמָשׁוֹת הוּא סְפֵק לַיְלָה. (מִיהוּ אִם עָבְרוּ וְאָמְרוּ "מְקֻדָּשׁ" בְּבֵין הַשְּׁמָשׁוֹת, וְעָבְרוּ אוֹ שָׁכְחוּ וְלֹא אָמְרוּ בְּיוֹם ל׳ "מְקֻדָּשׁ" — נִרְאֶה דִּלְפִי זֶה הֲוֵי קִדּוּשׁ דְּבֵין הַשְּׁמָשׁוֹת קִדּוּשׁ, וְאֵין הַחֹדֶשׁ מְעֻבָּר; וְצָרִיךְ עִיּוּן.) וְאִם הוּא בְּבֵין הַשְּׁמָשׁוֹת דְּיוֹם ל״א — דְּאִם לֹא יְקַדְּשׁוּ עַכְשָׁיו יִהְיֶה הַחֹדֶשׁ מְעֻבָּר (וַאֲפִלּוּ לְפִי דִּבְרֵי הָאוֹמֵר דִּמְקַדְּשִׁים לְמַפְרֵעַ — דְּזֶהוּ רַק כְּשֶׁבָּאוּ עֵדִים אַחַר כָּךְ; אֲבָל עֵדִים שֶׁבָּאוּ בִּזְמַנּוֹ, וְנִדְחוּ, וְלֹא קִדְּשׁוּ בִּזְמַנּוֹ — אֵין מְקַדְּשִׁים אַחַר כָּךְ, כְּשֶׁהִפְסִיק זְמַן שֶׁאֵין רָאוּי לְקִדּוּשׁ בֵּינְתַיִם) — הֲרֵי הֵם מְקַדְּשִׁים בְּבֵין הַשְּׁמָשׁוֹת, דְּזוֹ קֻלָּא הוּא, לְהַדְחוֹת רֹאשׁ חֹדֶשׁ מִזְּמַנּוֹ וְלַעֲשׂוֹתוֹ בְּיוֹם ל״א, לְפִי שֶׁרָאוּי לוֹ לָחוּשׁ לִהְיוֹת בִּזְמַנּוֹ; וּלְפִיכָךְ בְּבֵין הַשְּׁמָשׁוֹת, דְּהוּא סָפֵק — מַחְמִירִין מִסָּפֵק לְקַדְּשׁוֹ, וְלַעֲשׂוֹת רֹאשׁ חֹדֶשׁ בִּזְמַנּוֹ (זֶהוּ דַּעַת הָרַלְבָּ״ח. וְעַיֵּן בְּתוֹסְפוֹת יוֹם טוֹב (רֵישׁ פֶּרֶק ג׳ דְּרֹאשׁ הַשָּׁנָה), שֶׁכָּתַב דְּאֶפְשָׁר לוֹמַר דַּעַת הָרַמְבַּ״ם, דְּגַם בְּל׳ אֵין מְקַדְּשִׁין אֶלָּא עַד בֵּין הַשְּׁמָשׁוֹת דְּל״א, לֹא יוֹתֵר. וְכֵן נִרְאֶה לַעֲנִיּוּת דַּעְתִּי, דְּאֵין לָהֶם לְהַכְנִיס עַצְמָן לִסְפֵק: דְּאִם לֹא יְקַדְּשׁוּ בְּבֵין הַשְּׁמָשׁוֹת, וְיַעֲשׂוּ רֹאשׁ חֹדֶשׁ בְּיוֹם ל״א — יִהְיֶה כַּדִּין, דַּהֲרֵי אֲפִלּוּ מְזִידִים לְקַדְּשׁוֹ שֶׁלֹּא כַּדִּין, מִכָּל מָקוֹם הֲרֵי זֶה מְקֻדָּשׁ, כְּמוֹ שֶׁיִּתְבָּאֵר; אֲבָל אִם יְקַדְּשׁוּ בְּבֵין הַשְּׁמָשׁוֹת — שֶׁמָּא לַיְלָה הוּא, וְאֵין קִדּוּשָׁם קִדּוּשׁ, וְאֵין יְכוֹלִים לְקַדֵּשׁ כְּלָל, וְנִמְצָא הוּא שֶׁלֹּא כַּדִּין. וְעַיֵּן בְּרַמְבַּ״ם (פֶּרֶק ג׳ הֲלָכָה ו׳): "סָמוּךְ לִשְׁקִיעַת הַחַמָּה" וְכוּלֵי — נַמִּי מַשְׁמַע כַּנִּזְכָּר לְעֵיל, דְּלָא כְּהָרַלְבָּ״ח, עַיֵּן שָׁם. וְאוּלַי מוֹדֶה הָרַלְבָּ״ח דִּלְקַבֵּל עֵדוּת לְכַתְּחִלָּה בְּבֵין הַשְּׁמָשׁוֹת דְּלָא, וְצָרִיךְ לְעַיֵּן. שׁוּב מָצָאתִי בְּלֶחֶם מִשְׁנֶה שָׁם הֵעִיר בָּזֶה, וְהִנִּיחַ בְּצָרִיךְ עִיּוּן, עַיֵּן שָׁם):

**וְכָל** זֶה — לְאַחַר תַּקָּנַת רַבִּי יוֹחָנָן בֶּן זַכַּאי, שֶׁיִּתְבָּאֵר לְהַלָּן; אֲבָל מִקֹּדֶם — יִתְבָּאֵר דְּהָיְתָה תַּקָּנָה בְּלָאו הָכֵי שֶׁלֹּא לְקַבֵּל עֵדוּת אַחַר הַמִּנְחָה. וְכָל זֶה לְעִנְיַן אֲמִירַת רֹאשׁ בֵּית דִּין "מְקֻדָּשׁ"; אֲבָל בַּעֲנִיַּת הָעָם ב׳ פְּעָמִים "מְקֻדָּשׁ
