~ אֲסוּרִים מִשּׁוּם גִּיד הַנָּשֶׁה" וְכוּלֵי; "הַיָּד אוֹ הַכַּף" וְכוּלֵי — "בְּרֹאשָׁהּ יֵשׁ גִּיד אֶחָד כְּצוּרַת ט׳, וְשֶׁבְּאֶמְצָעִיתָהּ כְּצוּרַת ש׳, וּבִמְקוֹם עָבְיָהּ יֵשׁ גִּיד אֶחָד גַּם כֵּן כְּצוּרַת ש׳ אַחַת, וְקוֹרִין אוֹתוֹ אֶצְבַּע אֲגוּדָל; וּבִמְקוֹם דַּק שֶׁתַּחְתָּיו יֵשׁ גִּיד אֶחָד כְּצוּרַת קֶרֶן הַצְּבִי, וְאֵצֶל גִּיד זֶה יֵשׁ עוֹד גִּיד אֶחָד כְּצוּרַת געפלי״, וְכֻלָּן אֲסוּרִין מִשּׁוּם גִּיד הַנָּשֶׁה, וּצְרִיכִין חִטּוּט" וְכוּלֵי; "הבייט״ש — מְקַלְּפִין הַצַּד אַחַת עִם הַמַּרְאֶה כֶּסֶף מִשּׁוּם גִּיד הַנָּשֶׁה הָרוֹבֵץ עָלָיו, וּבְצַד הַב׳ מְגָרְרִין הַמַּרְאֶה כֶּסֶף הֵיטֵב, וּבְתוֹכָן יֵשׁ גִּידִין לָאֹרֶךְ הַנִּרְאִים לָעֵינַיִם, וְהֵן הֵן גִּיד הַנָּשֶׁה עַצְמָן; וּמִצַּד א׳ בִּפְנִים, בִּמְקוֹם הָרָחָב, הוֹלֵךְ סוֹף גִּיד הַנָּשֶׁה בְּצוּרַת ד׳, וּמְגָרְרִין הַמַּרְאֶה כֶּסֶף מִתּוֹכָן, וּמְצַדְּדִין" וְכוּלֵי; "הַמייג״ל — מְקַלְּפִין מִבַּחוּץ הֵיטֵב מְשֻׁקָּע, וּבְתוֹכָהּ יֵשׁ גִּידִין, וְהֵן גִּיד הַנָּשֶׁה בְּעַצְמָן, וְהֵן נִרְאֶה לָעֵינַיִם, וּצְרִיכִין חִטּוּט עַד סוֹפָן" וְכוּלֵי, עַד כָּאן לְשׁוֹנוֹ. וְאֵין לִלְמוֹד סֵדֶר הַנִּקּוּר רַק בְּמַרְאִית הָעַיִן, מִן הַמֻּמְחֶה הַבָּקִי בְּנִקּוּרוֹ, וְיִוָּדַע כָּל דָּבָר עַל מְכוֹנוֹ:

**כָּל** אָדָם נֶאֱמָן עַל נִקּוּר הַיָּרֵךְ מִגִּיד הַנָּשֶׁה, וְכֵן הַטַּבָּחִים עַצְמָם נֶאֱמָנִים. וּמִיהוּ אֵין לוֹקְחִין בָּשָׂר מִכָּל טַבָּח שֶׁשּׁוֹחֵט לְעַצְמוֹ וּמוֹכֵר לַאֲחֵרִים, אֶלָּא אִם כֵּן הֻחְזַק בְּכַשְׁרוּת, כְּמוֹ שֶׁיִּתְבָּאֵר. וְהַנָּשִׁים אֵין נֶאֱמָנוֹת לְנַקֵּר בְּעַצְמָם, לְפִי שֶׁטּוֹרַח גָּדוֹל בַּדָּבָר; אֲבָל נֶאֱמָנוֹת לוֹמַר שֶׁרָאוּ שֶׁנִּקְּרוֹ אָדָם כָּשֵׁר. וְיֵשׁ אוֹמְרִים שֶׁנֶּאֱמָנוֹת לְנַקֵּר בְּעַצְמָן גַּם כֵּן, אַף עַל פִּי שֶׁיֵּשׁ בּוֹ טוֹרַח, בְּוַדַּאי אִסּוּר; אֲבָל אִם הוּא סְפֵק מְנֻקָּר, וְצָרִיךְ לְנַקֵּר מִסָּפֵק, וְכַהַאי גַּוְנָא שֶׁיֵּשׁ צַד לְהָקֵל — אָז אֵין נֶאֱמָנוֹת, דְּדַעְתָּן קַלּוֹת, וְיָקִילוּ בָּזֶה (עַיֵּן בִּשְׁאֵלוֹת וּתְשׁוּבוֹת הָרַ״ב סִימָן ס״א). וְכָל יָרֵךְ שֶׁיָּצְאָה מִתַּחַת יְדֵי יִשְׂרָאֵל מְחֻתֶּכֶת כְּדֶרֶךְ שֶׁחוֹתְכִין אוֹתָהּ מְנַקְּרֵי בָּשָׂר — הִיא בְּחֶזְקַת שֶׁנִּטַּל גִּידָהּ, דַּחֲתִיכָתָהּ מְעִידָה עָלֶיהָ, וְאֵין לָנוּ לָחוּשׁ שֶׁמָּא לֹא הִשְׁלִים נִקּוּרוֹ; דְּמִסְּתָמָא, כֵּיוָן שֶׁהִתְחִיל — סִיֵּם, וַחֲזָקָה שֶׁלֹּא הוֹצִיא מִתַּחַת יָדוֹ דָּבָר שֶׁאֵינוֹ מְתֻקָּן. וּלְפִיכָךְ אֵין לִתֵּן אוֹ לִמְכּוֹר לְעוֹבֵד כּוֹכָבִים יָרֵךְ שֶׁגִּיד הַנָּשֶׁה בְּתוֹכוֹ כְּשֶׁהוּא חֲתוּכָה, אִם אוֹמֵר לוֹ שֶׁכְּשֵׁרָה הִיא בִּפְנֵי יִשְׂרָאֵל; וְכָל שֶׁכֵּן כְּשֶׁיִּשְׂרָאֵל עַצְמוֹ יוֹדֵעַ שֶׁהוּא מַכְשִׁירָהּ — דְּאָסוּר, שֶׁמָּא יַחֲזוֹר וְיִקְנֶה אוֹתוֹ יָרֵךְ מֵעוֹבֵד כּוֹכָבִים, וְיִרְאֶנָּה חֲתוּכָה, וְיִסְמוֹךְ שֶׁכְּבָר נוֹקַר הַגִּיד. וּמִי שֶׁנִּיקֵּר וְהָלַךְ לוֹ, וְאֵין יָדוּעַ אִם הוּא בָּקִי בְּנִקּוּר, וְאֵין כָּאן מִי שֶׁבָּקִי בְּנִקּוּר לְהַרְאוֹת לוֹ — מֻתָּר, דְּרֹב מְצוּיִין אֵצֶל נִקּוּר מֻמְחִין. מִיהוּ אִם אֶפְשָׁר לְהַרְאוֹת לְבָקִי — טוֹב לְהַחְמִיר לְהַרְאוֹת לוֹ, וְשֶׁלֹּא לִסְמוֹךְ עַל הַנִּקּוּר, כֵּיוָן דְּלָא יָדְעִינַן אִם הוּא בָּקִי. מִיהוּ לְאַטְרוּחֵי בִּשְׁבִיל זֶה — וַדַּאי אֵין צָרִיךְ. וְדַוְקָא שֶׁיּוֹדְעִים שֶׁהִשְׁלִים נִקּוּרוֹ — כְּגוֹן שֶׁעֲדַיִן שָׁהָה קְצָת אַחַר נִקּוּרוֹ, וְלֹא הָלַךְ תֵּכֶף;
