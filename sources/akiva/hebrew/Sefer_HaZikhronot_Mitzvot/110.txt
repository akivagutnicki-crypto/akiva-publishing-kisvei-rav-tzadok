~ ג׳ מֵאוֹת (נִרְאֶה דְּצָרִיךְ לוֹמַר אֶלֶף — הַמַּגִּיהַּ) שָׁנָה בְּקֵרוּב — חִלּוּק בְּמַעֲלָה אַחַת; וְאִם כֵּן, כָּל שֶׁכֵּן דְּאֵין לָנוּ לָחוּשׁ לְג׳ שְׁנִיִּים שֶׁל אֶלֶף יוֹם, שֶׁאֵינָם מִתְקַבְּצִים לְמַעֲלָה אַחַת אֶלָּא בְּעֵרֶךְ לְה׳ מֵאוֹת שָׁנָה (נִרְאֶה דְּצָרִיךְ לוֹמַר לְג׳ אֲלָפִים — הַמַּגִּיהַּ). וּכְבָר כָּתַב הָרַמְבַּ״ם (פֶּרֶק י״א, הֲלָכָה ה׳): "שֶׁלֹּא דִּקְדַּקְנוּ" וְכוּלֵי; וּבְפֶרֶק י״ג (הֲלָכָה י׳): "וְאַל תִּפְנֶה אֶל הַשְּׁנִיּוֹת" וְכוּלֵי:

**מִיהוּ**, מִכָּל מָקוֹם, לְפִי זֶה, עַכְשָׁיו שֶׁכְּבָר עָבְרוּ קָרוֹב לְז׳ מֵאוֹת שָׁנָה מִן הָעִקָּר שֶׁל הָרַמְבַּ״ם זִכְרוֹנוֹ לִבְרָכָה — אִם כֵּן נִתּוֹסַף יוֹתֵר מֵחֲצִי מַעֲלָה מִן הַל״ה רְבִיעִיִּים הַנִּזְכָּרִים לְעֵיל; וְגַם מִתּוֹסֶפֶת ג׳ שְׁנִיִּים לְאֶלֶף יוֹם יִתְקַבֵּץ יוֹתֵר מִשִּׁשִּׁית מַעֲלָה; אִם כֵּן, יִהְיֶה הֶבְדֵּל וּמֶרְחָק רַב. וְצָרִיךְ עִיּוּן עַל הַמְּפָרֵשׁ, וְכֵן עַל בַּעַל שוב״י, שֶׁלֹּא יָרְדוּ לְדַקְדֵּק בָּזֶה, וְצָרִיךְ לְעַיֵּן. וּלְפִי זֶה רָאוּי לָנוּ לַחְשׁוֹב בָּזֶה — לְדַקְדֵּק הַכֹּל בְּחֶשְׁבּוֹן אַלְבַּתָּאנִי הַנִּזְכָּר, בִּשְׁלִישִׁיִּים וּבִרְבִיעִיִּים:

**אֲבָל** מַה שֶּׁצָּרִיךְ לְעַיֵּן לְדַעַת אַלְבַּתָּאנִי: כִּי הִנֵּה הוּא סוֹבֵר מֶשֶׁךְ הַשָּׁנָה פָּחוֹת מִשס״ה יוֹם וּרְבִיעַ — בְּג׳ חֲלָקִים מִש״ס בַּיּוֹם, וּב׳ שְׁלִישֵׁי חֵלֶק; כְּמוֹ שֶׁכָּתַב בַּמְּפָרֵשׁ, שֶׁהֵם י״א חֶלְקֵי תתר״ף בַּיּוֹם (נִרְאֶה דְּצָרִיךְ לוֹמַר בְּשָׁעָה — הַמַּגִּיהַּ), שֶׁהֵם י״א פְּעָמִים כ״ד חֲלָקִים, שֶׁהֵם רס״ד חֲלָקִים מֵחֶלְקֵי תתר״ף בְּשָׁעָה (נִרְאֶה דְּצָרִיךְ לוֹמַר בַּיּוֹם — הַמַּגִּיהַּ). וְנִמְצָא מִדַּת הַשָּׁנָה לְפִי חֶשְׁבּוֹנוֹ — שס״ה יוֹם, ה׳ שָׁעוֹת, תתי״ו חֲלָקִים; וְכֵן כָּתַב לְהֶדְיָא הַמְּפָרֵשׁ (פֶּרֶק ו׳, הֲלָכָה ד׳). וְהִנֵּה מַהֲלַךְ הַיּוֹם — נ״ט ח׳ כ׳ ל״ה, כַּנִּזְכָּר לְעֵיל; אִם כֵּן, מַהֲלַךְ רֶבַע הַיּוֹם — י״ד חֲלָקִים, מ״ז שְׁנִיּוֹת, ה׳ שְׁלִישִׁיּוֹת, ח׳ רְבִיעִיּוֹת, מ״ה חֲמִישִׁיּוֹת; וּמַהֲלַךְ שָׁעָה אַחַת — ב׳ חֲלָקִים, כ״ז שְׁנִיּוֹת, נ׳ שְׁלִישִׁיִּים, נ״א רְבִיעִיִּים, כ״ז חֲמִישִׁיִּים, ל׳ שִׁשִּׁיִּים; וּמַהֲלַךְ רֶבַע שָׁעָה — ל״ו שְׁנִיִּים, נ״ז שְׁלִישִׁיִּים, מ״ב רְבִיעִיִּים, נ״א חֲמִישִׁיִּים, נ״ב שִׁשִּׁיִּים, ל׳ שְׁבִיעִיִּים; וּמַהֲלַךְ מִינוּ״ט — שֶׁהוּא אֶחָד מִשִּׁשִּׁים בְּשָׁעָה, שֶׁהוּא ח״י חֲלָקִים מֵחֶלְקֵי תתר״ף בְּשָׁעָה — הוּא ב׳ שְׁנִיִּים, כ״ז שְׁלִישִׁיִּים, נ׳ רְבִיעִיִּים, נ״א חֲמִישִׁיִּים, כ״ז שִׁשִּׁיִּים, ל׳ שְׁבִיעִיִּים; וּמַהֲלַךְ שְׁלִישׁ מִינוּ״ט — מ״ט שְׁלִישִׁיִּים, ט״ז רְבִיעִיִּים, נ״ז חֲמִישִׁיִּים, ט׳ שִׁשִּׁיִּים, י׳ שְׁבִיעִיִּים. וְהִנֵּה מַהֲלַךְ שס״ה יוֹם — שֶׁהוּא שס״ה פַּעַם נ״ט ח׳ כ׳ ל״ה — עוֹלֶה שנ״ט מַעֲלוֹת, מ״ה חֲלָקִים, מ״ה שְׁנִיּוֹת, י״ב שְׁלִישִׁיּוֹת, נ״ה רְבִיעִיּוֹת. וּמַהֲלַךְ ה׳ שָׁעוֹת תתי״ו חֲלָקִים: אִלּוּ הָיָה ו׳ שָׁעוֹת, שֶׁהוּא רֶבַע הַיּוֹם — כְּבָר יָדַעְנוּ מַהֲלָכוֹ; אֲבָל חָסֵר רס״ד חֲלָקִים; וְהִנֵּה ר״ע חֲלָקִים הוּא רֶבַע שָׁעָה, וְחָסֵר מ׳ חֲלָקִים, שֶׁהוּא שְׁלִישׁ מִינוּ״ט; נְנַכֶּה שְׁלִישׁ מִינוּ״ט — שֶׁכְּבָר יָדַעְנוּ מַהֲלָכוֹ — מִמַּהֲלַךְ רֶבַע שָׁעָה, שֶׁגַּם כֵּן נִתְבָּאֵר לְמַעְלָה, וְיִשָּׁאֵר ל״ו שְׁנִיִּים, ח׳ שְׁלִישִׁיִּים, כ״ה רְבִיעִיִּים, נ״ד חֲמִישִׁיִּים, מ״ג שִׁשִּׁיִּים, כ׳ שְׁבִיעִיִּים. נְנַכֶּה זֶה
