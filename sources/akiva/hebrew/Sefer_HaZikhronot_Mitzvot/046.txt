~ כְּשֶׁיִּבְדְּקוּהוּ, אוֹ שֶׁיִּפָּסֵל מֵעֲבֵירָה, וְכַהַאי גַּוְנָא, וְיִהְיֶה הוּא רָאוּי לְהָעִיד. וְכֵן רַשָּׁאִים ב׳ כִּתֵּי עֵדִים לֵילֵךְ בְּיַחַד וּלְחַלֵּל שַׁבָּת, דִּשְׁמָּא תִּפָּסֵל כַּת אַחַת. וְנִרְאֶה דִּכִתּוֹת הַרְבֵּה — וַדַּאי דְּאֵין רַשָּׁאִים, דְּאִי אֶפְשָׁר דִּבְכָל הָנֵי לֹא יִמְצָא אַחַת הַכְּשֵׁרָה (כְּמוֹ שֶׁכָּתַבְתִּי לְמַעְלָה אַדִּבְרֵי רַשִׁ״י בְּאַרְבָּעִים זוּג); אֶלָּא דִּלְפִי זֶה צָרִיךְ עִיּוּן: אִם כֵּן נָתַתָּ דְּבָרֶיךָ לְשִׁעוּרִים, וְאֵיזֶה נִקְרָא כִּתּוֹת הַרְבֵּה? וְלָכֵן אֶפְשָׁר דִּבְכָל גַּוְנֵי שַׁפִּיר דָּמֵי. וְאִם הָיָה הָעֵד שֶׁרָאָה אֶת הַחֹדֶשׁ בְּלֵילֵי שַׁבָּת חוֹלֶה, שֶׁאֵין יָכוֹל לֵילֵךְ — מַרְכִּיבִין אוֹתוֹ עַל הַחֲמוֹר, וַאֲפִלּוּ בְּמִטָּה. וְאִם יֵשׁ לָהֶם אוֹרְבִים בַּדֶּרֶךְ מִן הַצְּדוֹקִים הָאֶפִּיקוֹרְסִים לְעַכֵּב הֲלִיכָתָם — לוֹקְחִין בְּיָדָם כְּלֵי זַיִן, אַף אוֹתָם שֶׁאָסְרוּ לְטַלְטְלָם בְּשַׁבָּת. וְאִם הָיָה דֶּרֶךְ רְחוֹקָה (כְּמַהֲלַךְ)..... (נִמְחַק, וְנִרְאֶה דְּהָיָה כָּתוּב "יוֹם וָלַיְלָה", וְאוּלַי הָיָה כָּתוּב "מִיל") — לוֹקְחִים בְּיָדָם מְזוֹנוֹת. (וְנִרְאֶה לִי: כָּל דְּבָרִים אֵלּוּ, אִם אֶפְשָׁר לַעֲשׂוֹתָם עַל יְדֵי עוֹבֵד כּוֹכָבִים — וַדַּאי יַעֲשׂוּ עַל יָדוֹ, וְלֹא עַל יְדֵי יִשְׂרָאֵל, כֵּיוָן דְּאֵינוֹ אֶלָּא מַכְשִׁירֵי מִצְוָה; וּבְכָל עִנְיָן שֶׁיִּרְאֶה שֶׁיּוּכַל לְהַקְטִין הָאִסּוּר — מִדְּאוֹרַיְתָא לִדְרַבָּנָן, וְכַהַנֵּ״ל — וַדַּאי יַעֲשֶׂה):

**וּבָרִאשׁוֹנָה** לֹא הָיוּ זָזִין כָּל הַיּוֹם מִן הֶחָצֵר שֶׁבָּאוּ בּוֹ לְהָעִיד שָׁם, כְּדִין הַיּוֹצֵא לַתְּחוּם; עַד שֶׁבָּא רַבָּן גַּמְלִיאֵל הַזָּקֵן וְהִתְקִין שֶׁיִּהְיוּ מְהַלְּכִין אַלְפַּיִם אַמָּה לְכָל רוּחַ, כְּדֵי שֶׁלֹּא יִמָּנְעוּ מִלָּבוֹא; וְכֵן בְּכָל יוֹצֵא בִּרְשׁוּת, כְּמוֹ שֶׁיִּתְבָּאֵר בִּמְקוֹמוֹ. וּבְכָל עִנְיָן צְרִיכִים הָעֵדִים לֵילֵךְ לְהָעִיד בְּשַׁבָּת, בֵּין שֶׁרָאוּהוּ קָטָן בֵּין שֶׁרָאוּהוּ גָּדוֹל. וְנִרְאֶה: לְכָל לֹא יֹאמְרוּ "כְּשֵׁם שֶׁרְאִינוּהוּ אֲנַחְנוּ, רָאוּהוּ אֲחֵרִים, וְאֵין אָנוּ צְרִיכִים לְחַלֵּל הַשַּׁבָּת"; אֶלָּא מְחַלְּלִים כָּל מִי שֶׁיִּרְאֶה הַלְּבָנָה, וְרָאוּי לְהָעִיד, וְיִהְיֶה בֵּינוֹ וּבֵין מְקוֹם הַבֵּית דִּין לַיְלָה וָיוֹם אוֹ פָּחוֹת — מִצְוָה עָלָיו לְחַלֵּל אֶת הַשַּׁבָּת וְלֵילֵךְ לְהָעִיד. (נִרְאֶה לְטַעְמַאי: אֲפִלּוּ קָרוֹב מְאֹד לִמְקוֹם בֵּית דִּין, דִּשְׁמָּא שָׁם עָבִים כִּסּוּהָ וְלֹא נִרְאֵית.) וְנִרְאֶה לִי: אִם בַּדֶּרֶךְ הִילּוּכוֹ מָצְאוּ אֶחָד הַהוֹלֵךְ מִמְּקוֹם הַבֵּית דִּין (עַל פִּי עֵרוּב, אוֹ בִּרְשׁוּת לְצֹרֶךְ), וְאָמַר לוֹ שֶׁכְּבָר קִדְּשׁוּ בֵּית דִּין עַל פִּי עֵדִים — דְּאֶפְשָׁר דְּלֹא יְחַלֵּל שׁוּב לֵילֵךְ, אִם הוּא עֲדַיִן יוֹתֵר מִתְּחוּם; אֲבָל אִם אֵינוֹ יוֹתֵר מִתְּחוּם — וַדַּאי רַשַּׁאי, כַּנִּזְכָּר לְעֵיל, דְּהִתְקִין רַבָּן גַּמְלִיאֵל הַזָּקֵן שֶׁיֵּלֵךְ אַלְפַּיִם אַמָּה. וְצָרִיךְ עִיּוּן, דְּאֶפְשָׁר גַּם בָּזֶה חָיְישִׁינַן שֶׁמָּא יִמָּנַע:

**לֹא** בָּאוּ עֵדִים — עוֹשִׂים חֹדֶשׁ מְעֻבָּר. וּבָרִאשׁוֹנָה הָיוּ מְקַבְּלִים עֵדוּת הַחֹדֶשׁ כָּל הַיּוֹם שְׁלֹשִׁים. פַּעַם אַחַת נִשְׁתָּהוּ הָעֵדִים מִלָּבוֹא עַד בֵּין הָעַרְבַּיִם, וְנִתְקַלְקְלוּ בַּמִּקְדָּשׁ, וְלֹא יָדְעוּ מַה יַּעֲשׂוּ: אִם יַעֲשׂוּ עוֹלָה שֶׁל בֵּין הָעַרְבַּיִם —
