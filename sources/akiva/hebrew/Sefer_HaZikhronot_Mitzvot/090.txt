~ שְׁנַת ז׳ — שֶׁהוּא הַתְחָלַת ב׳ שָׁנִים הַשְּׁנִיִּים מִן ד׳ הַשְּׁנִיִּים — בִּתְחִלַּת יוֹם ד׳. וּכְפִי סִימָן אזג״י ואט״ד — יָצָא לָנוּ תְּקוּפַת תִּשְׁרֵי שֶׁל שְׁנַת ב׳ בְּט׳ שָׁעוֹת בְּלֵיל ו׳, בְּהֶחְלֵף הַיּוֹם לַלַּיְלָה, כְּמוֹ שֶׁנִּתְבָּאֵר. וְכֵן הַדִּין לְכָל הַשָּׁנִים וּלְכָל הַתְּקוּפוֹת:

**וְאִם** לֹא יָדַעְנוּ מִן סִימָנִים דבי״ז הגא״ו, וד״ב זהג״א, אזג״י ואט״ד הַנִּזְכָּרִים — יֵשׁ דֶּרֶךְ אַחֵר יָשָׁר וְנָקֵל לַחְשׁוֹב הַתְּקוּפוֹת: בִּקַּחַת לְכָל שָׁנָה שֶׁנִּשְׁאֲרָה מִכ״ח שָׁנָה יוֹם אֶחָד וְשִׁשָּׁה שָׁעוֹת (שֶׁהוּא הַמֶּרְחָק מִשָּׁנָה, כַּנִּזְכָּר לְעֵיל), וּנְקַבֵּץ הַכֹּל, וְנוֹסִיף עוֹד ג׳ יָמִים — שֶׁעַד תְּחִלַּת תְּקוּפָה רִאשׁוֹנָה, שֶׁהוּא בִּתְחִלַּת לֵיל ד׳ — וְנַשְׁלִיךְ הַכֹּל ז׳ ז׳, וְהַנִּשְׁאָר — מִן הַיָּמִים וּמִן הַשָּׁעוֹת: כָּךְ וְכָךְ שָׁעוֹת, וְכָךְ וְכָךְ יָמִים בַּשָּׁבוּעַ — הוּא תְּקוּפַת נִיסָן שֶׁל שָׁנָה שֶׁאַחַר כָּךְ; וּתְקוּפַת תַּמּוּז לְאַחֲרָיו — בְּז׳ שָׁעוֹת וּמֶחֱצָה, וְתִשְׁרֵי — בְּט״ו שָׁעָה, וְטֵבֵת — בְּכ״ב שָׁעָה וּמֶחֱצָה. וּלְפִי חֶשְׁבּוֹן זֶה — תְּקוּפַת נִיסָן אֵינָהּ לְעוֹלָם אֶלָּא בְּרֹאשׁ הַלַּיְלָה אוֹ הַיּוֹם, אוֹ בַּחֲצִי הַלַּיְלָה אוֹ הַיּוֹם; וְתַמּוּז לְעוֹלָם אֵינָהּ אֶלָּא בְּשָׁעָה וּמֶחֱצָה, אוֹ בְּז׳ וּמֶחֱצָה, בֵּין בַּיּוֹם בֵּין בַּלַּיְלָה; וְתִשְׁרֵי אֵינָהּ אֶלָּא מִג׳ שָׁעוֹת אוֹ בְּט׳, בֵּין בַּיּוֹם בֵּין בַּלַּיְלָה; וְטֵבֵת — בְּד׳ וּמֶחֱצָה אוֹ בְּי׳ וּמֶחֱצָה, בֵּין בַּיּוֹם בֵּין בַּלַּיְלָה. וְזֶה נִמְשָׁךְ מִמַּה שֶּׁנִּתְבָּאֵר:

**הִנֵּה** בֵּאַרְנוּ חֶשְׁבּוֹן הַתְּקוּפָה בִּימֵי הַשָּׁבוּעַ. אֲבָל כֵּיצַד נַחְשְׁבֶנָּה בִּימֵי הַחֹדֶשׁ? הִנֵּה כְּבָר בֵּאַרְנוּ שֶׁתְּקוּפַת תִּשְׁרֵי רִאשׁוֹנָה הָיְתָה בִּתְחִלַּת שָׁעָה ד׳ עַל יוֹם ד׳; גַּם בֵּאַרְנוּ כִּי מוֹלַד תִּשְׁרֵי הַהוּא הָיָה בִּתְחִלַּת שָׁעָה ג׳ עַל יוֹם ו׳; נִמְצָא קָדְמָה הַתְּקוּפָה לְהַמּוֹלָד יוֹם א׳ וְכ״ג שָׁעוֹת. וְהִנֵּה, תְּחִלַּת חֶשְׁבּוֹנֵנוּ לַתְּקוּפוֹת הוּא מִנִּיסָן שֶׁל תֹּהוּ, שֶׁהָיְתָה אָז הַתְּקוּפָה תְּחִלַּת לֵיל ד׳, כְּמוֹ שֶׁנִּתְבָּאֵר. וְהִנֵּה כְּבָר זָכַרְנוּ כַּמָּה פְּעָמִים כִּי יִתְרוֹן שְׁנַת הַחַמָּה עַל שְׁנַת הַלְּבָנָה הוּא י׳ יָמִים, כ״א שָׁעוֹת, ר״ד חֲלָקִים; אִם כֵּן, יִתְרוֹן חֲצִי שְׁנַת הַחַמָּה עַל חֲצִי שְׁנַת הַלְּבָנָה הוּא ה׳ יָמִים, י׳ שָׁעוֹת, תרמ״ב חֲלָקִים. וּנְחַבֵּר יִתְרוֹן זֶה אֶל א׳ וְכ״ג שָׁעוֹת — שֶׁקָּדְמָה בְּתִשְׁרֵי חַמָּה לַלְּבָנָה — יֵצֵא יִתְרוֹן קְדִימַת תְּקוּפַת חַמָּה לְמוֹלַד הַלְּבָנָה בְּנִיסָן עִם ז׳ יָמִים, ט׳ שָׁעוֹת, תרמ״ב חֲלָקִים:

**וְאַחַר** הוֹדַע לָנוּ תְּקוּפָה רִאשׁוֹנָה שֶׁל נִיסָן קְדִימָתָהּ — הִנֵּה מִמֶּנָּה נִשְׁפּוֹט עַל כָּל הַתְּקוּפוֹת, עִם הַמִּתְבָּאֵר אֵלֵינוּ מִכְּבָר, כִּי יִתְרוֹן הַשָּׁנָה י׳ כ״א ר״ד, וּמֶחֱצִיתָהּ ה׳ י׳ תרמ״ב, וּרְבִיעִיתָהּ — שֶׁהוּא תְּקוּפָה — יִתְרוֹנָהּ ב׳
