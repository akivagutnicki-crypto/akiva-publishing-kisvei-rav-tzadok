~ לַחֹדֶשׁ. לְפִיכָךְ כָּל הַיּוֹם — אַף קֹדֶם הַמּוֹלָד — הוּא רֹאשׁ חֹדֶשׁ; וְנִמְצָא יְמֵי הַחֹדֶשׁ שֶׁעָבַר רַק כ״ט, וְהַשָּׁעוֹת הַיְּתֵרוֹת מִכְּלַל הַחֹדֶשׁ שֶׁאַחַר כָּךְ. וּלְפִי שֶׁהַשָּׁעוֹת הֵן יוֹתֵר מֵחֲצִי הַיּוֹם, כְּמוֹ שֶׁנִּתְבָּאֵר — אִם כֵּן, בְּהִתְקַבֵּץ שְׁעוֹת שֶׁל חֹדֶשׁ רִאשׁוֹן עִם שְׁעוֹת חֹדֶשׁ ב׳ — יִהְיֶה יוֹתֵר מִיּוֹם שָׁלֵם, וְיִהְיֶה הַחֹדֶשׁ הַב׳ ל׳ יוֹם עַד הַמּוֹלָד שֶׁל חֹדֶשׁ ג׳. לָכֵן, לְפִי חֶשְׁבּוֹן זֶה, לְעוֹלָם סִדְרֵי הַחֳדָשִׁים כָּךְ הוּא: חֹדֶשׁ א׳ מִכ״ט יוֹם, וְחֹדֶשׁ א׳ מִל׳ יוֹם. וְזֶה שֶׁל כ״ט יוֹם נִקְרָא "חָסֵר", וְשֶׁל ל׳ יוֹם "מָלֵא". וְיִהְיֶה לְפִי סֵדֶר זֶה: תִּשְׁרֵי — לְעוֹלָם מָלֵא, חֶשְׁוָן — חָסֵר, כִּסְלֵו — מָלֵא, טֵבֵת — חָסֵר, שְׁבָט — מָלֵא, אֲדָר — חָסֵר, נִיסָן — מָלֵא, אִיָּר — חָסֵר, סִיוָן — מָלֵא, תַּמּוּז — חָסֵר, אָב — מָלֵא, אֱלוּל — חָסֵר. וְאִלּוּ הָיָה הַחֹדֶשׁ, מִמּוֹלָד רִאשׁוֹן לְשֶׁל אַחֲרָיו, רַק כ״ט יוֹם וּמֶחֱצָה — הָיָה דַּי בָּזֶה לְהַשְׁווֹת יְמֵי חָדְשֵׁי הַשָּׁנָה לְמוֹלְדֵי הַלְּבָנָה, וְהָיָה יְמֵי שְׁנַת הַלְּבָנָה שנ״ד יוֹם. וּלְפִי שֶׁיֵּשׁ תשצ״ג חֲלָקִים בְּכָל חֹדֶשׁ יוֹתֵר מִן כ״ט יוֹם וּמֶחֱצָה, וּמֵאוֹתָם הַחֲלָקִים מִתְקַבֵּץ לַשָּׁנָה שְׁמוֹנָה שָׁעוֹת תתע״ו חֲלָקִים, כְּמוֹ שֶׁנִּתְבָּאֵר; וּבְג׳ שָׁנִים יִהְיֶה יוֹתֵר מִיּוֹם שָׁלֵם — לְפִיכָךְ הָיָה הַהֶכְרֵחַ לַעֲשׂוֹת פְּעָמִים חֹדֶשׁ אֶחָד מֵאֵלּוּ הַחֲסֵרִים מָלֵא, כְּדֵי מִדַּת יְמֵי הַשָּׁנָה. וְרָצוּ לְתַקֵּן זֶה בִּתְחִלַּת הַשָּׁנָה; לְפִיכָךְ תִּקְּנוּ בַּחֹדֶשׁ הָרִאשׁוֹן הֶחָסֵר — וְהוּא מַרְחֶשְׁוָן — לְמַלְּאוֹתוֹ כְּשֶׁמִּצְטָרְכִים, וְלַעֲשׂוֹת הַשָּׁנָה שנ״ה יוֹם:

**וּלְפִי** שֶׁעַל יְדֵי הָאַרְבָּעָה דְּחִיּוֹת שֶׁזָּכַרְנוּ — מִמֵּילָא, כַּמָּה פְּעָמִים נִדְחֶה רֹאשׁ הַשָּׁנָה מִיּוֹם הַמּוֹלָד לְיוֹם שֶׁלְּאַחֲרָיו, וְשֶׁלְּאַחַר אַחֲרָיו, כְּמוֹ שֶׁנִּתְבָּאֵר, וּמִתְגַּדֶּלֶת הַשָּׁנָה שֶׁעָבְרָה — לְפִיכָךְ לֹא הֻצְרְכוּ לְתַקֵּן לַעֲשׂוֹת בְּכָל ג׳ שָׁנָה, בְּדִמְיוֹן, חֹדֶשׁ חֶשְׁוָן מָלֵא, כְּדֵי לְמַלְּאוֹת הַחִסָּרוֹן שֶׁזָּכַרְנוּ; אֶלָּא מִמֵּילָא הִיא מִתְמַלֵּאת עַל יְדֵי דְּחִיַּת רֹאשׁ הַשָּׁנָה, וְאִי אֶפְשָׁר לַעֲשׂוֹת אֱלוּל לְל״א יוֹם. אֶלָּא מְחַשְּׁבִין תְּחִלַּת הַשָּׁנָה, וְרוֹאִין: אִם רֹאשׁ הַשָּׁנָה צְרִיכָה לִדְחוֹת מִיּוֹם הַמּוֹלָד עַל יְדֵי אַחַת מִד׳ דְּחִיּוֹת שֶׁזָּכַרְנוּ, וְתִהְיֶה הַשָּׁנָה שנ״ה יוֹם — מוֹסִיפִים אוֹתוֹ יוֹם עַל חֹדֶשׁ חֶשְׁוָן, וְיִהְיֶה מָלֵא. וְאַף עַל פִּי שֶׁעַל יְדֵי דְּחִיּוֹת אֵלּוּ הַשָּׁנָה מִתְמַלֵּאת פְּעָמִים רַבּוֹת יוֹתֵר מִכְּפִי הָרָאוּי לְהַשְׁלִים הַח׳ שָׁעוֹת וְתתע״ו חֲלָקִים — מִכָּל מָקוֹם, נֶגֶד זֶה, פְּעָמִים הַרְבֵּה עַל יְדֵי דְּחִיּוֹת אֵלּוּ מִתְקַצֶּרֶת הַשָּׁנָה: כְּשֶׁרֹאשׁ הַשָּׁנָה הָרִאשׁוֹן נִדְחֶה, וְהַשֵּׁנִי עוֹמֵד בִּמְקוֹמוֹ — שֶׁהַשָּׁנָה מִתְקַצֶּרֶת וְאֵינָהּ אֶלָּא שנ״ג יוֹם. וְאָז גַּם כֵּן אֵין עוֹשִׂין אֱלוּל כ״ח יוֹם, שֶׁאֵין חֹדֶשׁ פָּחוֹת מִכ״ט יוֹם וְלֹא יֶתֶר עַל ל׳ יוֹם; אֶלָּא מְחַשְּׁבִין זֶה מִתְּחִלַּת הַשָּׁנָה, וּכְשֶׁרוֹאִין שֶׁתִּתְקַצֵּר — עוֹשִׂין חֹדֶשׁ הָרִאשׁוֹן הַמָּלֵא חָסֵר; וְתִשְׁרֵי, מִפְּנֵי הַמּוֹעֲדוֹת שֶׁבּוֹ, יָדוּעַ לַכֹּל
