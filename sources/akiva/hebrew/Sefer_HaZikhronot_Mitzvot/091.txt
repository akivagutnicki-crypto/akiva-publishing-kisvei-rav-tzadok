~ י״ז שכ״א; וְיִתְרוֹן הַמַּחֲזוֹר קָטָן — שָׁעָה אַחַת וְתפ״ה חֲלָקִים. וְאִם כֵּן, אִם נַחְפּוֹץ לֵידַע תְּקוּפַת נִיסָן שֶׁל אֵיזוֹ שָׁנָה — נַחְשׁוֹב לְכָל מַחֲזוֹר קָטָן א׳ תפ״ה הַנִּזְכָּר לְעֵיל, וּלְכָל שָׁנָה י׳ כ״א ר״ד, וּלְקִבּוּצָם כֻּלָּם; וְנִגְרַע כָּל חֹדֶשׁ עִבּוּר שֶׁעָבַר בְּמַחֲזוֹר קָטָן זֶה; וּמֵהַמּוֹתָר נִגְרַע ז׳ ט׳ תרמ״ב, שֶׁקָּדַם לְמוֹלָד רִאשׁוֹן; וְהַמּוֹתָר — נִתְּנֵהוּ עַל מוֹלַד אוֹתוֹ חֹדֶשׁ, וְכָךְ וְכָךְ יָמִים וְשָׁעוֹת בַּחֹדֶשׁ יִהְיֶה הַתְּקוּפָה. וְכֵן לִתְקוּפַת תַּמּוּז — נוֹסִיף עוֹד יִתְרוֹן הַתְּקוּפָה, שֶׁהוּא ב׳ י״ז שכ״א; וְכֵן לִשְׁאָר הַתְּקוּפוֹת:

**כֵּיצַד?** הֲרֵי שֶׁנִּרְצֶה לֵידַע תְּקוּפַת תִּשְׁרֵי שֶׁל שְׁנַת הַתר״ב לִבְרִיאַת עוֹלָם, הַבָּאָה עָלֵינוּ לְטוֹבָה — כַּמָּה תִּהְיֶה בַּחֹדֶשׁ. וְהִיא שְׁנַת ט״ז לְמַחֲזוֹר קָטָן רצ״ה לִבְרִיאַת עוֹלָם; וּלְפִי שֶׁלַּתְּקוּפוֹת אָנוּ חוֹשְׁבִין מִנִּיסָן — אִם כֵּן הִיא מֶחֱצָה שְׁנַת הַט״ו לַמַּחֲזוֹר. וְנִקַּח לְרצ״ד מַחֲזוֹרִים שְׁלֵמִים שֶׁעָבְרוּ, לְכָל מַחֲזוֹר קָטָן — שָׁעָה אַחַת וְתפ״ה חֲלָקִים; וְלִי״ד שָׁנִים שְׁלֵמִים שֶׁעָבְרוּ — לְכָל שָׁנָה י׳ כ״א ר״ד; וְלַחֲצִי שְׁנַת ט״ו — נִקַּח ה׳ י׳ תרמ״ב. תְּהֵא קְבוּצָתָם יַחַד: קע״ה יָמִים, י״ג שָׁעוֹת, רפ״ה חֲלָקִים. נִגְרַע מֵהֶם ה׳ חָדְשֵׁי עִבּוּר שֶׁעָבְרוּ בְּט״ו שָׁנִים אֵלּוּ — וְהֵם גו״ח א״ד, לְפִי סֵדֶר גו״ח אדז״ט מְעֻבָּרוֹת — וְכָל חֹדֶשׁ כ״ט י״ב תשצ״ג; אִם כֵּן קְבוּצָתָם: קמ״ז יָמִים, ט״ו שָׁעוֹת, תשכ״ה חֲלָקִים. וְגַם נִגְרַע ז׳ ט׳ תרמ״ב — וִיהִי הַיִּתְרוֹן כ׳ יָמִים, י״ב שָׁעוֹת, וְחֵלֶק אֶחָד. וְתִתְּנֵהוּ עַל הַמּוֹלָד, וּבְכָךְ יָמִים לַחֹדֶשׁ הוּא הַתְּקוּפָה:

**וְהִנֵּה** כְּבָר חִשַּׁבְנוּ לְעֵיל, בְּחֶשְׁבּוֹן הַמּוֹלָדוֹת, מוֹלַד תִּשְׁרֵי שֶׁל שָׁנָה זוֹ — שֶׁסִּימָנוֹ ד׳ ב׳ אֶלֶף ע״ט; וְנוֹסִיף עָלָיו כ׳ י״ב א׳, וְנַשְׁלִיךְ הַתתר״ף חֲלָקִים, וְנַעֲשֶׂה שָׁעָה; וְכָל ז׳ יָמִים — שָׁבוּעַ; יֵצֵא זְמַן הַתְּקוּפָה בִּימֵי הַשָּׁבוּעַ: בְּיוֹם ד׳, בְּסוֹף שָׁעָה ט״ו; וּבְיוֹם הַכ׳ לַחֹדֶשׁ (כִּי רֹאשׁ חֹדֶשׁ יִהְיֶה בְּיוֹם הַמּוֹלָד, בְּיוֹם ה׳). וְהוּא מְכֻוָּן לְחֶשְׁבּוֹן אזג״י ואט״ד שֶׁזָּכַרְנוּ לְמַעְלָה: כִּי תְּקוּפַת נִיסָן תר״א הָיְתָה הַתְחָלַת מַחֲזוֹר גָּדוֹל שֶׁל כ״ח שָׁנָה, וְהָיָה תְּקוּפָתָהּ בִּתְחִלַּת לֵיל ד׳; אִם כֵּן, תְּקוּפַת תִּשְׁרֵי — בְּסוֹף ג׳ שָׁעוֹת עַל יוֹם ד׳, שֶׁהוּא, עִם י״ב שְׁעוֹת הַלַּיְלָה, ט״ו שָׁעוֹת. וְאֵין צֹרֶךְ מֵעַתָּה לְחֶשְׁבּוֹן אזג״י ואט״ד; אֶלָּא לְפִי שֶׁזֶּה חֶשְׁבּוֹן אָרוֹךְ, לְדַקְדֵּק בַּחֲלָקִים וְשָׁעוֹת כָּל כָּךְ — לְפִיכָךְ טוֹב יוֹתֵר לַחְשׁוֹב מְקוֹם יוֹם וְשָׁעָה שֶׁל הַתְּקוּפָה בִּימֵי הַשָּׁבוּעַ עַל פִּי חֶשְׁבּוֹן אזג״י ואט״ד הַקָּצָר; וְאַחַר כָּךְ שׁוּב לֹא יִצְטָרֵךְ לְדַקְדֵּק כָּל כָּךְ בַּחֲלָקִים וְשָׁעוֹת — כִּי אִם יִפְקְדוּ אֵיזֶה יָמִים מֵהַחֶשְׁבּוֹן, יוּכַל לָדַעַת כְּשֶׁלֹּא יַגִּיעוּ לִימֵי הַשָּׁבוּעַ, וְיוֹסִיף עַד שֶׁיַּגִּיעַ לְאוֹתוֹ יוֹם בִּימֵי הַשָּׁבוּעַ. וּבְעִתִּים
