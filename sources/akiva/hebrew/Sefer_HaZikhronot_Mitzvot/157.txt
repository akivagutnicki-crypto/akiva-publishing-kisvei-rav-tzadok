~ חַטָּאתָם אֲשֶׁר עָשׂוּ". וְזֶהוּ מִצְוַת הַתְּשׁוּבָה, שֶׁהוּכְפַּל אַזְהָרָתָהּ בַּתּוֹרָה (דְּבָרִים ד׳ ל׳): "וְשַׁבְתָּ עַד ה׳ אֱלֹקֶיךָ" וְגוֹ׳; וְשָׁם (ל׳ ב׳): "וְשַׁבְתָּ עַד ה׳ אֱלֹקֶיךָ וְשָׁמַעְתָּ בְקֹלוֹ" וְגוֹ׳, "וְאַתָּה תָשׁוּב" וְגוֹ׳; וּבִשְׁאָר מְקוֹמוֹת. וְכָל הַנְּבִיאִים מַזְהִירִים עָלֶיהָ: יְשַׁעְיָה עָלָיו הַשָּׁלוֹם מַכְרִיז וְאוֹמֵר (יְשַׁעְיָה א׳): "רַחֲצוּ הִזַּכּוּ" וְגוֹ׳; וְאוֹמֵר (שָׁם כ״א): "אִם תִּבְעָיוּן בְּעָיוּ, שֻׁבוּ אֵתָיוּ"; וְאוֹמֵר (שָׁם ל״א): "שׁוּבוּ לַאֲשֶׁר הֶעְמִיקוּ סָרָה"; (שָׁם מ״ד): "שׁוּבָה אֵלַי כִּי גְאַלְתִּיךָ"; וְעוֹד בִּשְׁאָר הַמְּקוֹמוֹת. וְיִרְמְיָה עָלָיו הַשָּׁלוֹם אוֹמֵר (יִרְמְיָה ג׳): "וְשׁוֹב אֵלַי, נְאֻם ה׳" וְגוֹ׳; "אֵלַי תָּשׁוּב" וְגוֹ׳; "שׁוּבָה מְשֻׁבָה יִשְׂרָאֵל" וְגוֹ׳; "אַךְ דְּעִי עֲוֹנֵךְ" וְגוֹ׳; "שׁוּבוּ בָנִים שׁוֹבָבִים, נְאֻם ה׳" וְגוֹ׳; "שׁוּבוּ בָּנִים שׁוֹבָבִים, אֶרְפָּה מְשׁוּבֹתֵיכֶם" וְגוֹ׳; וְשָׁם (ד׳): "אִם תָּשׁוּב יִשְׂרָאֵל, נְאֻם ה׳, אֵלַי תָּשׁוּב" וְגוֹ׳; "הִמֹּלוּ לַה׳" וְגוֹ׳; "כַּבְּסִי מֵרָעָה לִבֵּךְ" וְגוֹ׳; וְעוֹד בִּשְׁאָר מְקוֹמוֹת הַרְבֵּה. וִיחֶזְקֵאל — בְּכַמָּה מְקוֹמוֹת מִסִּפְרוֹ, וּבְיִחוּד (יְחֶזְקֵאל י״ח) בֵּאֵר מַאֲמָר אָרֹךְ בְּעִנְיַן הַתְּשׁוּבָה, בְּמַאֲמַר "וְהָרָשָׁע כִּי יָשׁוּב מִכָּל חַטֹּאתָיו" וְגוֹ׳; וְכֵן (שָׁם ל״ג): "אִם אֶחְפֹּץ בְּמוֹת הָרָשָׁע, כִּי אִם בְּשׁוּב" וְגוֹ׳. וְכֵן הוֹשֵׁעַ (ו׳): "לְכוּ וְנָשׁוּבָה" וְגוֹ׳; (שָׁם י״ד): "שׁוּבָה יִשְׂרָאֵל עַד ה׳ אֱלֹקֶיךָ" וְגוֹ׳. וְיוֹאֵל (ב׳): "וְשׁוּבוּ אֶל ה׳ אֱלֹקֵיכֶם" וְגוֹ׳. וְעָמוֹס (ה׳): "כֹּה אָמַר ה׳ לְבֵית יִשְׂרָאֵל: דִּרְשׁוּנִי וִחְיוּ". וְיוֹנָה (ג׳): "כִּי שָׁבוּ מִדַּרְכָּם הָרָעָה, וַיִּנָּחֶם הָאֱלֹקִים עַל הָרָעָה" וְגוֹ׳. וְכֵן כָּל הַנְּבִיאִים כֻּלָּם וְהַכְּתוּבִים — מְלֵאִים מֵעִנְיַן הַתְּשׁוּבָה; כִּי הוּא עִקַּר הַכֹּל, "כִּי אָדָם אֵין צַדִּיק בָּאָרֶץ אֲשֶׁר יַעֲשֶׂה טּוֹב וְלֹא יֶחֱטָא". וְחָשְׁבוּ רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה רַק ד׳ שֶׁמֵּתוּ בְּעֶטְיוֹ שֶׁל נָחָשׁ (פֶּרֶק קַמָּא דְּבָבָא בַּתְרָא); אֲבָל לְכָל הָאָדָם (שַׁבָּת נ״ה) — אֵין מִיתָה בְּלֹא חֵטְא; וְאִם כֵּן, צֹרֶךְ הַתְּשׁוּבָה לְכָל אָדָם:

**וְלָכֵן** חָשְׁבוּ זִכְרוֹנָם לִבְרָכָה (פֶּרֶק "מְקוֹם שֶׁנָּהֲגוּ", וּבִבְרֵאשִׁית רַבָּה פָּרָשָׁה א׳) הַתְּשׁוּבָה מֵהַדְּבָרִים שֶׁנִּבְרְאוּ קֹדֶם בְּרִיאַת הָעוֹלָם — לְפִי שֶׁלֹּא הָיָה לָעוֹלָם קִיּוּם בִּלְעָדֶיהָ. וְזָכְרוּ (פֶּרֶק בַּתְרָא דְּיוֹמָא): "גְּדוֹלָה תְּשׁוּבָה, שֶׁמַּגַּעַת עַד כִּסֵּא הַכָּבוֹד, שֶׁנֶּאֱמַר: 'שׁוּבָה יִשְׂרָאֵל עַד ה׳ אֱלֹקֶיךָ'". וְהוּא — לְמַה שֶּׁחִלְּקוּ חַכְמֵי הַמֶּחְקָר הָעוֹלָמוֹת בִּכְלָלָם לְג׳, שֶׁהֵם: עוֹלַם הַכִּסֵּא, וְעוֹלַם הַמַּלְאָכִים, וְעוֹלַם הַגַּלְגַּלִּים. וְהִנֵּה, כְּבָר נוֹדָע מַה שֶּׁאָמְרוּ רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה (אֵיכָה רַבָּתִי, פָּסוּק "וַיֵּלְכוּ בְלֹא כֹחַ"): "בִּזְמַן שֶׁיִּשְׂרָאֵל עוֹשִׂין רְצוֹנוֹ שֶׁל מָקוֹם — מוֹסִיפִין כֹּחַ בַּפָּמַלְיָא שֶׁל מַעְלָה, שֶׁנֶּאֱמַר 'בֵּאלֹקִים נַעֲשֶׂה חָיִל'; וּבִזְמַן שֶׁאֵין עוֹשִׂין רְצוֹנוֹ שֶׁל מָקוֹם — כִּבְיָכוֹל מַתִּישִׁין כֹּחַ שֶׁל מַעְלָה, שֶׁנֶּאֱמַר 'צוּר יְלָדְךָ תֶּשִׁי'". וְזֶה הַכֹּל בְּעוֹלַם הַמַּלְאָכִים. אֲבָל בְּעוֹלַם הַכִּסֵּא — שֶׁהִיא, כְּפִי כִּנּוּי הַמֶּחְקָרִים, לִכְבוֹדוֹ יִתְבָּרַךְ וְעַצְמוּתוֹ — אֶצְלוֹ וַדַּאי לֹא שַׁיָּךְ שׁוּם תּוֹסֶפֶת וְתִשּׁוּי כֹּחַ; כְּמוֹ שֶׁאָמַר הַכָּתוּב: "אִם צָדַקְתָּ מַה תִּתֶּן לוֹ" וְגוֹ׳. וְזֶהוּ מַה שֶּׁזָּכְרוּ זִכְרוֹנָם לִבְרָכָה (יְרוּשַׁלְמִי פֶּרֶק ב׳ דְּמַכּוֹת): כִּי שָׁאֲלוּ לַחָכְמָה וְלַנְּבוּאָה וְלַתּוֹרָה — מָה עֹנֶשׁ הַחוֹטֵא?
