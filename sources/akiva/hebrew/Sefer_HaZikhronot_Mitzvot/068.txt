~ שֶׁלֹּא רָצָה לְהַטְרִיחַ עֵדִים, שֶׁאֵין יוֹדְעִים אָנָה מְקוֹם רֹאשׁ בֵּית דִּין; אֲבָל לְעִבּוּר שָׁנָה, דְּאֵין צָרִיךְ הֲלִיכַת עֵדִים — לֹא תִּיקֵּן. וְעוֹד יֵשׁ לוֹמַר: לְפִי מַה שֶּׁכָּתַב הוּא בְּעַצְמוֹ, דִּצְרִיךְ שֶׁיַּסְכִּים בְּאוֹתוֹ יוֹם — אִם כֵּן בְּקִדּוּשׁ הַחֹדֶשׁ, דְּאִי אֶפְשָׁר כְּשֶׁהוּא בְּמָקוֹם אַחֵר שֶׁיַּסְכִּים בְּאוֹתוֹ יוֹם — לְכָךְ תִּיקֵּן; מַה שֶּׁאֵין כֵּן בְּעִבּוּר, דְּיֵשׁ זְמַן מְרֻבֶּה. כֵּן נִרְאֶה לַעֲנִיּוּת דַּעְתִּי; וְכֵן מַשְׁמַע מִלְּשׁוֹן רַשִׁ״י, כְּמוֹ שֶׁכָּתַב הַטּוּרֵי אֶבֶן בְּעַצְמוֹ):

**וְאַחַר** הַחֻרְבָּן, שֶׁבָּטְלָה סַנְהֶדְרִין — בֵּית דִּין הַסְּמוּכִים בְּאֶרֶץ יִשְׂרָאֵל, סָמוּךְ מִפִּי סָמוּךְ עַד מֹשֶׁה רַבֵּינוּ עָלָיו הַשָּׁלוֹם, כְּדִין הַסְּמִיכָה שֶׁיִּתְבָּאֵר בִּמְקוֹמוֹ — הֵם שֶׁנָּתְנוּ לָהֶם סַנְהֶדְרִין רְשׁוּת לְקַדֵּשׁ. זוּלָתָם — אֵין שׁוּם בֵּית דִּין רַשַּׁאי לְקַדֵּשׁ; שֶׁכָּךְ נֶאֱמַר לְמֹשֶׁה רַבֵּינוּ עָלָיו הַשָּׁלוֹם וּלְאַהֲרֹן: "הַחֹדֶשׁ הַזֶּה לָכֶם רֹאשׁ חֳדָשִׁים", וּמִפִּי הַשְּׁמוּעָה לָמְדוּ שֶׁפֵּרוּשׁ הַדָּבָר: עֵדוּת זוֹ תְּהֵא מְסוּרָה לָכֶם, וְכָל הָעוֹמֵד אַחֲרֵיכֶם בִּמְקוֹמְכֶם (כֵּן הוּא דַּעַת הָרַמְבַּ״ם, רֵישׁ פֶּרֶק ה׳. וְעַיֵּן בִּדְבָרָיו בְּסֵפֶר הַמִּצְוֹת (מִצְוַת עֲשֵׂה קנ״ג), וְעַיֵּן בְּרַמְבַּ״ן שָׁם, הֶאֱרִיךְ לְהַשִּׂיג עָלָיו; וּמַסְקָנַת דַּעְתּוֹ זִכְרוֹנוֹ לִבְרָכָה: שֶׁאֵין זֶה מָסוּר לְסַנְהֶדְרֵי גְּדוֹלָה, אֶלָּא לְכָל בֵּית דִּין סְמוּכִים; וּפֵרוּשׁ "לָכֶם" — סְמוּכִים כְּמוֹתְכֶם; וְאֵין צָרִיךְ לְסַנְהֶדְרִין אֲפִלּוּ בִּזְמַן שֶׁבֵּית הַמִּקְדָּשׁ קַיָּם מִן הַתּוֹרָה. אֶלָּא שֶׁמִּכָּל מָקוֹם, בִּזְמַן שֶׁבֵּית הַמִּקְדָּשׁ קַיָּם, שֶׁהָיָה בֵּית דִּין הַגָּדוֹל בִּמְקוֹמָם — וְלָהֶם הָרְשׁוּת הַנְּתוּנָה מִן הַכָּתוּב, "עַל פִּי הַדָּבָר אֲשֶׁר יוֹרוּךָ" — לֹא הָיָה רְשׁוּת לְשׁוּם אָדָם לְעַבֵּר אוֹ לְקַדֵּשׁ אֶלָּא לָהֶם אוֹ בִּרְשׁוּתָן; לֹא לִהְיוֹת שְׁאָר הַסְּמוּכִים פְּסוּלִים, אֶלָּא לְפִי שֶׁהָיָה רְשׁוּת וְהַסְכָּמַת כָּל יִשְׂרָאֵל נְתוּנָה לָהֶם וּנְטוּלָה מִכָּל בֵּית דִּין אַחֵר. אֲבָל כֵּיוָן שֶׁגָּלוּ וּבָטַל כֹּחַ הַמִּשְׁפָּט מִסַּנְהֶדְרִין, בְּבִטּוּל הַמָּקוֹם, וְנִיטְּלָה רְשׁוּת מֵהֶם — חָזַר הָרְשׁוּת לְקַדֵּשׁ לְכָל ג׳ סְמוּכִין מִיִּשְׂרָאֵל; וְאִם יֵשׁ בֵּית דִּין גָּדוֹל בְּיִשְׂרָאֵל — בֵּין שֶׁיִּהְיֶה בְּאֶרֶץ יִשְׂרָאֵל, בֵּין בְּחוּץ לָאָרֶץ — הָרְשׁוּת נְתוּנָה לְאוֹתוֹ בֵּית דִּין, עַד כָּאן תֹּכֶן דְּבָרָיו. וְהִנֵּה, לְפִי מַה שֶּׁנִּתְבָּאֵר לְעֵיל בְּדַעַת הָרַמְבַּ״ם, דְּכָל סְמוּכִים עָבְדֵי שְׁלִיחוּתַיְהוּ דְּסַנְהֶדְרִין — נִתְיַשְּׁבוּ כַּמָּה קֻשְׁיוֹת הָרַמְבַּ״ן שָׁם; וְעַיֵּן שָׁם בְּסֵפֶר מְגִלַּת אֶסְתֵּר):

**וְאֵין** מְחַשְּׁבִין (זֶהוּ דַּעַת הָרַמְבַּ״ם; אֲבָל תּוֹסָפוֹת סַנְהֶדְרִין (י״א עַמּוּד ב׳), דִּבּוּר הַמַּתְחִיל "אֵין", כָּתְבוּ דְּחִשּׁוּב בְּעָלְמָא שַׁפִּיר דָּמֵי, בִּלְבַד שֶׁיִּקְבְּעוּ בְּאֶרֶץ יִשְׂרָאֵל) וְקוֹבְעִין חֳדָשִׁים אֶלָּא בְּאֶרֶץ יִשְׂרָאֵל, שֶׁנֶּאֱמַר: "כִּי מִצִּיּוֹן תֵּצֵא תוֹרָה, וּדְבַר ה׳ מִירוּשָׁלָיִם". וְאִם הָיָה אָדָם גָּדוֹל בְּחָכְמָה, וְנִסְמַךְ בְּאֶרֶץ יִשְׂרָאֵל, וְיָצָא לְחוּץ לָאָרֶץ, וְלֹא הִנִּיחַ בְּאֶרֶץ יִשְׂרָאֵל כְּמוֹתוֹ — הֲרֵי זֶה מְחַשֵּׁב וְקוֹבֵעַ חֳדָשִׁים בְּחוּץ לָאָרֶץ. אֲבָל אִם הִנִּיחַ כְּמוֹתוֹ, אוֹ שֶׁהִנִּיחוֹ קָטָן וְנִתְגַּדֵּל בַּתּוֹרָה וְנַעֲשָׂה כְּמוֹתוֹ, וְאֵין צָרִיךְ לוֹמַר גָּדוֹל מִמֶּנּוּ — הֲרֵי זֶה אָסוּר לִקְבּוֹעַ וּלְעַבֵּר בְּחוּץ לָאָרֶץ. וְאִם
