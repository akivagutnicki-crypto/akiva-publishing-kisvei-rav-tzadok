~ שֶׁהֲרֵי אַף בְּאִשְׁתּוֹ מְשַׁמֵּשׁ כְּשֶׁמְּעֻבֶּרֶת, שֶׁאֵינָהּ רְאוּיָה לֵילֵד. וְכֵן מְשַׁמֵּשׁ בְּאִשְׁתּוֹ בִּיאָה שֶׁאֵינָהּ רְאוּיָה לֵילֵד, כְּגוֹן מֵאֲחוֹרֶיהָ, שֶׁאֵינָהּ רְאוּיָה לֵילֵד בְּבִיאָה זוֹ, כָּל שֶׁהוּא זוֹרֶה בִּפְנִים; וּבִלְבַד שֶׁלֹּא יַתְמִיד כֵּן, שֶׁנִּמְצָא מְבַטֵּל מִפְּרִיָּה וּרְבִיָּה. וְיֵשׁ לוֹ לִיזָּהֵר כְּשֶׁמְּשַׁמֵּשׁ לִשְׁהוֹת עַל הָרֶחֶם עַד שִׁימוּט הָאֵבֶר, וְלֹא יִשָּׁאֲרוּ טִפֵּי זֶרַע. וְיֵשׁ אוֹמְרִים שֶׁמֻּתָּר לוֹ לָבוֹא עַל אִשְׁתּוֹ שֶׁלֹּא כְּדַרְכָּהּ רַק בַּנֶּקֶב הָאָחוֹר, וְאַף עַל פִּי שֶׁמּוֹצִיא שָׁם זֶרַע וְהוּא לְבַטָּלָה; וּבִלְבַד בְּאַקְרַאי, וְלֹא יַרְגִּיל לַעֲשׂוֹת כֵּן. וְיֵשׁ חוֹלְקִים וְאוֹמְרִים דַּאֲפִלּוּ בְּאַקְרַאי אָסוּר לְהוֹצִיא זֶרַע, רַק שֶׁלֹּא בְּהוֹצָאַת זֶרַע, שֶׁיִּמָּהֵר כְּשֶׁיָּבוֹא לִידֵי הוֹצָאַת זֶרַע לְהַכְנִיסוֹ לָרֶחֶם — מֻתָּר. וּמִכָּל מָקוֹם לֹא יַרְגִּיל בָּזֶה, שֶׁמָּא יָבוֹא לִידֵי הוֹצָאַת זֶרַע. וּמִכָּל מָקוֹם אֲפִלּוּ דְּבָרִים הַמֻּתָּרִים — כָּל הַמְקַדֵּשׁ עַצְמוֹ בַּמֻּתָּר "קָדוֹשׁ" יֵאָמֵר לוֹ. וּלְצֹרֶךְ מִצְוָה, כְּגוֹן שֶׁמִּתְיָרֵא שֶׁיִּכָּשֵׁל בְּאֵשֶׁת אִישׁ אוֹ בְּנִדָּה וְכַהַאי גַּוְנָא — טוֹב לוֹ לְהוֹצִיא זֶרַע לְבַטָּלָה. וּמִכָּל מָקוֹם לֹא יִנָּקֶה רָע, וְצָרִיךְ לַעֲשׂוֹת תְּשׁוּבָה (לְהִתְעַנּוֹת מ׳ יוֹם בִּימֵי הַקַּיִץ, אוֹ לֵישֵׁב בַּקּוֹר בִּימֵי הַחֹרֶף) — כֵּן כָּתוּב בְּסֵפֶר חֲסִידִים. וְכָתַב הַבֵּית שְׁמוּאֵל לְפִי זֶה מַה שֶּׁכָּתוּב בַּזֹּהַר וּבְשֻׁלְחָן עָרוּךְ דְּעָוֹן שִׁכְבַת זֶרַע חָמוּר מִכָּל לָאו — דַּוְקָא, עַיֵּן שָׁם. וּפָשׁוּט דְּוַדַּאי אֵשֶׁת אִישׁ וּשְׁאָר חַיָּבֵי מִיתוֹת חָמוּר. וּבְפֶרֶק הֶעָרֵל אָמְרִינַן: אִם נִקַּב הַגִּיד, בּוֹדְקִין אִם נִתְרַפֵּא, וּמַנִּיחִים בֵּי פּוֹקְרָא עַד שֶׁיּוֹצִיא זֶרַע, עַיֵּן שָׁם; מַשְׁמָע דִּלְצֹרֶךְ שַׁפִּיר דָּמֵי. וְעַיֵּן בְּבֵית שְׁמוּאֵל (סִימָן כ״ה) לְעִנְיַן כְּשֶׁצָּרִיךְ לִבְדּוֹק אִם הַדָּם מֵהָאִישׁ, אֶפְשָׁר דְּרַשַּׁאי גַּם כֵּן מֵהַאי טַעְמָא. וְצָרִיךְ עִיּוּן אִם הָאִסּוּר חָמוּר כָּל כָּךְ, אֵיךְ הִתִּירוּ כַּהַאי גַּוְנָא:

**אָסוּר** לִרְכּוֹב עַל בְּהֵמָה בְּלֹא אֻכָּף, אִם יוֹשֵׁב יָרֵךְ מִכָּאן וְיָרֵךְ מִכָּאן, דְּאֵיבָרוֹ מִתְחַמֵּם בִּבְשַׂר הַבְּהֵמָה. עַיֵּן בְּרַשִׁ״י, מַשְׁמָע דַּוְקָא בְּאֻכָּף שֶׁל עֵץ. וְיֵשׁ לְעַיֵּן דִּידַן, שֶׁעוֹשִׂין שֶׁל עוֹר, אֶפְשָׁר אֵין מוֹעִיל, וְצָרִיךְ לְעַיֵּן. מִיהוּ מִי שֶׁאֵינוֹ מֻרְגָּל לִרְכּוֹב, וּבָעִית, אִי נַמִּי הָרוֹכֵב בִּמְקוֹם סַכָּנָה, אוֹ בַּלַּיְלָה דְּאִיכָּא בִּיעֲתוּתָא — נִרְאֶה דְּשַׁפִּיר דָּמֵי. וְכֵן הָרוֹכֵב לְמָקוֹם קָרוֹב, אֶפְשָׁר דְּשַׁפִּיר דָּמֵי, דִּבִזְמַן מוּעָט לֹא יִתְחַמֵּם; אֶלָּא דְּאִם כֵּן נָתַתָּ דְּבָרֶיךָ לְשִׁעוּרִין, וְצָרִיךְ עִיּוּן. וּפָשׁוּט דְּהַנּוֹתֵן חֲמוֹר בְּלֹא אֻכָּף לְאַחֵר יִשְׂרָאֵל רוֹכֵב, בִּדְלֵיכָּא אַחֵר — עוֹבֵר בְּ"לִפְנֵי עִוֵּר"; וַאֲפִלּוּ בְּדְאִיכָּא אַחֵר וּבְלָאו הָכֵי הֲוֵי מָצֵי עָבַר — מִכָּל מָקוֹם מְסַיֵּעַ יְדֵי עוֹבְרֵי עֲבֵירָה הוּא, כְּמוֹ שֶׁכָּתְבוּ תּוֹסָפוֹת (רֵישׁ שַׁבָּת ג׳ עַמּוּד א׳). אֲבָל לְצַוּוֹת לְאַחֵר לִרְכּוֹב בִּסְתָם, אַף דְּיָדַע דְּהָאַחֵר רוֹכֵב בְּלֹא אֻכָּף — פָּשׁוּט דְּלֵית לַן בָּהּ:

**אָסוּר** לֶאֱחוֹז בָּאַמָּתוֹ, וַאֲפִלּוּ עַל יְדֵי מַטְלִית רַכָּה, שֶׁמָּא יָבוֹא לִידֵי הוֹצָאַת זֶרַע לְבַטָּלָה; וַאֲפִלּוּ לְצֹרֶךְ קְצָת, כְּגוֹן לְחַכֵּךְ אוֹ לִבְדּוֹק אִם רָאָה קֶרִי, אוֹ דְּאָזִיל לִנְהָרָא
