~ אֲבָל לֹא מִמִּנְיָנָם וּמִכְּלָלָם, עַד כָּאן דְּבָרָיו; וְעַיֵּן שָׁם. (וְעַיֵּן מַה שֶּׁכָּתַב רַבִּי יַעֲבֵ״ץ בְּ"מִגְדּוֹל אוֹצַר הַטּוֹב", מוּסָד א׳, אוֹת ט״ו.) וְהָרַשְׁבַּ״ץ בְּ"זֹהַר הָרָקִיעַ" הֵבִיא רְאָיָה לָזֶה מִגְּמָרָא הוֹרָיוֹת (ח׳ עַמּוּד ב׳): "אֵיזוֹהִי מִצְוָה שֶׁנֶּאֶמְרָה בַּתְּחִלָּה — זוֹ עֲבוֹדַת כּוֹכָבִים"; וּלְהָרַמְבַּ״ם — הֲרֵי קוֹדֵם לָהּ "אָנֹכִי", עַד כָּאן. וּבְסֵפֶר מְגִלַּת אֶסְתֵּר, בְּסֵפֶר הַמִּצְוֹת שָׁם, כָּתַב עָלָיו דְּלָא דָּק בִּלְשׁוֹן רַשִׁ״י, שֶׁכָּתַב "'אָנֹכִי' וְ'לֹא יִהְיֶה לְךָ'" — דְּהַיְנוּ עֲבוֹדַת כּוֹכָבִים, עַד כָּאן; וְעַיֵּן שָׁם:

**וּלְפִי** עֲנִיּוּת דַּעְתִּי — דִּבְרֵי הָרַשְׁבַּ״ץ נְכוֹנִים. כִּי מַה שֶּׁכָּתַב הַסֵּפֶר מְגִלַּת אֶסְתֵּר בְּהֶבְדֵּל בֵּין לְשׁוֹן "בַּתְּחִלָּה" לְ"בָרִאשׁוֹנָה" — תָּמוּהַּ; אִם כֵּן, מַאי פָּרֵיךְ מִי׳ מִצְוֹת שֶׁנִּצְטַוּוּ בְּמָרָה? וְגַם כְּבָר כָּתַב בְּעַצְמוֹ, דְּלֹא מַשְׁמַע חִלּוּק זֶה בַּגְּמָרָא. וּמַה שֶּׁכָּתַב, דְּגַם "אָנֹכִי" הַיְנוּ עֲבוֹדַת כּוֹכָבִים — לְפִי עֲנִיּוּת דַּעְתִּי, לֹא כִּי; וְתַרְתֵּי מִילֵּי נִינְהוּ, וְאֵין "אָנֹכִי" וְ"לֹא יִהְיֶה לְךָ" מִתְאַחֲדִים בְּאַזְהָרָתָם. כִּי הוֹרָאַת "אָנֹכִי" — לְהַאֲמִין בַּה׳ יִתְבָּרַךְ שְׁמוֹ; וְאוֹתָם הַכּוֹפְרִים בִּמְצִיאוּת ה׳, וְאוֹמְרִים שֶׁ"עוֹלָם כְּמִנְהָגוֹ נוֹהֵג" בְּלִי שׁוּם מַנְהִיג וֶאֱלוֹקַּ — לֹא עָבְרוּ עַל "לֹא יִהְיֶה לְךָ", וְלֹא עָבְדוּ עֲבוֹדָה זָרָה, אֲבָל כָּפְרוּ בְּ"אָנֹכִי". וְכֵן הִפּוּךְ הַדְּבָרִים: אוֹתָם הַמּוֹדִים בּוֹ יִתְבָּרַךְ, אֲבָל אוֹמְרִים שֶׁהוּא אֱלֹקֵי הָאֱלֹקִים, וְעָזַב הָאָרֶץ לַעֲבוֹדָה זָרָה לְהַנְהִיג — וּכְגוֹן דּוֹר אֱנוֹשׁ — לֹא יִכְפְּרוּ בְּ"אָנֹכִי", אֲבָל יַעַבְרוּ עַל "לֹא יִהְיֶה", וְיַעַבְדוּ עֲבוֹדָה זָרָה. וְכֵן כָּתַב הָרָאָ״ה בְּסֵפֶר הַחִנּוּךְ (מִצְוָה כ״ו), וְזֶה לְשׁוֹנוֹ: "אִם קִבֵּל בֶּאֱלוֹקַּ אֶחָד מִן הַנִּבְרָאִים, וַאֲפִלּוּ מוֹדֶה שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא שַׁלִּיט בּוֹ וּבֶאֱלָקָהוּ — עָבַר עַל 'לֹא יִהְיֶה'", עַד כָּאן; כַּוָּנָתוֹ לְהַבְדִּיל בֵּין זֶה לְמִצְוַת עֲשֵׂה דְּ"אָנֹכִי":

**וְעוֹד** נִרְאֶה לִי לְהָבִיא רְאָיָה מֵרַשִׁ״י, אַדְּרַבָּא — דְּלֹא כְּהָרַמְבַּ״ם — לְמַה שֶּׁכָּתַב בְּסֵפֶר מְגִלַּת אֶסְתֵּר שָׁם (אוֹת ב׳), לְיַשֵּׁב קֻשְׁיַית הָרַמְבַּ״ן, שֶׁהִקְשָׁה לְהָרַמְבַּ״ם: אִם כֵּן, שָׁמַעְנוּ מִפִּי הַגְּבוּרָה יוֹתֵר מִב׳! וְתֵרֵץ, דְּרַק עַד "לֹא תַעֲשֶׂה לְךָ" נֶאֱמַר לָנוּ מִפִּי הַגְּבוּרָה, עַד כָּאן. וְהִנֵּה, רַשִׁ״י הוֹרָיוֹת שָׁם (סוֹף עַמּוּד א׳), דִּבּוּר הַמַּתְחִיל "אֵיזֶהוּ", כָּתַב: "וְצִוָּה עַל יְדֵי מֹשֶׁה בְּכַמָּה פָּרָשִׁיּוֹת שֶׁבַּתּוֹרָה, כְּגוֹן 'לֹא תִשְׁתַּחֲוֶה לְאֵל אַחֵר'" וְכוּלֵי, עַד כָּאן. וּלְהָרַמְבַּ״ם — לָמָּה לוֹ לְהָבִיא מִמֶּרְחָק, מִקְּרָא דְּפָרָשַׁת כִּי תִשָּׂא (ל״ד, פָּסוּק י״ד)? לַיְתֵי טְפֵי מֵהָךְ גּוּפֵיהּ — דְּ"לֹא תַעֲשֶׂה לְךָ" וְכוּלֵי "לֹא תִשְׁתַּחֲוֶה" וְכוּלֵי — דְּמִפִּי מֹשֶׁה שָׁמַעְנוּ וְלֹא מִפִּי הַגְּבוּרָה, לְהָרַמְבַּ״ם הַנִּזְכָּר לְעֵיל! אֶלָּא עַל כָּרְחֲךָ כְּהָרַמְבַּ״ן — דְּגַם זֶה מִפִּי הַגְּבוּרָה שְׁמַעְנוּם. (וּמִיהוּ, מִכָּל מָקוֹם קְצָת צָרִיךְ עִיּוּן, דְּלֹא מַיְיתֵי רַשִׁ״י מֵהַמֻּקְדָּם בְּפָרָשַׁת מִשְׁפָּטִים, כ״ג פָּסוּק כ״ד, עַיֵּן שָׁם.) וּמַה שֶּׁכָּתַב רַשִׁ״י "'אָנֹכִי' וְ'לֹא יִהְיֶה לְךָ'" — תַּרְוַיְיהוּ, פָּשׁוּט דְּלִישָּׁנָא דְּתַלְמוּדָא נָקַט; וְעַיֵּן בָּרַמְבַּ״ן טַעַם דְּנָקְטִינְהוּ תַּלְמוּדָא, עַיֵּן שָׁם; וְכֵן רַשִׁ״י נָקַט הַשִּׁיגְרוּת הַלָּשׁוֹן:
