~ אַחַת הוּא. וְעוֹד, סֻכּוֹת נַמִּי תִּסְגֵּי בְּחַד בְּבָבֶל וּשְׁאָר מְקוֹמוֹת שֶׁאֵין הַשְּׁלוּחִים מַגִּיעִים, דְּנִסְמוֹךְ אַרֹב — אֱלוּל אֵין מְעֻבָּר; וְכֵן, לְפִי מַה שֶּׁכָּתַבְתִּי לְעֵיל דְּגַם רֹב אֲדָר אֵין מְעֻבָּר — אִם כֵּן גַּם פֶּסַח וְשָׁבוּעוֹת כֵּן. אֶלָּא נִרְאֶה פָּשׁוּט, דְּאַף רַשִׁ״י יוֹדֶה דְּוַדַּאי בִּירוּשָׁלַיִם גּוּפָהּ וּסְבִיבוֹתֶיהָ (כְּהָךְ דְּעֵירוּבִין דְּמַיְיתֵי הַתּוֹסָפוֹת בִּתְנָאֵי לְעִנְיַן עֵרוּב) — דְּחָיְישִׁינַן שֶׁמָּא יְעַבְּרוּ, כֵּיוָן דְּאֶפְשָׁר לְבָרֵר; אִי נַמִּי בְּבָבֶל וּשְׁאָר אֲרָצוֹת רְחוֹקוֹת, דִּלְעוֹלָם יַעֲשׂוּ מִסָּפֵק ב׳ יָמִים. עַל כָּרְחֲךָ לֹא קָאָמַר רַשִׁ״י אֶלָּא בְּמָקוֹם שֶׁמַּגִּיעִים שְׁלוּחֵי נִיסָן וְתִשְׁרֵי, דְּעָבְדֵי כָּל יוֹם טוֹב רַק יוֹם אֶחָד (עַיֵּן בְּסֵפֶר טוּרֵי אֶבֶן (דַּף כ׳ עַמּוּד ב׳), דִּבּוּר הַמַּתְחִיל "גְּזֵרָה נִיסָן"); דְּוַדַּאי, כֵּיוָן דְּאִי אֶפְשָׁר לָהֶם לְהִתְבָּרֵר — אֵין לָהֶם לִשְׁנוֹת גַּם רֹאשׁ הַשָּׁנָה, וְיִסְמְכוּ אַרֻבָּא שֶׁלֹּא לְשַׁנּוֹתוֹ, וְלַעֲשׂוֹתוֹ גַּם כֵּן יוֹם אֶחָד. זֶה נִרְאֶה לַעֲנִיּוּת דַּעְתִּי בָּרוּר בְּדַעַת רַשִׁ״י, וְנִתְיַשְּׁבוּ קֻשְׁיוֹת הַתּוֹסָפוֹת:

**וּבְסֵפֶר** טוּרֵי אֶבֶן הִקְשָׁה עוֹד אַרַשִׁ״י מִלְּקַמָּן (כ״ג עַמּוּד א׳), רַשִׁ״י דִּבּוּר הַמַּתְחִיל "אָמַר אַבָּיֵי" וְכוּלֵי: "ב׳ יָמִים" וְכוּלֵי. וְלָא קָשֶׁה מִידֵּי: דְּאִם נֹאמַר דְּלֹא יַעֲשׂוּ אֶלָּא יוֹם אֶחָד — וַדַּאי דְּאִיצְטְרִיךְ לְמַשּׂוּאוֹת בְּחָסֵר וְלֹא בְּמָלֵא, כְּדֵי שֶׁאִם יִרְאוּ שֶׁלֹּא יַעֲשׂוּ בְּמוֹצָאֵי יוֹם רִאשׁוֹן — יֵדְעוּ דִּמְלֵא, וְיַעֲשׂוּ רֹאשׁ הַשָּׁנָה בְּיוֹם ב׳; וּבְלָאו הָכֵי לֹא יַעֲשׂוּ בְּיוֹם ב׳ כְּלָל, כֵּיוָן דְּלֹא יֵדְעוּ דִּמְלֵא אֶלָּא בְּמַשּׂוּאוֹת דְּמוֹצָאֵי יוֹם ב׳, וְנִמְצָא מְקַלְקְלִים בְּרֹאשׁ הַשָּׁנָה — וּפְשִׁיטָא דַּעֲדִיף. אֶלָּא דְּעַל זֶה יֵשׁ לוֹמַר: יַשִּׂיאוּ מַשּׂוּאוֹת בְּמָלֵא, וְיִתְקְנוּ דְּיַעֲשׂוּ עוֹשֵׂי חוּץ לָאָרֶץ ב׳ יָמִים — וּמַאי אִיכְפַּת לָן? וְעַל זֶה אָמַר: דְּאֵין זֶה תַּקָּנָה, דְּמִתְבַּטְּלִים ב׳ יָמִים מִמְּלָאכָה; וְאָתֵי שַׁפִּיר, וּפָשׁוּט. וּמַה שֶּׁהִקְשָׁה עוֹד מִפְּסָחִים (נ״א עַמּוּד ב׳), רַשִׁ״י דִּבּוּר הַמַּתְחִיל "לִקְבִיעָא" — בְּיוֹם הַכִּפּוּרִים, עַיֵּן שָׁם — מִלְּבַד דְּיֵשׁ לוֹמַר כַּנִּזְכָּר לְעֵיל, וְגַם הָא יוֹם הַכִּפּוּרִים דָּמֵי מַמָּשׁ לְסֻכּוֹת; וְתִקְשֶׁה לִי טְפֵי מִסֻּכּוֹת, דִּמְבֹאָר בְּכַמָּה דּוּכְתֵּי בַּשַּׁ״ס. עוֹד נִרְאֶה לִי לְתָרֵץ זֶה: דְּרַשִׁ״י (י״ט עַמּוּד ב׳), דִּבּוּר הַמַּתְחִיל "הָא אִיצְטְרִיךְ" וְכוּלֵי, עַיֵּן שָׁם, וּבַתּוֹסָפוֹת דִּבּוּר הַמַּתְחִיל "וְאַל", הִקְשׁוּ: הָא יֵדְעוּ דְּתִדָּחֶה מִפְּנֵי אד״ו וְכוּלֵי. וְיֵשׁ לוֹמַר דְּלָא קָשֶׁה מִידֵּי לְמַה שֶּׁכָּתוּב בִּמְקוֹמוֹ: דִּכְשֶׁמְּקַדְּשִׁין עַל פִּי הָרְאִיָּה, כַּמָּה פְּעָמִים חָל אד״ו רֹאשׁ הַשָּׁנָה; אִם כֵּן אַכַּתִּי לֹא יֵדְעוּ. וּלְפִי זֶה, אִי חָל בְּאד״ו — הֲוֵי סְפֵיקָא; וּבְזוֹ וַדַּאי לֹא שַׁיָּךְ רֹב אֱלוּל אֵין מְעֻבָּר, דְּהָא יָצָא מִכְּלַל רֹב, לִהְיוֹתוֹ בְּאד״ו. וְהָנֵי, דְּוַדַּאי לֹא הֲוֵי כַּנִּזְכָּר לְעֵיל — מִכָּל מָקוֹם, מִכְּלַל סְפֵיקָא לֹא נָפֵיק, דִּלְמָא מְעַבְּרָא לֵיהּ מִשּׁוּם יַרְקָא וּמִתַּיָּא. וְאִם כֵּן, שַׁפִּיר נַפְקָא מִינָּהּ בְּיוֹם הַכִּפּוּרִים בִּפְסָחִים שָׁם לְכַהַאי גַּוְנָא. וּמְדֻקְדָּק בִּלְשׁוֹן רַשִׁ״י פְּסָחִים שָׁם: "כִּי מִסְּפֵקָא" וְכוּלֵי; וְלֹא אָמְרִינַן בְּחוּץ לָאָרֶץ אֶלָּא כִּי מִסְּפֵקָא — מַשְׁמַע בְּאֹפֶן דְּמִסְּפֵקָא:

**וּבָזֶה** יְבֹאַר דְּלָא קָשֶׁה מִידֵּי קֻשְׁיַת תּוֹסָפוֹת שָׁם, דִּבּוּר הַמַּתְחִיל "לִקְבִיעָא", אַרַשִׁ״י, עַיֵּן שָׁם. דְּבִירוּשַׁלְמִי (פֶּרֶק א׳ דְּרֹאשׁ הַשָּׁנָה הֲלָכָה ד׳) קָאָמַר טַעְמָא דְּיוֹם הַכִּפּוּרִים אֵין
