~ הָאִילָן. וְצָרִיךְ לוֹמַר: דְּשָׁאנֵי הָנֵי — כֵּיוָן דְּיֵשׁ צֹרֶךְ בַּדָּבָר, אֵין לָחוּשׁ לְשֶׁמָּא יִטְעוּ, דְּכָךְ הָיְתָה תַּקָּנַת חֲכָמִים לְעַבֵּר מִפְּנֵי הַצֹּרֶךְ, וְאֵין לָחוּשׁ; מַה שֶּׁאֵין כֵּן אָבִיב וּפֵרוֹת הָאִילָן — גְּזֵרַת הַכָּתוּב, שַׁפִּיר חָיְישִׁינַן; כֵּן נִרְאֶה מֻכְרָח. וּבָזֶה נִיחָא מַה שֶּׁכָּתְבוּ תּוֹסָפוֹת שָׁם לְמַעְלָה מִזֶּה — וְקָשֶׁה, דְּאָבִיב נַמִּי דְּאוֹרַיְתָא וְכוּלֵי. וְהִנֵּה בֶּאֱמֶת יֵשׁ לוֹמַר: דְּרַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל סָבִירָא לֵיהּ דְּאָבִיב וּפֵרוֹת הָאִילָן דְּרַבָּנָן, וְלָא דָּרֵישׁ כְּלָל "שָׁמוֹר" וְכוּלֵי לְחִיּוּבָא; וּכְמוֹ דְּלָא דָּרְשֵׁי אֲחֵרִים "חַג הָאָסִיף" לְחִיּוּבָא; רַק הַכָּתוּב קוֹרְאוֹ "אָבִיב" לְפִי שֶׁכֵּן הוּא בְּרֹב הַפְּעָמִים. וּתְקוּפָה דְּקָאָמַר — הַיְנוּ תְּקוּפַת תִּשְׁרֵי, מֵ"חַג הָאָסִיף" — זֶה דָּרֵישׁ, וַהֲוֵי דְּאוֹרַיְתָא; אֲבָל אָבִיב רַק דְּרַבָּנָן, מִפְּנֵי צֹרֶךְ הָעֹמֶר. וְאַף עַל גַּב דְּרַשִׁ״י, דִּבּוּר הַמַּתְחִיל "עַל", דָּחָה לוֹמַר כֵּן מִבָּרַיְתָא דְּעַל ג׳ אֲרָצוֹת — אֲבָל זֶהוּ אַלִּיבָּא דְּאֱמֶת דְּתַנָּא קַמָּא; אֲבָל רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל — אֶפְשָׁר בֶּאֱמֶת לֵית לֵיהּ אִידָךְ בָּרַיְתָא, וּבִשְּׁלָה בִּיהוּדָה דַּי. רַק דִּלְפִי זֶה, דְּטַעְמָא דְּאָבִיב דְּרַבָּנָן מִשּׁוּם צֹרֶךְ — אִם כֵּן, כָּל שֶׁכֵּן דְּדַי בְּאַחַת לְחוּד, כְּמוֹ בְּכָל הָנֵי דְּמִשּׁוּם צֹרֶךְ, כַּנִּזְכָּר לְעֵיל; וְנִיחָא. וּלְפִי זֶה אָתֵי שַׁפִּיר, דְּלָא קָשֶׁה קֻשְׁיַת הַתּוֹסָפוֹת: דְּיֵשׁ לוֹמַר רַבִּי לָא דָּרֵישׁ "שָׁמוֹר אֶת חֹדֶשׁ הָאָבִיב" (אִי נַמִּי דָּרֵישׁ, וְסָבִירָא לֵיהּ מִשּׁוּם צֹרֶךְ נַמִּי), וְסָבִירָא לֵיהּ אָבִיב דְּרַבָּנָן מִשּׁוּם צֹרֶךְ; אִם כֵּן שַׁפִּיר — אַף בְּאָבִיב לְחוּד מְעַבְּרִין, כִּבְכָל מִילֵּי דְּמִשּׁוּם צֹרֶךְ, כַּנִּזְכָּר לְעֵיל:

**וּלְפִי** זֶה, נִיחָא דְּלָא קָשֶׁה קֻשְׁיַת הָרמ״ךְ — דְּכֵיוָן דְּעַל כָּרְחָךְ לְרַבִּי טַעַם דְּאָבִיב מִשּׁוּם צֹרֶךְ — שַׁפִּיר קָאָמַר שֶׁאֵין מְעַבְּרִין בִּשְׁנַת בַּצֹּרֶת, וְאָתֵי שַׁפִּיר. וְעַיֵּן בְּכֶסֶף מִשְׁנֶה, בְּשֵׁם הָרַ״א; וְלַעֲנִיּוּת דַּעְתִּי — דְּבָרָיו תְּמוּהִים, דְּהָרַמְבַּ״ם עַצְמוֹ (הֲלָכָה ז׳ וַהֲלָכָה ח׳) כָּתַב דְּאֵין מוֹעִיל גְּדָיִים רַכִּים לְאָבִיב בִּלְבַד, אֶלָּא צָרִיךְ אָבִיב וּפֵרוֹת הָאִילָן, עַיֵּן שָׁם. וְתֵימָה עַל הָרמ״ךְ, שֶׁהֻצְרַךְ לְהַאֲרִיךְ לְהָשִׁיב עָלָיו. וּמַה שֶּׁכָּתוּב רְאָיָתוֹ מִכָּל שֶׁכֵּן — זִמְנָא דַּאֲבִיבָא לֹא מָטָא; עַיֵּן בְּסֵפֶר תּוֹרַת חַיִּים, שֶׁיִּישֵּׁב: דְּ"זִמְנָא דַּאֲבִיבָא" הַיְנוּ תְּקוּפָה, וְלָכֵן לֹא אָמַר "דַּאֲבִיבָא", עַיֵּן שָׁם):

**וּכְשֶׁמְּעַבְּרִין** בֵּית דִּין הַשָּׁנָה — שׁוֹלְחִין אִגְּרוֹת לְכָל הַמְּקוֹמוֹת הָרְחוֹקִים, וּמוֹדִיעִים אוֹתָן שֶׁעִבְּרוּהָ, וּמִפְּנֵי מָה עִבְּרוּ. וְעַל לְשׁוֹן הַנָּשִׂיא נִכְתָּבוּ, וְהוּא אוֹמֵר לָהֶן בְּמִכְתָּבוֹ: "יָדוּעַ לָכֶם שֶׁהִסְכַּמְתִּי אֲנִי וַחֲבֵרַי, וְהוֹסַפְנוּ עַל שָׁנָה זוֹ כָּךְ וְכָךְ יוֹם". וְתָלוּי בִּרְצוֹן הַבֵּית דִּין כַּמָּה יוֹם לְהוֹסִיף: רָצוּ — מוֹסִיפִין כ״ט יוֹם, רָצוּ — ל׳ יוֹם (בַּגְּמָרָא רֹאשׁ הַשָּׁנָה י״ט עַמּוּד ב׳ מַשְׁמַע לְהֶדְיָא דְּהַיְנוּ אִם לַעֲשׂוֹת הַח׳ הָעוֹבֵר מְעֻבָּר אוֹ בִּזְמַנּוֹ, עַיֵּן שָׁם; וְיֵשׁ לְפָרֵשׁ גַּם לְשׁוֹן הָרַמְבַּ״ם כֵּן
