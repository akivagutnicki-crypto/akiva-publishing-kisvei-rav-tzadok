~ לְקַדֵּשׁ בְּמוֹצָאֵי תַּעֲנִית; אִם לֹא כְּשֶׁאִי אֶפְשָׁר לְקַדֵּשׁ אַחַר כָּךְ, אוֹ יָחִיד הַמִּתְעַנֶּה וְרוֹאֶה צִבּוּר הַמְקַדְּשִׁין, וְאַחַר כָּךְ לֹא יִמְצָא צִבּוּר — דִּשְׁרֵי, מִשּׁוּם "בְּרָב עָם"; אֲבָל בְּלָאו הָכֵי — לֹא. וּבְמוֹצָאֵי תִּשְׁעָה בְּאָב — יֵשׁ אוֹמְרִים לְקַדֵּשׁ, מִשּׁוּם דְּאָז נוֹלַד הַגּוֹאֵל, וְהוּא עֵת שִׂמְחָה; וְיֵשׁ חוֹלְקִין; וְנָהֲגוּ לְקַדֵּשׁ. וְעַל כָּל פָּנִים, כְּשֶׁמְּקַדֵּשׁ בְּמוֹצָאֵי תַּעֲנִית — נָכוֹן לִטְעוֹם מִקֹּדֶם; אִם לֹא שֶׁחָשׁ שֶׁבֵּין כָּךְ יְקַדְּשׁוּ הַצִּבּוּר:

**וּבְמוֹצָאֵי** תַּעֲנִית אֶסְתֵּר — רַשַּׁאי לְקַדֵּשׁ, כֵּיוָן שֶׁאֵין לוֹ זְמַן אֶלָּא לַיְלָה אַחַת, וְגַם הוּא לֵיל פּוּרִים, דִּשְׁכִיחָא שִׁכְרוּת. וְקוֹדֶמֶת לִקְרִיאַת הַמְּגִלָּה, דִּתְדִירָא; וְגַם זְמַנָּהּ יוּכַל לִהְיוֹת עוֹבֵר, אִם תְּכֻסֶּה בָּעֲנָנִים (נוֹדָע בִּיהוּדָה קַמָּא, סִימָן מ״א). וְעַיֵּן עוֹד שָׁם, דְּאִם כָּלָה אָז הַחֲצִי מִכ״ט י״ב תשצ״ג, וְכָל הַלֵּילוֹת לֹא נִרְאֵית הַלְּבָנָה כִּי כִּסּוּהָ עֲנָנִים, וּבְאֶמְצַע קְרִיאַת הַמְּגִלָּה נִרְאֵית — רַשַּׁאי לְהַפְסִיק בְּאֶמְצַע, לֵילֵךְ וּלְחַדֵּשׁ, כֵּיוָן דְּאֵין עוֹד זְמַן אֶלָּא אוֹתָהּ לַיְלָה, וְשֶׁמָּא יָשׁוּב לְהִתְכַּסּוֹת בָּעֲנָנִים אִם נַמְתִּין עַד אַחַר קְרִיאַת מְגִלָּה. וַאֲפִלּוּ בְּאֶמְצַע קְרִיאַת שְׁמַע וּבִרְכוֹתֶיהָ — רַשַּׁאי לְהַפְסִיק, אִם אֵין עוֹד זְמַן. וּלְעִנְיַן לְהַקְדִּים לִקְרִיאַת שְׁמַע — וַדַּאי קְרִיאַת שְׁמַע קוֹדֶמֶת, דִּתְדִירָא וּדְאוֹרַיְתָא. מִיהוּ, אִם הָיָה רֹב הַלֵּילוֹת מְכֻסֶּה בָּעֲנָנִים, וּבְלַיְלָה אַחַת נִרְאֵית, וְיֵשׁ לוֹ לָחוּשׁ פֶּן בֵּין כָּךְ וּבֵין כָּךְ תִּתְכַּסֶּה — רַשַּׁאי לְחַדֵּשׁ קֹדֶם קְרִיאַת שְׁמַע וּתְפִלָּה, אַף שֶׁיֵּשׁ עוֹד זְמַן ב׳ אוֹ ג׳ לֵילוֹת:

**וְאֵין** מְקַדְּשִׁין הַלְּבָנָה קֹדֶם תִּשְׁעָה בְּאָב, מִשּׁוּם דִּשְׁרוּיִים בַּאֲבֵלוּת; וְקֹדֶם יוֹם הַכִּפּוּרִים — דִּשְׁרוּיִים בְּצַעַר מְחִילַת עֲוֹנוֹת. וּבְמוֹצָאֵי יוֹם הַכִּפּוּרִים — מְחַדֵּשׁ; וְנִרְאֶה נַמִּי, דְּאִם לֹא יְאַבֵּד הַצִּבּוּר — יִטְעוֹם מִקֹּדֶם. וְאֵין מְקַדְּשִׁים אֶלָּא בַּלַּיְלָה וְלֹא בַּיּוֹם, דְּאֵין אוֹרָהּ מוֹעִיל — דִּשְׁרַגָּא בְּטִיהֲרָא מַאי מְהַנֵּי? מִיהוּ, בְּבֵין הַשְּׁמָשׁוֹת — אִם נִרְאֶה לָעַיִן שֶׁאוֹרָהּ מוֹעִיל, וְנֶהֱנִין מֵאוֹרָהּ — נִרְאֶה דִּמְחַדְּשִׁין. וְיֵשׁ לְהַדֵּר בְּכָל הָאֶפְשָׁר אַחַר עֲשָׂרָה לְחַדֵּשׁ, מִשּׁוּם "בְּרָב עָם הַדְרַת מֶלֶךְ"; וְיוּכַל לְהַמְתִּין בַּקִּדּוּשׁ אֵיזֶה לֵילוֹת, עַד י׳ אוֹ י״א, עַל עֲשָׂרָה. וְנִרְאֶה דְּמוּטָב לְקַדֵּשׁ בְּחֹל בַּעֲשָׂרָה, מִלְּהַמְתִּין עַל מוֹצָאֵי שַׁבָּת. וְאָבֵל — בְּאֹפֶן דְּרַשַּׁאי לְקַדֵּשׁ בַּאֲבֵלוּתוֹ, כְּמוֹ שֶׁנִּתְבָּאֵר לְעֵיל — יְקַדֵּשׁ בְּבֵיתוֹ; אִם לֹא דְּיֵשׁ עֲשָׂרָה — אָז יוֹצֵא לַחוּץ. מִיהוּ, אִם חוֹשֵׁשׁ פֶּן לֹא תִּתְרָאֶה אַחַר כָּךְ בְּלֵילוֹת אֲחֵרוֹת, וּתְכֻסֶּה בָּעֲנָנִים — אֵין צָרִיךְ לְהַמְתִּין עַל עֲשָׂרָה, וּפָשׁוּט. וּלְדַעַת ר״י עֶמְדִּין הַנִּזְכָּר לְעֵיל — לֹא יַמְתִּין עַל הַצִּבּוּר אֶלָּא עַד לֵיל ז׳. כֵּן נִרְאֶה:

**וְזֶה** לְשׁוֹן "הַנְהָגוֹת הַגְּרָ״א": "קִדּוּשׁ לְבָנָה — אֵין מַחְמִיצִין הַמִּצְוָה עַד ז׳ אַחַר הַמּוֹלָד, אוֹ עַד מוֹצָאֵי שַׁבָּת; רַק תֵּכֶף אַחַר ג׳ יָמִים מֵהַמּוֹלָד (נִרְאֶה
