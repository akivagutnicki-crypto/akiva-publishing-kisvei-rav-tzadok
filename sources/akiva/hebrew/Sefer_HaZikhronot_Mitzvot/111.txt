~ מִן רֶבַע הַיּוֹם, שֶׁיִּשָּׁאֵר י״ד חֲלָקִים, י׳ שְׁנִיִּים, נ״ו שְׁלִישִׁיִּים, מ״ב רְבִיעִיִּים, נ׳ חֲמִישִׁיִּים, י״ו שִׁשִּׁיִּים, מ׳ שְׁבִיעִיִּים. נְחַבֵּר זֶה אֶל קְבוּצַת שס״ה יוֹם הַקּוֹדֵם — וְיֶחְסַר עֲדַיִן לְש״ס מַעֲלוֹת שְׁלֵמוֹת ג׳ שְׁנִיִּים, נ׳ שְׁלִישִׁיִּים, כ״ב רְבִיעִיִּים, ט׳ חֲמִישִׁיִּים, מ״ג שִׁשִּׁיִּים, כ׳ שְׁבִיעִיִּים. וּכְשֶׁנַּעֲשֶׂה מֵהַשְּׁנִיִּים וְהַשְּׁלִישִׁיִּים רְבִיעִיִּים — יִהְיֶה קָרוֹב לְי״ד אֶלֶף רְבִיעִיִּים; יַגִּיעַ יוֹתֵר מִל״ה רְבִיעִיִּים לְכָל יוֹם מֵהַשס״ה יוֹם. וְכֵיוָן שֶׁאַלְבַּתָּאנִי יָרַד לְדַקְדֵּק בִּרְבִיעִיִּים, בִּמְנוֹתוֹ ל״ה רְבִיעִיִּים — מַדּוּעַ לֹא אָמַר דְּיֵשׁ עוֹד, זוּלַת זֶה, יוֹתֵר מִל״ה רְבִיעִיִּים, כַּנִּזְכָּר לְעֵיל? וְצָרִיךְ עִיּוּן לַעֲנִיּוּת דַּעְתִּי בָּזֶה:

**גַּם** מַה שֶּׁכָּתַב הָרַמְבַּ״ם בְּמַהֲלַךְ שָׁנָה סְדוּרָה — שֶׁהוּא שנ״ד יוֹם — שמ״ח נ״ה ט״ו, גַּם כֵּן הִנִּיחַ הָרַלְבַּ״ח בְּצָרִיךְ עִיּוּן, שֶׁהוֹסִיף כְּפִי הַחֶשְׁבּוֹן יוֹתֵר מִן הָרָאוּי בְּז׳ שְׁנִיִּים, עַד כָּאן. וְזוֹ בֶּאֱמֶת תְּמִיהָה חֲזָקָה — שֶׁהוֹסִיף יוֹתֵר מִז׳ שְׁנִיִּים לְפִי הַחֶשְׁבּוֹן. וַאֲפִלּוּ אִם נָבוֹא לְדַקְדֵּק לְחֶשְׁבּוֹן שֶׁל אַלְבַּתָּאנִי בְּל״ה רְבִיעִים, כַּנִּזְכָּר לְעֵיל — יַגִּיעַ לְי״ג שְׁנִיּוֹת, כ״ו שְׁלִשִׁים, ל׳ רְבִיעִיִּים, וַעֲדַיִן לֹא יֻשְׁלְמוּ הַט״ו שְׁנִיּוֹת שֶׁל הָרַמְבַּ״ם. וְהָרַלְבַּ״ח כָּתַב שֶׁעָלוּ עַל לִבּוֹ קְצָת דְּבָרִים לְיַשֵּׁב; וַאֲנִי, בַּעֲנִיּוּתִי, יָגַעְתִּי וְלֹא מָצָאתִי, וְאַאֲמִין שֶׁלֹּא אֶמְצָא נֹפֶשׁ לְנַפְשִׁי בְּקֻשְׁיָא זוֹ. וְהַתֵּימָה עַל הַמְּפָרֵשׁ, שֶׁכָּתַב "וְכֵן הַדִּין בְּחֹדֶשׁ וְשָׁנָה", וְלֹא הִרְגִּישׁ בְּחֶשְׁבּוֹן זֶה:

**וְדַע**, לְקֻשְׁיָא הָרִאשׁוֹנָה שֶׁל הָרַלְבַּ״ח — נִרְאֶה לִי לִכְאוֹרָה כָּעֵת: דִּלְדַעַת הָרַמְבַּ״ם מַהֲלַךְ הַשֶּׁמֶשׁ בְּיוֹם אֶחָד — נ״ט ח׳ י״ט מ״ח, דְּלֹא כְּאַלְבַּתָּאנִי הַנִּזְכָּר לְעֵיל; וְעֵת הַשָּׁנָה לְפִי זֶה — שס״ה ה׳ תתע״ט בְּקֵרוּב. וְהַחֶשְׁבּוֹן מְכֻוָּן: שֶׁנִּשְׁאָר מִי׳ יָמִים ח״י שְׁלִישִׁיִּים, וּבְק׳ יָמִים נִתְקַבְּצוּ לְק״פ שְׁלִישִׁיִּים; וְלָכֵן הוֹסִיף ג׳ שְׁנִיִּים; וְשׁוּב לֹא הוֹסִיף מִן ק׳ וָהָלְאָה, כִּי לֹא נִשְׁאַר מְאוּמָה:

**אִיבְּרָא**, עֲדַיִן צָרִיךְ לְעַיֵּן גָּדוֹל עַל הַמְּפָרֵשׁ, שֶׁכָּתַב שֶׁדַּעַת הָרַמְבַּ״ם כְּאַלְבַּתָּאנִי. וְצָרִיךְ עִיּוּן: שֶׁבְּפֶרֶק י׳, הֲלָכָה א׳, זָכַר תְּקוּפַת רַב אַדָּא, שֶׁהַשָּׁנָה שס״ה ה׳ תתקצ״ז מ״ח, וְכָתַב שָׁם בַּהֲלָכָה ו׳ שֶׁחֶשְׁבּוֹן זֶה קָרוֹב מִן הַדְּבָרִים שֶׁנִּתְבָּאֲרוּ בְּאִצְטַגְנִינוּת, וְעַל חֶשְׁבּוֹן זֶה הָיוּ בֵּית דִּין הַגָּדוֹל סוֹמְכִין, עַיֵּן שָׁם. וְזֶה רָחוֹק הַרְבֵּה מִדַּעַת אַלְבַּתָּאנִי; אֲבָל קָרוֹב הוּא לְדַעַת בַּטְלְמְיוּס, שֶׁזָּכַר בַּמְּפָרֵשׁ — שֶׁחֵלֶק אֶחָד מִש׳ בַּיּוֹם, שֶׁהוּא ד׳ מִינוּ״ט וְד׳ חוּמְשֵׁי מִינוּ״ט, שֶׁהֵם פ״ו חֲלָקִים וּב׳ חוּמְשֵׁי חֵלֶק מֵחֶלְקֵי תתר״ף — בָּזֶה הוּא פָּחוֹת מֵרְבִיעַ הַיּוֹם; וְנִמְצָא מִדַּת הַשָּׁנָה שס״ה ה׳ תתקצ״ג וְג׳ חוּמְשִׁין, וְזֶהוּ קָרוֹב לְהַנִּזְכָּר לְעֵיל. אֲבָל לְפִי דַּעְתּוֹ — מַהֲלַךְ הַשֶּׁמֶשׁ בְּיוֹם אֶחָד נ״ט ח׳ י״ז י״ג י״ב ל״א; וּלְפִי זֶה יַגִּיעַ בְּי׳ יָמִים
