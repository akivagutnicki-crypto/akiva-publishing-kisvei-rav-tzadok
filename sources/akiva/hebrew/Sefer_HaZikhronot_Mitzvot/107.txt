~ צ׳, שֶׁהִיא תְּקוּפַת טָלֶה — הַהֶבְדֵּל קֶשֶׁת צ״ק וְקֶשֶׁת נ״ס, הַגְּדוֹלִים מִקֶּשֶׁת ד״ח; וּמִנְּקֻדַּת ז׳ לִנְקֻדַּת ר׳, תְּקוּפַת סַרְטָן — הַהֶבְדֵּל קֶשֶׁת ק״צ, הַקָּטָן מִקֶּשֶׁת ד״ח:

**וְהִנְּךָ** רוֹאֶה אֵיךְ בְּהִשְׁתַּנּוּת מְקוֹם גֹּבַהּ יִשְׁתַּנּוּ יְמֵי הַתְּקוּפוֹת וְעִתּוֹתֵיהֶם; שֶׁהוּא נִמְשָׁךְ מֵהַהֶבְדֵּל שֶׁבֵּין אוֹתוֹ רֹבַע שֶׁל גַּלְגַּל הַמַּזָּלוֹת וְהַדּוֹמֶה לְהַיּוֹצֵא מֶרְכָּז. וְכֵן תָּדִין לְכָל אֹפֶן שֶׁיִּהְיֶה בּוֹ מְקוֹם גֹּבַהּ — יִהְיֶה הֶבְדֵּל, בֵּין רַב לִמְעַט, בֵּין מְקוֹם חֶשְׁבּוֹן הָאֲמִתִּי, שֶׁהוּא בְּגַלְגַּל הַדּוֹמֶה, לְחֶשְׁבּוֹן אֶמְצָעִי, שֶׁהוּא בְּגַלְגַּל הַיּוֹצֵא מֶרְכָּז:

**וּכְפִי** מַה שֶּׁנִּתְבָּאֵר — מְבֹאָר הֱיוֹת הַקֶּשֶׁת ח״ד, שֶׁהוּא הַקֶּשֶׁת שֶׁבְּרֹבַע הַגַּלְגַּל — הוּא הַקֶּשֶׁת הַיּוֹתֵר גָּדוֹל שֶׁאֶפְשָׁר לְהִתְרַחֵק חֶשְׁבּוֹן הָאֲמִתִּי מֵחֶשְׁבּוֹן הָאֶמְצָעִי; וּמִשָּׁם וְאֵילָךְ מַתְחִיל לְהִתְקַטֵּן, עַד שֶׁבִּנְקֻדַּת ה׳ הֵם שְׁנֵיהֶם שָׁוִים. וְנִמְצָא: בִּשְׁנֵי מְקוֹמוֹת שְׁנֵיהֶם שָׁוִים — בִּנְקֻדַּת ג׳ וּבִנְקֻדַּת ה׳; וּבְב׳ מְקוֹמוֹת הֵם בַּמֶּרְחָק הַיּוֹתֵר גָּדוֹל זֶה מִזֶּה — וְהוּא בִּנְקֻדַּת ד׳ וּבִנְקֻדַּת ו׳, שֶׁאָז רָחוֹק זֶה מִזֶּה בְּקֶשֶׁת ד״ח אוֹ ו״י, שֶׁהוּא קֶשֶׁת הַיּוֹתֵר גָּדוֹל בְּהֶבְדֵּלָם. וּכְפִי דַּעַת הָרַמְבַּ״ם זִכְרוֹנוֹ לִבְרָכָה — שֶׁאַחֲרָיו אָנוּ נִמְשָׁכִים בִּדְבָרֵינוּ אֵלֶּה — קֶשֶׁת ד״ח אוֹ ו״י הַיּוֹתֵר גָּדוֹל הוּא מַעֲלָה וְנ״ט חֲלָקִים, וּכְמוֹ שֶׁיִּתְבָּאֵר לְקַמָּן (אֲבָל בַּטְלְמְיוּס סוֹבֵר שֶׁשִּׁעוּר קֶשֶׁת זֶה ב׳ מַעֲלוֹת וָחֵצִי. וְעַיֵּן בַּמְּפָרֵשׁ פֶּרֶק י״ב, דִּבּוּר הַמַּתְחִיל "וְדַע שֶׁהַשֶּׁמֶשׁ יֵשׁ לָהּ שִׁנּוּי א׳ וּמוּסְגָּל" וְכוּלֵי) — דְּבָרִים אֵלּוּ:

**וּמֵעַתָּה** נָבוֹא לְבָאֵר דֶּרֶךְ חֶשְׁבּוֹן יְדִיעַת מָקוֹם אֲמִתִּי שֶׁל הַשֶּׁמֶשׁ בְּכָל יוֹם, וּבְכָל עֵת, וּבְכָל שָׁעָה. וְרַק נַקְדִּים עוֹד דָּבָר אֶחָד צָרִיךְ לְדַעְתּוֹ, וְהוּא:

**כְּבָר** בֵּאַרְנוּ שֶׁיֵּשׁ הֶבְדֵּל בֵּין עִיר לְעִיר, וּמְדִינָה לִמְדִינָה, בֵּין הַיָּמִים — בְּגָדְלָם וְקָטְנָם, וּזְרִיחָתָם וּשְׁקִיעָתָם; וְהַדָּבָר תָּלוּי בִּמְקוֹם מַצַּב הֶעָרִים — רָחְקָם מִקַּו הַשָּׁוֶה לְצַד צָפוֹן אוֹ דָּרוֹם, וְרָחְקָם זֶה מִזֶּה מִן הַמִּזְרָח לְמַעֲרָב. וְהָעִקָּר הוּא מִמְּקוֹם מַצַּב הַשֶּׁמֶשׁ, שֶׁאֵינוֹ שָׁוֶה בְּכָל עִיר; וְהַדָּבָר מוּבָן מֵאֵלָיו בְּעִיר הָעוֹמֶדֶת רְחוֹקָה מֵחֲבֶרְתָּהּ. וּלְפִי שֶׁחֶשְׁבּוֹן לֹא יִהְיֶה אֶלָּא בְּאֶרֶץ יִשְׂרָאֵל בִּלְבַד, כְּמוֹ שֶׁנִּתְבָּאֵר לְמַעְלָה (בְּפֶרֶק א׳) — כָּל דַּרְכֵי הַחֶשְׁבּוֹן הַזֶּה בְּנוּיִים עַל עִיר יְרוּשָׁלַיִם, וְלִשְׁאָר הַמְּקוֹמוֹת הַסּוֹבְבִים בִּכְמוֹ מַהֲלַךְ ו׳ אוֹ ז׳ יָמִים, שֶׁבָּהֶם רוֹאִים אֶת הַיָּרֵחַ תָּמִיד, וּבָאִים וּמְעִידִים בְּבֵית דִּין:
