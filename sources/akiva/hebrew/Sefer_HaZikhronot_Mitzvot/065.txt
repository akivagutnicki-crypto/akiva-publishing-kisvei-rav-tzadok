**וְאִם** לֹא עִבְּרוּ כָּל חֹדֶשׁ אֲדָר, עַד שֶׁהִגִּיעַ יוֹם ל׳ בַּאֲדָר — שׁוּב לֹא יְעַבְּרוּ כְּלָל; שֶׁאוֹתוֹ הַיּוֹם רָאוּי לִהְיוֹת חֹדֶשׁ נִיסָן, וּמִשֶּׁנִּכְנַס נִיסָן וְלֹא עִבְּרוּ — אֵין יְכוֹלִים לְעַבֵּר. וְאִם עִבְּרוּ בְּיוֹם ל׳ שֶׁל אֲדָר — הֲרֵי זוֹ מְעֻבֶּרֶת; וְאִם בָּאוּ עֵדִים אַחַר כָּךְ, וְהֵעִידוּ עַל הַיָּרֵחַ — הֲרֵי אֵלּוּ מְקַדְּשִׁים הַחֹדֶשׁ בְּיוֹם ל׳, וְיִהְיֶה רֹאשׁ חֹדֶשׁ אֲדָר שֵׁנִי. וְאִם קִדְּשׁוּהוּ קֹדֶם שֶׁיְּעַבְּרוּ הַשָּׁנָה — שׁוּב לֹא הָיוּ מְעַבְּרִין, שֶׁאֵין מְעַבְּרִין בְּנִיסָן (וְעַיֵּן בַּגְּמָרָא: דְּעוּלָּא אָמַר אֵין מְקַדְּשִׁין הַחֹדֶשׁ; וּפֵרֵשׁ רַשִׁ״י: שֶׁלֹּא יְקַדְּשׁוּ אֲדָר שֵׁנִי כְּלָל, אֶלָּא שֶׁתִּקֵּי, וְכִי מָטֵי רֹאשׁ חֹדֶשׁ הָאַחֵר — מְקַדְּשִׁין אוֹתוֹ לְנִיסָן. וְאִם קִדְּשׁוּ — מַאי? פְּלִיגֵי רָבָא וְרַב נַחְמָן; וְרַב נַחְמָן אָמַר: מְקֻדָּשׁ וּמְעֻבָּר, וְכוּלֵי. וּפָסַק הָרַמְבַּ״ם — שֶׁכָּתַב "וְאִם בָּאוּ עֵדִים" וְכוּלֵי. וְצָרִיךְ עִיּוּן: לְפִי זֶה, מַדּוּעַ הִשְׁמִיט דִּבְרֵי עוּלָּא, דִּלְכַתְּחִלָּה אֵין מְקַדְּשִׁים — דְּמַשְׁמַע דְּעַל זֶה לֹא פָּלִיג רַב נַחְמָן? הֵן אֱמֶת, קְצָת צָרִיךְ עִיּוּן דִּבְרֵי רַשִׁ״י: אִם הוּא מְעֻבָּר — מַדּוּעַ לֹא יְקַדְּשׁוּ לַאֲדָר בְּל״א? וְאוּלַי, מִשּׁוּם דְּרֹב פְּעָמִים חָסֵר, כַּנִּזְכָּר לְעֵיל, מַיְירֵי בְּחָסֵר; וְצָרִיךְ עִיּוּן קְצָת):

**וְאֵין** מְעַבְּרִין אֶת הַשָּׁנָה בִּשְׁנַת רְעָבוֹן, שֶׁהַכֹּל רָצִין לְבֵית הַגְּרָנוֹת לֶאֱכוֹל וְלִחְיוֹת, וְאִי אֶפְשָׁר לְהוֹסִיף לָהֶם זְמַן לֶאֱסוֹר הֶחָדָשׁ. וְכֵן אֵין מְעַבְּרִין בַּשְּׁבִיעִית, לְפִי שֶׁיַּד הַכֹּל מְמַשְׁמֶשֶׁת עַל הַסְּפִיחִין, וְלֹא יִמְצְאוּ לְהַקְרִיב הָעֹמֶר וּשְׁתֵּי הַלֶּחֶם. (וְרַשִׁ״י כָּתַב הַטַּעַם: שֶׁלֹּא לְהַאֲרִיךְ עֲלֵיהֶם אִסּוּר עֲבוֹדַת קַרְקַע. וּבַגְּמָרָא אִיתָא עוֹד: "וְלֹא בְּמוֹצָאֵי שְׁבִיעִית"; וְהָרַמְבַּ״ם הִשְׁמִיטָה, דְּפָסַק כְּבֵית רַבָּן גַּמְלִיאֵל — דְּאָתָאן כְּרַבּוֹתֵינוּ, דְּהִתִּירוּ לְהָבִיא יָרָק מֵחוּץ לָאָרֶץ לְאֶרֶץ יִשְׂרָאֵל; וּרְגִילִין הָיוּ לְעַבֵּר בְּעֶרֶב שְׁבִיעִית — פֵּרֵשׁ רַשִׁ״י: הַטַּעַם, לְהוֹסִיף לָהֶם חֹדֶשׁ בַּעֲבוֹדַת קַרְקַע.) וְכָל זֶה כְּשֶׁרוֹצִים לְעַבֵּר מִפְּנֵי צֹרֶךְ הַדְּרָכִים וְהַגְּשָׁרִים, וְכַיּוֹצֵא בָּהֶם — נִמְנָעִים מִפְּנֵי שְׁנַת רְעָבוֹן אוֹ שְׁבִיעִית; אֲבָל אִם הָיְתָה הַשָּׁנָה רְאוּיָה לְהִתְעַבֵּר מִפְּנֵי הַתְּקוּפָה, אוֹ מִפְּנֵי הָאָבִיב וּפֵרוֹת הָאִילָן — מְעַבְּרִין לְעוֹלָם, בְּכָל זְמַן:

**וְזֶהוּ** דַּעַת הָרַמְבַּ״ם; וְהָרמ״ךְ הִשִּׂיג מִגְּמָרָא (י״ב עַמּוּד א׳): "וְאִישׁ בָּא מִבַּעַל שָׁלִישָׁה" וְכוּלֵי, "וּמִכָּל מָקוֹם לֹא עִבְּרָהּ" וְכוּלֵי — וְשָׁם הוּא מִפְּנֵי אָבִיב וּפֵרוֹת הָאִילָן. וְנִרְאֶה לִי לְתָרֵץ דְּתוֹסָפוֹת, דִּבּוּר הַמַּתְחִיל "רָאוּי" וְכוּלֵי, כָּתְבוּ: "וְאִם תֹּאמַר — מְנָא לֵיהּ" וְכוּלֵי, עַיֵּן שָׁם. וְיֵשׁ לְתָרֵץ דְּהַתּוֹסָפוֹת (י״א עַמּוּד ב׳), דִּבּוּר הַמַּתְחִיל "אוֹ", "דִּלְמָא", כָּתְבוּ; וְנִרְאֶה דְּהַיְנוּ טַעְמָא דִּתְקוּפָה וְכוּלֵי, יְעֻיַּן שָׁם. וְלִכְאוֹרָה, הִנֵּה יִשְּׁבוּ שִׁנּוּיָא דִּתְקוּפָה מֵאָבִיב — מִשּׁוּם דִּתְקוּפָה יָדוּעַ וּבָרוּר. אֲבָל מָה יַעֲנוּ אַדִּבְרֵי צֹרֶךְ, שֶׁמְּעַבְּרִין בִּשְׁבִילָן, דְּכָתְבוּ בְּעַצְמָם (שָׁם עַמּוּד א׳), דִּבּוּר הַמַּתְחִיל "מִפְּנֵי הַדְּרָכִים", דְּדַי בְּאַחַת? וַהֲרֵי הָנָךְ דָּמוּ לְאָבִיב וּפֵרוֹת
