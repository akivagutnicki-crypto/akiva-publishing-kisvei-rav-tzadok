~ לְפִי זֶה. וּלְפִי זֶה, בֵּין כָּל תְּקוּפָה לִתְקוּפָה — צ״א יוֹם, וְז׳ שָׁעוֹת, וְתקי״ט חֲלָקִים, וְל״א רְגָעִים; וְיִתְרוֹן הַתְּקוּפָה עַל ג׳ חֳדָשִׁים שֶׁל לְבָנָה — ב׳ יָמִים, י״ז שָׁעוֹת, ש׳ חֲלָקִים, ל״א רְגָעִים, לְפִי חֶשְׁבּוֹן זֶה. וּלְפִי שֶׁכַּוָּנַת חֶשְׁבּוֹן זֶה לְהַשְׁווֹת הַחַמָּה עִם הַלְּבָנָה — לְפִיכָךְ גַּם כֵּן בָּזֶה לֹא נַחְשׁוֹב קְדִימַת תְּקוּפַת נִיסָן רִאשׁוֹנָה שֶׁל יְצִירָה לְמוֹלְדוֹ בְּז׳ יָמִים, ט׳ תרמ״ב, כְּמוֹ שֶׁנִּתְבָּאֵר לְעֵיל לְפִי חֶשְׁבּוֹן דַּעַת שְׁמוּאֵל; כִּי אִם כֵּן, יִקְדַּם עֲדַיִן כָּל כָּךְ שְׁנַת הַחַמָּה, וְלֹא יִשְׁתַּווּ אַף בְּי״ט שָׁנָה. לְפִיכָךְ סָבוּר בַּעַל הַחֶשְׁבּוֹן זֶה — כְּדַעַת הָאוֹמֵר שֶׁבְּנִיסָן נִבְרָא הָעוֹלָם, וְהָיְתָה תְּקוּפָה רִאשׁוֹנָה שֶׁל חַמָּה בִּתְחִלַּת לֵיל ד׳, שֶׁאָז נִבְרֵאת הַחַמָּה, וְאָז הָיְתָה תְּקוּפָתָהּ (וְאֵינוֹ סוֹבֵר כִּי נִבְרֵאת בְּסוֹף ג׳ שָׁעוֹת עַל הַיּוֹם, רַק בִּתְחִלַּת הַלַּיְלָה); וּמוֹלַד הַלְּבָנָה שֶׁל אוֹתוֹ חֹדֶשׁ — לְפִי חֶשְׁבּוֹן בהר״ד (שֶׁקִּבְּלוּ שֶׁהָיָה כֵּן מוֹלַד תִּשְׁרֵי שֶׁל שְׁנַת תֹּהוּ) — הָיָה מוֹלַדְתּוֹ בְּאוֹתָהּ לַיְלָה, בְּט׳ שָׁעוֹת וְתרמ״ב חֲלָקִים, כְּשֶׁנּוֹסִיף א׳ י״ב תשצ״ג לְכָל חֹדֶשׁ, כְּפִי הַחֶשְׁבּוֹן שֶׁבֵּאַרְנוּ לְמַעְלָה בְּחֶשְׁבּוֹן הַמּוֹלָדוֹת. וְנִמְצָא: קָדְמָה תְּקוּפַת נִיסָן שֶׁל חַמָּה לַלְּבָנָה בְּט׳ שָׁעוֹת וְתרמ״ב חֲלָקִים; וְכֵן הוּא תְּקוּפַת נִיסָן שֶׁל תְּחִלַּת כָּל מַחֲזוֹר קָטָן — קֹדֶם לְמוֹלְדוֹ:

**וְאִם** נִרְצֶה לֵידַע כָּל תְּקוּפָה בְּכָל שָׁנָה מִשְּׁנֵי הַמַּחֲזוֹר קָטָן — נִקַּח לְכָל שָׁנָה מֵהֶן י׳ כ״א קכ״א מ״ח, הַיִּתְרוֹן מִשָּׁנָה; וּלְכָל תְּקוּפָה — יִתְרוֹנָהּ, וְהוּא ב׳ י״ז ש׳ ל״א; וּנְקַבְּצֵם יַחַד — כָּל הָרְגָעִים לַחֲלָקִים, וְכָל הַחֲלָקִים שָׁעוֹת, וְהַשָּׁעוֹת יָמִים, כְּמִשְׁפָּט הָאָמוּר לְמַעְלָה; וְנִגְרַע מִן הַכֹּל חָדְשֵׁי הָעִבּוּר שֶׁעָבְרוּ בִּשְׁנוֹת הַמַּחֲזוֹר קָטָן — כָּל חֹדֶשׁ כ״ט י״ב תשצ״ג; וְגַם נִגְרַע ט׳ שָׁעוֹת תרמ״ב חֲלָקִים הַקְּדוּמָה; וְהַנִּשְׁאָר — נוֹסִיפוֹ עַל מוֹלַד שֶׁל אוֹתוֹ חֹדֶשׁ, וְהוּא זְמַנָּהּ שֶׁל תְּקוּפָה הַנִּרְצֵית, בִּימֵי הַחֹדֶשׁ וְהַשָּׁבוּעַ:

**וְכָל** מַחֲזוֹר קָטָן — שֶׁהוּא י״ט שָׁנָה — חוֹזֵר חֲלִילָה הַמֶּרְחָק מִן הַתְּקוּפָה לְהַמּוֹלָד, שֶׁבְּנִיסָן הָרִאשׁוֹן תָּמִיד רָחוֹק ט׳ תרמ״ב; וְכֵן כָּל הֶחֳדָשִׁים שָׁוִים בְּכָל מַחֲזוֹר קָטָן. וְעַיֵּן בַּמְּפָרֵשׁ שֶׁכָּתַב: "וְזוֹ הַתְּקוּפָה שֶׁל רַב אַדָּא לֹא תָּבוֹא עַל חֶשְׁבּוֹנָהּ תְּקוּפַת נִיסָן בְּט״ז לְעוֹלָם, וְתִהְיֶה הַשָּׁנָה מְעֻבֶּרֶת, וְתִהְיֶה הַתְּקוּפָה בַּאֲדָר; וְהַיְנוּ דְּאָמַר ר״א: כַּד חָזְיָא דְּמָשְׁכָא תְּקוּפַת נִיסָן עַד שִׁיתְסַר" וְכוּלֵי, עַד כָּאן. וּדְבָרָיו תְּמוּהִים, דְּהָא דִּבְרֵי ר״א דְּשָׁם — עד״ש בַּגְּמָרָא רֹאשׁ הַשָּׁנָה (כ״א עַמּוּד א׳), עַיֵּן שָׁם (לְפָנֵינוּ הַגִּרְסָא "רַב הוּנָא בַּר אָבִין", וְכֵן הֶעְתִּיק בְּרמ״ךְ בְּהַשָּׂגוֹת רֵישׁ פֶּרֶק ד׳; וְאִם כֵּן לֵיתָא) — קָאֵי אַזְּמַן שֶׁקִּדְּשׁוּ עַל פִּי הָרְאִיָּה; וְהַיְנוּ מִפְּנֵי הַתְּקוּפָה הַנִּזְכָּר בְּפֶרֶק קַמָּא דְּסַנְהֶדְרִין, וְהֵן הֵן דִּבְרֵי הָרַמְבַּ״ם
