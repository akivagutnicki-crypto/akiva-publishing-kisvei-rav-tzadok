~ בֵּינוֹ וּבֵין יְרוּשָׁלַיִם מַהֲלַךְ יֶתֶר עַל עֲשָׂרָה יָמִים — עוֹשִׂין ב׳ יָמִים לְעוֹלָם, כְּמִנְהָגָם מִקֹּדֶם; שֶׁאֵין שְׁלוּחֵי כָּל תִּשְׁרֵי וְתִשְׁרֵי מַגִּיעִים אֶלָּא לְמָקוֹם שֶׁיֵּשׁ בֵּינוֹ וּבֵין יְרוּשָׁלַיִם מַהֲלַךְ עֲשָׂרָה יָמִים אוֹ פָּחוֹת. וּמָקוֹם שֶׁהוּא, כַּנִּזְכָּר לְעֵיל, שֶׁאֶפְשָׁר שֶׁהָיוּ שְׁלוּחִים מַגִּיעִים אֵלָיו — רוֹאִים: אִם אוֹתוֹ הַמָּקוֹם מֵאֶרֶץ יִשְׂרָאֵל, אוֹ אֲפִלּוּ מִחוּץ לָאָרֶץ, וּבִלְבַד שֶׁיּוֹדְעִים שֶׁהָיוּ שְׁלוּחִים מַגִּיעִים שָׁם — כְּגוֹן שֶׁהָיוּ בָּהּ יִשְׂרָאֵל בִּשְׁעַת הָרְאִיָּה, בְּכִבּוּשׁ הַב׳, כְּגוֹן אוּשָׁא, וְשַׁפְרְעָם, וְלוּד, וְיַבְנֶה, וּטְבֶרְיָא (וּצְפַת וְחֶבְרוֹן), וְכַיּוֹצֵא בָּהֶם — עוֹשִׂין יוֹם אֶחָד בִּלְבַד. וְאִם אוֹתוֹ הַמָּקוֹם מִסּוּרְיָא — כְּגוֹן צוֹר וְדַמֶּשֶׂק וְאַשְׁקְלוֹן, וְכַיּוֹצֵא בָּהֶם — אוֹ מִחוּץ לָאָרֶץ — כְּגוֹן מִצְרַיִם וְעַמּוֹן וּמוֹאָב, וְכַיּוֹצֵא בָּהֶן — עוֹשִׂין כְּמִנְהַג אֲבוֹתֵיהֶם: אִם יוֹם אֶחָד — יוֹם אֶחָד, וְאִם ב׳ יָמִים — ב׳ יָמִים. וְאִם אֵין לָהֶם מִנְהָג, אוֹ שֶׁהִיא עִיר שֶׁנִּתְחַדְּשָׁה בְּמִדְבַּר אֶרֶץ יִשְׂרָאֵל, אוֹ שֶׁהוּא מָקוֹם שֶׁיָּשְׁבוּ בּוֹ יִשְׂרָאֵל עַתָּה, וְלֹא יָשְׁבוּ אָז — עוֹשִׂין ב׳ יָמִים, כְּמִנְהַג רֹב הָעוֹלָם:

(**לֹא** יָדַעְתִּי מַדּוּעַ — הָא לֹא שַׁיָּךְ חֲשָׁשָׁא דְּיַחֲזוֹר הַדָּבָר לְקִלְקוּלוֹ. וְאוּלַי סָבִירָא לֵיהּ: מִכָּל מָקוֹם, "מִנְהַג אֲבוֹתֵינוּ בִּידֵינוּ" — אַף בְּמָקוֹם דְּלֹא שַׁיָּךְ הַאי טַעְמָא; וְצָרִיךְ עִיּוּן. וְדַע, עוֹד צָרִיךְ עִיּוּן לְפִי עֲנִיּוּת דַּעְתִּי: מֵאַיִן נַחְשׁוֹב מַהֲלַךְ עֲשָׂרָה יוֹם? הָא לֹא תָּמִיד קָבְעוּ וְקִדְּשׁוּ בִּירוּשָׁלַיִם — רַק פְּעָמִים בְּאוּשָׁא, וּפְעָמִים בְּעֵין טַב, וּבִשְׁאָר מְקוֹמוֹת. וְאוּלַי, לְפִי שֶׁבִּזְמַן שֶׁבֵּית הַמִּקְדָּשׁ קַיָּם לֹא קִדְּשׁוּ אֶלָּא בִּירוּשָׁלַיִם, מְקוֹם מוֹשַׁב הַסַּנְהֶדְרִין, כְּמוֹ שֶׁנִּתְבָּאֵר לְעֵיל — לְכָךְ חָשְׁבִינַן מַהֲלַךְ הָעֲשָׂרָה יוֹם מִירוּשָׁלַיִם, לֹא מִזּוּלַת הָעֲיָרוֹת; וְצָרִיךְ עִיּוּן):

# שְׁאָר דִּינִים יִתְבָּאֵר בְּהִלְכוֹת יוֹם טוֹב

**כֵּיצַד** הוּא הַחֶשְׁבּוֹן? הַשָּׁבוּעַ הוּא ז׳ יָמִים, וְהַיּוֹם נֶחְלָק לְכ״ד שָׁעוֹת (וּבְנִיסָן וְתִשְׁרֵי — הַיּוֹם י״ב שָׁעוֹת וְהַלַּיְלָה י״ב שָׁעוֹת, שֶׁשְּׁנֵיהֶם שָׁוִים; וּבְזוּלָתָם מֵהֶחֳדָשִׁים — שֶׁפְּעָמִים הַיּוֹם גָּדוֹל מֵהַלַּיְלָה, וּפְעָמִים הַלַּיְלָה גְּדוֹלָה). הַשָּׁעוֹת עַל ב׳ פָּנִים: שָׁעוֹת שָׁווֹת וְשָׁעוֹת זְמַנִּיּוֹת. הַשָּׁווֹת — שָׁוִים בְּשִׁעוּרָם, אֲבָל לֹא בְּמִנְיָנָם בַּיּוֹם וּבַלַּיְלָה, שֶׁהֵם כָּל אַחַת מִכ״ד בְּמֵעֵת לְעֵת; וְנִמְצָא, כְּשֶׁהַלַּיְלָה קְצָרָה — הֵם בַּלַּיְלָה פָּחוֹת מִי״ב שָׁעוֹת, וּבַיּוֹם יֶתֶר עַל י״ב שָׁעוֹת, וּלְהֵפֶךְ כְּשֶׁהַלַּיְלָה גְּדוֹלָה. וְהַזְּמַנִּיּוֹת — הֵם לְהֵפֶךְ: שָׁוִים בְּמִנְיָנָם בַּיּוֹם וּבַלַּיְלָה, אֲבָל לֹא בְּשִׁעוּרָם; שֶׁהֵם תָּמִיד נֶחְשָׁבִים י״ב שָׁעוֹת לַיּוֹם וְי״ב שָׁעוֹת לַלַּיְלָה; וּכְשֶׁהַלַּיְלָה קְטַנָּה — נִמְצְאוּ שְׁעוֹת הַלַּיְלָה קְטַנִּים בְּשִׁעוּרָם מִשֶּׁל יוֹם, וּלְהֵפֶךְ אִם הִיא גְּדוֹלָה. וּפֹה, אִם נְדַבֵּר מִשָּׁעוֹת — יִהְיֶה כַּוָּנָתֵנוּ
