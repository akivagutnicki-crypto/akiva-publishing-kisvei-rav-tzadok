~ גַּוְנָא שֶׁאֵינוֹ אָנוּס. וְהֲגַם דְּרַשִׁ״י דָּחַק לְפָרֵשׁ כֵּן לְשׁוֹן "חַיָּב" — מִכָּל מָקוֹם יֵשׁ לוֹמַר, דִּסְבִירָא לֵיהּ לְהַתּוֹסָפוֹת: דְּבֶאֱמֶת, הֵיכִי דְּאָנוּס — פָּטוּר מִשּׁוּם "בְּדֶרֶךְ רְחוֹקָה"; דַּהֲרֵי לֹא בֵּאֲרָה תּוֹרָה אֵיזֶהוּ דֶּרֶךְ רְחוֹקָה, וְהַכֹּל לְפִי מַה שֶּׁהוּא אָדָם; רַק דַּחֲכָמִים שִׁעֲרוּ מִן הַמּוֹדִיעִים וְלִפְנִים, כְּפִי דֶּרֶךְ מַהֲלַךְ אָדָם בֵּינוֹנִי, שֶׁזֶּהוּ הַשִּׁעוּר לְרֹב בְּנֵי אָדָם; אֲבָל לַזָּקֵן וּכְהַאי גַּוְנָא — סְבִירָא לֵיהּ לְהַתּוֹסָפוֹת דְּזֶהוּ דֶּרֶךְ רְחוֹקָה, וְלֹא מִשּׁוּם אָנוּס לְחוּד הוּא פָּטוּר, אֶלָּא מִשּׁוּם דֶּרֶךְ רְחוֹקָה. וְהַנַּפְקָא מִינַּהּ: דְּאִי מִשּׁוּם אָנוּס — אִם הָיָה יָכוֹל לְסַלֵּק אָנְסוֹ, לֵילֵךְ בִּקְרוֹן אוֹ בְּסוּסִים וּבִפְרָדִים — חַיָּב; אֲבָל מִשּׁוּם דֶּרֶךְ רְחוֹקָה — פָּטוּר; וְכִמְפֹרָשׁ בַּגְּמָרָא: "הָיָה" וְכוּלֵי "וְיָכוֹל לִיכָּנֵס בְּסוּסִים וּפְרָדִים — יָכוֹל יְהֵא חַיָּב? תַּלְמוּד לוֹמַר" וְכוּלֵי. וּלְפִי זֶה, שַׁפִּיר כָּתְבוּ הַתּוֹסָפוֹת: זָקֵן שֶׁאֵין יָכוֹל לֵילֵךְ, דְּפָטוּר מֵרְאִיָּה — דְּהַיְנוּ שֶׁאֵין יָכוֹל לֵילֵךְ מִירוּשָׁלַיִם לְהַר הַבַּיִת — פָּטוּר גַּם כֵּן מִפֶּסַח, כַּנִּזְכָּר לְעֵיל, דְּאֵין צָרִיךְ לִיכָּנֵס בְּסוּסִים וְכוּלֵי. וְלֹא קָאֵי מַה שֶּׁכָּתְבוּ הַתּוֹסָפוֹת "דְּפָטוּר מִפֶּסַח" וְכוּלֵי אֶלָּא אַהָךְ תֵּרוּץ דְּזָקֵן — לֹא אַתֵּרוּץ דְּ"אֵין לוֹ קַרְקַע וְחוּץ לָאָרֶץ"; וְהָךְ תֵּרוּץ דְּ"אֵין לוֹ קַרְקַע" — הוּא רַק אַלַּעֲלוֹת לָרֶגֶל, דִּכְמוֹ כֵן קֻשְׁיַית הַתּוֹסָפוֹת לֹא הָיְתָה אֶלָּא עַל זֶה; דְּמִפֶּסַח לֹא קָשֶׁה לְהוּ כְּלָל, דְּפָשׁוּט יֵשׁ לוֹמַר דְּהָיָה בְּדֶרֶךְ רְחוֹקָה, חוּץ לַמּוֹדִיעִים; וְרַק אַלַּעֲלוֹת לָרֶגֶל, דְּחִיּוּבָא הוּא מִכָּל אֶרֶץ יִשְׂרָאֵל — עַל זֶה הִקְשׁוּ, וְתֵרְצוּ הָנָךְ תֵּרוּצִים. וּבְהָךְ תֵּרוּץ דְּזָקֵן, דְּנִיחָא גַּם בְּפֶסַח, אֲפִלּוּ הָיָה מִן הַמּוֹדִיעִים וְלִפְנִים — כָּתְבוּ דִּבְזֶה נִיחָא גַּם בְּפֶסַח, אֲפִלּוּ הָיָה תּוֹךְ יְרוּשָׁלַיִם, כָּל שֶׁכֵּן רָחוֹק קְצָת. כֵּן נִרְאֶה לַעֲנִיּוּת דַּעְתִּי בְּדִבְרֵי הַתּוֹסָפוֹת. וְעַיֵּן בַּצְּלָ״ח מַה שֶּׁכָּתַב בְּיִשּׁוּב דִּבְרֵיהֶם, וּדְבָרָיו דְּחוּקִים בְּמַשְׁמָעוּת לְשׁוֹנָם):

**וְיֵשׁ** חוֹלְקִים עַל זֶה (רַמְבַּ״ם פֶּרֶק ה׳, הֲלָכָה ט׳), וְאוֹמְרִים: דְּכֵיוָן שֶׁאֵינוֹ בְּדֶרֶךְ רְחוֹקָה (כְּשִׁעוּר שֶׁיִּתְבָּאֵר בְּפָרָשַׁת בְּהַעֲלֹתְךָ), אַף עַל פִּי שֶׁהוּא אֵינוֹ יָכוֹל לַעֲלוֹת — לָא מִקְּרֵי גַּבֵּיהּ דֶּרֶךְ רְחוֹקָה, אֶלָּא הָוֵי לֵיהּ אָנוּס, וּפָטוּר מִפְּנֵי אֲנִיסוּתוֹ — אִם אִי אֶפְשָׁר לוֹ לְתַקֵּן; וְאִם אֶפְשָׁר לוֹ לַעֲלוֹת בְּסוּסִים וּבִפְרָדִים — חַיָּב לַעֲלוֹת, כֵּיוָן דְּאֵינוֹ אָנוּס מֵעַתָּה; וְכֵן עִקָּר:

**מַה** הוּא מֵבִיא לְקָרְבַּן פֶּסַח? זָכָר (וְהַמַּקְדִּישׁ נְקֵבָה לְקָרְבַּן פֶּסַח — יִתְבָּאֵר בִּשְׁאָר קָרְבָּנוֹת, דְּדִינוֹ שָׁוֶה לָהֶם), בֶּן שָׁנָה (בְּשָׁעוֹת; וְאִם עָבְרָה שְׁנָתוֹ — אֲפִלּוּ אַחַר שְׁחִיטָה, קֹדֶם זְרִיקָה — פָּסוּל; וְיִתְבָּאֵר בִּשְׁאָר קָרְבָּנוֹת פְּרָטֵי דִּינוֹ), תָּם (וְלֹא בַּעַל מוּם; וְיִתְבָּאֵר בִּמְקוֹמוֹ דִּינֵי הַבַּעַל מוּם), מִן הַכְּבָשִׂים אוֹ מִן הָעִזִּים — אֵיזֶה שֶׁיִּרְצֶה (גְּמָרָא נ״ז עַמּוּד ב׳, וּמִשְׁנָה שִׁלְהֵי כְּרִיתוֹת); וְאֵין לְאֶחָד קְדִימָה וּבְחִירָה עַל חֲבֵרוֹ, אֶלָּא שְׁנֵיהֶם שָׁוִים. וּמִכָּל מָקוֹם (כֵּן מַשְׁמַע בַּגְּמָרָא פ״ח עַמּוּד ב׳), רֹב
