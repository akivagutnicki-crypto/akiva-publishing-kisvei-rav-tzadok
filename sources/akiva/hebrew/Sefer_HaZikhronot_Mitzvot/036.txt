~ בְּסוֹף פֶּרֶק ג׳ דְּאָבוֹת: "תְּקוּפוֹת וְגִימַטְרִיָּאוֹת — פַּרְפְּרָאוֹת לַחָכְמָה". וְעַיֵּן שָׁם בְּתוֹסְפוֹת יוֹם טוֹב כָּתַב: "וְצָרִיךְ לוֹמַר שֶׁדְּבָרָיו עַל הַיְדִיעָה מִן הַמַּהֲלָכִים בִּלְבַד, וְלֹא עַל הַנִּמְשָׁךְ מֵהֶם, שֶׁהוּא קִדּוּשׁ הַחֹדֶשׁ וְעִבּוּר שָׁנָה, שֶׁלְּעִנְיָן זֶה וַדַּאי שֶׁהֵן הֵן גּוּפֵי הֲלָכוֹת", עַד כָּאן. וְאִם מְבָרְכִים עַל מִצְוָה זוֹ — עַיֵּן אַבּוּדַרְהַם (דַּף ז׳ עַמּוּד א׳), וְיֵשׁ לְפַלְפֵּל בִּדְבָרָיו; וְעַיֵּן בְּמַסֶּכֶת סוֹפְרִים (פֶּרֶק י״ט הֲלָכָה ט׳), עַיֵּן שָׁם הֵיטֵב. וְעַיֵּן עוֹד בִּשְׁמוֹת רַבָּה פָּרָשָׁה ט״ו (דַּף מ״ח עַמּוּד ב׳). וְעַיֵּן בְּשַׁבָּת (ע״ה עַמּוּד א׳): אָמַר רַב זוּטְרָא בַּר טוֹבִיָּה אָמַר רַב וְכוּלֵי: הַיּוֹדֵעַ לַחְשׁוֹב בִּתְקוּפוֹת וּמַזָּלוֹת וְאֵינוֹ חוֹשֵׁב — אָסוּר לְסַפֵּר הֵימֶנּוּ. וְאָמַר רַבִּי שִׁמְעוֹן בֶּן פָּזִי, אָמַר רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי מִשּׁוּם בַּר קַפָּרָא: כָּל הַיּוֹדֵעַ לַחְשׁוֹב בִּתְקוּפוֹת וּמַזָּלוֹת וְאֵינוֹ חוֹשֵׁב — עָלָיו הַכָּתוּב אוֹמֵר: "וְאֵת פֹּעַל ה׳ לֹא יַבִּיטוּ, וּמַעֲשֵׂה יָדָיו לֹא רָאוּ". וְאָמַר רַבִּי שְׁמוּאֵל בַּר נַחְמָנִי, אָמַר רַבִּי יוֹחָנָן: מִנַּיִן שֶׁמִּצְוָה עַל הָאָדָם לַחְשׁוֹב בִּתְקוּפוֹת וּמַזָּלוֹת? שֶׁנֶּאֱמַר: "כִּי הִיא חָכְמַתְכֶם וּבִינַתְכֶם לְעֵינֵי הָעַמִּים" — אֵיזוֹ חָכְמָה שֶׁלְּעֵינֵי הָעַמִּים? זֶה חִשּׁוּב תְּקוּפוֹת וּמַזָּלוֹת. (עַיֵּן שָׁם בְּרַשִׁ״י, שֶׁפֵּרֵשׁ לְפִי שֶׁחָכְמָה הַנִּכֶּרֶת הִיא, שֶׁמַּרְאָה לָהֶם סִימָן לִדְבָרָיו, בְּהִלּוּךְ הַחַמָּה וְהַמַּזָּלוֹת, שֶׁמְּעִידִין שֶׁהוּא אוֹמֵר "שָׁנָה זוֹ שְׁחוּנָה", וְהוּא כֵּן, וְכוּלֵי. וְלִי נִרְאֶה פָּשׁוּט, לְפִי שֶׁגַּם מֵאָז, מִימֵי קֶדֶם, הֶחֱזִיקוּ הָעוֹבְדֵי כּוֹכָבִים בְּחָכְמָה זוֹ — חָכְמַת הַתְּכוּנָה וְחִשּׁוּב תְּקוּפוֹת וּמַזָּלוֹת — כַּיָּדוּעַ, וּמִימֵיהֶם נִתְפַּלְּאוּ עַל חֶשְׁבּוֹנֵנוּ בִּקְבִיעוּת חֳדָשִׁים וְשָׁנִים, כִּי הוּא אֲמִתִּי וְנִכְבָּד מְאֹד, כַּיָּדוּעַ; וְזֶהוּ "חָכְמָה שֶׁלְּעֵינֵי הָעַמִּים", וּפָשׁוּט.) וּמַשְׁמַע דְּזֶה קָאֵי אִידִיעַת הַמַּהֲלָךְ בִּלְבַד, לֹא אַמִּצְוָה הַנִּמְשֶׁכֶת מִמֶּנּוּ; דְּאִם כֵּן, מַאי קָאָמַר "מִנַּיִן שֶׁמִּצְוָה" וְכוּלֵי. וְכֵן הוּא בְּסֵפֶר מִצְוֹת גָּדוֹל (עֲשִׂין מ״ז), הוּבָא בְּהַגָּהוֹת מַיְמוֹנִיּוֹת, מוֹנֶה ב׳ מִצְוֹת: א׳ — פְּרָטֵי לְבֵית דִּין; ב׳ — כְּלָלֵי, לְחַשֵּׁב תְּקוּפוֹת וּמַזָּלוֹת, עַיֵּן שָׁם. וְאוּלַי מִצְוָה זוֹ אַף עַכְשָׁיו; אוֹ אֶפְשָׁר רַק בִּזְמַן שֶׁהָיוּ קוֹבְעִים הַחֳדָשִׁים עַל פִּי זֶה הַחֶשְׁבּוֹן, לֹא עַכְשָׁיו; וְצָרִיךְ לְעַיֵּן. וּבִשְׁמוֹת רַבָּה (פָּרָשָׁה ט״ו): "מַגִּיד דְּבָרָיו לְיַעֲקֹב, חֻקָּיו וּמִשְׁפָּטָיו לְיִשְׂרָאֵל" — זֶה קִדּוּשׁ הַחֹדֶשׁ, שֶׁיֵּשׁ בָּהֶם חֻקִּים וּמִשְׁפָּטִים: חֻקִּים — אֵלּוּ הַמּוֹעֲדוֹת; וּמִשְׁפָּטִים — שֶׁכָּךְ הַדַּיָּנִין בּוֹדְקִים בִּשְׁטָר (וְכֵן עֵדִים) בְּאֵיזֶה חֹדֶשׁ, בְּכַמָּה בַּחֹדֶשׁ, וְעַל פִּי כָךְ הֵם דָּנִין:

**עִנְיַן** קִדּוּשׁ הַחֹדֶשׁ וְעִבּוּר הַשָּׁנָה הוּא — כִּי חָדְשֵׁי הַשָּׁנָה הֵם חָדְשֵׁי הַלְּבָנָה, שֶׁיִּשְׂרָאֵל מוֹנִין לַלְּבָנָה, שֶׁנֶּאֱמַר: "הַחֹדֶשׁ הַזֶּה לָכֶם"; וְאָמְרוּ חֲכָמִים: הֶרְאָה לוֹ הַקָּדוֹשׁ בָּרוּךְ הוּא לְמֹשֶׁה בִּנְבוּאָה דְּמוּת לְבָנָה בְּחִדּוּשָׁהּ, וְאָמַר לוֹ: "כָּזֶה רְאֵה וְקַדֵּשׁ". וְהַשָּׁנִים שֶׁאָנוּ מְחַשְּׁבִין הֵם שְׁנֵי הַחַמָּה, שֶׁנֶּאֱמַר: "שָׁמוֹר אֶת חֹדֶשׁ הָאָבִיב"; וְכֵן כָּל הַמּוֹעֲדִים תְּלוּיִם בַּחַמָּה — חַג הַשָּׁבוּעוֹת בִּזְמַן הַקָּצִיר, וְחַג הַסֻּכּוֹת בִּזְמַן הָאָסִיף. וּשְׁנַת הַחַמָּה, כְּפִי מַה שֶּׁיִּתְבָּאֵר לְהַלָּן, יָמֶיהָ שס״ה וּרְבִיעַ, וּלְדַעַת יֵשׁ אוֹמְרִים פָּחוֹת מִזֶּה, כְּמוֹ שֶׁיִּתְבָּאֵר. וּשְׁנַת הַלְּבָנָה — וְהוּא
