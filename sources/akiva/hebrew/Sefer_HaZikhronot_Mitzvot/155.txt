**וְהוּא** שֶׁשָּׁהָה וְהוּא עוֹמֵד עַל קַרְקָעִית הָעֲזָרָה; אֲבָל תָּלָה עַצְמוֹ בַּאֲוִיר הָעֲזָרָה וְשָׁהָה — אַף עַל גַּב דְּנִתְבָּאֵר לְעֵיל, בַּסִּימָן שֶׁלִּפְנֵי זֶה, דַּאֲוִיר עֲזָרָה כַּעֲזָרָה — מִכָּל מָקוֹם בָּעֵינַן שְׁהִיָּה הָרְאוּיָה לְהִשְׁתַּחֲוָאָה, וּבִקְרָא דְּהִשְׁתַּחֲוָה (חִדּוּשֵׁי הָרַשְׁבָּ״א) דִּלְעֵיל "אַרְצָה עַל הָרִצְפָּה" כְּתִיב; אֲבָל בָּאֲוִיר אֵין דֶּרֶךְ הִשְׁתַּחֲוָאָה, וְלָא הֲוֵי שְׁהִיָּה הָרְאוּיָה לְהִשְׁתַּחֲוָאָה; וּלְפִיכָךְ פָּטוּר (שְׁבוּעוֹת י״ז עַמּוּד א׳ — בְּעָיָא דְּלֹא אִפְּשִׁיטָא, וּלְהָקֵל. וְצָרִיךְ עִיּוּן עַל הָרַמְבַּ״ם, שֶׁהִשְׁמִיט זֶה; לֹא אֵדַע טַעְמוֹ):

**וַאֲפִלּוּ** לֹא שָׁהָה כְּשִׁעוּר הַנִּזְכָּר לְעֵיל, אֶלָּא שֶׁהִשְׁתַּחֲוָה (שָׁם ט״ז עַמּוּד ב׳) הִשְׁתַּחֲוָאָה שֶׁאֵין בָּהּ שִׁעוּר שְׁהִיָּה הַנִּזְכָּר לְעֵיל — וְהִיא כְּרִיעָה בְּעָלְמָא — וּבִלְבַד שֶׁהֶחֱזִיר פָּנָיו לַהֵיכָל וְהִשְׁתַּחֲוָה — הֲרֵי זֶה חַיָּב כָּרֵת וּמַלְקוֹת בְּמֵזִיד, וּבְשׁוֹגֵג קָרְבָּן. אֲבָל אִם לֹא הֶחֱזִיר פָּנָיו, אֶלָּא הִשְׁתַּחֲוָה דֶּרֶךְ יְצִיאָתוֹ כְּלַפֵּי חוּץ — הֲרֵי זֶה פָּטוּר, אִם לֹא שָׁהָה כְּשִׁעוּר. (זֶהוּ לְשׁוֹן הָרַמְבַּ״ם, וְנִמְשַׁךְ בִּלְשׁוֹנוֹ אַחַר דִּבְרֵי רַבּוֹ הָרִ״י ן׳ מִיגַּשׁ, שֶׁהֵבִיא בְּחִדּוּשֵׁי הָרַשְׁבָּ״א, דִּמְפָרֵשׁ: "כְּלַפֵּי פְּנִים" — שֶׁהֶחֱזִיר פָּנָיו לִפְנִים, וְהוּא הָיָה צָרִיךְ לָשׂוּם פָּנָיו כְּלַפֵּי חוּץ וְלָצֵאת — אַף עַל פִּי שֶׁאָנַס עַצְמוֹ וְלֹא שָׁהָה (נִרְאֶה לִי, דְּרוֹצֶה לוֹמַר: לֹא שָׁהָה כְּלָל בַּהַחְזָרַת פָּנִים, וּכְאִלּוּ לֹא הֶחֱזִיר; דְּאִי אַשִּׁעוּר שְׁהִיָּה — מַאי "אָנַס" דְּקָאָמַר?) — חַיָּב; וּ"כְלַפֵּי חוּץ" — הַיְנוּ דֶּרֶךְ יְצִיאָתוֹ, וְכֵיוָן שֶׁלֹּא שָׁהָה כְּלוּם — פָּטוּר, עַד כָּאן. וּלְפִי דְּבָרָיו, אִם הֶחֱזִיר פָּנָיו לְאֵיזֶה צַד שֶׁיִּהְיֶה לְהִשְׁתַּחֲווֹת — וּבִלְבַד שֶׁלֹּא כְּדֶרֶךְ יְצִיאָתוֹ — חַיָּב. אֲבָל רַשִׁ״י פֵּרֵשׁ: "כְּלַפֵּי פְּנִים" — לְצַד מַעֲרָב, שֶׁזֶּהוּ דֶּרֶךְ הִשְׁתַּחֲוָאָה; וּ"כְלַפֵּי חוּץ" — לְצַד מִזְרָח, שֶׁאֵינוֹ דֶּרֶךְ הִשְׁתַּחֲוָאָה; וְלֹא בְּדֶרֶךְ יְצִיאָתוֹ תַּלְיָא. וְהוּא הַדִּין, הֶחֱזִיר פָּנָיו לְצַד צָפוֹן וּכְהַנִּזְכָּר לְעֵיל — שֶׁלֹּא כְּלַפֵּי הַהֵיכָל — וְהִשְׁתַּחֲוָה, דְּלָאו דֶּרֶךְ הִשְׁתַּחֲוָאָה — אַף עַל פִּי שֶׁהֶחֱזִיר פָּנָיו, פָּטוּר):

**וַאֲפִלּוּ** לֹא שָׁהָה וְלֹא הִשְׁתַּחֲוָה, אֶלָּא שֶׁכְּשֶׁבָּא לָצֵאת מִן הַמִּקְדָּשׁ — שֶׁלֹּא יִשְׁהֶה שִׁעוּר שְׁהִיָּה הַנִּזְכָּר לְעֵיל — הָלַךְ בַּדֶּרֶךְ אֲרוּכָּה — הֲרֵי זֶה חַיָּב. וְאֵיזֶהוּ דֶּרֶךְ אֲרוּכָּה? כָּל שֶׁאֶפְשָׁר לוֹ לָצֵאת מִן הַמִּקְדָּשׁ בִּקְצָרָה מִמֶּנָּה. יָצָא בִּקְצָרָה — אַף עַל פִּי שֶׁלֹּא רָץ, אֶלָּא הָלַךְ עָקֵב בְּצַד גּוּדָל, אֲפִלּוּ כָּל הַיּוֹם — פָּטוּר. יָצָא בַּאֲרוּכָּה — אַף עַל פִּי שֶׁרָץ וְדָחַק עַצְמוֹ בְּכָל כֹּחוֹ, וְנִמְצָא שִׁעוּר זְמַן שֶׁהָלַךְ בּוֹ בַּאֲרוּכָּה פָּחוֹת מִשִּׁעוּר הַזְּמַן שֶׁמְּהַלֵּךְ כָּל אֶחָד בִּקְצָרָה — הוֹאִיל וְיָצָא בַּאֲרוּכָּה, חַיָּב: בְּמֵזִיד — כָּרֵת; וּבְשׁוֹגֵג (בְּהֶעְלֵם טֻמְאָה אוֹ מִקְדָּשׁ, אִי נַמִּי שֶׁחָשַׁב שֶׁזֶּהוּ דֶּרֶךְ הַקְּצָרָה) — חַטָּאת. וְאִם יָצָא בִּקְצָרָה, וְהָלַךְ מְעַט וְעָמַד וְשָׁהָה מְעַט, וְכֵן עָשָׂה כַּמָּה פְּעָמִים, עַד שֶׁנִּצְטָרֵף מִכָּל הַשְּׁהִיּוֹת כְּשִׁעוּר שֶׁנִּתְבָּאֵר — פָּטוּר; אֲבָל לוֹקֶה מַכַּת מַרְדּוּת, אִם עָשָׂה בְּמֵזִיד. זֶהוּ דַּעַת הָרַמְבַּ״ם. וְנִרְאֶה לִי, דְּהוּא הַדִּין לְכָל הָנָךְ פְּטוּרֵי דִּלְעֵיל — דְּלוֹקֶה מַכַּת מַרְדּוּת מִיהַת:
