~ בִּפְנֵי הָעָם, וְרוֹצֶה לְכַסּוֹת הָעֶרְוָה בְּיָדוֹ, וְכַהַאי גַּוְנָא — אָסוּר לוֹ לִיגַּע בָּהּ, אִם לֹא עַל יְדֵי מַטְלִית עָבָה. (וְאִם נִזְדַּעְזְעוּ אֵיבָרָיו וְרָאָה קֶרִי — רַשַּׁאי לִיגַּע אַף בְּמַטְלִית רַכָּה; וְנִרְאֶה לִי אַף בְּיָדוֹ, דְּכָל אַחְמוּמֵי וַהֲדַר אִיחְמוּמֵי בְּשַׁעְתֵּיהּ לָא שְׁכִיחַ.) וְדַוְקָא בָּאַמָּה עַצְמָהּ, אֲבָל לְמַעְלָה מֵאַמָּתוֹ — בַּבֵּיצִים אוֹ בַּשֵּׂעָר — אוֹ מֵעֲטָרָה וּלְמַטָּה שַׁפִּיר דָּמֵי. (מִיהוּ הַתּוֹסָפוֹת רֵישׁ פֶּרֶק כָּל הַיָּד כָּתְבוּ: אִי נַמִּי אֲפִלּוּ מֵעֲטָרָה וּלְמַטָּה לָא שְׁרֵי אֶלָּא לְהַשְׁתִּין וְכוּלֵי, אֲבָל לִבְדּוֹק לָא, וְצָרִיךְ עִיּוּן.) מִיהוּ לֶאֱחוֹז בְּכַח — כָּתְבוּ הַתּוֹסָפוֹת רֵישׁ פֶּרֶק כָּל הַיָּד דַּאֲפִלּוּ מֵעֲטָרָה וּלְמַטָּה אָסוּר; וְנִרְאֶה דִּלְמַעְלָה מֵהָאַמָּה אֲפִלּוּ בְּכַח שַׁפִּיר דָּמֵי. וּבִמְקוֹם דְּאִית לֵיהּ בִּיעֲתוּתָא, דִּנְהָרָא וְכַהַאי גַּוְנָא, אוֹ שֶׁהוּא בַּלַּיְלָה וְיֵשׁ לוֹ בִּיעֲתוּתָא, דִּלֵּילָא — נִרְאֶה דְּשַׁפִּיר דָּמֵי. וְאִיתָא בְּשַׁבָּת פֶּרֶק כָּל כִּתְבֵי, דְּרַבֵּינוּ הַקָּדוֹשׁ לֹא הִכְנִיס יָדוֹ לְמַטָּה מֵאַבְנֵטוֹ, וּמַשְׁמָע רַק הוּא דַּעֲבִיד הָכֵי וְלֹא שְׁאָר תַּנָּאִים. אֲבָל בְּשֻׁלְחָן עָרוּךְ כָּתַב זֶה לְמִדַּת חֲסִידוּת לְכָל אָדָם. מִיהוּ מֵהַתּוֹסָפוֹת רֵישׁ פֶּרֶק כָּל הַיָּד מַשְׁמָע דִּמְפָרְשֵׁי אַף רַבֵּינוּ הַקָּדוֹשׁ לֹא לְמַטָּה מֵאַבְנֵטוֹ מַמָּשׁ, אֶלָּא קָאֵי אָאַמָּה, וּרְבוּתָא דִּידֵיהּ לְהַשְׁתִּין דְּשַׁפִּיר דָּמֵי בְּנָשׂוּי, כְּמוֹ שֶׁיִּתְבָּאֵר. וְנִרְאֶה אַף הַנּוֹהֲגִים מִדַּת חֲסִידוּת וּקְדֻשָּׁה כְּרַבֵּינוּ הַקָּדוֹשׁ — מִכָּל מָקוֹם עַל יְדֵי מַטְלִית אַף רַכָּה אֶפְשָׁר שַׁפִּיר דָּמֵי. וְכָל זֶה בְּלֹא צֹרֶךְ קְצָת; אֲבָל אִם רוֹצֶה לְהַשְׁתִּין, וְיָרֵא שֶׁמָּא יִפְּלוּ הַרְבֵּה נִיצוֹצוֹת עַל רַגְלָיו וְיִהְיֶה נִרְאֶה כִּכְרוּת שָׁפְכָה (וְאֵינוֹ עוֹמֵד בְּמָקוֹם גָּבֹהַּ וּמַשְׁתִּין בְּמָקוֹם נָמוּךְ, וְכֵן אֵינוֹ מַשְׁתִּין בֶּעָפָר תִּיחוּחַ וְלֹא מְיֻשָּׁב; אִי נַמִּי עוֹמֵד בְּבֵית הַכִּסֵּא וּמַשְׁתִּין לַחוּץ, וְיָרֵא שֶׁמָּא יִפְּלוּ נִיצוֹצוֹת בְּבֵית הַכִּסֵּא — נַמִּי שַׁפִּיר דָּמֵי): אִם הוּא נָשׂוּי וְאִשְׁתּוֹ עִמּוֹ וְאֵינָהּ נִדָּה — רַשַּׁאי לֶאֱחוֹז בָּאַמָּתוֹ וּלְהַשְׁתִּין. וְאִם אֵינוֹ נָשׂוּי, אוֹ נָשׂוּי וְאֵין אִשְׁתּוֹ עִמּוֹ, אוֹ שֶׁהִיא נִדָּה — אָסוּר לֶאֱחוֹז בָּאַמָּתוֹ; וְאִם יָרֵא מִשּׁוּם נִיצוֹצוֹת, מְסַיֵּעַ בַּבֵּיצִים אוֹ אוֹחֵז מֵעֲטָרָה וּלְמַטָּה, דִּשְׁרֵי כַּנִּזְכָּר. וְאִיתָא בַּגְּמָרָא דְּהֵיכִי דְּאִיכָּא בִּיעֲתוּתָא — כְּגוֹן שֶׁהוּא בַּלַּיְלָה, אוֹ עוֹמֵד בְּאִיגְּרָא, אוֹ לִפְנֵי רַבּוֹ, וּבָעִית מִפָּנָיו; אִי נַמִּי הוֹלֵךְ לִנְהָרָא, וּבָעִית, אוֹ בְּמִלְחָמָה, אוֹ שֶׁעוֹמֵד בְּבֵית הַכְּנֶסֶת, מְקוֹם דִּקְדֻשָּׁה, וּבָעִית מִשְּׁכִינָה (אִי נַמִּי לְגַבְרָא רַבָּא דְּאִית לֵיהּ תָּמִיד אֵימְתָא דִּשְׁכִינָה) — בְּכָל הָנֵי שַׁפִּיר דָּמֵי, אַף לְשֶׁאֵינוֹ נָשׂוּי, לֶאֱחוֹז בָּאַמָּה עַצְמָהּ לְהַשְׁתִּין (וּכְבָר נִתְבָּאֵר הוּא הַדִּין שֶׁלֹּא לְהַשְׁתִּין). וּבְבֵית יוֹסֵף כָּתַב דְּאֵין אָנוּ בְּקִיאִים אֵיזֶהוּ בִּיעֲתוּתָא. וּלְפִי עֲנִיּוּת דַּעְתִּי, הַסּוֹמֵךְ לְהָקֵל כְּשֶׁיּוֹדֵעַ בְּעַצְמוֹ דְּבָעִית — לֹא הִפְסִיד. וְנִרְאֶה לִי פָּשׁוּט דְּאִם אֵינוֹ חוֹשֵׁשׁ לְנִיצוֹצוֹת, אַף לְהַשְׁתִּין אָסוּר הַנָּשׂוּי, אִם לֹא בָּעֲטָרָה וְכַנִּזְכָּר לְעֵיל. וּפָשׁוּט דְּלָאו דַּוְקָא לְהַשְׁתִּין, אֶלָּא הוּא הַדִּין לְכָל צֹרֶךְ גָּדוֹל כַּהַאי גַּוְנָא:

**אָסוּר** לִלְבּוֹשׁ מִכְנָסַיִם קְטַנִּים שֶׁדּוֹחֲקִים בְּשַׂר הַגִּיד, אִם אֵינָם עֲשׂוּיִים
