# מִצְוַת עֲשֵׂה — שִׁלּוּחַ טְמֵאִים
# סֵפֶר בְּמִדְבַּר, בְּעֶזְרַת הַשֵּׁם
# פָּרָשַׁת נָשֹׂא, סִימָן ש׳ — מִצְוַת עֲשֵׂה (פֶּרֶק ג׳ מִבִּיאַת מִקְדָּשׁ)

**(ז) מִצְוַת** עֲשֵׂה שֶׁצִּוָּה הַבּוֹרֵא יִתְבָּרַךְ לְשַׁלֵּחַ הַטְּמֵאִים חוּץ לְמַחֲנֵהוּ; שֶׁנֶּאֱמַר (ה׳, פָּסוּק ב׳ ג׳): "צַו אֶת בְּנֵי יִשְׂרָאֵל וִישַׁלְּחוּ מִן הַמַּחֲנֶה כָּל צָרוּעַ וְכָל זָב וְכֹל טָמֵא לָנָפֶשׁ; מִזָּכָר עַד נְקֵבָה תְּשַׁלֵּחוּ, אֶל מִחוּץ לַמַּחֲנֶה תְּשַׁלְּחוּם, וְלֹא יְטַמְּאוּ אֶת מַחֲנֵיהֶם" וְגוֹ׳, עַד כָּאן. וְנִכְפְּלָה הָאַזְהָרָה: בְּצָרוּעַ — בְּפָרָשַׁת תַּזְרִיעַ (וַיִּקְרָא י״ג מ״ו): "בָּדָד יֵשֵׁב, מִחוּץ לַמַּחֲנֶה מוֹשָׁבוֹ"; וּבְנִדָּה וְזָבָה — הוּא אוֹמֵר בְּרֵישׁ פָּרָשַׁת תַּזְרִיעַ (שָׁם י״ב ד׳): "וְאֶל הַמִּקְדָּשׁ לֹא תָבֹא עַד מְלֹאת יְמֵי טָהֳרָהּ"; וּבְסוֹף פָּרָשַׁת מְצֹרָע (ט״ו ל״א): "וְהִזַּרְתֶּם אֶת בְּנֵי יִשְׂרָאֵל מִטֻּמְאָתָם, וְלֹא יָמֻתוּ בְּטֻמְאָתָם בְּטַמְּאָם אֶת מִשְׁכָּנִי אֲשֶׁר בְּתוֹכָם"; וּבִטְמֵא לְנֶפֶשׁ מֵת — הוּא אוֹמֵר בְּפָרָשַׁת חֻקַּת (י״ט י״ג): "כָּל הַנֹּגֵעַ בְּמֵת בְּנֶפֶשׁ הָאָדָם אֲשֶׁר יָמוּת וְלֹא יִתְחַטָּא — אֶת מִשְׁכַּן ה׳ טִמֵּא" וְגוֹ׳; וְשָׁם (פָּסוּק כ׳): "וְאִישׁ אֲשֶׁר יִטְמָא" וְגוֹ׳ "כִּי אֶת מִקְדַּשׁ ה׳ טִמֵּא"; וּבְבַעַל קֶרִי — נִכְפְּלָה אַזְהָרָה בְּפָרָשַׁת תֵּצֵא, כְּמוֹ שֶׁיִּתְבָּאֵר שָׁם בְּסִיַּעְתָּא דִּשְׁמַיָּא. וְכָתַב הָרָאָ״ה: כִּי בְּהִכָּפֵל הָאַזְהָרוֹת בַּמִּצְוֹת — הוֹרָאָה קְצָת בְּחֹמֶר הַמִּצְוָה, שֶׁעַל כֵּן הִזְהִיר וְחָזַר וְהִזְהִיר, לְחֶפְצוֹ בְּטוֹבַת יִשְׂרָאֵל, וְלִהְיוֹת הַדָּבָר צֹרֶךְ רַב לָהֶם; כְּדֶרֶךְ בְּנֵי אָדָם, כְּשֶׁיַּזְהִיר לְמִי שֶׁחָפֵץ בְּטוֹבָתוֹ דָּבָר שֶׁסַּכָּנָתוֹ עֲצוּמָה אִם יִמָּנַע מִמֶּנּוּ — שֶׁיַּרְבֶּה לְהַזְהִירוֹ. וְעַיֵּן שֶׁכָּתַב בְּטַעַם מִצְוָה זוֹ: לִהְיוֹת הַטֻּמְאָה מַחֲלַאת הַנֶּפֶשׁ, וּכְעִנְיָן שֶׁדָּרְשׁוּ חֲכָמֵינוּ זִכְרוֹנָם לִבְרָכָה "וְנִטְמֵתֶם בָּם" — אֵין רָאוּי לַמְלֻכְלָךְ וְנִבְזֶה לָבוֹא לְמַחֲנֵהוּ יִתְבָּרַךְ, עַל דֶּרֶךְ שֶׁנֶּאֱמַר: "כִּי אֵין לָבוֹא אֶל שַׁעַר הַמֶּלֶךְ בִּלְבוּשׁ שָׂק", עַד כָּאן. וְרַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה אָמְרוּ גַּם כֵּן, לְחַד מַאן דְּאָמַר: "כָּל מָקוֹם שֶׁנֶּאֱמַר 'צַו' — אֵינוֹ אֶלָּא זֵרוּז", וּבִנְיַן אָב לְכֻלָּם — שֶׁנֶּאֱמַר (דְּבָרִים ג׳): "צַו אֶת יְהוֹשֻׁעַ וְחַזְּקֵהוּ וְאַמְּצֵהוּ". וְכֵן כָּאן, שֶׁאָמַר "צַו" וְגוֹ׳ — מוֹרֶה עַל הַזֵּרוּז שֶׁצִּוָּהוּ יִתְבָּרַךְ לְמֹשֶׁה, לְזָרְזָם לִבְנֵי יִשְׂרָאֵל עַל הַשִּׁלּוּחַ הַזֶּה; וְזֶה לְחֻמְרוֹ שֶׁל הָעִנְיָן:

**וְנוֹהֶגֶת** מִצְוָה זוֹ בְּכָל זְמַן — בֵּין בִּפְנֵי הַבַּיִת בֵּין שֶׁלֹּא בִּפְנֵי הַבַּיִת; דְּקַיְמָא לָן כְּמַאן דְּאָמַר (מְגִלָּה דַּף י׳): "קְדֻשָּׁה רִאשׁוֹנָה קִדְּשָׁה לִשְׁעָתָהּ וְקִדְּשָׁה לֶעָתִיד לָבוֹא"; וְאַף עַל פִּי שֶׁהַבַּיִת חָרֵב — בִּקְדֻשָּׁתוֹ הוּא עֲדַיִן. וְכָל פְּרָטֵי דִּינֵי שִׁלּוּחַ הַטְּמֵאִים, כְּפִי שֶׁיִּתְבָּאֵר — נוֹהֲגוֹת גַּם עַכְשָׁיו. וְהִנֵּה עַכְשָׁיו, בַּעֲוֹנוֹתֵינוּ הָרַבִּים, שֶׁאָנוּ כֻּלָּנוּ טְמֵאֵי מֵתִים (וְעַיֵּן מַה שֶּׁכָּתַב הָרַב חַיִּים כֹּהֵן בְּתוֹסָפוֹת דְּנָזִיר (דַּף
