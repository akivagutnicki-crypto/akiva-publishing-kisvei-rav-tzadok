~ עָבַר וְקָבַע וְעִבֵּר — לֹא עָשָׂה וְלֹא כְּלוּם. (מִיהוּ אִם נָתְנוּ לוֹ רְשׁוּת בֵּית דִּין שֶׁל אֶרֶץ יִשְׂרָאֵל — נִרְאֶה דְּשַׁפִּיר דָּמֵי. וְאִם אֵין יְכוֹלִים בֵּית דִּין שֶׁל אֶרֶץ יִשְׂרָאֵל לִקְבּוֹעַ, מֵחֲמַת גְּזֵרַת הַמַּלְכוּת וְכַהַאי גַּוְנָא — נִרְאֶה נַמִּי דְּרַשָּׁאִים נַמִּי בֵּית דִּין חוּץ לָאָרֶץ, אֲפִלּוּ קְטַנִּים, לְקַדֵּשׁ; וְכָל בֵּית דִּין שֶׁיִּהְיוּ סְמוּכִים. וְנִרְאֶה לִי נַמִּי: דְּבֵית דִּין דְּחוּץ לָאָרֶץ, אֲפִלּוּ שֶׁהוּא גְּדוֹל הַדּוֹר — דְּרַשַּׁאי אַף בְּחוּץ לָאָרֶץ, כְּמוֹ שֶׁנִּתְבָּאֵר; מִכָּל מָקוֹם, אִם יָכוֹל לִיכָּנֵס לְאֶרֶץ יִשְׂרָאֵל — צָרִיךְ שֶׁיִּכָּנֵס לְאֶרֶץ יִשְׂרָאֵל לְקַדֵּשׁ. וְאֶפְשָׁר אִם עָבַר וְקִדֵּשׁ בְּחוּץ לָאָרֶץ, אִם הָיָה אֶפְשָׁר לוֹ לִיכָּנֵס לְאֶרֶץ יִשְׂרָאֵל לְקַדֵּשׁ — לֹא עָשָׂה כְּלוּם; עַיֵּן בִּירוּשַׁלְמִי, פֶּרֶק קַמָּא דְּסַנְהֶדְרִין, וְהוּבָא בְּרַמְבַּ״ן שָׁם; וְצָרִיךְ עִיּוּן בָּזֶה):

**וְכֵן** בְּעִבּוּר הַשָּׁנָה — אֵין מְעַבְּרִין אֶלָּא בְּאֶרֶץ יִשְׂרָאֵל; וְאֵין מְעַבְּרִין אֶלָּא בְּאֶרֶץ יְהוּדָה, שֶׁהַשְּׁכִינָה בְּתוֹכָהּ, שֶׁנֶּאֱמַר: "לְשִׁכְנוֹ תִדְרְשׁוּ". וְאִם עִבְּרוּהָ בַּגָּלִיל — מְעֻבֶּרֶת (בַּגְּמָרָא, י״א עַמּוּד ב׳, אִיתָא: הֵעִיד חֲנַנְיָא אִישׁ אוֹנוֹ דְּאֵין מְעֻבֶּרֶת; מַאי טַעְמָא? דְּתַנְיָא: "לְשִׁכְנוֹ תִדְרְשׁוּ". אֲבָל בְּסֵפֶר תּוֹרַת חַיִּים כָּתַב: דִּבַּתּוֹסֶפְתָּא הַגִּרְסָא "מְעֻבֶּרֶת", וְשֶׁכֵּן עִקָּר, עַיֵּן שָׁם. וּבָזוֹ נִיחָא דְּלֹא קָשֶׁה קֻשְׁיַת תּוֹסָפוֹת שָׁם, סוֹף דִּבּוּר הַמַּתְחִיל "אֵין", שֶׁהִנִּיחוּ בְּקֻשְׁיָא. וּלְמַה שֶּׁכָּתַבְתִּי נִיחָא: דִּ"לְשִׁכְנוֹ" — הוּא רַק לְעִנְיַן לְכַתְּחִלָּה, וְעַל יְהוּדָה; אֲבָל שָׁם קָאֵי לְעִנְיַן כְּלַל אֶרֶץ יִשְׂרָאֵל, וְזֶהוּ אַף דִּיעֲבַד. וְיֵשׁ לְהִסְתַּפֵּק אִם גַּם אַחַר הַחֻרְבָּן — דְּלֹא הָיְתָה הַשְּׁכִינָה בִּיהוּדָה — אִי דַּוְקָא יְהוּדָה. וְעַיֵּן בְּמַהַרְשָׁ״א, בְּחִדּוּשֵׁי אַגָּדוֹת, פֶּרֶק הָרוֹאֶה (ס״ב עַמּוּד ב׳), בְּמָה שֶּׁכָּתַב לְתָרֵץ קֻשְׁיַת תּוֹסָפוֹת דְּכָאן מִזֶּה, עַיֵּן שָׁם.) וְכֵן אֵין מְעַבְּרִין אֶלָּא סַנְהֶדְרִין, וּבִכְלָלָם הָרֹאשׁ בֵּית דִּין, כְּמוֹ שֶׁנִּתְבָּאֵר (אוֹ סְמוּכִים):

**הָיָה** רֹאשׁ בֵּית דִּין בְּדֶרֶךְ רְחוֹקָה — אֵין מְעַבְּרִין אוֹתָהּ אֶלָּא עַל תְּנַאי: אִם יִרְצֶה הָרֹאשׁ בֵּית דִּין — בָּא וְרָצָה — הֲרֵי זוֹ מְעֻבֶּרֶת; לֹא רָצָה — הֲרֵי זוֹ אֵינָהּ מְעֻבֶּרֶת. (וְכָתַב בְּסֵפֶר טוּרֵי אֶבֶן, רֹאשׁ הַשָּׁנָה (ל״א עַמּוּד ב׳), דִּבּוּר הַמַּתְחִיל "שֶׁלֹּא יִהְיוּ": דַּוְקָא כְּשֶׁהוֹדִיעוּ לְרֹאשׁ בֵּית דִּין וְנִתְרַצָּה קֹדֶם נִיסָן; אֲבָל אַחַר נִיסָן — כֵּיוָן שֶׁאֵין מְעַבְּרִין לְמַפְרֵעַ, הָכֵי נַמִּי בְּהוֹדָעַת רִיצּוּי רֹאשׁ בֵּית דִּין לֹא אָמְרִינַן "אִגְּלַאי מִלְּתָא לְמַפְרֵעַ". וְעַיֵּן שָׁם שֶׁכָּתַב דִּלְמַאן דְּאָמַר יֵשׁ בְּרֵירָה — אֶפְשָׁר הָכֵי נַמִּי יוֹעִיל, עַיֵּן שָׁם):

**וְאֵין** מוֹשִׁיבִין לְעִבּוּר הַשָּׁנָה לֹא מֶלֶךְ וְלֹא כֹּהֵן גָּדוֹל. מֶלֶךְ — מִפְּנֵי חַיָּלוֹתָיו וּמִלְחֲמוֹתָיו, שֶׁמָּא דַּעְתּוֹ נוֹטָה בִּשְׁבִילָן לְעַבֵּר אוֹ שֶׁלֹּא לְעַבֵּר (וְעַיֵּן בְּרַשִׁ״י — פֵּרֵשׁ רַק לְעַבֵּר); וְכֹהֵן גָּדוֹל — מִפְּנֵי הַצִּנָּה, שֶׁמָּא לֹא תְּהֵא דַּעְתּוֹ נוֹטָה לְעַבֵּר, כְּדֵי שֶׁלֹּא יָבוֹא תִּשְׁרֵי בַּיּוֹם הַקָּר, וְהוּא טוֹבֵל בְּיוֹם הַכִּפּוּרִים ה׳ טְבִילוֹת,
