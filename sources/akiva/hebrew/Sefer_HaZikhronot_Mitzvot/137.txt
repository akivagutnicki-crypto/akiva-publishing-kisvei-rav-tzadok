~ מָקוֹם לַתּוֹרָה וּמִצְוֹת? וְאָמְרוּ רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה, שִׁלְהֵי מַכּוֹת (כ״ד עַמּוּד א׳): "בָּא עָמוֹס וְהֶעֱמִידָן עַל אַחַת, שֶׁנֶּאֱמַר 'דִּרְשׁוּנִי וִחְיוּ'; מַתְקִיף" וְכוּלֵי, "אֶלָּא: בָּא חֲבַקּוּק וְהֶעֱמִידָן עַל אַחַת, שֶׁנֶּאֱמַר 'וְצַדִּיק בֶּאֱמוּנָתוֹ יִחְיֶה'". וּלְפִי הַמַּסְקָנָא — גַּם כַּוָּנַת עָמוֹס וַדַּאי כֵּן הוּא, וּשְׁנֵיהֶם לְדָבָר אֶחָד נִתְכַּוְּנוּ. וְתַכְלִית הַכַּוָּנָה — שֶׁנִּתְחַבְּטוּ בָּהּ גְּדוֹלֵי הָרִאשׁוֹנִים זִכְרוֹנָם לִבְרָכָה בְּבֵאוּר הַסּוּגְיָא, עַיֵּן שָׁם בְּדִבְרֵי הָרָמָ״ה וּשְׁאָר מְפָרְשִׁים — לְדַעְתִּי, הַכַּוָּנָה עַל דֶּרֶךְ אוֹתוֹ גֵּר בְּפֶרֶק "בַּמֶּה מַדְלִיקִין", שֶׁבִּקֵּשׁ שֶׁיְּלַמְּדוּהוּ כָּל הַתּוֹרָה כֻּלָּהּ עַל רֶגֶל אַחַת, וְאָמַר לוֹ: "דַּעֲלָךְ" וְכוּלֵי, "זֶהוּ כָּל הַתּוֹרָה, וְאִידָךְ פֵּירוּשָׁא — זִיל גְּמוֹר". כֵּן אוֹתָם הַנְּבִיאִים, שֶׁרָאוּ כִּשְׁלוֹן דּוֹרָם, וְכָל אֶחָד רוֹצֶה לְקַיֵּם כָּל הַתּוֹרָה בְּקַיְּמוֹ "חַד כַּרְעָא אַאַרְעָא וְחַד אַמַּבְרָא", לְהָקֵל עַל נַפְשָׁם כְּאוֹתוֹ גֵּר — הֵם הֶעֱמִידוּ עַל אַחַת, וְהִיא הָאֱמוּנָה וְהַדְּרִישָׁה אֵלָיו יִתְבָּרַךְ: לְהַאֲמִין שֶׁיֵּשׁ מָצוּי אֶחָד, הַמַּמְצִיא וּמַשְׁגִּיחַ וּמוֹשֵׁל בַּכֹּל. כִּי אַחַר הֱיוֹת הָאֱמוּנָה "אֵזוֹר חֲלָצָיו", וּמַאֲמִין בְּחֵי הָעוֹלָמִים — הֲלֹא לְבָבוֹ מוּכָן לְקַיֵּם כָּל צִוּוּיָיו יִתְבָּרַךְ; וְאִידָךְ, כָּל הַתּוֹרָה כֻּלָּהּ — פֵּירוּשָׁא הִיא, לְהָבִין וְלָדַעַת מָה הֵמָּה צִוּוּיָיו יִתְבָּרַךְ. וְאָמְרוּ בְּפֶרֶק "בַּמֶּה מַדְלִיקִין" (ל״א עַמּוּד א׳): "'אֱמוּנַת' — זֶה סֵדֶר זְרָעִים"; כִּי הַסֵּדֶר הַזֶּה כּוֹלֵל חֲלֻקַּת תְּרוּמוֹת וּמַעַשְׂרוֹת, וְחַלָּה וְעָרְלָה, וּשְׁאָר עִנְיָנִים — כֻּלָּם דְּבָרִים שֶׁיֵּשׁ בָּהֶם חֶסְרוֹן כִּיס, שֶׁבְּיוֹתֵר צָרִיךְ זֵרוּז, וַעֲלֵיהֶם הָיָה קֹרַח מַפְלִיג, כִּמְבֹאָר בְּדִבְרֵי רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה. אֲבָל "אִישׁ אֱמוּנוֹת" — הַמַּאֲמִין בִּמְצִיאוּת הַבּוֹרֵא יִתְבָּרַךְ, וְהַשְׁגָּחָתוֹ עַל כָּל דָּבָר — "קָטֹן וְגָדוֹל שָׁם הוּא" — הִנֵּה נָקֵל הוּא לוֹ לִשְׁמוֹר וְלַעֲשׂוֹת וּלְקַיֵּם כָּל דְּבָרִים אֵלּוּ:

**וּפְרָטֵי** אֱמוּנָה זוֹ — הֵם שְׁנַיִם; וּזְכָרָם הַכָּתוּב, כְּמוֹ שֶׁנֶּאֱמַר: "אָנֹכִי ה׳ אֱלֹקֶיךָ". כִּי ה׳ — וְהוּא שֵׁם הֲוָיָ״ה, שֶׁמַּשְׁמָעוּתוֹ וּפֵירוּשׁ כַּוָּנָתוֹ הוּא "הָיָה וְהֹוֶה וְיִהְיֶה", כְּמוֹ שֶׁאִיתָא בְּאֹרַח חַיִּים (סִימָן ה׳) — מַזְהִיר לְהַאֲמִין עַל הֱיוֹתוֹ יִתְבָּרַךְ נִצְחִיִּי, וְהוּא רִאשׁוֹן לְכָל הַיְּצוּר, וְהוּא אַחֲרִיתָם, וּמְהַוֶּה הַכֹּל. וֶ"אֱלֹקֶיךָ" — וְהוּא שֵׁם אֱלֹקִים, שֶׁבֵּאוּרוֹ מַנְהִיג וּבַעַל הַכֹּחוֹת כֻּלָּם, שֶׁבְּכָל כֹּחוֹתָיו מַנְהִיג עוֹלָמוֹ — וּלְהַאֲמִין שֶׁהוּא יִתְבָּרַךְ מַשְׁגִּיחַ בְּכָל פְּרָטֵי הַיְּצוּר כֻּלָּם, וְכָל הַמִּתְהַוֶּה הַנֶּאֱרָע בְּאֶחָד מֵהֶם — מִמֶּנּוּ יִתְבָּרַךְ הוּא נִתְהַוָּה. וּבֵאֵר זֶה הַכָּתוּב, בַּמֶּה שֶּׁאָמַר אַחַר כָּךְ: "אֲשֶׁר הוֹצֵאתִיךָ מֵאֶרֶץ מִצְרַיִם" — רוֹצֶה לוֹמַר: לִרְאָיָה וּלְמוֹפֵת שֶׁ"אָנֹכִי אֱלֹקֶיךָ", מַנְהִיגְךָ וּמַשְׁגִּיחֲךָ, הוּא "אֲשֶׁר הוֹצֵאתִיךָ" וְגוֹ׳; שֶׁאָז נִרְאֵית הַשְׁגָּחָתוֹ יִתְבָּרַךְ וְיִתְעַלֶּה בַּמּוּחָשׁ לְעֵין כֹּל, כִּי "לָקַח לוֹ גוֹי מִקֶּרֶב גּוֹי בְּמַסֹּת בְּאֹתֹת" וְכוּלֵי (וְיִהְיֶה מִלַּת "אֲשֶׁר" — בֵּאוּרוֹ "בַּעֲבוּר", כְּמוֹ (מְלָכִים א׳ ט״ו): "אֲשֶׁר עָשְׂתָה מִפְלֶצֶת"). וּבֵאֵר הַכָּתוּב, שֶׁיְּדִיעַת הָאָדָם אֱמוּנָה זוֹ בַּמּוּחָשׁ — הוּא הַיְּצִיאָה מֵאֶרֶץ מִצְרַיִם:
