~ בּוֹ בְּנוֹתֵן טַעַם. אֲבָל עִם הַקְּנוּקְנוֹת וְשׁוּמָן — יֵשׁ בָּהֶן טַעַם; וְאַף עַל פִּי שֶׁאֵין אוֹסְרִין מִשּׁוּם בְּרִיָּה, אוֹסְרִין תַּעֲרוֹבְתָּן בִּפְלִיטָתָן מִשּׁוּם נוֹתֵן טַעַם, עַד שֶׁיִּהְיֶה שִׁשִּׁים נֶגְדָּם. לְפִיכָךְ יָרֵךְ שֶׁנִּתְבַּשֵּׁל עִם גִּידוֹ — אִם מַכִּירוֹ, זוֹרְקוֹ, וְהַכֹּל מֻתָּר, אִם יֵשׁ בּוֹ שִׁשִּׁים נֶגֶד שׁוּמָנוֹ. וְאִם אֵין בַּיָּרֵךְ שִׁשִּׁים נֶגֶד שׁוּמָנוֹ, אֲבָל יֵשׁ בִּשְׁאָר הַחֲתִיכוֹת שִׁשִּׁים — יֵשׁ אוֹמְרִים דְּאִם הַשּׁוּמָן דָּבוּק בַּיָּרֵךְ, הוּא תָּלוּי בְּמָה שֶׁנֶּחְלְקוּ פּוֹסְקִים אִם בְּכָל הָאִסּוּרִים חֲתִיכָה נַעֲשֵׂית נְבֵלָה בְּאִסּוּר דָּבוּק, וְיִתְבָּאֵר בִּמְקוֹמוֹ. דְּאָנַן קַיְימָא לַן בְּכָל הָאִסּוּרִים בְּאִסּוּר דָּבוּק חֲתִיכָה נַעֲשֵׂית נְבֵלָה; אִם כֵּן הָכָא צָרִיךְ שֶׁיִּהְיֶה בַּיָּרֵךְ עַצְמוֹ שִׁשִּׁים נֶגֶד הַשּׁוּמָן, וְאִם אֵין בּוֹ — הֲרֵי כֻּלּוֹ נַעֲשֶׂה נְבֵלָה, וְצָרִיךְ בִּשְׁאָר חֲתִיכוֹת שִׁשִּׁים נֶגֶד כָּל הַיָּרֵךְ. וְיֵשׁ חוֹלְקִים עַל זֶה וְאוֹמְרִים, דְּכֵיוָן דְּאִסּוּר שׁוּמַן הַגִּיד אֵינוֹ אֶלָּא דְּיִשְׂרָאֵל קְדוֹשִׁים נָהֲגוּ בּוֹ אִסּוּר — וְנִרְאֶה דְּהוּא הַדִּין בִּקְנוּקְנוֹת, כֵּיוָן דְּאֵינוֹ אֶלָּא אִסּוּר (וְעַיֵּן בִּכְרֵתִי וּפְלֵתִי לְהַדְיָא, עַיֵּן שָׁם) דְּרַבָּנָן — מְצָרְפִין כָּל הַחֲתִיכוֹת לְשִׁשִּׁים נֶגֶד הַשּׁוּמָן וְהַקְּנוּקְנוֹת, אַף שֶׁדָּבוּק בַּיָּרֵךְ. וְאֵין לִסְמוֹךְ אַדֵּעָה זוֹ אֶלָּא בִּשְׁעַת הַדְּחָק וְהֶפְסֵד מְרֻבֶּה וְצֹרֶךְ גָּדוֹל. וְאִם אֵינוֹ מַכִּירוֹ — אָז הַחֲתִיכוֹת כֻּלָּם אֲסוּרוֹת; וְאִם יֵשׁ שִׁשִּׁים נֶגֶד הַשּׁוּמָן — הָרוֹטֶב מֻתָּר, כַּנִּזְכָּר לְעֵיל, וְאִם לָאו — גַּם רוֹטֶב אָסוּר. וְנִרְאֶה פָּשׁוּט דִּכְשֶׁאֵין שִׁשִּׁים גַּם גִּיד עַצְמוֹ בָּלַע שׁוּמָנוֹ, וְאִם חָזַר וְנִתְעָרֵב בְּאַחֶרֶת — אוֹסֵר עַד שִׁשִּׁים בִּפְלִיטָתוֹ; דְּאַף עַל גַּב דְּעֵץ הוּא, מִכָּל מָקוֹם בּוֹלֵעַ כְּעֵץ — וּפָשׁוּט. וְאִם נִמּוֹחַ הַגִּיד וְאֵינוֹ נִכָּר — דְּתוּ לָא הֲוֵי בְּרִיָּה — מִכָּל מָקוֹם, כֵּיוָן שֶׁלֹּא נְטָלוּהוּ וּזְרָקוּהוּ, הֲרֵי מַמָּשׁוֹ שָׁם, וְהַתּוֹרָה אָסְרָה מַמָּשׁוֹ, אַף עַל פִּי שֶׁאֵינוֹ נוֹתֵן טַעַם; וּלְפִיכָךְ צָרִיךְ שִׁשִּׁים לְבַטְּלוֹ. וְיֵשׁ אוֹמְרִים דְּדַי בְּרֹב, וְיֵשׁ לִסְמוֹךְ עַל זֶה בְּעֵת הַדְּחָק וְהֶפְסֵד מְרֻבֶּה; אִי נַמִּי כְּשֶׁנִּשְׁפַּךְ, וְלֹא יָדַע אִי הֲוֵי בּוֹ שִׁשִּׁים. אִי נַמִּי נִרְאֶה לְהָקֵל כְּשֶׁהָיָה דָּבוּק בַּחֲתִיכָה, וְלֹא הָיָה בַּחֲתִיכָה עַצְמָהּ ס׳, אֶלָּא בְּצֵרוּף; דְּאַף עַל גַּב דְּזוֹ אִסּוּר דְּאוֹרַיְתָא, מִכָּל מָקוֹם הָא דַּחֲתִיכָה נַעֲשֵׂית נְבֵלָה בְּאִסּוּר דָּבוּק בְּכָל הָאִסּוּרִים — כְּבָר נִתְבָּאֵר דְּמַחְלֹקֶת פּוֹסְקִים הוּא, וְגַם לְדִבְרֵי הָאוֹסְרִים הוּא רַק דְּרַבָּנָן — יֵשׁ לְהָקֵל, כַּנִּזְכָּר לְעֵיל. אַף עַל פִּי שֶׁיִּשְׂרָאֵל קְדוֹשִׁים וְהֶחְמִירוּ בְּשׁוּמָנוֹ שֶׁל גִּיד, לֹא הֶחְמִירוּ בְּסַכִּינוֹ לַחְתּוֹךְ דְּבָרִים אֲחֵרִים (טוּר סִימָן ס״ד):

**מֻתָּר** לֵהָנוֹת מֵהַגִּיד הַנָּשֶׁה, דְּרַק אֲכִילָה אָסְרָה תּוֹרָה וְלֹא הֲנָאָה. וּלְפִיכָךְ שׁוֹלֵחַ אָדָם יָרֵךְ לְנָכְרִי וְגִיד הַנָּשֶׁה בְּתוֹכוֹ, אֲפִלּוּ הִיא שְׁלֵמָה; אַף עַל גַּב שֶׁהַיָּרֵךְ נִרְאֶה יוֹתֵר גָּדוֹל כְּשֶׁהַגִּיד בְּתוֹכוֹ, וְהַגּוֹי מַחֲזִיק לוֹ טוֹבָה יוֹתֵר בְּסִבַּת הַגִּיד, שֶׁחוֹשְׁבוֹ לְמַתָּנָה — אֵינוֹ חוֹשֵׁשׁ, שֶׁהֲרֵי מֻתָּר בַּהֲנָאָה. וְיֵשׁ חוֹלְקִים וְאוֹמְרִים, דְּכָל מָקוֹם שֶׁנֶּאֱמַר "לֹא יֹאכְלוּ" — אֶחָד אִסּוּר אֲכִילָה וְאֶחָד אִסּוּר הֲנָאָה בְּמַשְׁמָע; וּלְפִיכָךְ אָסוּר בַּהֲנָאָה, וְאִם עָבַר וְנֶהֱנָה — לוֹקֶה (לְכוּלֵּי עָלְמָא, עַיֵּן פְּרִי מְגָדִים)
