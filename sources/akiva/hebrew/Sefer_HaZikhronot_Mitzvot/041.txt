~ מְקֻדָּשׁ" — אַף עַל גַּב דִּלְכַתְּחִלָּה צָרִיךְ גַּם כֵּן שֶׁיִּהְיֶה בַּיּוֹם, מִכָּל מָקוֹם אִם לֹא הָיָה אֶלָּא בַּלַּיְלָה — כֵּיוָן שֶׁהָרֹאשׁ בֵּית דִּין אָמַר בְּסוֹף הַיּוֹם "מְקֻדָּשׁ" — שַׁפִּיר דָּמֵי (סֵפֶר טוּרֵי אֶבֶן, רֵישׁ פֶּרֶק ג׳, מִדִּיּוּקָא דְּמַתְנִיתִין). וְהֵיכִי דְּאִי אֶפְשָׁר בְּעִנְיָן אַחֵר — פָּשׁוּט דְּאַף לְכַתְּחִלָּה כְּדִיעֲבַד. אֲבָל צָרִיךְ עִיּוּן לַעֲנִיּוּת דַּעְתִּי, הָא אֵין מְקַדְּשִׁין אֶלָּא בִּשְׁלֹשָׁה, כְּמוֹ שֶׁנִּתְבָּאֵר, וַאֲפִלּוּ יָחִיד מֻמְחֶה אֵין מוֹעִיל; וַהֲרֵי רַק הָרֹאשׁ בֵּית דִּין לְבַדּוֹ אוֹמֵר "מְקֻדָּשׁ", וּב׳ הָאֲחֵרִים הֵם בִּכְלַל כָּל הָעָם הָעוֹנִים "מְקֻדָּשׁ מְקֻדָּשׁ"; וְכֵיוָן שֶׁהֵם עוֹנִים בַּלַּיְלָה — לֹא הוֹעִילָה אֲמִירַת רֹאשׁ בֵּית דִּין דְּיוֹם כְּלָל. וְאוּלַי הוֹאִיל וְהַתְחָלַת גְּמַר דִּין הָיָה בַּיּוֹם, גּוֹמְרִין הַגְּמַר דִּין בַּלַּיְלָה; וְשַׁאנִי עֵדִים, דְּהֵם הַתְחָלַת דִּין, לֹא הַתְחָלַת גְּמַר דִּין; וְצָרִיךְ עִיּוּן לַעֲנִיּוּת דַּעְתִּי. וְיֵשׁ חוֹלְקִים עַל כָּל זֶה, וְאוֹמְרִים דְּאֵין קִדּוּשׁ הַחֹדֶשׁ מְעַכֵּב, וַאֲמִירַת "מְקֻדָּשׁ" — אַף דְּרֹאשׁ בֵּית דִּין — אֵינָהּ אֶלָּא לְמִצְוָה, אוֹ לְפִרְסוּמֵי מִלְּתָא בְּעָלְמָא; אֲבָל מִכֵּיוָן שֶׁהִסְכִּימָה דַּעַת בֵּית דִּין שֶׁזֶּה הַחֹדֶשׁ מָלֵא אוֹ חָסֵר — "קוֹרֵא אֲנִי בּוֹ: אֲשֶׁר תִּקְרְאוּ אֹתָם" (וְכֵן בְּעִבּוּר שָׁנִים); וּבְחֻשְׁבָּנָא וּשְׁמִיעַת עֵדִים תַּלְיָא מִלְּתָא (הָרַמְבַּ״ן, בְּסֵפֶר הַמִּצְוֹת, סוֹף מִצְוַת עֲשֵׂה קנ״ג, עַיֵּן שָׁם); דְּאֵין רְאָיוֹתָיו מֻכְרָחוֹת. וְצָרִיךְ עִיּוּן אֵיךְ יְפָרֵשׁ הַךְ דִּינֵי דִּרְאוּהוּ בֵּית דִּין, וְלֹא הִסְפִּיקוּ הַנִּזְכָּר; וְצָרִיךְ לוֹמַר דִּיפָרֵשׁ הַכֹּל עַל הַחֶשְׁבּוֹן וְהַהַסְכָּמָה שֶׁלֹּא הִסְפִּיקוּ לְהַסְכִּים. אֲבָל לִישָׁנָא לֹא מַשְׁמַע הָכֵי לַמְעַיֵּן בְּכוּלֵּי סוּגְיָא. וְעַיֵּן שָׁם בִּמְגִלַּת אֶסְתֵּר, כָּתַב גַּם כֵּן דְּלָא כְּהָרַמְבַּ״ן; וְכֵן נִרְאֶה עִקָּר, וְכַנִּרְאֶה לִכְאוֹרָה בַּגְּמָרָא (כ״ד עַמּוּד א׳), וְעַיֵּן בִּירוּשַׁלְמִי:

**אֵין** כָּשֵׁר לְעֵדוּת הַחֹדֶשׁ אֶלָּא ב׳ אֲנָשִׁים כְּשֵׁרִים, הָרְאוּיִים לְהָעִיד בְּכָל דָּבָר; וְכָל הַפָּסוּל לְעֵדוּת, כְּמוֹ שֶׁיִּתְבָּאֵר בִּמְקוֹמוֹ — פָּסוּל לְעֵדוּת הַחֹדֶשׁ, בֵּין פְּסוּל דְּבַר תּוֹרָה בֵּין פְּסוּל דְּרַבָּנָן, בֵּין פְּסוּל דִּקְרוּבָה בֵּין דַּעֲבֵירָה אוֹ שְׁאָר פְּסוּלִים. (נִרְאֶה לִי נוֹגֵעַ נַמִּי פָּסוּל; וּמֵהַאי טַעְמָא אֵין מוֹשִׁיבִין מֶלֶךְ וְכֹהֵן גָּדוֹל בְּעִבּוּר הַשָּׁנָה, כְּמוֹ שֶׁיִּתְבָּאֵר, דַּהֲוֵי לְהוּ נוֹגְעִים. הֵיכִי דָּמֵי נוֹגֵעַ? שֶׁיֵּשׁ לוֹ טוֹבָה שֶׁיּוּקְדְּמוּ יְמֵי הֶחָג, וִיצוּיַּיר הַטּוֹבָה בְּכַמָּה גַּוְונֵי; וְעוֹד יְצוּיַּיר נוֹגֵעַ בְּגַוְונֵי טוֹבֵי.) וד״ת שֶׁאֵין מְדַקְדְּקִים בְּעֵדוּת הַחֹדֶשׁ — (יְרוּשַׁלְמִי, פֶּרֶק ג׳ הֲלָכָה א׳): אר״א בְּשֵׁם ריב״פ: "קִדְּשׁוּהוּ" וְכוּלֵי, כֵּן א״א בְּשֵׁם ר״י וְכוּלֵי; וְעַיֵּן עוֹד בִּירוּשַׁלְמִי (פֶּרֶק ב׳ הֲלָכָה ז׳), עַיֵּן שָׁם הֵיטֵב; וְעַיֵּן בְּלֶחֶם מִשְׁנֶה, דָּחַק לִמְצוֹא מָקוֹר — וְנֶעְלַם מִמֶּנּוּ זֶה — שֶׁאֲפִלּוּ קִדְּשׁוּ אֶת הַחֹדֶשׁ עַל פִּי עֵדִים וְנִמְצְאוּ זוֹמְמִין בְּעֵדוּת זוֹ, הֲרֵי זֶה מְקֻדָּשׁ. לְפִיכָךְ הָיוּ בָּרִאשׁוֹנָה מְקַבְּלִין עֵדוּת הַחֹדֶשׁ מִכָּל אָדָם מִיִּשְׂרָאֵל, שֶׁכָּל יִשְׂרָאֵל בְּחֶזְקַת כַּשְׁרוּת, עַד שֶׁיִּוָּדַע לְךָ שֶׁזֶּה פָּסוּל. (עַיֵּן בַּמְּפָרֵשׁ, מַה שֶּׁהִקְשָׁה מִפֶּרֶק י״א דְּעֵדוּת, וְתֵרוּצוֹ; וּבְלֶחֶם מִשְׁנֶה כָּתַב דְּתֵרוּצוֹ תָּמוּהַּ, וְלֹא דַּק, דְּכַוָּנַת תֵּרוּצוֹ דְּשָׁם הוּא מִדְּרַבָּנָן מִתַּקָּנַת
