~ כְּאֶחָד; וְלִכְאוֹרָה, אַדְּרַבָּא — מַשְׁמַע מִדְּבָרָיו לְהֵפֶךְ בְּשָׁחַט זֶה אַחַר זֶה, וְסַתְרֵי דִּיּוּקֵי דְּרֵישָׁא וְסֵיפָא בִּדְבָרָיו אַהֲדָדֵי, כַּנִּזְכָּר לְעֵיל; וְעַל כָּל פָּנִים, אֵין מְבֹאָר בְּהֶדְיָא בִּדְבָרָיו, עַד שֶׁיֹּאמַר "וְכֵן פָּסַק הָרַמְבַּ״ם". וּמַה שֶּׁכָּתַב: "בִּזְבָחִים" וְכוּלֵי "קָשֶׁה, דִּילְמָא שָׁאנֵי פֶּסַח, דִּכְתִיב בֵּיהּ 'בָּעֶרֶב' וּ'בֵין הָעַרְבַּיִם'" וְכוּלֵי — לָא קַשְׁיָא מִידֵּי; דִּמְדַיֵּק גְּמָרָא מִפֶּסַח, דְּלֵיכָּא סְבָרָא דְּתָדִיר, כַּנִּזְכָּר לְעֵיל מֵהַתּוֹסָפוֹת, רַק סְבָרָא דְּ"בָעֶרֶב" וּ"בֵין הָעַרְבַּיִם"; וּכְמוֹ דְּהַדִּין שָׁם — כֵּן הַדִּין בִּסְבָרָא דְּתָדִיר, דְּמַאי שְׁנָא; וְכֵן הָרַמְבַּ״ם בְּהִלְכוֹת תְּמִידִין וּמוּסָפִין יָלֵיף מִתָּדִיר לִמְקֻדָּשׁ, עַיֵּן שָׁם — דְּמַאי שְׁנָא, וּפָשׁוּט. וּבְגוּף דְּבָרָיו — לֹא יָדַעְתִּי מָה הֶאֱרִיךְ כָּל כָּךְ, וְגוּפָא דְּמִלְּתָא מְפֹרָשׁ בַּגְּמָרָא, כַּנִּזְכָּר לְעֵיל; יְעַיֵּן שָׁם הֵיטֵב:

**וִיהֵא** אַחֵר מְמָרֵס בְּדַם הַפֶּסַח עַד שֶׁיִּזְרוֹק דַּם הַתָּמִיד, וְאַחַר כָּךְ יִזְרוֹק דַּם הַפֶּסַח אַחֲרָיו (וְנִרְאֶה, אֲבָל אֵין צָרִיךְ לְהַמְתִּין עַד אַחַר קְטֹרֶת וַהֲטָבַת נֵרוֹת. וְעַיֵּן בַּיְרוּשַׁלְמִי (פֶּרֶק ה׳, סוֹף הֲלָכָה ג׳): "תַּמָּן אַתְּ אָמַר: דַּם הַתָּמִיד וְאֵבָרָיו קוֹדְמִין" וְכוּלֵי, "אָמַר רַבִּי אִילָא: כָּאן בְּחַי, כָּאן בְּשָׁחוּט"; עַיֵּן שָׁם בְּקָרְבַּן הָעֵדָה, בְּפֵירוּשׁ ב׳ — דִּבְעָבַר וְהִקְדִּים שְׁחִיטָתוֹ לַפֶּסַח, אֵין צָרִיךְ לְהַמְתִּין גַּם אַחַר הַקְרָבַת אֵבְרֵי תָּמִיד, אֶלָּא בִּזְרִיקַת דָּמוֹ סַגִּי). וְאִם נִזְרַק דַּם הַפֶּסַח קֹדֶם דַּם הַתָּמִיד — כָּשֵׁר; לְפִי שֶׁנֶּאֱמַר בּוֹ "וְשָׁחֲטוּ" וְכוּלֵי "בֵּין הָעַרְבָּיִם", וְכָפַל עוֹד (דְּבָרִים ט״ז): "תִּזְבַּח" וְכוּלֵי "בָּעָרֶב". וְהָיוּ מַקְדִּימִין שְׁחִיטַת הַתָּמִיד וַעֲשִׂיָּתוֹ בְּעֶרֶב פֶּסַח, כְּדֵי לְהַקְדִּים שְׁחִיטַת הַפֶּסַח, כְּמוֹ שֶׁיִּתְבָּאֵר בִּמְקוֹמוֹ; לְפִי שֶׁהַרְבֵּה פְּסָחִים הָיוּ בְּיִשְׂרָאֵל, וּצְרִיכִים שֶׁיִּהְיֶה הַרְבֵּה לְשָׁחֳטָם וְלַעֲשׂוֹתָם כְּמִשְׁפָּטָם, שֶׁיִּתְבָּאֵר. וְהָיוּ מַתְחִילִין אָז לִשְׁחוֹט אֶת הַפְּסָחִים, וְעַד סוֹף הַיּוֹם (נִרְאֶה — עַד צֵאת הַכּוֹכָבִים; וְצָרִיךְ עִיּוּן בְּבֵין הַשְּׁמָשׁוֹת. וְנִרְאֶה, דִּבְעֶרֶב שַׁבָּת לֹא הָיוּ עוֹשִׂין עַד סוֹף הַיּוֹם בְּמִלּוּאוֹ, אֶלָּא כְּדֵי שֶׁיִּשָּׁאֵר עוֹד זְמַן לִצְלִיָּתוֹ וַהֲדָחַת קְרָבָיו; עַיֵּן בְּרֵישׁ פֶּרֶק "תָּמִיד נִשְׁחָט", וְעַיֵּן מַה שֶּׁכָּתַבְתִּי בְּחִדּוּשַׁי לְיוֹרֶה דֵּעָה (סִימָן ס״ט), עַיֵּן שָׁם, בְּסִיַּעְתָּא דִּשְׁמַיָּא) הָיוּ שׁוֹחֲטִים וְזוֹרְקִים, כְּמִשְׁפְּטֵי כָּל הַפְּסָחִים שֶׁבְּיִשְׂרָאֵל:

# מִצְוַת עֲשֵׂה — אָנֹכִי
# פָּרָשַׁת יִתְרוֹ — מִצְוַת עֲשֵׂה

**(ו) מִצְוַת** עֲשֵׂה לְהַאֲמִין שֶׁיֵּשׁ אֱלוֹקַּ אֶחָד, הַמַּמְצִיא כָּל נִמְצָא, וְהוּא רִאשׁוֹן וְהוּא אַחֲרוֹן, וְיָדוֹ בַּכֹּל מָשָׁלָה, וּמִבַּלְעָדָיו אֵין אֱלֹקִים; שֶׁנֶּאֱמַר: "אָנֹכִי ה׳ אֱלֹקֶיךָ אֲשֶׁר הוֹצֵאתִיךָ מֵאֶרֶץ מִצְרַיִם" וְגוֹ׳. זֶהוּ דַּעַת הָרַמְבַּ״ם. אֲבָל הָרַמְבַּ״ן, בְּסֵפֶר הַמִּצְוֹת מִצְוַת עֲשֵׂה א׳, כָּתַב שֶׁהַבַּהֲ״ג לֹא מָנָה זֶה בַּמִּנְיָן; וְדַעְתּוֹ שֶׁלֹּא לִמְנוֹת זֶה, שֶׁהוּא הָעִקָּר וְהַיָּתֵד שֶׁכָּל הַמִּצְוֹת תְּלוּיִים בּוֹ — וְזֶהוּ שֹׁרֶשׁ מִכָּל הַמִּצְוֹת,
