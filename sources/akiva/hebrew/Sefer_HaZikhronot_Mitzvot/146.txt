~ אַף לִטְבוּל יוֹם; שֶׁנֶּאֱמַר (דִּבְרֵי הַיָּמִים ב׳ כ׳): "וַיַּעֲמֹד יְהוֹשָׁפָט בִּקְהַל יְהוּדָה וִירוּשָׁלַיִם בְּבֵית ה׳, לִפְנֵי הֶחָצֵר הַחֲדָשָׁה" — מַאי "חֲדָשָׁה"? שֶׁחִדְּשׁוּ בָּהּ דְּבָרִים, וְאָמְרוּ: טְבוּל יוֹם אַל יִכָּנֵס בֶּחָצֵר — שֶׁהוּא הָעֶזְרַת נָשִׁים, שֶׁנִּקְרָא "חָצֵר" בִּיחֶזְקֵאל (מ׳) (וְכֻלְּהוּ טְבוּלֵי יוֹם בְּמַשְׁמָע — אֲפִלּוּ דְּשֶׁרֶץ וּדְנְבֵלָה; תּוֹסָפוֹת זְבָחִים ל״ב עַמּוּד ב׳, דִּבּוּר הַמַּתְחִיל "וְרַבִּי יְהוּדָה"). וְהַטַּעַם שֶׁהֶחְמִירוּ יוֹתֵר בְּעֶזְרַת נָשִׁים — מִפְּנֵי שֶׁהִיא (תּוֹסָפוֹת זְבָחִים ל״ב עַמּוּד א׳) מְקוֹם כְּנִיסָה וִיצִיאָה לְכָל אָדָם בְּיוֹתֵר; לְפִיכָךְ אָסְרוּ בּוֹ טְבוּל יוֹם, וְלֹא הֶעֱמִידוּ עַל דִּבְרֵי תּוֹרָה אֶלָּא מְחֻסַּר כִּפּוּרִים בִּלְבַד, שֶׁאֵינוֹ נֶאֱסָר אֶלָּא מִפֶּתַח עֶזְרַת יִשְׂרָאֵל בִּלְבַד:

**וְהוֹסִיפוּ** עוֹד חֲכָמִים זִכְרוֹנָם לִבְרָכָה מַעֲלוֹת וְחֻמְרוֹת יְתֵרוֹת בֶּעָרִים הַמֻּקָּפִים חוֹמָה, מִשּׁוּם גֶּדֶר הַרְחָקַת טֻמְאָה מִמַּחֲנֵה יִשְׂרָאֵל: שֶׁלֹּא לִקְבּוֹר בְּתוֹכָם מֵת — מֵעִירָן, אוֹ מֵעִיר אַחֶרֶת שֶׁהֱבִיאוּהוּ לְשָׁם — עַד שֶׁיִּרְצוּ ז׳ טוֹבֵי הָעִיר, אוֹ כָּל אַנְשֵׁי הָעִיר (אוּלַי זֶהוּ קָאֵי הֵיכִי דְּלֵיכָּא ז׳ טוֹבֵי הָעִיר). וְאִם יָצָא הַמֵּת חוּץ לָעִיר — אֵין מַחֲזִירִין אוֹתוֹ לְתוֹכָהּ, אַף עַל פִּי שֶׁרָצוּ כֻּלָּן לְהַחֲזִירוֹ. זֶהוּ דַּעַת הָרַמְבַּ״ם. אֲבָל הָרַאֲבַ״ד כָּתַב, שֶׁאֵין מוֹעִיל כְּלָל רְצוֹן בְּנֵי הָעִיר, אֶלָּא לְעוֹלָם אֵין קוֹבְרִין בָּהֶם (אוּלַי רוֹצֶה לוֹמַר: רַק מֵת מֵעִיר אַחֶרֶת, אֲבָל לֹא מֵאוֹתָהּ הָעִיר); אֲבָל אִם רָצוּ לְסַבְּבוֹ בָּעִיר, כְּדֵי לְהַרְבּוֹת הֶסְפֵּדוֹ וּכְבוֹדוֹ, עַד שֶׁיִּרְצוּ קְרוֹבָיו לְהוֹצִיאוֹ — מְסַבְּבִין, עַד כָּאן. וְאִם רָצוּ בְּנֵי הָעִיר לְהוֹצִיא הַקֶּבֶר הַנִּמְצָא שָׁם בָּעִיר, לְהוֹצִיאוֹ חוּץ לָעִיר — מְפַנִּין אוֹתוֹ; חוּץ מִקֶּבֶר הַמֶּלֶךְ אוֹ הַנָּבִיא, שֶׁאֵין מְפַנִּין. וְאִם הִקִּיפָה הָעִיר לְהַקֶּבֶר מִד׳ אוֹ מִב׳ רוּחוֹת, זוֹ כְּנֶגֶד זוֹ — אִם הָיְתָה בֵּינוֹ וּבֵין הָעִיר יוֹתֵר מִנ׳ אַמָּה (זֶהוּ מִמַּסֶּכֶת א״ר; אֲבָל בַּיְרוּשַׁלְמִי אִיתָא ע׳ אַמָּה וְשִׁירַיִם) לְכָאן, וְנ׳ אַמָּה לְכָאן — אֵין מְפַנִּין אוֹתוֹ עַד שֶׁיִּרְצוּ כֻּלָּן (נִרְאֶה, הוּא הַדִּין רֻבָּן); פָּחוֹת מִכָּאן — מְפַנִּין אוֹתוֹ:

**וְאֵיזֶהוּ** עִיר מֻקֶּף חוֹמָה (פֶּרֶק בַּתְרָא דַּעֲרָכִין)? ג׳ חֲצֵרוֹת שֶׁל ב׳ ב׳ בָּתִּים, מֻקָּפִים חוֹמָה מִימוֹת יְהוֹשֻׁעַ בִּן נוּן — כְּגוֹן: קַצְרָה הַיְשָׁנָה שֶׁל צִפּוֹרִי, וְחַקְרָה שֶׁבְּגוּשׁ חָלָב, וְיוֹדְפַת הַיְשָׁנָה, וְגַמְלָא, וּגְדוּד, וְחָדִיד, וְאוֹנוֹ, וְלוֹד; וְלֹא אֵלּוּ בִּלְבַד, אֶלָּא כָּל כַּיּוֹצֵא בָּהֶם — שֶׁיֵּשׁ קַבָּלָה בָּהֶם וּמָסֹרֶת שֶׁהֵם מֻקָּפִים חוֹמָה מִימוֹת יְהוֹשֻׁעַ בִּן נוּן — כָּל הַדְּבָרִים הָאֵלּוּ נוֹהֲגִים בּוֹ. וּבִטְבֶרְיָא — יֵשׁ לְעַיֵּן; בְּפֶרֶק קַמָּא דִּמְגִלָּה, חִזְקִיָּה מְסַפְּקָא לֵיהּ — עַיֵּן שָׁם ה׳ עַמּוּד ב׳; וְעַיֵּן שָׁם בְּהָרַ״ן, אִם הוּא סָפֵק הַשָּׁקוּל, אוֹ אָזְלִינַן בָּתַר רֹב — דְּלָאו מֻקָּפִים חוֹמָה; וְעַיֵּן בְּסֵפֶר טוּרֵי אֶבֶן שָׁם:
