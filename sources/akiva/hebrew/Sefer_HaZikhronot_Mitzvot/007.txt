# פֶּרֶק שֵׁנִי

**כָּל** מִי שֶׁמְּחֻיָּב בְּמִצְוַת פְּרִיָּה וּרְבִיָּה אָסוּר לוֹ לְהַשְׁחִית זַרְעוֹ לָאָרֶץ (תּוֹסָפוֹת סַנְהֶדְרִין נ״ט עַמּוּד ב׳). וְעַיֵּן בְּתוֹסָפוֹת כְּתֻבּוֹת (ל״ט עַמּוּד א׳) וְנִדָּה (י״ג עַמּוּד א׳) בְּעִנְיַן הַנָּשִׁים, דְּגַם כֵּן אֲסוּרוֹת (עַיֵּן שָׁם). וַחֲמִיר לָאו זֶה מִכָּל הָאִסּוּרִים שֶׁבַּתּוֹרָה, שֶׁהוּא כְּאִלּוּ מֵבִיא מַבּוּל לָעוֹלָם, שֶׁזֶּה הָיָה עֲוֹן דּוֹר הַמַּבּוּל, כְּמוֹ שֶׁאָמְרוּ: בְּרוֹתְחִין קִלְקְלוּ. וּמוּטָב יֵצֵא לַעַז עַל בָּנָיו, וְאַל יוֹצִיא זֶרַע לְבַטָּלָה. וְכָל הַמּוֹצִיא זֶרַע לְבַטָּלָה חַיָּב מִיתָה, וּכְאִלּוּ שׁוֹפֵךְ דָּמִים, וּכְאִלּוּ עוֹבֵד עֲבוֹדַת כּוֹכָבִים. וּמוּטָב תִּבָּקַע כְּרֵסוֹ, וְאַל יָבִיא עַצְמוֹ לְהוֹצִיא זֶרַע לְבַטָּלָה (הַכֹּל בְּרֵישׁ פֶּרֶק כָּל הַיָּד). וּבְמַסֶּכֶת כַּלָּה אִיתָא: כָּל הַמְחַמֵּם עַצְמוֹ לְהוֹצִיא זֶרַע לְבַטָּלָה חָשׁוּב כִּבְהֵמָה וְכוּלֵי — מָה בְּהֵמָה אֵינָהּ אֶלָּא לִשְׁחִיטָה וְאֵין לָהּ חֵלֶק לָעוֹלָם הַבָּא, אַף הוּא אֵין לוֹ חֵלֶק לָעוֹלָם הַבָּא. וּבַזֹּהַר הַקָּדוֹשׁ בְּרֵאשִׁית (נ״ז עַמּוּד א׳): "רַע לָא אִקְרֵי אֶלָּא מַאן דִּמְחַבֵּל אָרְחֵיהּ" וְכוּלֵי, "וְלָא עָאל בְּפַלְטְרִין, וְלָא חָמֵי אַפֵּי שְׁכִינְתָּא". וְכֵן הוּא כָּפוּל בְּפָרָשַׁת נֹחַ (ס״ט עַמּוּד א׳): "מַאן דִּמְחַבֵּל אָרְחֵיהּ" וְכוּלֵי. וְעוֹד בְּהַרְבֵּה מְקוֹמוֹת הֶאֱרִיכוּ בַּזֹּהַר בְּחֹמֶר הֶעָוֹן מְאֹד מְאֹד, עַד שֶׁאָמְרוּ שֶׁאֵין הַתְּשׁוּבָה מוֹעֶלֶת לוֹ. וְעַיֵּן בְּפָרָשַׁת נֹחַ (ס״ב עַמּוּד א׳) נִתּוֹסַף: "בַּר בִּתְשׁוּבָה סַגִּי"; וְכֵן כָּתוּב בְּסֵפֶר רֵאשִׁית חָכְמָה, שַׁעַר הַקְּדֻשָּׁה (פֶּרֶק י״ז), בְּשֵׁם סְפָרִים יְשָׁנִים כְּתַב יָד, כָּתוּב בָּהֶם: "בַּר בְּחֵילָא דִּתְיוּבְתָּא תַּקִּיפָא". וְעַיֵּן לְהַגָּאוֹן יַעְבֵּ״ץ בְּסֵפֶר מִטְפַּחַת, וְעַיֵּן בְּסֵפֶר תּוֹלְדוֹת אָדָם, שֶׁהִרְעִישׁוּ עַל זֶה מִמַּאֲמַר רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה: "אֵין לְךָ דָּבָר שֶׁעוֹמֵד בִּפְנֵי הַתְּשׁוּבָה". וְעַיֵּן בְּסֵפֶר רֵאשִׁית חָכְמָה דָּחַק לְיַשֵּׁב, כִּי צָרִיךְ תְּשׁוּבָה תַּקִּיפָא — "תְּיוּבְתָּא דְּאִתְחַזְיָא לְחַפָּאָה עַל כָּל עוֹבָדוֹי", עַיֵּן שָׁם בַּאֲרִיכוּת. וּבְתַנָּא דְּבֵי אֵלִיָּהוּ, הוּבָא בְּיַלְקוּט מִשְׁפָּטִים (רֶמֶז שס״א): "אַשְׁרֵי" וְכוּלֵי "וְלֹא יָצְתָה מִמֶּנּוּ שִׁכְבַת זֶרַע חִנָּם, וַאֲפִלּוּ הוּא יִשְׂרָאֵל רָאוּי לְהַעֲלוֹת עוֹלוֹת" וְכוּלֵי, עַיֵּן שָׁם. וְיַעֲקֹב אָבִינוּ עָלָיו הַשָּׁלוֹם בָּזֶה נִשְׁתַּבַּח, שֶׁאָמַר: "כֹּחִי וְרֵאשִׁית אוֹנִי". וְעַיֵּן סִפְרֵי מוּסָר, הֶאֱרִיכוּ בָּזֶה עַד בְּלִי דָּי:

**אֵיזֶהוּ** הוֹצָאַת זֶרַע לְבַטָּלָה? אֵלּוּ הַמְנָאֲפִים בַּיָּד אוֹ בָּרֶגֶל וּשְׁאָר דְּרָכִים, עַד שֶׁנִּמְצָא מוֹצִיא זֶרַע לְבַטָּלָה. וְלֹא זוֹ בִּלְבַד, אֶלָּא אַף הַבָּא עַל אִשְׁתּוֹ וְדָשׁ בִּפְנִים וְזוֹרֶה מִבַּחוּץ — הֲרֵי זֶה מוֹצִיא זֶרַע לְבַטָּלָה, שֶׁכֵּן הָיָה מַעֲשֵׂה עֵר, וְנֶאֱמַר: "וַיְהִי עֵר בְּכוֹר יְהוּדָה רַע". וּלְפִיכָךְ אָסוּר לְשַׁמֵּשׁ עִם אִשָּׁה שֶׁיֵּשׁ לָהּ אֹטֶם בָּרֶחֶם, שֶׁנִּמְצָא כְּשֶׁבּוֹעֲלָהּ זוֹרֶה מִבַּחוּץ. אֲבָל מֻתָּר לְשַׁמֵּשׁ בִּקְטַנָּה וְאַיְלוֹנִית וּזְקֵנָה שֶׁאֵין רְאוּיִים לֵילֵד, אַף עַל פִּי שֶׁאָסוּר לְשָׂאָם לְכַתְּחִלָּה מִפְּנֵי שֶׁמְּבַטֵּל מִצְוַת פְּרִיָּה וּרְבִיָּה, אִם לֹא שֶׁעֲדַיִן אֵינוֹ בֶּן כ׳ (וּבְאַתְרָא דְּלָא נְהִיגֵי בְּחֵרֶם דְּרַבֵּינוּ גֵּרְשׁוֹם מְאוֹר הַגּוֹלָה; אִי נַמִּי בִּקְטַנָּה, וְתִגְדַּל עַד שֶׁיִּהְיֶה בֶּן כ׳). מִכָּל מָקוֹם, אִם נָשָׂא — מֻתָּר לְשַׁמֵּשׁ
