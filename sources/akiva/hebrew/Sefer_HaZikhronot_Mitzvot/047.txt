~ שֶׁמָּא יָבוֹאוּ הָעֵדִים, וְאִי אֶפְשָׁר שֶׁיַּקְרִיבוּ מוּסַף הַיּוֹם אַחַר תָּמִיד שֶׁל בֵּין הָעַרְבַּיִם (כֵּן מַשְׁמַע לְשׁוֹן הַבָּרַיְתָא (ל׳ עַמּוּד ב׳): "כְּדֵי שֶׁיְּהֵא שָׁהוּת לְהַקְרִיב" וְכוּלֵי — מַשְׁמַע דְּחָשַׁשׁ מֵהַקְרָבָה, וְתָנָא דְּמַתְנִיתִין חֲדָא מִתַּרְתֵּי נָקַט; וְהָרַמְבַּ״ם בְּשִׁיר סָמַךְ עַל "וְנִסְכֵּיהֶם", שֶׁהַשִּׁיר בִּמְקוֹם שֶׁצָּרִיךְ הוּא בִּכְלַל הַנְּסָכִים; כֵּן נִרְאֶה פָּשׁוּט. וְעַיֵּן בְּלֶחֶם מִשְׁנֶה וְתוֹסְפוֹת יוֹם טוֹב. וְעַיֵּן בְּתוֹסָפוֹת, דִּבּוּר הַמַּתְחִיל "וְנִתְקַלְקְלוּ", הִקְשׁוּ, וְכָתְבוּ: "וְיֵשׁ לוֹמַר" וְכוּלֵי, "דְּאָתֵי עֲשֵׂה דְּרַבִּים וְדָחֵי עֲשֵׂה דְּהַשְׁלָמָה" וְכוּלֵי. וְלֹא הֵבַנְתִּי, דְּהָא אַחַר כָּךְ כָּתְבוּ "וְנִרְאֶה לִי דִּלְכַתְּחִלָּה" וְכוּלֵי; וּלְכַתְּחִלָּה וַדַּאי גַּם זֶה אָסוּר לְהָבִיא עַצְמוֹ אַף בְּדָחֵי בְּדִין, מִכָּל מָקוֹם דְּחוּיָה לֹא הוּתְּרָה. וְגַם אֶפְשָׁר דְּזֶה מִקְרִי לַעֲשׂוֹת שְׁנֵיהֶם כְּשֶׁיִּתְקָנוּ, כַּנִּזְכָּר לְעֵיל, דְּהָא אִתֶּם אֲפִלּוּ מְזִידִין. וּבֶאֱמֶת מַשְׁמָעוּת הַבָּרַיְתָא כֵּן כַּנִּזְכָּר לְעֵיל, וּמַתְנִיתִין חֲדָא נָקְטָא, כַּנִּזְכָּר לְעֵיל — כֵּן הוּא פָּשׁוּט). עָמְדוּ בֵּית דִּין וְהִתְקִינוּ שֶׁלֹּא יִהְיוּ מְקַבְּלִים עֵדוּת הַחֹדֶשׁ אֶלָּא עַד זְמַן הַמִּנְחָה קְטַנָּה, שֶׁהוּא עַד ט׳ שָׁעוֹת וּמֶחֱצָה שֶׁל הַיּוֹם (סֵפֶר טוּרֵי אֶבֶן, דִּבּוּר הַמַּתְחִיל "הִתְקִינוּ"), כְּדֵי שֶׁיְּהֵא שָׁהוּת בַּיּוֹם לְהַקְרִיב מוּסָפִין, וְתָמִיד שֶׁל בֵּין הָעַרְבַּיִם, וְנִסְכֵּיהֶם. וְאִם הִגִּיעַ מִנְחָה וְלֹא בָּאוּ עֵדִים — עוֹשִׂין אוֹתוֹ הַיּוֹם קֹדֶשׁ (לָא יָדְעִנָא — לְפִי מַה דְּאָמְרֵי בְּכָל חֳדָשִׁים, מַאי "קֹדֶשׁ"? וְאַכּוּלֵּי יוֹמָא קָאֵי. וְנִרְאֶה: דְּיֵשׁ לוֹמַר אַהֶסְפֵּד וְתַעֲנִית), וּמַקְרִיבִין מוּסָף לְמָחָר, לְפִי שֶׁלֹּא הָיוּ מְקַדְּשִׁין אוֹתוֹ אַחַר מִנְחָה. מִשֶּׁחָרַב בֵּית הַמִּקְדָּשׁ הִתְקִין רַבָּן יוֹחָנָן בֶּן זַכַּאי שֶׁיִּהְיוּ מְקַבְּלִין עֵדוּת הַחֹדֶשׁ כָּל הַיּוֹם כֻּלּוֹ; וַאֲפִלּוּ בָּאוּ עֵדִים גַּם ל׳ סָמוּךְ לִשְׁקִיעַת הַחַמָּה — מְקַבְּלִין עֵדוּתָן, וּמְקַדְּשִׁין יוֹם ל׳ בִּלְבַד (וְזֶהוּ דַּעַת הָרַמְבַּ״ם; אֲבָל מִדִּבְרֵי רַשִׁ״י וְתוֹסָפוֹת וּשְׁאָר מְפָרְשִׁים מַשְׁמַע לְהֶדְיָא, דְּרַק בְּרֹאשׁ הַשָּׁנָה הָיְתָה תַּקָּנָה רִאשׁוֹנָה שֶׁלֹּא לְקַבֵּל עֵדוּת מִן הַמִּנְחָה וּלְמַעְלָה, מִשּׁוּם שִׁיר, וְנוֹהֲגִין אוֹתוֹ יוֹם קֹדֶשׁ מִמְּלָאכָה — לֹא בִּשְׁאָר חֳדָשִׁים. וְעַיֵּן בְּסֵפֶר טוּרֵי אֶבֶן, יִישֵּׁב עַל פִּי זֶה הָא דְּנָקַט שִׁיר וְלֹא מִשּׁוּם מוּסָף, עַיֵּן שָׁם):

**וּכְשֶׁמְּעַבְּרִין** אֶת הַחֹדֶשׁ, כְּשֶׁלֹּא בָּאוּ עֵדִים כָּל הַיּוֹם ל׳ — הָיוּ עוֹלִין לְמָקוֹם מוּכָן, וְעוֹשִׂין בּוֹ סְעוּדָה בְּלֵיל ל״א, שֶׁהוּא רֹאשׁ חֹדֶשׁ. וְאֵין עוֹלִין לְשָׁם בַּלַּיְלָה, אֶלָּא מִשְּׁקִיעַת הַחַמָּה בְּיוֹם ל׳, בְּעוֹד יוֹם קְצָת, כְּדֵי שֶׁיִּרְאוּם עוֹבְרֵי דְּרָכִים עוֹלִים; וְעִקַּר הַסְּעוּדָה תְּהֵא בַּלַּיְלָה. וְאֵין עוֹלִין לִסְעוּדָה זוֹ פָּחוֹת מֵעֲשָׂרָה בְּנֵי אָדָם; וְאֵין עוֹלִין לָהּ בְּבָשָׂר וּשְׁאָר מַעֲדַנִּים, אֶלָּא בְּפַת דָּגָן וְקִטְנִית בִּלְבַד. וְזוֹ הִיא סְעוּדַת מִצְוָה, וְאֵינָהּ לֹא מִן הַתּוֹרָה, אֶלָּא תַּקָּנַת חֲכָמִים הָיְתָה לְפַרְסֵם הַדָּבָר (עַיֵּן תּוֹסְפוֹת יוֹם טוֹב פֶּרֶק ח׳ דְּסַנְהֶדְרִין מִשְׁנָה ב׳, וּבְתוֹסָפוֹת חֲדָשִׁים). וּכְשֶׁיּוֹרְדִים — יָרְדוּ בְּהַשְׁכָּמָה, כְּדֵי שֶׁהָרוֹאֶה אוֹתָם יוֹרְדִים יֵדַע שֶׁיָּשְׁבוּ שָׁם כָּל הַלַּיְלָה, וְלֹא יַחְשׁוֹב שֶׁעַכְשָׁיו עָלוּ, וְיִתְפַּרְסֵם
