~ הַקְּדוֹשִׁים לֹא פֵּרְשׁוּ כֵּן. וּבֶאֱמֶת צָרִיךְ עִיּוּן לְרַבִּי אֶלְעָזָר בְּרַבִּי צָדוֹק, דְּאֵין מְקַדְּשִׁין מְעֻבָּר — עַד אֵימָתַי יַמְתִּין בִּמְעֻבָּר? וְאוּלַי עַד שִׁעוּר שֶׁיּוּכַל לְקַדֵּשׁ, וּמִשּׁוּם גְּזֵרָה, כַּנִּזְכָּר לְעֵיל; אִי נַמִּי עַד סְעוּדַת הָעִבּוּר; וְצָרִיךְ עִיּוּן. וְלָכֵן נִרְאֶה דְּרַשִׁ״י וְתוֹסָפוֹת נַמִּי אָזְלֵי דְּלָא כְּרַבִּי אֶלְעָזָר בְּרַבִּי צָדוֹק — לֹא לְדִידַן, דְּאָתֵי שַׁפִּיר הֲלָכָה כְּרַבִּי אֶלְעָזָר בְּרַבִּי צָדוֹק; וַעֲדַיִן צָרִיךְ עִיּוּן):

**וְכָל** מָקוֹם שֶׁהָיוּ שְׁלוּחִים מַגִּיעִים — הָיוּ עוֹשִׂין אֶת הַמּוֹעֲדוֹת יוֹם טוֹב אֶחָד, כַּכָּתוּב בַּתּוֹרָה. וּבַמְּקוֹמוֹת הָרְחוֹקִים, שֶׁאֵין הַשְּׁלוּחִים מַגִּיעִים אֲלֵיהֶם — הָיוּ עוֹשִׂים ב׳ יָמִים, מִפְּנֵי הַסָּפֵק, לְפִי שֶׁאֵין יוֹדְעִים יוֹם שֶׁקָּבְעוּ בֵּית דִּין הַחֹדֶשׁ אֵיזֶהוּ — אִם ל׳ אוֹ ל״א. כָּךְ הָיוּ עוֹשִׂין בְּכָל יוֹם טוֹב; אֲבָל יוֹם הַכִּפּוּרִים לֹא עָשׂוּ אֶלָּא יוֹם אֶחָד, מִפְּנֵי שֶׁהוּא סַכָּנָה לְהִתְעַנּוֹת ב׳ יָמִים בְּבַת אַחַת וּלְהַפְסִיק מִקֹּדֶם; וּלְפִיכָךְ סָמְכוּ עַל הַחֲזָקָה שֶׁאֵין בֵּית דִּין מִתְעַצְּלִים מִלְּקַדֵּשׁ הַחֹדֶשׁ בִּזְמַנּוֹ. (יְרוּשַׁלְמִי, כְּמוֹ שֶׁכָּתַבְתִּי לְעֵיל פֵּרוּשׁוֹ; וְעַיֵּן מַה שֶּׁכָּתַבְתִּי לְעֵיל בְּדִבְרֵי רַשִׁ״י דִּפְסָחִים, שֶׁבְּאד״ו הֲוֵי סָפֵק, וְלֹא שַׁיָּךְ חֲזָקָה זוֹ, וְצָרִיךְ יוֹם הַכִּפּוּרִים ב׳ יָמִים.) וּמִכָּל מָקוֹם יֵשׁ מֵאַנְשֵׁי מַעֲשֶׂה הָיוּ מִתְעַנִּים יוֹם הַכִּפּוּרִים ב׳ יָמִים, מִסְּפֵיקָא דְּיוֹמָא (כ״א עַמּוּד א׳: "רָבָא" וְכוּלֵי, וּבִירוּשַׁלְמִי):

**וּבַחֲנֻכָּה** וּפוּרִים — נִרְאֶה לִי פָּשׁוּט, בְּמָקוֹם שֶׁאֵין הַשְּׁלוּחִים מַגִּיעִים לֹא יַעֲשׂוּ אֶלָּא יוֹם אֶחָד, דְּמִדְּרַבָּנָן הוּא, וּסְפֵיקוֹ לְהָקֵל; וְתֵדַע, דְּלֹא עָבְדִינַן הָאִידָּנָא מִסְּפֵיקָא דְּיוֹמָא בַּחֲנֻכָּה וּפוּרִים. וְעַיֵּן בְּסֵפֶר טוּרֵי אֶבֶן (י״ח עַמּוּד ב׳), דִּבּוּר הַמַּתְחִיל "בִּזְמַן שֶׁבֵּית הַמִּקְדָּשׁ קַיָּם" — לֹא כָּתַב כֵּן, וְטַעְמוֹ: דְּלֹא שַׁיָּךְ סְפֵיקָא דְּרַבָּנָן לְהָקֵל, דְּיִתְבַּטֵּל כָּל הַדְּרַבָּנָן, עַיֵּן שָׁם בַּאֲרִיכוּת. וַאֲנִי אוֹמֵר: בַּחֲנֻכָּה פָּשׁוּט, דְּלֹא יִתְבַּטֵּל כְּלָל כָּל הַדְּרַבָּנָן אֶלָּא יוֹם אֶחָד, וְאֵין זֶה קָרוּי בִּטּוּל כָּל הַדְּרַבָּנָן. מִיהוּ, בְּלָאו הָכֵי לֹא קָשֶׁה מִידֵּי בֵּין בַּחֲנֻכָּה בֵּין בְּפוּרִים — דְּאַטּוּ בְּנֵי בָבֶל לֹא קָבְעוּ שׁוּם קְבִיעוּת עַד שֶׁיֵּדְעוּ מֵאֶרֶץ יִשְׂרָאֵל? זֶה אֵין הַדַּעַת סוֹבַלְתּוֹ. וְעַיֵּן (בְּדַף כ״א רֵישׁ עַמּוּד א׳) וּבַתּוֹסָפוֹת, דִּבּוּר הַמַּתְחִיל "לֵוִי" — נַמִּי לֹא מַשְׁמַע כֵּן; אֶלָּא בְּנֵי בָבֶל הָיוּ קוֹבְעִין לָהֶם כְּפִי רְאוּת עֵינֵיהֶם, וּמוֹנִים לָהֶם; אֲבָל לֹא לְעִקָּר, עַד שֶׁיּוֹדִיעוּ לָהֶם מִבֵּית דִּין שֶׁל אֶרֶץ יִשְׂרָאֵל. וְאִם כֵּן, וַדַּאי קָבְעוּ גַּם חֲנֻכָּה וּפוּרִים לְפִי קְבִיעָתָם, וְלֹא שַׁיָּךְ דְּמִסָּפֵק יִתְבַּטֵּל הַכֹּל — דְּהָא לֹא יִתְבַּטֵּל, כַּנִּזְכָּר לְעֵיל; וְעַיֵּן מַה שֶּׁכָּתַבְתִּי לְעֵיל עַל דִּבְרֵי הַטּוּרֵי אֶבֶן. וּקְצָת רְאָיָה לְמַה שֶּׁכָּתַבְתִּי מִדִּבְרֵי רַשִׁ״י (כ״ג עַמּוּד א׳), דִּבּוּר הַמַּתְחִיל "אָמַר אַבָּיֵי", שֶׁפֵּרֵשׁ אַרֹאשׁ הַשָּׁנָה; וְעַיֵּן שָׁם בַּתּוֹסָפוֹת, דִּבּוּר הַמַּתְחִיל "מִשּׁוּם" וְכוּלֵי, וּלְחִנָּם דָּחֲקוּ וְכוּלֵי בְּרֹאשׁ חֹדֶשׁ וְכוּלֵי, עַיֵּן שָׁם. וּלְמַה שֶּׁכָּתַבְתִּי — יֵשׁ לוֹמַר: דַּהֲלֹא רֹאשׁ חֹדֶשׁ דְּרַבָּנָן, וְלֹא שַׁיָּךְ מִסָּפֵק יַחְמִיר; וְגַם אַוַּדַּאי לְתַקֵּן — זֶה אֵין לְהִזָּקֵק, כֵּיוָן דְּאֵינוֹ חִיּוּבָא בֶּאֱמֶת, אֶלָּא מִנְהָגָא; וְרַק
