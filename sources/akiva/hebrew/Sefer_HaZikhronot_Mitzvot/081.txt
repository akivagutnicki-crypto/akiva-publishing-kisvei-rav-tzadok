~ פִּי הָרְאִיָּה — לְפִיכָךְ קָבְעוּ יָמִים שֶׁיִּדָּחֶה רֹאשׁ חֹדֶשׁ, וְלֹא יִקָּבַע אִם יִפּוֹל הַמּוֹלָד בָּהֶם, כְּדֵי לִפְגּוֹעַ בְּיוֹם קִבּוּץ חַמָּה וּלְבָנָה הָאֲמִתִּי. וְקָבְעוּ כֵּן מֵרֵאשִׁית הַשָּׁנָה, בְּרֹאשׁ חֹדֶשׁ תִּשְׁרֵי, לְתַקֵּן זֶה. כֵּיצַד? חָל בְּג׳ — קוֹבְעִין בְּד׳; דּוֹחִין בְּה׳ — קוֹבְעִין בְּו׳; דּוֹחִין בְּשַׁבָּת — קוֹבְעִין בְּא׳; דּוֹחִין בְּב׳ — קוֹבְעִין. וְלָמָּה בָּחֲרוּ בְּג׳ יָמִים אֵלּוּ — שֶׁהֵם א׳ ד׳ ו׳ — לִדְחוֹת רֹאשׁ הַשָּׁנָה אִם חָל בָּהֶם? מִפְּנֵי טְעָמִים שֶׁהָיוּ לָהֶם: בְּא׳ — מִפְּנֵי שֶׁאִם יִהְיֶה רֹאשׁ הַשָּׁנָה בְּא׳, יָחוּל יוֹם הוֹשַׁעְנָא רַבָּה בְּשַׁבָּת, וְאֵין חוֹבְטִין הָעֲרָבָה בְּשַׁבָּת, שֶׁאֵין חֲבִיטַת הָעֲרָבָה — שֶׁהוּא רַק מִנְהַג נְבִיאִים — דּוֹחֶה שַׁבָּת, וְתִשְׁתַּכַּח מִצְוַת חֲבִיטַת הָעֲרָבָה; לְפִיכָךְ עָשׂוּ חֲכָמִים חִזּוּק לְדִבְרֵיהֶם, שֶׁלֹּא יָחוּל לְעוֹלָם בְּשַׁבָּת. וּבְד׳ ו׳ — מִפְּנֵי שֶׁאָז יָחוּל יוֹם הַכִּפּוּרִים בְּו׳ אוֹ בְּא׳, וְיִהְיֶה יוֹם הַכִּפּוּרִים סָמוּךְ לְשַׁבָּת, לְפָנָיו אוֹ לְאַחֲרָיו; וְאֵין עוֹשִׂים ב׳ שַׁבָּתוֹת סְמוּכִים זֶה לָזֶה, מִשּׁוּם שֶׁמָּא יָמוּת אֶחָד בֵּינְתַיִם, וִיהֵא אָסוּר לְקָבְרוֹ אֲפִלּוּ עַל יְדֵי עוֹבֵד כּוֹכָבִים, וְיַסְרִיחַ; וְעוֹד, מִשּׁוּם יְרָקוֹת שֶׁלֹּא יִתְעַפְּשׁוּ, כְּשֶׁיְּהֵא אָסוּר לְבַשְּׁלָם ב׳ יָמִים זֶה אַחַר זֶה:

**וְאַף עַל פִּי** שֶׁהַטְּעָמִים הַלָּלוּ חַלּוּשִׁים לִדְחוֹת בִּשְׁבִילָם רֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים וּמוֹעֲדוֹת מִמְּקוֹמָן — כְּבָר אָמַרְנוּ, שֶׁעִקַּר הַטַּעַם לְהַשְׁווֹת חֶשְׁבּוֹן זֶה הָאֶמְצָעִי עִם חֶשְׁבּוֹן הָאֲמִתִּי, וּדְבָרִים אֵלּוּ הֵם סַעַד לִהְיוֹת הַדְּחִיָּה בְּיָמִים אֵלּוּ, לֹא בְּזוּלָתָם. (וּמַשְׁמַע מִזֶּה, דִּבִזְמַן שֶׁהָיוּ מְקַדְּשִׁין עַל פִּי הָרְאִיָּה — לֹא הָיוּ דּוֹחִין בְּאד״ו. וְעַיֵּן פְּסָחִים (נ״ט עַמּוּד ב׳), בְּרַשִׁ״י וְתוֹסָפוֹת, דִּבּוּר הַמַּתְחִיל "ב׳ אֵלּוּ", עַיֵּן שָׁם; וּבְסֵפֶר צְלִיל״ת שָׁם, עַיֵּן שָׁם. וְעַיֵּן בְּהַגָּהַת לְבוּשׁ, בְּשֵׁם הָרַב רַבִּי שִׁמְשׁוֹן בְּרַבִּי בְּצַלְאֵל — כֵּן. וְעַיֵּן מַהַרְשָׁ״א פְּסָחִים (ע״א עַמּוּד א׳), בַּתּוֹסָפוֹת דִּבּוּר הַמַּתְחִיל "לֵילֵי". וְעַיֵּן סֵפֶר יְסוֹד עוֹלָם (מַאֲמָר ד׳ פֶּרֶק ה׳), וּבְסֵפֶר בֵּית אַבְרָהָם, בְּקֻנְטְרֵס מִנְחָה בְּלוּלָה, בְּסוֹפוֹ, בְּחִדּוּשֵׁי מְגִלָּה (ה׳ עַמּוּד א׳), וּבִכְנֶסֶת יְחֶזְקֵאל (סִימָן י״ט) עַל פּוּרִים שֶׁחָל בְּשַׁבָּת. וְעַיֵּן בַּמְּפָרֵשׁ לְכָאן):

**וְעוֹד** יֵשׁ ג׳ דְּחִיּוֹת אֲחֵרוֹת לְרֹאשׁ חֹדֶשׁ, וְהֵם: (א) כְּשֶׁהַמּוֹלָד חָל בְּי״ח שָׁעוֹת אוֹ אַחַר כָּךְ — אֲפִלּוּ חָל בבגה״ז — נִדְחֶה לְמָחֳרָתוֹ; וְזוֹ הַנִּקְרָא בְּפִי חַכְמֵי הָעִבּוּר "מוֹלָד זָקֵן". לְפִי שֶׁכְּשֶׁהַמּוֹלָד חָל בְּסוֹף י״ח שָׁעוֹת — וְהוּא רֶגַע חִבּוּר הַלְּבָנָה עִם הַחַמָּה — אִי אֶפְשָׁר לִרְאוֹת אוֹר הַלְּבָנָה אֶלָּא עַד ו׳ שָׁעוֹת אַחַר כָּךְ (כְּמוֹ שֶׁנִּתְבָּאֵר לְעֵיל, רֵישׁ פֶּרֶק א׳, מִגְּמָרָא דְּרֹאשׁ הַשָּׁנָה); וְאִם כֵּן, לֹא הָיָה אֶפְשָׁר, בִּזְמַן שֶׁהָיוּ מְקַדְּשִׁים עַל פִּי הָרְאִיָּה, לִרְאוֹת הַלְּבָנָה בְּאוֹתוֹ יוֹם הַמּוֹלָד. וְאִם חָל יוֹם אד״ו לְמָחֳרָתוֹ — כְּגוֹן שֶׁחָל הַמּוֹלָד בְּשַׁבָּת בְּסוֹף י״ח שָׁעוֹת, וּלְמָחֳרָתוֹ יוֹם א׳ — הֲרֵי הוּא נִדְחֶה לְמָחֳרַת מָחֳרָתוֹ, שֶׁהוּא יוֹם
