~ הָיְתָה מֵעָבִים (עַיֵּן בְּפֵרוּשׁ הַמִּשְׁנָה לְהָרַמְבַּ״ם, וְלִכְאוֹרָה מְפָרֵשׁ כְּרַשִׁ״י; אֲבָל עַיֵּן בְּהָרַ״ע בַּרְטְנוּרָא וְתוֹסְפוֹת יוֹם טוֹב, בְּמָה שֶּׁכָּתְבוּ "וְנִסְתֶּרֶת בְּאוֹר הַשֶּׁמֶשׁ בִּכְמוֹ ב׳ יָמִים" וְכוּלֵי — נִרְאֶה דִּמְפָרֵשׁ כְּרַבּוֹתָיו דְּרַשִׁ״י; וּלְפִי פֵּרוּשָׁם דְּרַבָּן גַּמְלִיאֵל מִלְּתָא אַחֲרִיתִי — נִיחָא קֻשְׁיַת רַשִׁ״י עֲלֵיהֶם. אֲבָל מֵהַיְרוּשַׁלְמִי שֶׁהֵבִיאוּ תּוֹסָפוֹת (כ״ד עַמּוּד ב׳) דִּבּוּר הַמַּתְחִיל "רְאִינוּהוּ", שֶׁכָּתַב "וְלָמָּה קִבְּלָהּ רַבָּן גַּמְלִיאֵל? שֶׁכֵּן מְסֹרֶת" וְכוּלֵי — לֹא מַשְׁמַע כֵּן, וְיְעֻיַּן שָׁם בַּיְרוּשַׁלְמִי עַצְמוֹ). וְכֵן אִם רָאוּהוּ בִּזְמַנּוֹ, וּבְלֵיל עִבּוּרוֹ לֹא נִרְאָה — הֲרֵי אֵלּוּ נֶאֱמָנִים, שֶׁאֵין אָנוּ אַחְרָאִים אֶלָּא לִרְאִיַּת לֵיל ל׳ בִּלְבַד; וּמַה שֶּׁלֹּא רָאוּ לֵיל ל״א — אֵין לָחוּשׁ, דְּ"לֹא רְאִינוּ" אֵין רְאָיָה; וְהַבֵּית דִּין מְקַדְּשִׁים יוֹם ל׳ כְּעֵדוּת הָעֵדִים, וְאֵין לָהֶם לָחוּשׁ כְּלָל:

**אֵין** הָעֵדִים הוֹלְכִים לְהָעִיד אֶלָּא כְּשֶׁיֵּשׁ בֵּינֵיהֶם וּבֵין מְקוֹם הַבֵּית דִּין מַהֲלַךְ לַיְלָה וָיוֹם אוֹ פָּחוֹת. (וְנִרְאֶה: דְּרָאוּ הַלְּבָנָה בִּתְחִלַּת לֵיל ל׳; אֲבָל רָאוּהָ בִּתְחִלַּת יוֹם ל׳ — צָרִיךְ שֶׁיִּהְיֶה רַק מַהֲלַךְ יוֹם. וְדַוְקָא אַחַר תַּקָּנַת רַבִּי יוֹחָנָן בֶּן זַכַּאי, שֶׁיִּתְבָּאֵר לְהַלָּן, לְקַדֵּשׁ אַחַר הַמִּנְחָה; אֲבָל מִקַּמֵּי הָכֵי צָרִיךְ שֶׁיִּהְיֶה מֶרְחָק שֶׁיַּגִּיעַ לְבֵית דִּין קֹדֶם מִנְחָה, דְּאַחַר כָּךְ לֹא יְקַבְּלוּ עֵדוּתָם.) אֲבָל הָיָה בֵּינֵיהֶם יוֹתֵר עַל כָּךְ — לֹא יֵלְכוּ לְהָעִיד, הוֹאִיל וּכְבָר נִתְעַבֵּר הַחֹדֶשׁ. וְאִם הָיָה בֵּינֵיהֶם מַהֲלָךְ הַנִּזְכָּר — הֲרֵי הֵם מְחַלְּלִים שַׁבָּת, אִם חָל בְּשַׁבָּת, לֵילֵךְ לְהָעִיד (וַאֲפִלּוּ מַכְשִׁירֵי דָּחוּ שַׁבָּת). וַאֲפִלּוּ לֹא הָיָה נַפְקָא מִינָּהּ לְעִנְיַן קָרְבָּן — מְחַלְּלִין, דְּמִצְוָה לְקַדֵּשׁ עַל הָרְאִיָּה (דַּף כ׳ רֵישׁ עַמּוּד א׳). וְעַיֵּן בְּחִדּוּשֵׁי הָרַשְׁבָּ״א (כ״א עַמּוּד ב׳) דִּבּוּר הַמַּתְחִיל "עַל קְרִיאָתָן", וְעַיֵּן שָׁם דְּמוּכָח דַּאֲפִלּוּ לֹא קִדְּשׁוּ בְּיוֹם ל׳ — דְּעַל כָּל פָּנִים יְקַדְּשׁוּ בְּל״א — מְחַלְּלִין הָעֵדִים, דְּמִצְוָה לְקַדֵּשׁ עַל הָרְאִיָּה, עַיֵּן שָׁם. וְדַעַת הָרַמְבַּ״ם רֵישׁ פֶּרֶק א׳ אֵינוֹ כֵּן, עַיֵּן מַה שֶּׁכָּתַבְתִּי לְעֵיל):

**וּבִזְמַן** שֶׁבֵּית הַמִּקְדָּשׁ הָיָה קַיָּם — הָיוּ מְחַלְּלִין עַל כָּל הַחֳדָשִׁים, דְּנַפְקָא מִינָּהּ לְעִנְיַן קָרְבַּן מוּסָף דְּרֹאשׁ חֹדֶשׁ, לָדַעַת אֵימָתַי; וְקָרְבַּן מוּסָף דּוֹחֶה שַׁבָּת, וּלְכָךְ גַּם זוֹ הַהֲלִיכָה דּוֹחָה שַׁבָּת. וּלְאַחַר שֶׁחָרַב בֵּית הַמִּקְדָּשׁ הִתְקִין רַבִּי יוֹחָנָן בֶּן זַכַּאי (בַּגְּמָרָא כ״א עַמּוּד ב׳ אִיתָא "הִתְקִינוּ" וְכוּלֵי; מִיהוּ שָׁם ל״א עַמּוּד ב׳ מְבֹאָר דְּרַבִּי יוֹחָנָן בֶּן זַכַּאי תִּקְּנָהּ) שֶׁלֹּא יְחַלְּלוּ הַשַּׁבָּת אֶלָּא עַל רֹאשׁ חֹדֶשׁ נִיסָן וְרֹאשׁ חֹדֶשׁ תִּשְׁרֵי בִּלְבַד, מִפְּנֵי תַּקָּנַת הַמּוֹעֲדוֹת (כֵּן מַשְׁמָע בַּגְּמָרָא וְרַשִׁ״י וְתוֹסָפוֹת. אֲבָל דִּבְרֵי הָרַמְבַּ״ם, שֶׁכָּתַב "לְפִיכָךְ" וְכוּלֵי — נִרְאֶה לִכְאוֹרָה דִּסְבִירָא לֵיהּ כְּקֻשְׁיַת הַתּוֹסָפוֹת דִּבּוּר הַמַּתְחִיל "עַל ב׳", וְלֵית לֵיהּ תֵּרוּצָם, עַיֵּן שָׁם; רַק תַּקָּנָה הָיְתָה בִּזְמַן שֶׁבֵּית הַמִּקְדָּשׁ קַיָּם, וּכְשֶׁאֵין בֵּית הַמִּקְדָּשׁ קַיָּם הִתְקִין רַבִּי יוֹחָנָן בֶּן זַכַּאי שֶׁיַּחֲזוֹר הַדָּבָר לִמְקוֹמוֹ; וְעַיֵּן בְּלֶחֶם מִשְׁנֶה). וְאֶחָד עֵדֵי הָרְאִיָּה וְאֶחָד עֵדִים שֶׁמַּכִּירִים לְעֵדֵי הָרְאִיָּה שֶׁכְּשֵׁרִים — מְחַלְּלִין הַשַּׁבָּת,
