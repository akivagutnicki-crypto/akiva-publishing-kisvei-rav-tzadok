~ "מַתְנִיתִין" — כָּתְבוּ "וְאִם תֹּאמַר" וְכוּלֵי, עַיֵּן שָׁם. וּבַעֲנִיּוּתִי לֹא יָדַעְתִּי קֻשְׁיָתָם: הָא מְפֹרָשׁ בְּפֶרֶק קַמָּא דְּסַנְהֶדְרִין (י״א עַמּוּד ב׳) שֶׁכּוֹתְבִין אִגְּרוֹת לְהוֹדִיעַ לִבְנֵי גּוֹלָה כֻּלָּם, וְכָל שֶׁכֵּן לְאוֹתָם שֶׁהַשְּׁלוּחִים מַגִּיעִים — בְּאִגְּרוֹת מַגִּיעִים לָהֶם מִקֹּדֶם; וְאַף דְּבָאִגְּרוֹת כָּתוּב גַּם כֵּן אִם מָלֵא אִם חָסֵר, אוֹ חֹדֶשׁ אוֹ ל׳ יוֹם — מִכָּל מָקוֹם מַתְנִיתִין, דְּלֹא חָשֵׁיב אֶלָּא בְּהָדֵי שִׁשָּׁה חֳדָשִׁים, וְלֹא זָכַר אִגְּרוֹת — וּמַאי שְׁנָא אִגְּרוֹת אוֹ שָׁלִיחַ? שְׁמַע מִינָּהּ דְּאֵין צָרִיךְ, וְשַׁפִּיר קָאָמַר. וְאִם כֵּן, הָרַמְבַּ״ם סָמִיךְ עַל מַה שֶּׁכָּתַב (פֶּרֶק ד׳ הֲלָכָה י״ז): כּוֹתְבִין אִגְּרוֹת לְכָל הַמְּקוֹמוֹת הָרְחוֹקִים וְכוּלֵי, עַיֵּן שָׁם. וְוַדַּאי עֲדִיפָא אִגְּרוֹת מִשְּׁלוּחִין; וְעַל כָּל פָּנִים הֲווּ כִּשְׁלוּחִין. כֵּן נִרְאֶה פָּשׁוּט וּבָרוּר):

**וּבִזְמַן** שֶׁבֵּית הַמִּקְדָּשׁ קַיָּם — יוֹצְאִין אַף עַל אִיָּר, מִפְּנֵי פֶּסַח שֵׁנִי. וְעַל שָׁבוּעוֹת אֵין צְרִיכִים לָצֵאת, מִפְּנֵי שֶׁהוּא תָּלוּי בִּסְפִירַת נ׳ יוֹם מִפֶּסַח, לֹא בִּימֵי הַחֹדֶשׁ. וְאֵימָתַי הַשְּׁלוּחִים יוֹצְאִים? בָּעֶרֶב, כְּשֶׁרוֹאִים שֶׁנִּרְאָה הַיָּרֵחַ, וּבְוַדַּאי יְקַדְּשׁוּהוּ מָחָר; אַף עַל פִּי שֶׁעֲדַיִן לֹא קִדְּשׁוּהוּ בֵּית דִּין, מִפְּנֵי שֶׁאֵין מְקַדְּשִׁין בַּלַּיְלָה — מִכָּל מָקוֹם יוֹצְאִים מִיָּד, שֶׁהֲרֵי וַדַּאי יְקַדְּשׁוּהוּ. בַּמֶּה דְּבָרִים אֲמוּרִים — בְּכָל הַחֳדָשִׁים; אֲבָל בְּנִיסָן וְתִשְׁרֵי אֵין יוֹצְאִים אֶלָּא בְּיוֹם רֹאשׁ חֹדֶשׁ, אַחַר שֶׁתַּעֲלֶה הַשֶּׁמֶשׁ, עַד שֶׁשָּׁמְעוּ מִפִּי בֵּית דִּין "מְקֻדָּשׁ" (וּבְתִשְׁרֵי, שֶׁהוּא רֹאשׁ הַשָּׁנָה — לֹא יָצְאוּ בַּיּוֹם אֶלָּא לְתוֹךְ הַתְּחוּם). וְאִם קִדְּשׁוּ בֵּית דִּין בְּסוֹף יוֹם כ״ט, כְּמוֹ שֶׁנִּתְבָּאֵר לְעֵיל, וְשָׁמְעוּ מִפִּי בֵּית דִּין "מְקֻדָּשׁ" — יוֹצְאִים בָּעֶרֶב. וְאִם לֹא קִדְּשׁוּ בֵּית דִּין יוֹם ל׳ — הֲרֵי הַחֹדֶשׁ מְעֻבָּר, יוֹצְאִים מִיָּד, תְּחִלַּת לֵיל ל״א, שֶׁאֵין הַחֹדֶשׁ יוֹתֵר מִל״א יוֹם, וְעַל כָּרְחָךְ יִהְיֶה רֹאשׁ חֹדֶשׁ יוֹם ל״א; וְאֵין הַבֵּית דִּין מְקַדְּשִׁין כְּלָל חֹדֶשׁ מְעֻבָּר, כְּמוֹ שֶׁנִּתְבָּאֵר לְעֵיל (כֵּן מַשְׁמַע דַּעַת הָרַמְבַּ״ם; וְרַשִׁ״י (כ״א עַמּוּד ב׳), דִּבּוּר הַמַּתְחִיל "שְׁלוּחִין" וְכוּלֵי, כָּתַב "בְּחֹדֶשׁ מְעֻבָּר" וְכוּלֵי. מִיהוּ, יֵשׁ לוֹמַר דְּאָתָאן דְּלָא כְּרַבִּי אֶלְעָזָר בְּרַבִּי צָדוֹק. וְעַיֵּן שָׁם בַּתּוֹסָפוֹת, דִּבּוּר הַמַּתְחִיל "עַל נִיסָן", בְּסוֹף הַדִּבּוּר: "וְאִם תֹּאמַר" וְכוּלֵי, "וְיֵשׁ לוֹמַר דִּגְזוֹרוּ הַאי אַטּוּ הַאי" וְכוּלֵי. וְאֶפְשָׁר לוֹמַר טַעַם זֶה גַּם לְרַבִּי אֶלְעָזָר בְּרַבִּי צָדוֹק, וְצָרִיךְ עִיּוּן לְדִינָא. וְעַיֵּן (שָׁם עַמּוּד א׳): "לֵוִי אִיקְלַע" וְכוּלֵי, "מִפִּי בֵּית דִּין מְקֻדָּשׁ" — עַיֵּן שָׁם בְּרַשִׁ״י, וְתוֹסָפוֹת דִּבּוּר הַמַּתְחִיל "לֵוִי", דְּשָׁם בְּל״א מַשְׁמַע — דַּאֲפִלּוּ הָכֵי אֵין יוֹצְאִין אֶלָּא כְּשֶׁשָּׁמַע מִבֵּית דִּין "מְקֻדָּשׁ"; וְעַיֵּן מַה שֶּׁכָּתַבְתִּי לְמַעְלָה מִזֶּה. וְעַיֵּן בְּסֵפֶר טוּרֵי אֶבֶן, שָׁם הֶאֱרִיךְ לְהַקְשׁוֹת אַתּוֹסָפוֹת, וְהֵבִיא דִּבְרֵי הָרַזָ״ה, שֶׁפֵּרֵשׁ בְּדֶרֶךְ אַחֵר, וּפֵרוּשׁוֹ — וּלְדִבְרֵיהֶם אֵין מִלּוּי רְאָיָה. וְלוּלֵי דְּמִסְתַּפִּינָא, הָיִיתִי מְפָרֵשׁ לַעֲנִיּוּת דַּעְתִּי: דְּהָכֵי קָאָמַר, אִי לָאו "לֹא שָׁמַעְתִּי" וְכוּלֵי — הַיְנוּ אַסְהֲדוּתֵיהּ; כֵּיוָן דְּקַיְימָא לֵיהּ כְּרַבִּי אֶלְעָזָר בְּרַבִּי צָדוֹק, דְּרַק בְּיוֹם ל׳ אוֹמֵר "מְקֻדָּשׁ", לֹא בְּל״א; וּכְשֶׁאָמַר לוֹ "אַסְהִיד" — הָעֵד, שֶׁלֹּא שָׁמַע מִפִּי בֵּית דִּין "מְקֻדָּשׁ"; וְאִם כֵּן הֲוֵי ל״א רֹאשׁ חֹדֶשׁ, שֶׁהֲרֵי הָעִקָּר בִּ"מְקֻדָּשׁ" תָּלוּי; וַאֲפִלּוּ לֹא הִסְפִּיקוּ בֵּית דִּין לוֹמַר "מְקֻדָּשׁ" — אֵינוֹ קָדוֹשׁ, וּכְמוֹ שֶׁנִּתְבָּאֵר לְעֵיל, וְהָבֵן. אֲבָל מָה אֶעֱשֶׂה, וְרַבּוֹתֵינוּ
