~ יִהְיֶה בֵּינָהּ לְרֹאשׁ הַשָּׁנָה הַבָּאָה אֶלָּא ד׳ יָמִים; כִּי אִם יִהְיֶה ה׳ יָמִים — יָבוֹא רֹאשׁ הַשָּׁנָה בְּיוֹם א׳, וְאִם ג׳ — יִהְיֶה בְּיוֹם ו׳; וּשְׁנֵי אֵלֶּה אִי אֶפְשָׁר לְפִי הַכְּלָל שֶׁבֵּאַרְנוּ, שֶׁלֹּא אד״ו רֹאשׁ; וְעַל כָּרְחָךְ יִהְיֶה רֹאשׁ הַשָּׁנָה הַבָּא בְּיוֹם ז׳, וּבֵינְתַיִם ד׳ יָמִים — וְהִיא כְּסִדְרָן. וְאִם הִיא מְעֻבֶּרֶת — הִנֵּה לִהְיוֹת ה׳ יָמִים בֵּינְתַיִם לֹא יִתָּכֵן, שֶׁאִם כֵּן יִהְיֶה רֹאשׁ הַשָּׁנָה בְּיוֹם א׳; וְעַל כָּרְחָךְ ו׳ אוֹ ז׳. וְכֵיוָן שֶׁיּוֹם ו׳ הוּא בִּשְׁנֵי — וְשָׁנָה מְעֻבֶּרֶת יִתְרוֹנָהּ הוּא ה׳ כ״א תקפ״ט, שֶׁהוּא קָרוֹב לְו׳ יָמִים — אִם כֵּן, לָמָּה נִדְחֵהוּ לְיוֹם ז׳?

**וְכָל** רֹאשׁ הַשָּׁנָה שֶׁיִּהְיֶה בְּשַׁבָּת אוֹ בְּב׳ — לֹא תִּהְיֶה כְּסִדְרָן לְעוֹלָם, בֵּין פְּשׁוּטָה בֵּין מְעֻבֶּרֶת, אֶלָּא אוֹ חֲסֵרָה אוֹ מְלֵאָה. וְזֶה מְבֹאָר לְפִי הַכְּלָל לֹא אד״ו רֹאשׁ: כִּי אִם יִהְיֶה כְּסִדְרָן, וְיִהְיֶה בֵּינְתַיִם ד׳ אִם הִיא פְּשׁוּטָה, אוֹ ו׳ אִם הִיא מְעֻבֶּרֶת — יָבוֹא רֹאשׁ הַשָּׁנָה הַבָּא בְּאֶחָד מִימֵי אד״ו, וְזֶה אִי אֶפְשָׁר, כְּמוֹ שֶׁנִּתְבָּאֵר. וְכָל רֹאשׁ הַשָּׁנָה שֶׁיִּהְיֶה בְּה׳ — אִם פְּשׁוּטָה הִיא, אִי אֶפְשָׁר שֶׁיִּהְיוּ חֳדָשִׁים חֲסֵרִים, שֶׁהוּא ג׳ יָמִים בֵּינְתַיִם; כִּי אִם כֵּן, יִהְיֶה רֹאשׁ הַשָּׁנָה הַבָּא בְּיוֹם א׳ — דָּבָר שֶׁאֵינוֹ. וְאִם מְעֻבֶּרֶת — אִי אֶפְשָׁר שֶׁיִּהְיוּ חֳדָשִׁים כְּסִדְרָן, דְּהַיְנוּ ו׳ יָמִים בֵּינְתַיִם, וְיָבוֹא רֹאשׁ הַשָּׁנָה בְּיוֹם ד׳ — שֶׁזֶּה גַּם כֵּן אֵינוֹ, כְּמוֹ שֶׁנִּתְבָּאֵר:

**וּמִדְּבָרֵינוּ** אַתָּה לָמֵד, כִּי י״ד אֳפַנֵּי קְבִיעוּת שָׁנָה יֵשׁ: ז׳ מֵהֶם לִפְשׁוּטָה, וְז׳ לִמְעֻבֶּרֶת. וְאוּלָם ז׳ קְבִיעוּת הַפְּשׁוּטָה הֵם אֵלּוּ: (א) כְּשֶׁיָּחוּל רֹאשׁ הַשָּׁנָה בְּיוֹם ב׳, וְהַשָּׁנָה חֲסֵרָה — וְאָז רֹאשׁ חֹדֶשׁ נִיסָן וּפֶסַח בְּיוֹם ג׳; וְהַסִּימָן: בח״ג. (ב) כַּנִּזְכָּר לְעֵיל, וְהַשָּׁנָה שְׁלֵמָה (אֲבָל כְּסִדְרָן אִי אֶפְשָׁר, כַּנִּזְכָּר לְעֵיל) — וְאָז נִיסָן וּפֶסַח בְּיוֹם ה׳; וְהַסִּימָן: בש״ה. (ג) בִּהְיוֹת רֹאשׁ הַשָּׁנָה בְּיוֹם ג׳, וְהַשָּׁנָה עַל כָּרְחָךְ כְּסִדְרָן — וְאָז נִיסָן וּפֶסַח גַּם כֵּן בְּיוֹם ה׳; וְהַסִּימָן: גכ״ה. (ד) בִּהְיוֹתוֹ בְּה׳, וְשָׁנָה כְּסִדְרָן — וְנִיסָן וּפֶסַח אָז בְּז׳; הַסִּימָן: הכ״ז. (ה) כַּנִּזְכָּר לְעֵיל, וְהַשָּׁנָה שְׁלֵמָה — וְנִיסָן וּפֶסַח בְּא׳; וְסִימָנוֹ: הש״א. (ו) בִּהְיוֹתוֹ בְּז׳, וַחֲסֵרָה — וְנִיסָן וּפֶסַח אָז גַּם כֵּן בְּא׳; סִימָנוֹ: זח״א. (ז) כַּנִּזְכָּר לְעֵיל, וּשְׁלֵמָה — וְנִיסָן וּפֶסַח בְּג׳; סִימָנוֹ: זש״ג. אֵלֶּה הֵם קְבִיעוּת שְׁנַת פְּשׁוּטָה:

**וּקְבִיעוּת** שְׁנַת מְעֻבֶּרֶת הֵם אֵלּוּ: (א) בְּחוּל רֹאשׁ הַשָּׁנָה בְּב׳, וְשָׁנָה חֲסֵרָה — וְאָז נִיסָן וּפֶסַח בְּה׳; הַסִּימָן: בח״ה. (ב) כַּנִּזְכָּר לְעֵיל, וּשְׁלֵמָה — וְנִיסָן וּפֶסַח אָז בְּז׳; הַסִּימָן: בש״ז. (ג) בְּחוּלוֹ בְּג׳, וּקְבִיעוּת הַשָּׁנָה עַל כָּרְחָךְ כְּסִדְרָן — וְאָז נִיסָן וּפֶסַח בְּז׳; סִימָנוֹ: גכ״ז. (ד) בְּחוּלוֹ בְּה׳, וַחֲסֵרָה — וְנִיסָן וּפֶסַח בְּא׳; סִימָנוֹ: הח״א.
