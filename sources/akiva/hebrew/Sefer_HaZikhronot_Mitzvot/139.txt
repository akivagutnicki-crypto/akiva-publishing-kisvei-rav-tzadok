**וְאֵין** צֹרֶךְ לְמַה שֶּׁנִּתְיַגְּעוּ הָרִאשׁוֹנִים זִכְרוֹנָם לִבְרָכָה — וּבְרֹאשָׁם עֲטֶרֶת תִּפְאֶרֶת יִשְׂרָאֵל, הָרַמְבַּ״ם זִכְרוֹנוֹ לִבְרָכָה — לְהָבִיא מוֹפְתִים עַל מְצִיאוּת הַבּוֹרֵא יִתְבָּרַךְ וְהַשְׁגָּחָתוֹ בָּזֶה הָעוֹלָם, מוֹפְתִים פִילוֹסוֹפִיִּים; בְּאָמְרָם שֶׁזֶּהוּ עִקַּר מִצְוַת "אָנֹכִי ה׳" וְגוֹ׳ — שֶׁיֵּדַע זֶה בִּידִיעַת הַלֵּב; וְאֵין יְדִיעַת הַלֵּב בְּעִקָּר וּבֶאֱמֶת אֶלָּא עַל יְדֵי הַמּוֹפֵת, וְלֹא עַל פִּי הַקַּבָּלָה מֵאֲבוֹתָיו. וְהֶאֱרִיכוּ בָּזֶה, וּמָשְׁלוּ מָשָׁל: לְמָה הַדָּבָר דּוֹמֶה — לְחֶבֶר בְּנֵי אָדָם נִסְמָכִים כָּל אֶחָד עַל גַּב חֲבֵרוֹ, אֲשֶׁר בִּהְיוֹת הָרִאשׁוֹן נוֹפֵל בַּפַּחַת — כֵּן כָּל הָעָם יִפְּלוּ אַחֲרָיו, בִּהְיוֹת הוּא לָהֶם הַסִּבָּה הָרִאשׁוֹנָה. וְעַיֵּן בְּסִפְרֵיהֶם, שֶׁהֶאֱרִיכוּ בַּהֲבָאַת מוֹפְתֵי הַפִילוֹסוֹפִים עַל אֵלּוּ הַדְּבָרִים, וְחִבְּרוּ סְפָרִים הַרְבֵּה עַל זֶה, וְהָלְכוּ בְּעִקְּבוֹת אֲרִיסְטוֹ יִמַּח שְׁמוֹ, וְיֶתֶר מִפִילוֹסוֹפֵי הָעַמִּים, אֲשֶׁר עָמְלוּ לְחַבֵּר סְפָרִים בְּמוֹפְתִים בְּעִנְיָנִים אֵלּוּ:

**וַאֲנִי** אוֹמֵר: הִתְכַּבְּדוּ, מְכֻבָּדִים קְדוֹשִׁים מְשָׁרְתֵי עֶלְיוֹן, וּתְנוּ כָּבוֹד לֵאלֹקֵיכֶם בְּדֶרֶךְ הַתּוֹרָה, וְלֹא בְּדֶרֶךְ פִילוֹסוֹפֵי הָעַמִּים! וְהָגַע עַצְמְךָ: אִלּוּ רָאוּ אֲרִיסְטוֹ, שַׂר״י, הַמַּמְטֶרֶת לֶחֶם מִן הַשָּׁמַיִם, וְכַדּוֹמֶה מִן הַנִּסִּים; אוֹ אִלּוּ רָאָה קוֹלוֹת מַתַּן תּוֹרָה, וְשָׁמַע "אָנֹכִי" וְ"לֹא יִהְיֶה" בְּדִבּוּר אֶחָד, וּשְׁאָר נִפְלָאוֹת אֲשֶׁר יָעִידוּן וְיַגִּידוּן כִּי יֵשׁ אֱלֹקִים בַּשָּׁמַיִם — הַאִם הָיָה חוֹקֵר אַחַר מוֹפְתִים עַל מְצִיאוּת הַשֵּׁם וְהַשְׁגָּחָתוֹ? הֲיֵשׁ מוֹפֵת גָּדוֹל מִזֶּה? אֶלָּא שֶׁהוּא, שַׂר״י, לֹא הֶאֱמִין בִּדְבָרִים אֵלּוּ. אֲבָל אֲנַחְנוּ, זֶרַע קֹדֶשׁ — אֲשֶׁר לְעֵינֵי אֲבוֹתֵינוּ הָיוּ הַדְּבָרִים הָאֵלֶּה, לְעֵינֵי שִׁשִּׁים רִבּוֹא אֲנָשִׁים, וְלֹא בַּסֵּתֶר — אִם כֵּן, נָבִיא מִזֶּה גּוּפָא מוֹפֵת. וְלֹא אוֹמַר מִמַּה שֶּׁכָּתוּב בַּתּוֹרָה מְצִיאוּת הַשֵּׁם — שֶׁזֶּהוּ מֵהַקַּבָּלָה, וְהֵם יַחְפְּצוּ בְּמוֹפֵת; אֲבָל אוֹמַר דֶּרֶךְ מוֹפֵת, מֵהַנִּסִּים וְהַנִּפְלָאוֹת שֶׁנַּעֲשׂוּ לַאֲבוֹתֵינוּ בַּיָּמִים הָהֵם בִּזְמַן הַזֶּה — שֶׁזֶּהוּ מוֹפֵת שֶׁאֵין לְמַעְלָה מִמֶּנּוּ עַל מְצִיאוּתוֹ:

**וְאִם** נֹאמַר שֶׁמֵּאֲנוּ בָּזֶה, בַּעֲבוּר שֶׁגַּם עַל זֶה גּוּפֵיהּ אֵין מוֹפֵת שֶׁהוּא אֱמֶת, אֲבָל רַק מֵהַקַּבָּלָה — חַס וְשָׁלוֹם לוֹמַר כֵּן! כִּי אִם בָּאנוּ לָזֶה — אַיֵּה מָקוֹם לַתּוֹרָה? אֲבָל הֲלֹא לָנוּ הַדָּבָר נוֹדָע, כִּי לֹא נָחֲלוּ לָנוּ אֲבוֹתֵינוּ שֶׁקֶר; בְּשַׁגַּם, אִי אֶפְשָׁר לוֹמַר שֶׁקֶר כָּזֶה — מַה שֶּׁהָיָה לְעֵינֵי כָּל הָעַמִּים. וְהַדָּבָר דּוֹמֶה לְמִי שֶׁבְּדִמְיוֹן תָּמִיד, מֵעֵת הִוָּלְדוֹ, הָיָה בְּבוֹר תַּחְתִּיּוֹת, וְלֹא רָאָה הֲלִיכַת הַגַּלְגַּלִּים, אֲבָל סִפְּרוּ אֵלָיו מִכָּל הָאָדָם שֶׁבָּאוּ אֵלָיו שֶׁכֵּן הוּא, בְּאֹפֶן לֹא יִסְתַּפֵּק בַּדָּבָר — הַאִם יִמָּנַע לְהָבִיא מִזֶּה מוֹפֵת עַל מְצִיאוּת הַבּוֹרֵא (כְּפִי דַּרְכֵיהֶם)? וְכֵן לֹא נִמָּנַע אֲנַחְנוּ מִלְּהָבִיא מוֹפֵת מִמַּה שֶּׁיָּדוּעַ אֵלֵינוּ בְּבֵרוּר, שֶׁאֵין אַחֲרָיו סָפֵק — שֶׁאָכְלוּ אֲבוֹתֵינוּ הַמָּן, וּשְׁאָר מוֹפְתִים וְנִסִּים שֶׁאֵרְעוּ לַאֲבוֹתֵינוּ, הַמְבֹאָרִים בַּכְּתוּבִים; וּבְיוֹתֵר נֵס דַּחֲנַנְיָה מִישָׁאֵל וַעֲזַרְיָה, בְּגוֹב אֲרָיוֹת וּכְבַשַׁן אֵשׁ
